package com.example.shoppingassistant.core.network

import io.ktor.client.HttpClient
import io.ktor.client.engine.okhttp.OkHttp
import io.ktor.client.plugins.HttpRequestRetry
import io.ktor.client.plugins.HttpTimeout
import io.ktor.client.plugins.HttpRedirect
import io.ktor.client.plugins.HttpResponseRedirectEvent
import io.ktor.client.plugins.defaultRequest
import io.ktor.client.plugins.compression.ContentEncoding
import io.ktor.client.plugins.contentnegotiation.ContentNegotiation
import io.ktor.client.request.header
import io.ktor.http.HttpHeaders
import io.ktor.http.HttpHeaders.ContentEncoding
import io.ktor.http.URLProtocol
import io.ktor.serialization.kotlinx.json.json
import io.ktor.util.AttributeKey
import kotlinx.serialization.json.Json

class IngestClient(
    val client: HttpClient,
)

private const val MaxRedirects = 4
private val RedirectCountKey = AttributeKey<Int>("IngestRedirectCount")

fun createIngestHttpClient(): IngestClient {
    val httpClient = HttpClient(OkHttp) {
        expectSuccess = false

        install(ContentNegotiation) {
            json(
                Json {
                    ignoreUnknownKeys = true
                    isLenient = true
                    encodeDefaults = false
                    prettyPrint = false
                },
            )
        }

        install(HttpTimeout) {
            requestTimeoutMillis = 15_000
            connectTimeoutMillis = 10_000
            socketTimeoutMillis = 15_000
        }

        install(HttpRedirect) {
            allowHttpsDowngrade = false
            checkHttpMethod = false
        }

        install(HttpRequestRetry) {
            retryOnException(maxRetries = 2)
            retryIf(maxRetries = 1) { _, response ->
                val code = response.status.value
                code == 408 || code == 425 || code == 429 || code in 500..504
            }
        }

        install(ContentEncoding) {
            gzip()
        }

        defaultRequest {
            header(HttpHeaders.UserAgent, "Mozilla/5.0 (compatible; ShoppingAssistantBot/1.0)")
            header(HttpHeaders.Accept, "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8")
            header(HttpHeaders.AcceptLanguage, "ru-RU,ru;q=0.9,en-US;q=0.8,en;q=0.7")
            header(HttpHeaders.AcceptEncoding, "gzip")
            url {
                protocol = protocol.takeIf { it == URLProtocol.HTTP || it == URLProtocol.HTTPS }
                    ?: URLProtocol.HTTPS
            }
        }
    }

    httpClient.monitor.subscribe(HttpResponseRedirectEvent) { response ->
        val attributes = response.call.request.attributes
        val redirects = attributes.getOrNull(RedirectCountKey) ?: 0
        val next = redirects + 1
        if (next > MaxRedirects) {
            throw IllegalStateException("Too many redirects ($next) for ${response.call.request.url}")
        }
        attributes.put(RedirectCountKey, next)
    }

    return IngestClient(httpClient)
}
