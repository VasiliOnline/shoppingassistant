package com.example.shoppingassistant.core.push

import android.app.NotificationChannel
import android.app.NotificationManager
import android.content.Context
import android.os.Build

object TrackingNotificationsChannel {
    // New channel id to avoid keeping the user-facing "Подписки" channel name from older installs.
    const val CHANNEL_ID: String = "tracking_alerts"
    private const val CHANNEL_NAME: String = "Отслеживания"
    private const val CHANNEL_DESC: String = "Уведомления о новых событиях по отслеживаниям"

    fun ensureCreated(context: Context) {
        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.O) return

        val manager = context.getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        val existing = manager.getNotificationChannel(CHANNEL_ID)
        if (existing != null) return

        val channel = NotificationChannel(
            CHANNEL_ID,
            CHANNEL_NAME,
            NotificationManager.IMPORTANCE_DEFAULT,
        ).apply {
            description = CHANNEL_DESC
        }

        manager.createNotificationChannel(channel)
    }
}

@Deprecated("Use TrackingNotificationsChannel")
typealias SubscriptionsNotificationsChannel = TrackingNotificationsChannel
