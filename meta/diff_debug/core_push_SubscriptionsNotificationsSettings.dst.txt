package com.example.shoppingassistant.core.push

import kotlinx.serialization.Serializable

@Serializable
enum class NotificationPriority {
    LOW,
    DEFAULT,
    HIGH,
}

@Serializable
data class SubscriptionsNotificationsSettings(
    val pushEnabled: Boolean = true,
    val quietHoursEnabled: Boolean = false,
    val quietHoursStart: Int = 22,
    val quietHoursEnd: Int = 8,
    val groupNotifications: Boolean = true,
    val priority: NotificationPriority = NotificationPriority.DEFAULT,
)

interface SubscriptionsNotificationsSettingsStorage {
    fun get(): SubscriptionsNotificationsSettings
    fun set(settings: SubscriptionsNotificationsSettings)
    fun update(block: (SubscriptionsNotificationsSettings) -> SubscriptionsNotificationsSettings) {
        set(block(get()))
    }
}
