package com.example.shoppingassistant.core.push

import android.content.Context
import androidx.work.ExistingPeriodicWorkPolicy
import androidx.work.ExistingWorkPolicy
import androidx.work.OneTimeWorkRequestBuilder
import androidx.work.PeriodicWorkRequestBuilder
import androidx.work.WorkManager
import java.util.concurrent.TimeUnit

class SubscriptionsPushSchedulerImpl(
    private val context: Context,
) : SubscriptionsPushScheduler {

    private val workManager get() = WorkManager.getInstance(context)

    override fun ensureScheduled() {
        SubscriptionsNotificationsChannel.ensureCreated(context)

        workManager.enqueueUniqueWork(
            ONE_TIME_WORK_NAME,
            ExistingWorkPolicy.KEEP,
            OneTimeWorkRequestBuilder<SubscriptionsPushWorker>()
                .build(),
        )

        workManager.enqueueUniquePeriodicWork(
            PERIODIC_WORK_NAME,
            ExistingPeriodicWorkPolicy.KEEP,
            PeriodicWorkRequestBuilder<SubscriptionsPushWorker>(15, TimeUnit.MINUTES)
                .build(),
        )
    }

    override fun cancel() {
        workManager.cancelUniqueWork(ONE_TIME_WORK_NAME)
        workManager.cancelUniqueWork(PERIODIC_WORK_NAME)
    }

    private companion object {
        const val ONE_TIME_WORK_NAME: String = "subscriptions_push_once"
        const val PERIODIC_WORK_NAME: String = "subscriptions_push_periodic"
    }
}
