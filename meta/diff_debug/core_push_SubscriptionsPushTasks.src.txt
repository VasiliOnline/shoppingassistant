package com.example.shoppingassistant.core.push

import kotlinx.serialization.Serializable

/**
 * Контракты "пуша" для отслеживаний.
 *
 * В проде это может быть FCM/APNs, но в текущей реализации уведомления доставляются
 * через периодический фоновой поллинг серверного inbox и показ системных нотификаций.
 */
interface TrackingPushScheduler {
    fun ensureScheduled()
    fun cancel()
}

interface TrackingPushStateStorage {
    fun getLastNotifiedId(): Long
    fun setLastNotifiedId(id: Long)
}

interface TrackingInboxRemoteDataSource {
    suspend fun listNotificationsPage(limit: Int, offset: Int): TrackingInboxPage
    suspend fun markRead(id: Long): Boolean
    suspend fun markAllRead(): Int
}

interface TrackingPushTokensRemoteDataSource {
    suspend fun registerToken(platform: String, token: String, deviceId: String? = null)
}

@Serializable
data class TrackingInboxPage(
    val items: List<TrackingInboxNotification>,
    val limit: Int,
    val offset: Int,
    val canLoadMore: Boolean,
)

@Serializable
data class TrackingInboxNotification(
    val id: Long,
    val alertId: Long,
    val offerId: Long,
    val priceHistoryId: Long,
    val createdAt: Long,
    val priceCollectedAt: Long,
    val isRead: Boolean,
    val message: String,
    val dropPercent: Double,
    val baselinePriceMinor: Long,
    val currentPriceMinor: Long,
    val currency: String,
)

@Deprecated("Use TrackingPushScheduler")
typealias SubscriptionsPushScheduler = TrackingPushScheduler

@Deprecated("Use TrackingPushStateStorage")
typealias SubscriptionsPushStateStorage = TrackingPushStateStorage

@Deprecated("Use TrackingInboxRemoteDataSource")
typealias SubscriptionsInboxRemoteDataSource = TrackingInboxRemoteDataSource

@Deprecated("Use TrackingPushTokensRemoteDataSource")
typealias SubscriptionsPushTokensRemoteDataSource = TrackingPushTokensRemoteDataSource

@Deprecated("Use TrackingInboxPage")
typealias SubscriptionsInboxPage = TrackingInboxPage

@Deprecated("Use TrackingInboxNotification")
typealias SubscriptionsInboxNotification = TrackingInboxNotification
