package com.example.shoppingassistant.core.push

import com.example.shoppingassistant.core.config.BackendConfig
import com.example.shoppingassistant.core.network.BackendClient
import com.example.shoppingassistant.domain.auth.AuthRepository
import io.ktor.client.request.header
import io.ktor.client.request.post
import io.ktor.client.request.setBody
import io.ktor.client.statement.HttpResponse
import io.ktor.http.ContentType
import io.ktor.http.HttpStatusCode
import io.ktor.http.contentType
import io.ktor.http.isSuccess
import kotlinx.serialization.Serializable

class SubscriptionsPushTokensRemoteDataSourceImpl(
    private val backendClient: BackendClient,
    private val authRepository: AuthRepository,
) : SubscriptionsPushTokensRemoteDataSource {

    private val baseUrl get() = BackendConfig.BASE_URL

    override suspend fun registerToken(platform: String, token: String, deviceId: String?) {
        val bearer = authRepository.currentToken() ?: return

        val response: HttpResponse = backendClient.client.post("$baseUrl/api/push/tokens") {
            header("Authorization", ensureBearer(bearer))
            contentType(ContentType.Application.Json)
            setBody(RegisterPushTokenRequest(platform = platform, token = token, deviceId = deviceId))
        }

        if (response.status == HttpStatusCode.Unauthorized) {
            authRepository.logout()
            return
        }
        if (!response.status.isSuccess()) {
            throw IllegalStateException("Failed to register push token: HTTP ${response.status.value}")
        }
    }

    private fun ensureBearer(token: String): String =
        if (token.startsWith("Bearer ")) token else "Bearer $token"
}

@Serializable
private data class RegisterPushTokenRequest(
    val platform: String,
    val token: String,
    val deviceId: String? = null,
)
