package com.example.shoppingassistant.core.rank

import com.example.shoppingassistant.domain.model.NormalizedQuery
import com.example.shoppingassistant.domain.model.ProductDto

/**
 * Единая «точка истины» для ранжирования кандидатов.
 * Детерминированные тай-брейки и централизованный расчёт avgPrice.
 */
class RankService(
    private val engine: ScoringEngine
) {

    /** Короткий путь: только DTO тройки лидеров. */
    fun top3Dtos(candidates: List<ProductDto>, query: NormalizedQuery): List<ProductDto> =
        topN(candidates, query, n = 3).map { (dto, _) -> dto }

    /** Короткий путь: вернуть тройку с разложением score. */
    fun top3(candidates: List<ProductDto>, query: NormalizedQuery): List<Pair<ProductDto, ScoreBreakdown>> =
        topN(candidates, query, n = 3)

    /**
     * Вернуть Top-N пар (ProductDto, ScoreBreakdown), отсортированных по убыванию score.
     * Тай-брейки для стабильности UI:
     *  1) score ↓
     *  2) price ↑  (дешевле — выше)
     *  3) deliveryTime ↑ (быстрее — выше)
     *  4) sellerRating ↓ (выше рейтинг — выше)
     */
    fun topN(
        candidates: List<ProductDto>,
        query: NormalizedQuery,
        n: Int = 3
    ): List<Pair<ProductDto, ScoreBreakdown>> {
        val prefiltered = candidates.filter { it.price != null } // без цены ранжировать нельзя
        val avg = prefiltered.mapNotNull { it.price }.let { if (it.isNotEmpty()) it.average() else 0.0 }

        return prefiltered
            .asSequence()
            .map { dto -> dto to engine.score(dto, query, avg) }
            .sortedWith(
                compareByDescending<Pair<ProductDto, ScoreBreakdown>> { it.second.score }
                    .thenBy { it.first.price ?: Double.MAX_VALUE }
                    .thenBy { it.first.deliveryTime ?: Int.MAX_VALUE }
                    .thenByDescending { it.first.sellerRating ?: Double.MIN_VALUE }
            )
            .take(n)
            .toList()
    }

    /**
     * Посчитать breakdown для уже отсортированных DTO без пересортировки (порядок сохраняется).
     * Полезно для чата, когда нужно добавить «почему эти 3».
     */
    fun breakdownFor(
        orderedDtos: List<ProductDto>,
        query: NormalizedQuery
    ): List<Pair<ProductDto, ScoreBreakdown>> {
        val prices = orderedDtos.mapNotNull { it.price }
        val avg = if (prices.isNotEmpty()) prices.average() else 0.0
        return orderedDtos.map { dto -> dto to engine.score(dto, query, avg) }
    }
}
