package com.example.shoppingassistant.core.rank

import com.example.shoppingassistant.core.config.FeatureFlagsConfig.FeatureFlags
import com.example.shoppingassistant.core.config.RankBoundsConfig.RankBounds
import com.example.shoppingassistant.domain.model.NormalizedQuery
import com.example.shoppingassistant.domain.model.ProductDto
import kotlin.math.max

class ScoringEngineImpl(
    private val weights: Weights,
    private val bounds: RankBounds,
    private val flags: FeatureFlags
) : ScoringEngine {

    override fun score(dto: ProductDto, q: NormalizedQuery, avgPrice: Double): ScoreBreakdown {
        val price = dto.price ?: bounds.price.endInclusive.toDouble()
        val deliveryDays = (dto.deliveryTime ?: bounds.deliveryDays.last).toDouble()
        val sellerRating = dto.sellerRating ?: bounds.sellerRating.start.toDouble()

        val priceScore = normalizeInverse(
            anchorToAvg(price, avgPrice, bounds.price.start.toDouble(), bounds.price.endInclusive.toDouble()),
            bounds.price.start.toDouble(), bounds.price.endInclusive.toDouble()
        )
        val deliveryScore = normalizeInverse(
            deliveryDays, bounds.deliveryDays.first.toDouble(), bounds.deliveryDays.last.toDouble()
        )
        val ratingScore = normalizeDirect(
            sellerRating, bounds.sellerRating.start.toDouble(), bounds.sellerRating.endInclusive.toDouble()
        )

        // Фич-флаги: данных для trust/stock в ProductDto нет → вклад = 0, пока не появятся поля/источники.
        val trustBoost = if (flags.useSellerTrustScore) computeSellerTrustBoost(dto) else 0.0
        val stockPenalty = if (flags.useStockAvailability) computeStockPenalty(dto) else 0.0

        val total = (weights.price * priceScore +
                weights.delivery * deliveryScore +
                weights.rating * ratingScore +
                trustBoost - stockPenalty)
            .coerceIn(0.0, 1.0)

        return ScoreBreakdown(
            price = priceScore.toFloat(),
            delivery = deliveryScore.toFloat(),
            rating = ratingScore.toFloat(),
            penalties = stockPenalty.toFloat(), // пока единственная «наказующая» часть
            score = total.toFloat()
        )
    }

    private fun normalizeDirect(value: Double, min: Double, max: Double): Double {
        val v = value.coerceIn(min, max)
        return if (max == min) 1.0 else (v - min) / (max - min)
    }

    private fun normalizeInverse(value: Double, min: Double, max: Double): Double {
        return 1.0 - normalizeDirect(value, min, max)
    }

    private fun anchorToAvg(value: Double, avg: Double, lo: Double, hi: Double): Double {
        if (avg <= 0.0) return value.coerceIn(lo, hi)
        val anchored = (value / max(1.0, avg)) * ((lo + hi) / 2.0)
        return anchored.coerceIn(lo, hi)
    }

    // Плейсхолдеры: без полей → возвращают 0. Когда появятся поля/репозитории — подключим здесь.
    private fun computeSellerTrustBoost(@Suppress("UNUSED_PARAMETER") dto: ProductDto): Double = 0.0
    private fun computeStockPenalty(@Suppress("UNUSED_PARAMETER") dto: ProductDto): Double = 0.0
}
