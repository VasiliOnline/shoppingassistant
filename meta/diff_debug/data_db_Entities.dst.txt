package com.example.shoppingassistant.core.data.db

import androidx.room.Entity
import androidx.room.ForeignKey
import androidx.room.Index
import androidx.room.PrimaryKey

@Entity(
    tableName = "products",
    indices = [
        Index("brand"),
        Index(value = ["brand", "model"]),
        Index("title"),
        Index(value = ["brandKey", "modelKey"])
    ]
)
data class ProductEntity(
    @PrimaryKey(autoGenerate = true) val id: Long = 0,
    val title: String,
    val brand: String?,
    val model: String?,
    val priceCents: Long?,      // цена в центах/копейках
    val deliveryDays: Int?,     // срок доставки в днях
    val sellerRating: Double?,  // рейтинг 0..5
    val currency: String,
    val sourceUrl: String,
    val brandKey: String?,      // машинные ключи для индексов/поиска
    val modelKey: String?,
    val imageUrls: List<String>?
)

@Entity(
    tableName = "product_attributes",
    foreignKeys = [
        ForeignKey(
            entity = ProductEntity::class,
            parentColumns = ["id"],
            childColumns = ["productId"],
            onDelete = ForeignKey.CASCADE
        )
    ],
    indices = [
        Index("productId"),
        Index(value = ["productId", "key"])
    ]
)
data class ProductAttributeEntity(
    @PrimaryKey(autoGenerate = true) val id: Long = 0,
    val productId: Long,
    val key: String,      // "memory", "color", ...
    val value: String     // "256GB", "blue", ...
)

@Entity(
    tableName = "category_offer_counts",
    indices = [Index("categoryCode")],
)
data class CategoryOfferCountEntity(
    @PrimaryKey val categoryCode: String,
    val offersCount: Int,
    val updatedAtMillis: Long,
)
