package com.example.shoppingassistant.core.data.menu

import com.example.shoppingassistant.domain.menu.MenuActionId
import com.example.shoppingassistant.domain.menu.MenuLayout
import com.example.shoppingassistant.domain.menu.MenuLayoutConfig

object MenuLayoutDefaults {
    val config = MenuLayoutConfig(
        rightSlots = 5,
        bottomSlots = 5,
    )

    fun defaultLayout(): MenuLayout = MenuLayout(
        right = listOf(
            MenuActionId.DASHBOARD,
            MenuActionId.SEARCH,
            MenuActionId.POST,
        ),
        bottom = listOf(
            MenuActionId.PROFILE,
            MenuActionId.MESSAGES,
            MenuActionId.NOTIFICATIONS,
            MenuActionId.SUBSCRIPTIONS,
        ),
        pinned = setOf(MenuActionId.DASHBOARD),
    )
}
