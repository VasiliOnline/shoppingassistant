package com.example.shoppingassistant.core.data.offers

import com.example.shoppingassistant.core.config.BackendConfig
import com.example.shoppingassistant.core.network.BackendClient
import com.example.shoppingassistant.domain.auth.AuthRepository
import com.example.shoppingassistant.domain.offers.CreateTrackedOfferResult
import com.example.shoppingassistant.domain.offers.CreateTrackedOfferStatus
import com.example.shoppingassistant.domain.offers.TrackedOfferInput
import com.example.shoppingassistant.domain.offers.TrackedOfferRepository
import com.example.shoppingassistant.domain.offers.RefreshTrackedOfferInput
import com.example.shoppingassistant.domain.offers.RefreshTrackedOfferResult
import com.example.shoppingassistant.domain.offers.RefreshTrackedOfferStatus
import io.ktor.client.call.body
import io.ktor.client.request.header
import io.ktor.client.request.post
import io.ktor.client.request.put
import io.ktor.client.request.setBody
import io.ktor.client.statement.HttpResponse
import io.ktor.http.ContentType
import io.ktor.http.HttpStatusCode
import io.ktor.http.isSuccess
import io.ktor.http.contentType

/**
 * HTTP-репозиторий для создания отслеживаемого оффера по нормализованным данным.
 */
class TrackedOfferRemoteRepository(
    private val backendClient: BackendClient,
    private val authRepository: AuthRepository,
) : TrackedOfferRepository {

    private val baseUrl get() = BackendConfig.BASE_URL

    override suspend fun createTrackedOffer(request: TrackedOfferInput): CreateTrackedOfferResult {
        val token = authRepository.currentToken()
            ?: return CreateTrackedOfferResult(
                status = CreateTrackedOfferStatus.INVALID_INPUT,
                message = "Unauthorized",
            )

        val response: HttpResponse = backendClient.client.post("$baseUrl/api/offers/tracked") {
            header("Authorization", ensureBearer(token))
            contentType(ContentType.Application.Json)
            setBody(request)
        }

        if (response.status.isSuccess()) {
            return response.body()
        }

        if (response.status == HttpStatusCode.Unauthorized) {
            authRepository.logout()
            return CreateTrackedOfferResult(
                status = CreateTrackedOfferStatus.INVALID_INPUT,
                message = "Unauthorized",
            )
        }

        return CreateTrackedOfferResult(
            status = CreateTrackedOfferStatus.INVALID_INPUT,
            message = "HTTP ${response.status.value}",
        )
    }

    override suspend fun refreshTrackedOffer(request: RefreshTrackedOfferInput): RefreshTrackedOfferResult {
        val token = authRepository.currentToken()
            ?: return RefreshTrackedOfferResult(
                status = RefreshTrackedOfferStatus.INVALID_INPUT,
                message = "Unauthorized",
            )

        val response: HttpResponse = backendClient.client.put("$baseUrl/api/offers/tracked/refresh") {
            header("Authorization", ensureBearer(token))
            contentType(ContentType.Application.Json)
            setBody(request)
        }

        if (response.status.isSuccess()) {
            return response.body()
        }

        if (response.status == HttpStatusCode.Unauthorized) {
            authRepository.logout()
            return RefreshTrackedOfferResult(
                status = RefreshTrackedOfferStatus.INVALID_INPUT,
                message = "Unauthorized",
            )
        }

        return RefreshTrackedOfferResult(
            status = when (response.status.value) {
                404 -> RefreshTrackedOfferStatus.NOT_FOUND
                else -> RefreshTrackedOfferStatus.INVALID_INPUT
            },
            message = "HTTP ${response.status.value}",
        )
    }

    private fun ensureBearer(token: String): String =
        if (token.startsWith("Bearer ")) token else "Bearer $token"
}
