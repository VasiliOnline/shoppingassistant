package com.example.shoppingassistant.core.data.profile

import android.content.Context
import androidx.security.crypto.EncryptedSharedPreferences
import androidx.security.crypto.MasterKey
import com.example.shoppingassistant.domain.profile.ExternalLink
import com.example.shoppingassistant.domain.profile.ExternalLinksRepository
import com.example.shoppingassistant.domain.profile.ProfileCacheRepository
import com.example.shoppingassistant.domain.profile.ProfileSettings
import com.example.shoppingassistant.domain.profile.ProfileSettingsRepository
import com.example.shoppingassistant.domain.profile.ProfileSnapshot
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import kotlinx.serialization.builtins.ListSerializer
import kotlinx.serialization.json.Json

class ProfilePreferencesStorage(
    context: Context,
) : ProfileSettingsRepository, ExternalLinksRepository, ProfileCacheRepository {

    private val json = Json {
        ignoreUnknownKeys = true
        encodeDefaults = true
    }

    private val prefs = EncryptedSharedPreferences.create(
        context,
        PREFS_NAME,
        MasterKey.Builder(context)
            .setKeyScheme(MasterKey.KeyScheme.AES256_GCM)
            .build(),
        EncryptedSharedPreferences.PrefKeyEncryptionScheme.AES256_SIV,
        EncryptedSharedPreferences.PrefValueEncryptionScheme.AES256_GCM,
    )

    override suspend fun getSettings(userId: String?): ProfileSettings = withContext(Dispatchers.IO) {
        val raw = prefs.getString(settingsKey(userId), null)
            ?: prefs.getString(KEY_LAST_SETTINGS, null)
        if (raw.isNullOrBlank()) return@withContext ProfileSettings()
        runCatching { json.decodeFromString(ProfileSettings.serializer(), raw) }
            .getOrElse { ProfileSettings() }
    }

    override suspend fun saveSettings(userId: String?, settings: ProfileSettings) = withContext(Dispatchers.IO) {
        val payload = json.encodeToString(ProfileSettings.serializer(), settings)
        prefs.edit()
            .putString(settingsKey(userId), payload)
            .putString(KEY_LAST_SETTINGS, payload)
            .apply()
    }

    override suspend fun getLinks(userId: String): List<ExternalLink> = withContext(Dispatchers.IO) {
        val raw = prefs.getString(linksKey(userId), null)
        if (raw.isNullOrBlank()) return@withContext emptyList()
        runCatching {
            json.decodeFromString(ListSerializer(ExternalLink.serializer()), raw)
        }.getOrElse { emptyList() }
    }

    override suspend fun saveLinks(userId: String, links: List<ExternalLink>) = withContext(Dispatchers.IO) {
        val sanitized = links
            .map { it.copy(title = it.title.trim(), url = it.url.trim()) }
            .filter { it.url.isNotBlank() }
        val payload = json.encodeToString(ListSerializer(ExternalLink.serializer()), sanitized)
        prefs.edit()
            .putString(linksKey(userId), payload)
            .apply()
    }

    override suspend fun clearLinks(userId: String) = withContext(Dispatchers.IO) {
        prefs.edit().remove(linksKey(userId)).apply()
    }

    override suspend fun getProfile(userId: String): ProfileSnapshot? = withContext(Dispatchers.IO) {
        val raw = prefs.getString(profileCacheKey(userId), null)
        if (raw.isNullOrBlank()) return@withContext null
        runCatching { json.decodeFromString(ProfileSnapshot.serializer(), raw) }
            .getOrNull()
    }

    override suspend fun saveProfile(profile: ProfileSnapshot) = withContext(Dispatchers.IO) {
        val payload = json.encodeToString(ProfileSnapshot.serializer(), profile)
        prefs.edit()
            .putString(profileCacheKey(profile.id), payload)
            .apply()
    }

    override suspend fun clearProfile(userId: String) = withContext(Dispatchers.IO) {
        prefs.edit().remove(profileCacheKey(userId)).apply()
    }

    private fun settingsKey(userId: String?): String =
        if (userId.isNullOrBlank()) KEY_SETTINGS_GLOBAL else "$KEY_SETTINGS_PREFIX$userId"

    private fun linksKey(userId: String): String =
        "$KEY_LINKS_PREFIX$userId"

    private fun profileCacheKey(userId: String): String =
        "$KEY_PROFILE_PREFIX$userId"

    private companion object {
        private const val PREFS_NAME = "profile.secure.prefs"
        private const val KEY_SETTINGS_GLOBAL = "settings.global"
        private const val KEY_SETTINGS_PREFIX = "settings.user."
        private const val KEY_LAST_SETTINGS = "settings.last"
        private const val KEY_LINKS_PREFIX = "links.user."
        private const val KEY_PROFILE_PREFIX = "profile.cache."
    }
}
