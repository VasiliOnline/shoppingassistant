package com.example.shoppingassistant.core.data.suggest

/**
 * Минимальные данные товара для подсказок (якорь продукта).
 */
data class ProductSuggestCandidate(
    val id: Long,
    val title: String,
    val brand: String?,
    val model: String?,
    val categoryCode: String? = null,
)

interface ProductSuggestRepository {
    suspend fun search(query: String, limit: Int = 6): List<ProductSuggestCandidate>
}
