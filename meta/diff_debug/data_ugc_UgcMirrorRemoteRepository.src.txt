package com.example.shoppingassistant.core.data.ugc

import com.example.shoppingassistant.core.config.BackendConfig
import com.example.shoppingassistant.core.network.BackendClient
import com.example.shoppingassistant.domain.ugc.UgcMirrorRepository
import com.example.shoppingassistant.domain.ugc.UgcMirrorResult
import io.ktor.client.call.body
import io.ktor.client.request.post
import io.ktor.client.request.setBody
import io.ktor.http.ContentType
import io.ktor.http.HttpStatusCode
import io.ktor.http.contentType
import kotlinx.serialization.Serializable

/**
 * HTTP-репозиторий для UGC-зеркала по ссылке.
 *
 * Ходит на backend по /api/ugc/mirror и возвращает нормализованные данные
 * в доменную модель UgcMirrorResult.
 */
class UgcMirrorRemoteRepository(
    private val backendClient: BackendClient,
) : UgcMirrorRepository {

    private val baseUrl: String
        get() = BackendConfig.BASE_URL

    override suspend fun mirrorByUrl(url: String): UgcMirrorResult {
        val trimmed = url.trim()

        if (trimmed.isEmpty()) {
            // Пустая строка → возвращаем только исходное значение, без сетевых вызовов.
            return UgcMirrorResult(sourceUrl = url)
        }

        val response = backendClient.client.post("$baseUrl/api/ugc/mirror") {
            contentType(ContentType.Application.Json)
            setBody(MirrorRequest(url = trimmed))
        }

        if (response.status != HttpStatusCode.OK) {
            // На MVP: мягкий фолбэк — возвращаем только ссылку.
            return UgcMirrorResult(sourceUrl = trimmed)
        }

        val dto: MirrorResponse = response.body()

        return UgcMirrorResult(
            sourceUrl = dto.sourceUrl ?: trimmed,
            title = dto.title,
            description = dto.description,
            priceRaw = dto.priceRaw,
            currency = dto.currency,
            imageUrls = dto.imageUrls ?: emptyList(),
            attributes = dto.attributes ?: emptyMap(),
        )
    }

    @Serializable
    private data class MirrorRequest(
        val url: String,
    )

    @Serializable
    private data class MirrorResponse(
        val sourceUrl: String? = null,
        val title: String? = null,
        val description: String? = null,
        val priceRaw: String? = null,
        val currency: String? = null,
        val imageUrls: List<String>? = null,
        val attributes: Map<String, String>? = null,
    )
}
