package com.example.shoppingassistant.core.data.vision

import android.content.Context
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import kotlinx.serialization.json.Json
import com.example.shoppingassistant.domain.vision.VisionUsage

class VisionUsageStorage(
    context: Context,
) {
    private val json = Json {
        ignoreUnknownKeys = true
        encodeDefaults = true
    }

    private val prefs = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)

    suspend fun getLast(userKey: String?): VisionUsage? = withContext(Dispatchers.IO) {
        val raw = prefs.getString(cacheKey(userKey), null) ?: return@withContext null
        runCatching { json.decodeFromString(VisionUsage.serializer(), raw) }.getOrNull()
    }

    suspend fun save(userKey: String?, usage: VisionUsage) = withContext(Dispatchers.IO) {
        val payload = json.encodeToString(VisionUsage.serializer(), usage)
        prefs.edit().putString(cacheKey(userKey), payload).apply()
    }

    private fun cacheKey(userKey: String?): String =
        if (userKey.isNullOrBlank()) KEY_GLOBAL else "$KEY_PREFIX$userKey"

    private companion object {
        private const val PREFS_NAME = "vision.usage.prefs"
        private const val KEY_GLOBAL = "vision.usage.global"
        private const val KEY_PREFIX = "vision.usage."
    }
}
