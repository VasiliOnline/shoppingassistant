private fun profileForCategory(category: Category): CategoryProfile {
    val attributes = attributesForCategory(category.code)
    val categoryAttributes = attributes.mapIndexed { index, def ->
        CategoryAttribute(
            categoryCode = category.code,
            attributeCode = def.code,
            uiOrder = index + 1,
            isRequiredForCategory = def.requiredForOffer,
        )
    }
    return CategoryProfile(
        category = category,
        attributes = attributes,
        categoryAttributes = categoryAttributes,
        valueDictionaries = dictsFor(attributes),
        requiredIfRules = requiredIfRulesForCategory(category.code),
    )
}

private fun buildBrowseNodes(
    categories: List<Category>,
    techStage21Nodes: List<BrowseNode>,
    applStage21Nodes: List<BrowseNode>,
    homeStage21Nodes: List<BrowseNode>,
    beautyStage21Nodes: List<BrowseNode>,
    kidsStage21Nodes: List<BrowseNode>,
    foodStage21Nodes: List<BrowseNode>,
    petsStage21Nodes: List<BrowseNode>,
    sportStage21Nodes: List<BrowseNode>,
): List<BrowseNode> {
    val childrenByParent: Map<String, List<Category>> = categories
        .filter { !it.parentCode.isNullOrBlank() }
        .groupBy { it.parentCode.orEmpty() }

    val roots = categories.filter { it.parentCode.isNullOrBlank() }
    return buildList {
        roots.forEachIndexed { rootOrder, root ->
            if (
                root.code == "TECH" ||
                root.code == "APPL" ||
                root.code == "HOME" ||
                root.code == "BEAUTY" ||
                root.code == "KIDS" ||
                root.code == "FOOD" ||
                root.code == "PETS" ||
                root.code == "SPORT"
            ) {
                return@forEachIndexed
            }
            val rootBrowseCode = "ROOT.${root.code}"
            add(
                BrowseNode(
                    browseCode = rootBrowseCode,
                    parentBrowseCode = null,
                    nodeKind = BrowseNodeKind.ROOT,
                    titleKey = "catalog.browse.${root.code.lowercase()}",
                    titleRu = root.title ?: root.code,
                    targetCategoryCode = null,
                    targetType = null,
                    order = rootOrder + 1,
                    availabilityScope = "ALL",
                    status = BrowseNodeStatus.ACTIVE,
                    tags = listOf("stage2.0", "root"),
                ),
            )

            childrenByParent[root.code].orEmpty().forEachIndexed { childOrder, child ->
                val childStatus = when (child.status) {
                    CategoryStatus.HIDDEN -> BrowseNodeStatus.HIDDEN
                    else -> BrowseNodeStatus.ACTIVE
                }
                add(
                    BrowseNode(
                        browseCode = "NODE.${child.code}",
                        parentBrowseCode = rootBrowseCode,
                        nodeKind = BrowseNodeKind.LEAF_LINK,
                        titleKey = "catalog.browse.${child.code.lowercase().replace('.', '_')}",
                        titleRu = child.title ?: child.code,
                        targetCategoryCode = child.code,
                        targetType = BrowseTargetType.CATEGORY,
                        order = childOrder + 1,
                        availabilityScope = "ALL",
                        status = childStatus,
                        tags = listOf("stage2.0", "leaf", root.code),
                    ),
                )
            }
        }
        addAll(homeStage21Nodes)
        addAll(beautyStage21Nodes)
        addAll(kidsStage21Nodes)
        addAll(foodStage21Nodes)
        addAll(petsStage21Nodes)
        addAll(sportStage21Nodes)
        addAll(applStage21Nodes)
        addAll(techStage21Nodes)
    }
}

private fun buildAliasEntries(
    legacyEntries: List<AliasEntry>,
    techStage21Entries: List<AliasEntry>,
    applStage21Entries: List<AliasEntry>,
    homeStage21Entries: List<AliasEntry>,
    beautyStage21Entries: List<AliasEntry>,
    kidsStage21Entries: List<AliasEntry>,
    foodStage21Entries: List<AliasEntry>,
    petsStage21Entries: List<AliasEntry>,
    sportStage21Entries: List<AliasEntry>,
): List<AliasEntry> {
    val legacyWithoutStage21Segments = legacyEntries.filterNot { entry ->
        val target = entry.targetCode.trim()
        target == "ROOT.TECH" ||
            target.startsWith("TECH.") ||
            target == "ROOT.APPL" ||
            target.startsWith("APPL.") ||
            target == "ROOT.HOME" ||
            target.startsWith("HOME.") ||
            target == "ROOT.BEAUTY" ||
            target.startsWith("BEAUTY.") ||
            target == "ROOT.KIDS" ||
            target.startsWith("KIDS.") ||
            target == "ROOT.FOOD" ||
            target.startsWith("FOOD.") ||
            target == "ROOT.PETS" ||
            target.startsWith("PETS.") ||
            target == "ROOT.SPORT" ||
            target.startsWith("SPORT.")
    }

    val merged = legacyWithoutStage21Segments +
        techStage21Entries +
        applStage21Entries +
        homeStage21Entries +
        beautyStage21Entries +
        kidsStage21Entries +
        foodStage21Entries +
        petsStage21Entries +
        sportStage21Entries
    val deduplicatedByLocaleAndNormalized = LinkedHashMap<String, AliasEntry>()
    merged.forEach { entry ->
        val key = "${entry.locale.lowercase()}|${entry.normalizedTerm.lowercase()}"
        val current = deduplicatedByLocaleAndNormalized[key]
        if (current == null || isPreferredAliasEntry(candidate = entry, current = current)) {
            deduplicatedByLocaleAndNormalized[key] = entry
        }
    }
    return deduplicatedByLocaleAndNormalized.values.toList()
}

private fun isPreferredAliasEntry(candidate: AliasEntry, current: AliasEntry): Boolean {
    val candidateScore = aliasEntryPreferenceScore(candidate)
    val currentScore = aliasEntryPreferenceScore(current)
    return candidateScore > currentScore
}

private fun aliasEntryPreferenceScore(entry: AliasEntry): Int {
    val kindRank = when (entry.kind) {
        AliasKind.CATEGORY -> 4
        AliasKind.BRAND -> 3
        AliasKind.BROWSE -> 2
        AliasKind.ATTRIBUTE_HINT -> 1
    }
    val blockedPenalty = if (entry.isBlocked) -1000 else 0
    return blockedPenalty + (entry.weight * 10) + kindRank
}

private fun leafCategoryCodeSet(categories: List<Category>): Set<String> {
    val parents = categories.mapNotNull { it.parentCode?.takeIf(String::isNotBlank) }.toSet()
    return categories
        .asSequence()
        .map { it.code }
        .filter { it !in parents }
        .toSet()
}
