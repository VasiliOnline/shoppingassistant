private fun smartphoneConstraints(): CatalogConstraints =
    CatalogConstraints(
        scope = ConstraintScope.CATEGORY,
        categoryCode = "TECH.SMARTPHONE",
        compatibilityRules = listOf(
            CompatibilityRule(
                whenAll = listOf(
                    AttributeCondition(
                        attributeCode = "brand",
                        op = AttributeConditionOp.EQUALS_ANY,
                        values = listOf("Apple"),
                    ),
                    AttributeCondition(
                        attributeCode = "model",
                        op = AttributeConditionOp.STARTS_WITH_ANY,
                        values = listOf("iPhone"),
                    ),
                ),
                apply = listOf(
                    AttributeValueConstraint(
                        attributeCode = "os_family",
                        allowedValues = listOf("iOS"),
                        reason = "Для iPhone доступна только iOS",
                    )
                ),
            )
        ),
    )

private fun dictsFor(defs: List<AttributeDef>): List<AttributeValueDict> =
    defs.mapNotNull { def -> def.valueDictCode?.let { baseDicts[it] } }
        .distinctBy { it.attributeCode }
