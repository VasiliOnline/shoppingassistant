package com.example.shoppingassistant.domain.catalog

import kotlinx.serialization.Serializable

@Serializable
enum class QueryRouteType {
    OPEN_CATEGORY,
    OPEN_BROWSE,
    RUN_SEARCH,
}

@Serializable
data class QueryRoutingResult(
    val routeType: QueryRouteType,
    val primaryTargetCode: String? = null,
    val extractedTokens: List<String> = emptyList(),
    val confidence: Double = 0.0,
)

interface QueryRouter {
    suspend fun route(query: String, locale: String = "ru-RU"): QueryRoutingResult
}

class RouteQueryTask(
    private val queryRouter: QueryRouter,
) {
    suspend operator fun invoke(
        query: String,
        locale: String = "ru-RU",
    ): QueryRoutingResult = queryRouter.route(query = query, locale = locale)
}

class RunStage21TechGoldenTask(
    private val goldenRunner: Stage21TechGoldenRunner,
) {
    suspend operator fun invoke(): Stage21TechGoldenReport = goldenRunner.run()
}

internal class RunStage21GoldenSuiteTask(
    private val goldenSuiteRunner: Stage21GoldenSuiteRunner,
) {
    operator fun invoke(): Stage21GoldenSuiteReport = goldenSuiteRunner.run()
}
