package com.example.shoppingassistant.domain.catalog

import kotlinx.serialization.SerialName
import kotlinx.serialization.Serializable

internal object Stage21ApplPackageLoader {
    private val descriptor = Stage21PackageDescriptor(
        l0Code = "APPL",
        basePath = "taxonomy/stage2/2.1/APPL",
        browseNodesFile = "browse_nodes.appl.tsv",
        aliasesFile = "aliases.appl.tsv",
        goldenQueriesFile = "queries_golden.appl.tsv",
        coverageFile = "coverage.appl.json",
        routingRulesFile = "routing_rules.appl.yaml",
        browseRootCode = "B.APPL",
        schemaFamily = Stage21SchemaFamily.APPL,
        supportsBlockedGoldenRouteKind = false,
    )

    private val packageData: Stage21ApplPackageData by lazy {
        GenericStage21PackageLoader.loadAppl(descriptor)
    }

    val browseNodes: List<BrowseNode> by lazy { packageData.browseNodes }
    val aliasSeedRows: List<ApplAliasSeedRow> by lazy { packageData.aliasSeedRows }
    val aliasEntries: List<AliasEntry> by lazy { aliasSeedRows.map { it.toAliasEntry() } }
    val goldenQueries: List<GoldenQuery> by lazy { packageData.goldenQueries }
    val coverageGate: Stage21ApplCoverageGate by lazy { packageData.coverageGate }
    val routingRulesYaml: String by lazy { packageData.routingRulesYaml }
}

data class ApplAliasSeedRow(
    val aliasId: String,
    val locale: String,
    val query: String,
    val normalizedQuery: String,
    val targetType: ApplAliasTargetType,
    val targetId: String,
    val flag: ApplAliasFlag,
    val weightOverride: Int?,
)

enum class ApplAliasTargetType {
    NODE,
    CANONICAL,
}

enum class ApplAliasFlag {
    POSITIVE,
    DISAMBIGUATE,
    NEGATIVE_FOR_APPL,
}

@Serializable
internal data class Stage21ApplCoverageGate(
    @SerialName("package") val packageName: String,
    val version: String,
    @SerialName("expected_canonical_codes") val expectedCanonicalCodes: List<String>,
    @SerialName("expected_leaf_nodes") val expectedLeafNodes: List<String>,
    @SerialName("min_aliases_per_leaf") val minAliasesPerLeaf: Int,
    @SerialName("min_golden_per_leaf") val minGoldenPerLeaf: Int,
    @SerialName("required_ambiguous_tests") val requiredAmbiguousTests: List<String>,
)

private fun ApplAliasSeedRow.toAliasEntry(): AliasEntry {
    val weight = weightOverride ?: when (flag) {
        ApplAliasFlag.POSITIVE -> 78
        ApplAliasFlag.DISAMBIGUATE -> 90
        ApplAliasFlag.NEGATIVE_FOR_APPL -> 94
    }
    return AliasEntry(
        locale = locale,
        term = query,
        normalizedTerm = normalizedQuery,
        kind = when (targetType) {
            ApplAliasTargetType.NODE -> AliasKind.BROWSE
            ApplAliasTargetType.CANONICAL -> AliasKind.CATEGORY
        },
        targetCode = targetId,
        weight = weight,
        matchKind = AliasMatchKind.EXACT,
        isBlocked = false,
        source = AliasSource.SEED,
        notes = "stage2.1.appl:${flag.name.lowercase()}:$aliasId",
    )
}
