package com.example.shoppingassistant.domain.catalog

class Stage21BeautyQueryRouter(
    aliases: List<HomeAliasSeedRow> = Stage21BeautyPackageLoader.aliasSeedRows,
    browseNodes: List<BrowseNode> = Stage21BeautyPackageLoader.browseNodes,
    routingRulesYaml: String = Stage21BeautyPackageLoader.routingRulesYaml,
) : QueryRouter {

    private val delegate = Stage21HomeQueryRouter(
        aliases = aliases,
        browseNodes = browseNodes,
        routingRulesYaml = routingRulesYaml,
    )

    override suspend fun route(query: String, locale: String): QueryRoutingResult =
        routeWithCandidates(query = query, locale = locale).result

    fun routeWithCandidates(
        query: String,
        locale: String = "ru-RU",
        topN: Int = 3,
    ): QueryRoutingDebugResult {
        val normalizedQuery = normalize(query)
        val queryTokens = tokenize(normalizedQuery)
        val delegateTopN = topN.coerceAtLeast(8)
        val base = delegate.routeWithCandidates(
            query = query,
            locale = locale,
            topN = delegateTopN,
        )

        val baseTarget = base.result.primaryTargetCode.orEmpty()
        if (base.result.routeType == QueryRouteType.OPEN_BROWSE && baseTarget == BEAUTY_BROWSE_ROOT) {
            return base.copy(topCandidates = base.topCandidates.take(topN.coerceAtLeast(1)))
        }
        if (base.result.routeType == QueryRouteType.OPEN_CATEGORY && !baseTarget.startsWith("BEAUTY")) {
            return base.copy(topCandidates = base.topCandidates.take(topN.coerceAtLeast(1)))
        }

        val heuristicCategory = heuristicCategoryFor(
            normalizedQuery = normalizedQuery,
            queryTokens = queryTokens,
        )

        if (!shouldOverride(baseTarget = baseTarget, heuristicCategory = heuristicCategory)) {
            return base.copy(topCandidates = base.topCandidates.take(topN.coerceAtLeast(1)))
        }

        val targetCode = heuristicCategory.orEmpty()
        val result = QueryRoutingResult(
            routeType = QueryRouteType.OPEN_CATEGORY,
            primaryTargetCode = targetCode,
            extractedTokens = queryTokens,
            confidence = maxOf(base.result.confidence, 0.82),
        )
        val candidates = overrideCandidates(
            baseCandidates = base.topCandidates,
            targetCode = targetCode,
            limit = topN.coerceAtLeast(1),
        )
        return QueryRoutingDebugResult(
            result = result,
            topCandidates = candidates,
            wasAmbiguous = base.wasAmbiguous,
        )
    }

    private fun shouldOverride(baseTarget: String, heuristicCategory: String?): Boolean {
        if (heuristicCategory.isNullOrBlank()) return false
        if (baseTarget == heuristicCategory) return false
        if (baseTarget == BEAUTY_CANONICAL_ROOT) return true
        if (baseTarget == BEAUTY_FRAGRANCE) return true
        if (baseTarget == BEAUTY_DEVICES && heuristicCategory == BEAUTY_HEALTH) return true
        return false
    }

    private fun heuristicCategoryFor(
        normalizedQuery: String,
        queryTokens: List<String>,
    ): String? {
        val hasPhrase: (String) -> Boolean = { phrase ->
            containsPhrase(normalizedQuery, normalize(phrase))
        }
        val hasToken: (String) -> Boolean = { token ->
            val normalizedToken = normalize(token)
            if (normalizedToken.isBlank()) {
                false
            } else if (' ' in normalizedToken) {
                containsPhrase(normalizedQuery, normalizedToken)
            } else {
                queryTokens.any { queryToken ->
                    queryToken == normalizedToken ||
                        (
                            queryToken.length > 2 &&
                                normalizedToken.length > 2 &&
                                (queryToken.startsWith(normalizedToken) || normalizedToken.startsWith(queryToken))
                            )
                }
            }
        }
        val hasLatinToken = queryTokens.any { token -> token.any { ch -> ch in 'a'..'z' } }
        val hasBuSignal = (queryTokens.contains("б") && queryTokens.contains("у")) || hasPhrase("бу")

        if (hasPhrase("техника для красоты и здоровья")) return BEAUTY_DEVICES
        if (hasPhrase("декоративная косметика")) return BEAUTY_MAKEUP
        if (hasPhrase("аптечка дорожная")) return BEAUTY_HEALTH

        val bodycareSignal = hasPhrase("мыло жидкое") ||
            hasPhrase("гель для душа") ||
            hasToken("дезодорант") ||
            hasToken("антиперспирант") ||
            hasPhrase("скраб для тела") ||
            hasPhrase("крем для тела") ||
            hasPhrase("воск для депиляции") ||
            hasPhrase("интимная гигиена")
        if (bodycareSignal) return BEAUTY_BODYCARE

        val makeupSignal = hasToken("помада") ||
            hasToken("консилер") ||
            hasToken("палетк") ||
            hasToken("тен")
        if (makeupSignal) return BEAUTY_MAKEUP

        val skincareSignal = hasToken("мицелляр") ||
            hasToken("пилинг") ||
            hasToken("ретинол") ||
            hasToken("ниацинамид") ||
            hasToken("spf")
        if (skincareSignal) return BEAUTY_SKINCARE

        if (hasToken("ирригатор") && hasBuSignal && !hasLatinToken) return BEAUTY_HEALTH

        val healthSignal = hasToken("аптечк") ||
            hasToken("бинт") ||
            hasToken("пластыр") ||
            hasToken("тонометр") ||
            hasToken("глюкометр") ||
            hasToken("пульсоксиметр")
        if (healthSignal) return BEAUTY_HEALTH

        return null
    }

    private fun overrideCandidates(
        baseCandidates: List<QueryRoutingCandidate>,
        targetCode: String,
        limit: Int,
    ): List<QueryRoutingCandidate> {
        val existing = baseCandidates.firstOrNull {
            it.routeType == QueryRouteType.OPEN_CATEGORY && it.targetCode == targetCode
        }
        val boosted = QueryRoutingCandidate(
            targetCode = targetCode,
            routeType = QueryRouteType.OPEN_CATEGORY,
            score = maxOf(existing?.score ?: 0, 95),
            matchedAlias = existing?.matchedAlias ?: "beauty-heuristic",
            matchKind = existing?.matchKind ?: AliasMatchKind.TOKEN,
        )
        val tail = baseCandidates
            .filterNot { it.routeType == boosted.routeType && it.targetCode == boosted.targetCode }
            .take((limit - 1).coerceAtLeast(0))
        return listOf(boosted) + tail
    }

    private fun containsPhrase(query: String, phrase: String): Boolean {
        if (phrase.isBlank()) return false
        if (query == phrase) return true
        val escaped = Regex.escape(phrase)
        return Regex("(^|\\s)$escaped(\\s|$)").containsMatchIn(query)
    }

    private fun normalize(input: String): String = Stage21QueryTextNormalizer.normalize(input)

    private fun tokenize(text: String): List<String> = Stage21QueryTextNormalizer.tokenize(text)

    private companion object {
        private const val BEAUTY_BROWSE_ROOT = "B.BEAUTY"
        private const val BEAUTY_CANONICAL_ROOT = "BEAUTY"
        private const val BEAUTY_FRAGRANCE = "BEAUTY.FRAGRANCE"
        private const val BEAUTY_BODYCARE = "BEAUTY.BODYCARE"
        private const val BEAUTY_DEVICES = "BEAUTY.DEVICES"
        private const val BEAUTY_HEALTH = "BEAUTY.HEALTH"
        private const val BEAUTY_MAKEUP = "BEAUTY.MAKEUP"
        private const val BEAUTY_SKINCARE = "BEAUTY.SKINCARE"
    }
}
