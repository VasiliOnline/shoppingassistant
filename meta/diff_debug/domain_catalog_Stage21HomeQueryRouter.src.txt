package com.example.shoppingassistant.domain.catalog

import kotlin.math.roundToInt

class Stage21HomeQueryRouter(
    aliases: List<HomeAliasSeedRow> = Stage21HomePackageLoader.aliasSeedRows,
    browseNodes: List<BrowseNode> = Stage21HomePackageLoader.browseNodes,
    routingRulesYaml: String = Stage21HomePackageLoader.routingRulesYaml,
) : QueryRouter {

    private val rules: HomeRoutingRules = HomeRoutingRulesParser.parse(routingRulesYaml)
    private val browseByCode: Map<String, BrowseNode> = browseNodes.associateBy { it.browseCode }
    private val knownCategoryCodes: Set<String> = CatalogSeed.categories.map { it.code }.toSet()

    private val runtimeAliases: List<HomeAliasRuntimeEntry> = aliases
        .asSequence()
        .mapNotNull { row ->
            val normalizedAlias = normalize(row.normalizedAlias)
            if (normalizedAlias.isBlank()) return@mapNotNull null
            HomeAliasRuntimeEntry(
                locale = row.locale,
                aliasText = row.aliasText,
                normalizedAlias = normalizedAlias,
                aliasTokens = tokenize(normalizedAlias),
                routeKind = row.routeKind,
                targetId = row.targetId,
                priority = row.priority.coerceIn(0, 100),
                matchType = row.matchType,
                negativeTokens = row.negativeTokens.map(::normalize).filter { it.isNotBlank() },
                isBlocked = row.routeKind == HomeAliasRouteKind.BLOCKED || !row.isActive,
                regex = compileRegexOrNull(normalizedAlias, row.matchType),
            )
        }
        .toList()

    private val blockedAliases: List<HomeAliasRuntimeEntry> = runtimeAliases.filter { it.isBlocked }
    private val routableAliases: List<HomeAliasRuntimeEntry> = runtimeAliases.filter { !it.isBlocked }
    private val blockedAliasesByLocale: Map<String, List<HomeAliasRuntimeEntry>> = blockedAliases.groupBy { it.locale.lowercase() }
    private val routableAliasMatcherIndex: HomeAliasMatcherIndex = HomeAliasMatcherIndex.build(routableAliases, ::tokenStem)

    override suspend fun route(query: String, locale: String): QueryRoutingResult =
        routeWithCandidates(query = query, locale = locale).result

    fun routeWithCandidates(
        query: String,
        locale: String = "ru-RU",
        topN: Int = 3,
    ): QueryRoutingDebugResult {
        val normalizedQuery = normalize(query)
        val queryTokens = tokenize(normalizedQuery)
        val queryTokensSet = queryTokens.toHashSet()

        if (normalizedQuery.isBlank()) {
            return QueryRoutingDebugResult(
                result = fallback(normalizedQuery = normalizedQuery, confidence = 0.0),
                topCandidates = emptyList(),
                wasAmbiguous = false,
            )
        }

        if (matchesBlockedAlias(normalizedQuery, queryTokens, queryTokensSet, locale)) {
            return QueryRoutingDebugResult(
                result = fallbackToRoot(normalizedQuery = normalizedQuery, confidence = 0.0),
                topCandidates = emptyList(),
                wasAmbiguous = false,
            )
        }

        val disambiguation = resolveDisambiguation(normalizedQuery, queryTokens)
        if (disambiguation != null) {
            return QueryRoutingDebugResult(
                result = disambiguation,
                topCandidates = listOf(
                    QueryRoutingCandidate(
                        targetCode = disambiguation.primaryTargetCode.orEmpty(),
                        routeType = disambiguation.routeType,
                        score = (disambiguation.confidence * 100.0).roundToInt(),
                        matchedAlias = "disambiguation",
                        matchKind = AliasMatchKind.EXACT,
                    ),
                ),
                wasAmbiguous = false,
            )
        }

        val scored = scoreCandidates(
            normalizedQuery = normalizedQuery,
            queryTokens = queryTokens,
            queryTokensSet = queryTokensSet,
            locale = locale,
        )

        if (scored.isEmpty()) {
            return QueryRoutingDebugResult(
                result = fallback(normalizedQuery = normalizedQuery, confidence = 0.0),
                topCandidates = emptyList(),
                wasAmbiguous = false,
            )
        }

        val top1 = scored.first()
        val top2 = scored.getOrNull(1)
        val margin = if (top2 == null) Int.MAX_VALUE else top1.score - top2.score

        val result = if (top1.confidence >= rules.minConfidenceToRoute) {
            top1.toResult(normalizedQuery)
        } else {
            fallback(normalizedQuery = normalizedQuery, confidence = top1.confidence)
        }

        val topCandidates = scored
            .take(topN.coerceAtLeast(1))
            .map { candidate ->
                QueryRoutingCandidate(
                    targetCode = candidate.targetCode,
                    routeType = candidate.routeType,
                    score = candidate.score,
                    matchedAlias = candidate.matchedAlias,
                    matchKind = candidate.matchKind,
                )
            }

        return QueryRoutingDebugResult(
            result = result,
            topCandidates = topCandidates,
            wasAmbiguous = margin <= 2,
        )
    }

    private fun resolveDisambiguation(
        normalizedQuery: String,
        queryTokens: List<String>,
    ): QueryRoutingResult? {
        val queryTokensSet = queryTokens.toHashSet()
        val matchedRule = rules.disambiguationRules.firstOrNull { rule ->
            val anyMatched = rule.ifAnyTokens.any { token ->
                matchesRuleToken(queryTokens, queryTokensSet, normalizedQuery, token)
            }
            if (!anyMatched) return@firstOrNull false
            val andMatched = rule.andAnyTokens.any { token ->
                matchesRuleToken(queryTokens, queryTokensSet, normalizedQuery, token)
            } || rule.andAnyTokens.isEmpty()
            val andNoneMatched = rule.andNoneTokens.none { token ->
                matchesRuleToken(queryTokens, queryTokensSet, normalizedQuery, token)
            }
            andMatched && andNoneMatched && passesRuleSpecificGuard(
                ruleId = rule.id,
                normalizedQuery = normalizedQuery,
                queryTokens = queryTokens,
                queryTokensSet = queryTokensSet,
            )
        } ?: return null

        val canonicalTarget = matchedRule.thenRouteCanonical?.trim()?.takeIf { it.isNotEmpty() }
        val browseTarget = matchedRule.thenRouteBrowseNode
            ?.trim()
            ?.takeIf { it.isNotEmpty() }
            ?.let(::normalizeBrowseCode)

        if (canonicalTarget != null) {
            return QueryRoutingResult(
                routeType = QueryRouteType.OPEN_CATEGORY,
                primaryTargetCode = canonicalTarget,
                extractedTokens = queryTokens,
                confidence = matchedRule.confidence.coerceIn(0.0, 1.0),
            )
        }

        if (browseTarget != null) {
            val canonicalFromBrowse = browseByCode[browseTarget]?.targetCategoryCode?.takeIf { it.isNotBlank() }
            if (canonicalFromBrowse != null) {
                return QueryRoutingResult(
                    routeType = QueryRouteType.OPEN_CATEGORY,
                    primaryTargetCode = canonicalFromBrowse,
                    extractedTokens = queryTokens,
                    confidence = matchedRule.confidence.coerceIn(0.0, 1.0),
                )
            }

            return QueryRoutingResult(
                routeType = QueryRouteType.OPEN_BROWSE,
                primaryTargetCode = browseTarget,
                extractedTokens = queryTokens,
                confidence = matchedRule.confidence.coerceIn(0.0, 1.0),
            )
        }

        return null
    }

    private fun scoreCandidates(
        normalizedQuery: String,
        queryTokens: List<String>,
        queryTokensSet: Set<String>,
        locale: String,
    ): List<HomeScoredCandidate> {
        val byTarget = LinkedHashMap<String, HomeScoredCandidate>()
        val candidateAliases = routableAliasMatcherIndex.selectCandidates(
            normalizedQuery = normalizedQuery,
            queryTokens = queryTokens,
        )

        candidateAliases.forEach { alias ->
            val matchKind = detectMatchKind(alias, normalizedQuery, queryTokens, queryTokensSet) ?: return@forEach
            val resolved = resolveRoute(alias) ?: return@forEach

            val overlap = alias.aliasTokens.count { aliasToken ->
                matchesRuleToken(queryTokens, queryTokensSet, normalizedQuery, aliasToken)
            }
            val overlapRatio = if (alias.aliasTokens.isEmpty()) {
                0.0
            } else {
                overlap.toDouble() / alias.aliasTokens.size.toDouble()
            }

            val phraseMatched = containsPhrase(normalizedQuery, alias.normalizedAlias)
            val exactMatched = normalizedQuery == alias.normalizedAlias
            val negativePenalty = if (alias.negativeTokens.any { matchesRuleToken(queryTokens, queryTokensSet, normalizedQuery, it) }) {
                0.65
            } else {
                0.0
            }
            val localePenalty = if (alias.locale.equals(locale, ignoreCase = true)) 0.0 else 0.05
            val leafBonus = if (resolved.routeType == QueryRouteType.OPEN_CATEGORY && resolved.targetCode.contains('.')) {
                rules.leafPreferenceBonus
            } else {
                0.0
            }
            val specificityBonus = ((alias.aliasTokens.size - 1).coerceAtLeast(0) * 0.05).coerceAtMost(0.2)
            val contextBonus = contextBonusFor(
                targetCode = resolved.targetCode,
                normalizedQuery = normalizedQuery,
                queryTokens = queryTokens,
            )

            val rawScore = (
                (alias.priority / 100.0) * rules.aliasMatchWeight +
                    (overlapRatio * rules.tokenOverlapWeight) +
                    (if (phraseMatched || exactMatched) rules.exactPhraseBonus else 0.0) +
                    specificityBonus +
                    leafBonus -
                    contextBonus -
                    negativePenalty -
                    localePenalty
                )

            if (rawScore <= 0.0) return@forEach

            val score = (rawScore * 100.0).roundToInt()
            val candidate = HomeScoredCandidate(
                targetCode = resolved.targetCode,
                routeType = resolved.routeType,
                score = score,
                confidence = rawScore.coerceIn(0.0, 1.0),
                matchedAlias = alias.aliasText,
                matchKind = matchKind,
            )
            val key = "${candidate.routeType}:${candidate.targetCode}"
            val current = byTarget[key]
            if (current == null || candidate.isBetterThan(current)) {
                byTarget[key] = candidate
            }
        }

        return byTarget.values
            .sortedWith(
                compareByDescending<HomeScoredCandidate> { it.score }
                    .thenBy { it.matchKind.ordinal }
                    .thenBy { it.targetCode },
            )
    }

    private fun detectMatchKind(
        alias: HomeAliasRuntimeEntry,
        normalizedQuery: String,
        queryTokens: List<String>,
        queryTokensSet: Set<String>,
    ): AliasMatchKind? {
        return when (alias.matchType) {
            HomeAliasMatchType.EXACT -> {
                if (normalizedQuery == alias.normalizedAlias) AliasMatchKind.EXACT else null
            }

            HomeAliasMatchType.CONTAINS -> {
                when {
                    normalizedQuery == alias.normalizedAlias -> AliasMatchKind.EXACT
                    containsPhrase(normalizedQuery, alias.normalizedAlias) -> AliasMatchKind.PREFIX
                    tokenContains(queryTokens, queryTokensSet, alias.aliasTokens) -> AliasMatchKind.TOKEN
                    else -> null
                }
            }

            HomeAliasMatchType.TOKEN -> {
                if (tokenContains(queryTokens, queryTokensSet, alias.aliasTokens)) {
                    AliasMatchKind.TOKEN
                } else {
                    null
                }
            }

            HomeAliasMatchType.REGEX -> {
                if (alias.regex?.containsMatchIn(normalizedQuery) == true) {
                    AliasMatchKind.FUZZY
                } else {
                    null
                }
            }
        }
    }

    private fun resolveRoute(alias: HomeAliasRuntimeEntry): HomeResolvedRoute? {
        return when (alias.routeKind) {
            HomeAliasRouteKind.CANONICAL -> HomeResolvedRoute(
                routeType = QueryRouteType.OPEN_CATEGORY,
                targetCode = alias.targetId,
            )

            HomeAliasRouteKind.BROWSE_NODE -> {
                val browseCode = normalizeBrowseCode(alias.targetId)
                val canonical = browseByCode[browseCode]
                    ?.targetCategoryCode
                    ?.takeIf { it.isNotBlank() }
                    ?: deriveCategoryFromBrowseCode(browseCode)
                if (canonical != null) {
                    HomeResolvedRoute(
                        routeType = QueryRouteType.OPEN_CATEGORY,
                        targetCode = canonical,
                    )
                } else {
                    HomeResolvedRoute(
                        routeType = QueryRouteType.OPEN_BROWSE,
                        targetCode = browseCode,
                    )
                }
            }

            HomeAliasRouteKind.BLOCKED -> null
        }
    }

    private fun matchesBlockedAlias(
        normalizedQuery: String,
        queryTokens: List<String>,
        queryTokensSet: Set<String>,
        locale: String,
    ): Boolean {
        val localeKey = locale.lowercase()
        val aliases = blockedAliasesByLocale[localeKey].orEmpty()
        return aliases.any { alias ->
            detectMatchKind(alias, normalizedQuery, queryTokens, queryTokensSet) != null
        }
    }

    private fun fallback(
        normalizedQuery: String,
        confidence: Double,
    ): QueryRoutingResult {
        val queryTokens = tokenize(normalizedQuery)
        val shouldRouteToRootBrowse = rules.ifOnlyGenericTokensRouteToRoot && hasOnlyGenericTokens(
            queryTokens = queryTokens,
            normalizedQuery = normalizedQuery,
        )
        return if (shouldRouteToRootBrowse) {
            fallbackToRoot(normalizedQuery = normalizedQuery, confidence = confidence)
        } else {
            QueryRoutingResult(
                routeType = QueryRouteType.OPEN_CATEGORY,
                primaryTargetCode = rules.defaultCanonical,
                extractedTokens = queryTokens,
                confidence = confidence.coerceIn(0.0, 1.0),
            )
        }
    }

    private fun fallbackToRoot(
        normalizedQuery: String,
        confidence: Double,
    ): QueryRoutingResult = QueryRoutingResult(
        routeType = QueryRouteType.OPEN_BROWSE,
        primaryTargetCode = rules.defaultBrowseNode,
        extractedTokens = tokenize(normalizedQuery),
        confidence = confidence.coerceIn(0.0, 1.0),
    )

    private fun hasOnlyGenericTokens(
        queryTokens: List<String>,
        normalizedQuery: String,
    ): Boolean {
        if (queryTokens.isEmpty()) return true
        if (rules.blockedNoiseTokens.isEmpty()) return false
        return queryTokens.all { token ->
            rules.blockedNoiseTokens.any { blocked ->
                val normalizedBlocked = normalize(blocked)
                if (normalizedBlocked.contains(' ')) {
                    containsPhrase(normalizedQuery, normalizedBlocked) &&
                        tokenize(normalizedBlocked).any { blockedToken -> tokensMatch(token, blockedToken) }
                } else {
                    tokensMatch(token, normalizedBlocked)
                }
            }
        }
    }

    private fun deriveCategoryFromBrowseCode(browseCode: String): String? {
        val parts = browseCode.split('.').filter { it.isNotBlank() }
        if (parts.size < 2 || parts.first() != "B") return null

        val rootCategory = parts.getOrNull(1) ?: return null
        val l1Category = parts.getOrNull(2)

        val candidates = listOfNotNull(
            l1Category?.let { "$rootCategory.$it" },
            rootCategory,
        )
        return candidates.firstOrNull { it in knownCategoryCodes }
    }

    private fun HomeScoredCandidate.toResult(normalizedQuery: String): QueryRoutingResult = QueryRoutingResult(
        routeType = routeType,
        primaryTargetCode = targetCode,
        extractedTokens = tokenize(normalizedQuery),
        confidence = confidence.coerceIn(0.0, 1.0),
    )

    private fun HomeScoredCandidate.isBetterThan(other: HomeScoredCandidate): Boolean {
        if (score != other.score) return score > other.score
        if (matchKind != other.matchKind) return matchKind.ordinal < other.matchKind.ordinal
        return targetCode < other.targetCode
    }

    private fun containsPhrase(query: String, phrase: String): Boolean {
        if (phrase.isBlank()) return false
        if (query == phrase) return true
        val escaped = Regex.escape(phrase)
        return Regex("(^|\\s)$escaped(\\s|$)").containsMatchIn(query)
    }

    private fun tokenContains(
        queryTokens: List<String>,
        queryTokensSet: Set<String>,
        aliasTokens: List<String>,
    ): Boolean {
        if (queryTokens.isEmpty() || aliasTokens.isEmpty()) return false
        return aliasTokens.all { aliasToken ->
            queryTokensSet.contains(aliasToken) || queryTokens.any { queryToken -> tokensMatch(queryToken, aliasToken) }
        }
    }

    private fun matchesRuleToken(
        queryTokens: List<String>,
        queryTokensSet: Set<String>,
        normalizedQuery: String,
        rawToken: String,
    ): Boolean {
        val token = normalize(rawToken)
        if (token.isBlank()) return false
        if (' ' in token) {
            return containsPhrase(normalizedQuery, token)
        }
        if (token in queryTokensSet) return true
        return queryTokens.any { queryToken -> tokensMatch(queryToken, token) }
    }

    private fun matchesRuleToken(
        queryTokens: List<String>,
        normalizedQuery: String,
        rawToken: String,
    ): Boolean = matchesRuleToken(
        queryTokens = queryTokens,
        queryTokensSet = emptySet(),
        normalizedQuery = normalizedQuery,
        rawToken = rawToken,
    )

    private fun normalizeBrowseCode(rawBrowseCode: String): String {
        val normalized = rawBrowseCode.trim()
        if (normalized.isBlank()) return normalized
        val upper = normalized.uppercase()
        if (upper.startsWith("B.")) return normalized
        if (upper.startsWith("BN.")) return "B.${normalized.substring(3)}"
        return normalized
    }

    private fun normalize(input: String): String = input
        .trim()
        .lowercase()
        .replace('ё', 'е')
        .replace("[-‐‑‒–—]+".toRegex(), " ")
        .replace("[^\\p{L}\\p{N}\\s]".toRegex(), " ")
        .replace("\\s+".toRegex(), " ")
        .trim()

    private fun tokenize(text: String): List<String> =
        text.split(" ").map { it.trim() }.filter { it.isNotEmpty() }

    private fun tokensMatch(leftRaw: String, rightRaw: String): Boolean {
        val left = leftRaw.trim()
        val right = rightRaw.trim()
        if (left.isEmpty() || right.isEmpty()) return false
        if (left == right) return true
        if (left.length <= 2 || right.length <= 2) return false
        if (left.startsWith(right) || right.startsWith(left)) return true

        val leftStem = tokenStem(left)
        val rightStem = tokenStem(right)
        if (leftStem.length < 4 || rightStem.length < 4) return false
        return leftStem == rightStem || leftStem.startsWith(rightStem) || rightStem.startsWith(leftStem)
    }

    private fun tokenStem(token: String): String {
        val lower = token.lowercase()
        val suffixes = listOf(
            "иями", "ями", "ами", "ов", "ев", "ей",
            "ого", "ему", "ому", "ыми", "ими",
            "ая", "яя", "ое", "ее", "ые", "ие",
            "ой", "ый", "ий", "ом", "ем", "ам", "ям",
            "ах", "ях", "ы", "и", "а", "я", "е", "о", "у", "ю",
        )
        val matchedSuffix = suffixes.firstOrNull { lower.length > it.length + 2 && lower.endsWith(it) }
        return if (matchedSuffix == null) lower else lower.removeSuffix(matchedSuffix)
    }

    private fun contextBonusFor(
        targetCode: String,
        normalizedQuery: String,
        queryTokens: List<String>,
    ): Double {
        val hasToken: (String) -> Boolean = { needle ->
            queryTokens.any { tokensMatch(it, needle) }
        }
        return when (targetCode) {
            "HOME.STORAGE" -> {
                if (
                    hasToken("органайзер") ||
                    hasToken("хранение") ||
                    hasToken("контейнер") ||
                    hasToken("коробка") ||
                    hasToken("корзина") ||
                    hasToken("вешалка")
                ) {
                    -0.20
                } else {
                    0.0
                }
            }

            "HOME.KITCHEN_DINING" -> {
                if (hasToken("контейнер") && hasToken("набор")) {
                    0.22
                } else {
                    0.0
                }
            }

            "HOME.FURNITURE" -> {
                if (hasToken("садов") && hasToken("мебел")) {
                    0.24
                } else if (hasToken("органайзер") && hasToken("шкаф")) {
                    0.15
                } else {
                    0.0
                }
            }

            "HOME.GARDEN" -> {
                if (hasToken("садов") && hasToken("мебел")) {
                    -0.20
                } else {
                    0.0
                }
            }

            else -> {
                if (
                    targetCode == "HOME.CLEANING" &&
                    (
                        hasToken("чистящ") ||
                        hasToken("моющ") ||
                        hasToken("стирк") ||
                        hasToken("уборк") ||
                        hasToken("средств")
                    )
                ) {
                    -0.12
                } else {
                    0.0
                }
            }
        }
    }

    private fun passesRuleSpecificGuard(
        ruleId: String,
        normalizedQuery: String,
        queryTokens: List<String>,
        queryTokensSet: Set<String>,
    ): Boolean {
        val hasToken: (String) -> Boolean = { needle ->
            matchesRuleToken(queryTokens, queryTokensSet, normalizedQuery, needle)
        }

        return when (ruleId) {
            "R.SMART_HOME_LIGHTING" -> {
                val hasStrongSmartSignal = hasToken("smart") ||
                    hasToken("zigbee") ||
                    hasToken("z-wave") ||
                    hasToken("homekit") ||
                    hasToken("alexa") ||
                    hasToken("google home")
                val hasSmartSwitchSignal = hasToken("умн") &&
                    (hasToken("выключател") || hasToken("датчик") || hasToken("розетк"))
                hasStrongSmartSignal || hasSmartSwitchSignal
            }

            "R.PLUMBING_FILTERS" -> {
                val hasCleaningContext = hasToken("чистящ") ||
                    hasToken("моющ") ||
                    hasToken("стирк") ||
                    hasToken("уборк")
                !hasCleaningContext
            }

            else -> true
        }
    }

    private fun compileRegexOrNull(pattern: String, matchType: HomeAliasMatchType): Regex? {
        if (matchType != HomeAliasMatchType.REGEX) return null
        return try {
            Regex(pattern, RegexOption.IGNORE_CASE)
        } catch (_: IllegalArgumentException) {
            null
        }
    }

}

private data class HomeAliasRuntimeEntry(
    val locale: String,
    val aliasText: String,
    val normalizedAlias: String,
    val aliasTokens: List<String>,
    val routeKind: HomeAliasRouteKind,
    val targetId: String,
    val priority: Int,
    val matchType: HomeAliasMatchType,
    val negativeTokens: List<String>,
    val isBlocked: Boolean,
    val regex: Regex?,
)

private data class HomeResolvedRoute(
    val routeType: QueryRouteType,
    val targetCode: String,
)

private data class HomeScoredCandidate(
    val targetCode: String,
    val routeType: QueryRouteType,
    val score: Int,
    val confidence: Double,
    val matchedAlias: String,
    val matchKind: AliasMatchKind,
)

private class HomeAliasMatcherIndex private constructor(
    private val aliases: List<HomeAliasRuntimeEntry>,
    private val exactByNormalizedAlias: Map<String, IntArray>,
    private val aliasIdsByToken: Map<String, IntArray>,
    private val aliasIdsByTokenPrefix: Map<String, IntArray>,
    private val aliasIdsByStem: Map<String, IntArray>,
    private val aliasIdsByStemPrefix: Map<String, IntArray>,
    private val regexAliasIds: IntArray,
    private val tokenStem: (String) -> String,
) {

    fun selectCandidates(
        normalizedQuery: String,
        queryTokens: List<String>,
    ): List<HomeAliasRuntimeEntry> {
        if (aliases.isEmpty()) return emptyList()

        val selected = BooleanArray(aliases.size)
        var selectedCount = 0

        fun collect(ids: IntArray?) {
            if (ids == null) return
            ids.forEach { aliasId ->
                if (aliasId !in selected.indices || selected[aliasId]) return@forEach
                selected[aliasId] = true
                selectedCount++
            }
        }

        collect(exactByNormalizedAlias[normalizedQuery])

        queryTokens.forEach { token ->
            if (token.isBlank()) return@forEach

            collect(aliasIdsByToken[token])

            if (token.length > 2) {
                collect(aliasIdsByTokenPrefix[token])
                for (prefixLength in 3 until token.length) {
                    collect(aliasIdsByToken[token.substring(0, prefixLength)])
                }
            }

            val stem = tokenStem(token)
            if (stem.isBlank()) return@forEach

            collect(aliasIdsByStem[stem])

            if (stem.length >= 4) {
                collect(aliasIdsByStemPrefix[stem])
                for (prefixLength in 4 until stem.length) {
                    collect(aliasIdsByStem[stem.substring(0, prefixLength)])
                }
            }
        }

        collect(regexAliasIds)

        if (selectedCount == 0) return emptyList()

        val result = ArrayList<HomeAliasRuntimeEntry>(selectedCount)
        aliases.indices.forEach { aliasId ->
            if (selected[aliasId]) {
                result += aliases[aliasId]
            }
        }
        return result
    }

    companion object {
        fun build(
            aliases: List<HomeAliasRuntimeEntry>,
            tokenStem: (String) -> String,
        ): HomeAliasMatcherIndex {
            val exactByNormalizedAlias = linkedMapOf<String, MutableList<Int>>()
            val aliasIdsByToken = linkedMapOf<String, MutableList<Int>>()
            val aliasIdsByTokenPrefix = linkedMapOf<String, MutableList<Int>>()
            val aliasIdsByStem = linkedMapOf<String, MutableList<Int>>()
            val aliasIdsByStemPrefix = linkedMapOf<String, MutableList<Int>>()
            val regexAliasIds = mutableListOf<Int>()

            fun append(map: MutableMap<String, MutableList<Int>>, key: String, aliasId: Int) {
                if (key.isBlank()) return
                map.getOrPut(key) { mutableListOf() }.add(aliasId)
            }

            aliases.forEachIndexed { aliasId, alias ->
                append(exactByNormalizedAlias, alias.normalizedAlias, aliasId)
                if (alias.matchType == HomeAliasMatchType.REGEX) {
                    regexAliasIds += aliasId
                }

                alias.aliasTokens.forEach { aliasToken ->
                    if (aliasToken.isBlank()) return@forEach
                    append(aliasIdsByToken, aliasToken, aliasId)

                    if (aliasToken.length > 2) {
                        for (prefixLength in 3..aliasToken.length) {
                            append(aliasIdsByTokenPrefix, aliasToken.substring(0, prefixLength), aliasId)
                        }
                    }

                    val stem = tokenStem(aliasToken)
                    append(aliasIdsByStem, stem, aliasId)

                    if (stem.length >= 4) {
                        for (prefixLength in 4..stem.length) {
                            append(aliasIdsByStemPrefix, stem.substring(0, prefixLength), aliasId)
                        }
                    }
                }
            }

            fun freeze(map: Map<String, MutableList<Int>>): Map<String, IntArray> = map.mapValues { (_, ids) -> ids.toIntArray() }

            return HomeAliasMatcherIndex(
                aliases = aliases,
                exactByNormalizedAlias = freeze(exactByNormalizedAlias),
                aliasIdsByToken = freeze(aliasIdsByToken),
                aliasIdsByTokenPrefix = freeze(aliasIdsByTokenPrefix),
                aliasIdsByStem = freeze(aliasIdsByStem),
                aliasIdsByStemPrefix = freeze(aliasIdsByStemPrefix),
                regexAliasIds = regexAliasIds.toIntArray(),
                tokenStem = tokenStem,
            )
        }
    }
}

private data class HomeRoutingRules(
    val blockedNoiseTokens: List<String>,
    val disambiguationRules: List<HomeDisambiguationRule>,
    val aliasMatchWeight: Double,
    val tokenOverlapWeight: Double,
    val exactPhraseBonus: Double,
    val leafPreferenceBonus: Double,
    val minConfidenceToRoute: Double,
    val defaultBrowseNode: String,
    val defaultCanonical: String,
    val ifOnlyGenericTokensRouteToRoot: Boolean,
)

private data class HomeDisambiguationRule(
    val id: String,
    val ifAnyTokens: List<String>,
    val andAnyTokens: List<String>,
    val andNoneTokens: List<String>,
    val thenRouteCanonical: String?,
    val thenRouteBrowseNode: String?,
    val confidence: Double,
    val notes: String?,
)

private object HomeRoutingRulesParser {
    private val keyValueRegex = Regex("""^([A-Za-z0-9_.-]+):\s*(.+?)\s*$""")

    fun parse(yaml: String): HomeRoutingRules {
        val lines = yaml
            .lineSequence()
            .map { it.trimEnd('\r') }
            .toList()

        val blockedNoiseTokens = parseBlockedNoiseTokens(lines)
        val disambiguationRules = parseDisambiguationRules(lines)
        val scoring = parseKeyValueSection(lines, "scoring:", stopHeader = "fallback:")
        val fallback = parseKeyValueSection(lines, "fallback:", stopHeader = null)

        return HomeRoutingRules(
            blockedNoiseTokens = blockedNoiseTokens,
            disambiguationRules = disambiguationRules,
            aliasMatchWeight = scoring["alias_match_weight"]?.toDoubleOrNull() ?: 1.0,
            tokenOverlapWeight = scoring["token_overlap_weight"]?.toDoubleOrNull() ?: 0.7,
            exactPhraseBonus = scoring["exact_phrase_bonus"]?.toDoubleOrNull() ?: 0.4,
            leafPreferenceBonus = scoring["leaf_preference_bonus"]?.toDoubleOrNull() ?: 0.2,
            minConfidenceToRoute = scoring["min_confidence_to_route"]?.toDoubleOrNull() ?: 0.55,
            defaultBrowseNode = normalizeBrowseCode(fallback["default_browse_node"] ?: "B.HOME"),
            defaultCanonical = fallback["default_canonical"] ?: "HOME",
            ifOnlyGenericTokensRouteToRoot = fallback["if_only_generic_tokens_route_to_root"]
                ?.toBooleanStrictOrNullCompat()
                ?: true,
        )
    }

    private fun parseBlockedNoiseTokens(lines: List<String>): List<String> {
        val start = lines.indexOfFirst { it.trim() == "blocked_noise_tokens:" }
        if (start < 0) return emptyList()

        val out = mutableListOf<String>()
        var i = start + 1
        while (i < lines.size) {
            val trimmed = lines[i].trim()
            if (!trimmed.startsWith("- ")) break
            out += unwrapYamlScalar(trimmed.removePrefix("- ").trim())
            i++
        }
        return out
    }

    private fun parseDisambiguationRules(lines: List<String>): List<HomeDisambiguationRule> {
        val start = lines.indexOfFirst { it.trim() == "disambiguation_ordered_rules:" }
        if (start < 0) return emptyList()

        val out = mutableListOf<HomeDisambiguationRule>()
        var i = start + 1
        while (i < lines.size) {
            val trimmed = lines[i].trim()
            if (trimmed == "scoring:") break
            if (!trimmed.startsWith("- id:")) {
                i++
                continue
            }

            val id = unwrapYamlScalar(trimmed.removePrefix("- id:").trim())
            val ifAnyTokens = mutableListOf<String>()
            val andAnyTokens = mutableListOf<String>()
            val andNoneTokens = mutableListOf<String>()
            var thenRouteCanonical: String? = null
            var thenRouteBrowseNode: String? = null
            var confidence = 1.0
            var notes: String? = null

            i++
            while (i < lines.size) {
                val line = lines[i]
                val inner = line.trim()
                if (inner.startsWith("- id:") || inner == "scoring:") break

                when {
                    inner == "if_any_tokens:" -> {
                        i++
                        while (i < lines.size) {
                            val tokenLine = lines[i].trim()
                            if (!tokenLine.startsWith("- ") || tokenLine.startsWith("- id:")) break
                            ifAnyTokens += unwrapYamlScalar(tokenLine.removePrefix("- ").trim())
                            i++
                        }
                        continue
                    }

                    inner == "and_any_tokens:" -> {
                        i++
                        while (i < lines.size) {
                            val tokenLine = lines[i].trim()
                            if (!tokenLine.startsWith("- ") || tokenLine.startsWith("- id:")) break
                            andAnyTokens += unwrapYamlScalar(tokenLine.removePrefix("- ").trim())
                            i++
                        }
                        continue
                    }

                    inner == "and_none_tokens:" -> {
                        i++
                        while (i < lines.size) {
                            val tokenLine = lines[i].trim()
                            if (!tokenLine.startsWith("- ") || tokenLine.startsWith("- id:")) break
                            andNoneTokens += unwrapYamlScalar(tokenLine.removePrefix("- ").trim())
                            i++
                        }
                        continue
                    }

                    inner.startsWith("then_route_canonical:") -> {
                        thenRouteCanonical = unwrapYamlScalar(inner.removePrefix("then_route_canonical:").trim())
                    }

                    inner.startsWith("then_route_browse_node:") -> {
                        thenRouteBrowseNode = normalizeBrowseCode(
                            unwrapYamlScalar(inner.removePrefix("then_route_browse_node:").trim()),
                        )
                    }

                    inner.startsWith("confidence:") -> {
                        confidence = inner.removePrefix("confidence:").trim().toDoubleOrNull() ?: confidence
                    }

                    inner.startsWith("notes:") -> {
                        notes = unwrapYamlScalar(inner.removePrefix("notes:").trim())
                    }
                }
                i++
            }

            out += HomeDisambiguationRule(
                id = id,
                ifAnyTokens = ifAnyTokens,
                andAnyTokens = andAnyTokens,
                andNoneTokens = andNoneTokens,
                thenRouteCanonical = thenRouteCanonical,
                thenRouteBrowseNode = thenRouteBrowseNode,
                confidence = confidence,
                notes = notes,
            )
        }
        return out
    }

    private fun parseKeyValueSection(
        lines: List<String>,
        sectionHeader: String,
        stopHeader: String?,
    ): Map<String, String> {
        val start = lines.indexOfFirst { it.trim() == sectionHeader }
        if (start < 0) return emptyMap()

        val out = LinkedHashMap<String, String>()
        var i = start + 1
        while (i < lines.size) {
            val trimmed = lines[i].trim()
            if (stopHeader != null && trimmed == stopHeader) break
            if (trimmed.endsWith(":") && keyValueRegex.matchEntire(trimmed) == null) break
            val match = keyValueRegex.matchEntire(trimmed)
            if (match != null) {
                out[match.groupValues[1]] = unwrapYamlScalar(match.groupValues[2])
            }
            i++
        }
        return out
    }

    private fun normalizeBrowseCode(rawBrowseCode: String): String {
        val normalized = rawBrowseCode.trim()
        if (normalized.isBlank()) return normalized
        val upper = normalized.uppercase()
        return if (upper.startsWith("BN.")) "B.${normalized.substring(3)}" else normalized
    }

    private fun unwrapYamlScalar(raw: String): String {
        val trimmed = raw.trim()
        return trimmed.removePrefix("\"").removeSuffix("\"").removePrefix("'").removeSuffix("'")
    }

    private fun String.toBooleanStrictOrNullCompat(): Boolean? = when (trim().lowercase()) {
        "true", "1", "yes" -> true
        "false", "0", "no" -> false
        else -> null
    }
}
