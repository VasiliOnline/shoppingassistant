package com.example.shoppingassistant.domain.catalog

class Stage21HybridQueryRouter(
    private val techRouter: Stage21TechQueryRouter = Stage21TechQueryRouter(),
    private val applRouter: Stage21ApplQueryRouter = Stage21ApplQueryRouter(),
    private val homeRouter: Stage21HomeQueryRouter = Stage21HomeQueryRouter(),
    private val fashRouter: Stage21FashQueryRouter = Stage21FashQueryRouter(),
    private val beautyRouter: Stage21BeautyQueryRouter = Stage21BeautyQueryRouter(),
    private val kidsRouter: Stage21KidsQueryRouter = Stage21KidsQueryRouter(),
    private val foodRouter: Stage21FoodQueryRouter = Stage21FoodQueryRouter(),
    private val petsRouter: Stage21PetsQueryRouter = Stage21PetsQueryRouter(),
    private val sportRouter: Stage21SportQueryRouter = Stage21SportQueryRouter(),
) : QueryRouter {

    override suspend fun route(query: String, locale: String): QueryRoutingResult {
        val techDebug = techRouter.routeWithCandidates(query = query, locale = locale, topN = 1)
        val applDebug = applRouter.routeWithCandidates(query = query, locale = locale, topN = 1)
        val homeDebug = homeRouter.routeWithCandidates(query = query, locale = locale, topN = 1)
        val fashDebug = fashRouter.routeWithCandidates(query = query, locale = locale, topN = 1)
        val beautyDebug = beautyRouter.routeWithCandidates(query = query, locale = locale, topN = 1)
        val kidsDebug = kidsRouter.routeWithCandidates(query = query, locale = locale, topN = 1)
        val foodDebug = foodRouter.routeWithCandidates(query = query, locale = locale, topN = 1)
        val petsDebug = petsRouter.routeWithCandidates(query = query, locale = locale, topN = 1)
        val sportDebug = sportRouter.routeWithCandidates(query = query, locale = locale, topN = 1)

        val decisions = listOf(
            HybridDecision(
                router = HybridRouter.TECH,
                debug = techDebug,
                topCandidate = techDebug.topCandidates.firstOrNull(),
                isFallback = isTechFallback(techDebug.result),
            ),
            HybridDecision(
                router = HybridRouter.APPL,
                debug = applDebug,
                topCandidate = applDebug.topCandidates.firstOrNull(),
                isFallback = isApplFallback(applDebug.result),
            ),
            HybridDecision(
                router = HybridRouter.HOME,
                debug = homeDebug,
                topCandidate = homeDebug.topCandidates.firstOrNull(),
                isFallback = isHomeFallback(homeDebug.result),
            ),
            HybridDecision(
                router = HybridRouter.FASH,
                debug = fashDebug,
                topCandidate = fashDebug.topCandidates.firstOrNull(),
                isFallback = isFashFallback(fashDebug.result),
            ),
            HybridDecision(
                router = HybridRouter.BEAUTY,
                debug = beautyDebug,
                topCandidate = beautyDebug.topCandidates.firstOrNull(),
                isFallback = isBeautyFallback(beautyDebug.result),
            ),
            HybridDecision(
                router = HybridRouter.KIDS,
                debug = kidsDebug,
                topCandidate = kidsDebug.topCandidates.firstOrNull(),
                isFallback = isKidsFallback(kidsDebug.result),
            ),
            HybridDecision(
                router = HybridRouter.FOOD,
                debug = foodDebug,
                topCandidate = foodDebug.topCandidates.firstOrNull(),
                isFallback = isFoodFallback(foodDebug.result),
            ),
            HybridDecision(
                router = HybridRouter.PETS,
                debug = petsDebug,
                topCandidate = petsDebug.topCandidates.firstOrNull(),
                isFallback = isPetsFallback(petsDebug.result),
            ),
            HybridDecision(
                router = HybridRouter.SPORT,
                debug = sportDebug,
                topCandidate = sportDebug.topCandidates.firstOrNull(),
                isFallback = isSportFallback(sportDebug.result),
            ),
        )

        val routable = decisions.filter { it.topCandidate != null }
        if (routable.isEmpty()) {
            return techDebug.result
        }

        if (looksLikePetQuery(query)) {
            val bestPetsDecision = routable
                .asSequence()
                .filter { isPetsTarget(it.debug.result.primaryTargetCode) }
                .sortedWith(
                    compareByDescending<HybridDecision> { it.topCandidate?.score ?: Int.MIN_VALUE }
                        .thenBy { if (it.isFallback) 1 else 0 }
                        .thenByDescending { it.debug.result.confidence }
                        .thenBy { it.router.priority },
                )
                .firstOrNull()
            if (bestPetsDecision != null && !bestPetsDecision.isFallback) {
                return bestPetsDecision.debug.result
            }
        }

        val best = routable
            .sortedWith(
                compareByDescending<HybridDecision> { it.topCandidate?.score ?: Int.MIN_VALUE }
                    .thenBy { if (it.isFallback) 1 else 0 }
                    .thenByDescending { it.debug.result.confidence }
                    .thenBy { it.router.priority },
            )
            .first()
        return best.debug.result
    }

    private fun isTechFallback(result: QueryRoutingResult): Boolean =
        result.routeType == QueryRouteType.OPEN_BROWSE && result.primaryTargetCode == "B.TECH"

    private fun isApplFallback(result: QueryRoutingResult): Boolean =
        result.routeType == QueryRouteType.OPEN_CATEGORY && result.primaryTargetCode == "APPL"

    private fun isHomeFallback(result: QueryRoutingResult): Boolean =
        (result.routeType == QueryRouteType.OPEN_BROWSE && result.primaryTargetCode == "B.HOME") ||
            (result.routeType == QueryRouteType.OPEN_CATEGORY && result.primaryTargetCode == "HOME")

    private fun isFashFallback(result: QueryRoutingResult): Boolean =
        (result.routeType == QueryRouteType.OPEN_BROWSE && result.primaryTargetCode == "B.FASH") ||
            (result.routeType == QueryRouteType.OPEN_CATEGORY && result.primaryTargetCode == "FASH")

    private fun isBeautyFallback(result: QueryRoutingResult): Boolean =
        (result.routeType == QueryRouteType.OPEN_BROWSE && result.primaryTargetCode == "B.BEAUTY") ||
            (result.routeType == QueryRouteType.OPEN_CATEGORY && result.primaryTargetCode == "BEAUTY")

    private fun isKidsFallback(result: QueryRoutingResult): Boolean =
        (result.routeType == QueryRouteType.OPEN_BROWSE && result.primaryTargetCode == "B.KIDS") ||
            (result.routeType == QueryRouteType.OPEN_CATEGORY && result.primaryTargetCode == "KIDS")

    private fun isFoodFallback(result: QueryRoutingResult): Boolean =
        (result.routeType == QueryRouteType.OPEN_BROWSE && result.primaryTargetCode == "B.FOOD") ||
            (result.routeType == QueryRouteType.OPEN_CATEGORY && result.primaryTargetCode == "FOOD")

    private fun isPetsFallback(result: QueryRoutingResult): Boolean =
        (result.routeType == QueryRouteType.OPEN_BROWSE && result.primaryTargetCode == "B.PETS") ||
            (result.routeType == QueryRouteType.OPEN_CATEGORY && result.primaryTargetCode == "PETS")

    private fun isSportFallback(result: QueryRoutingResult): Boolean =
        (result.routeType == QueryRouteType.OPEN_BROWSE && result.primaryTargetCode == "B.SPORT") ||
            (result.routeType == QueryRouteType.OPEN_CATEGORY && result.primaryTargetCode == "SPORT")

    private fun isPetsTarget(targetCode: String?): Boolean =
        targetCode?.let { it == "PETS" || it == "B.PETS" || it.startsWith("PETS.") || it.startsWith("B.PETS.") } == true

    private fun looksLikePetQuery(rawQuery: String): Boolean {
        val query = rawQuery
            .trim()
            .lowercase()
            .replace('ё', 'е')
        if (query.isEmpty()) return false
        val petSignals = listOf(
            "кот",
            "кошк",
            "котен",
            "котенок",
            "собак",
            "щен",
            "питом",
            "зоо",
            "хомяк",
            "грызун",
            "попуга",
            "аквариум",
            "рыбк",
        )
        return petSignals.any { signal -> query.contains(signal) }
    }
}

private data class HybridDecision(
    val router: HybridRouter,
    val debug: QueryRoutingDebugResult,
    val topCandidate: QueryRoutingCandidate?,
    val isFallback: Boolean,
)

private enum class HybridRouter(val priority: Int) {
    TECH(priority = 0),
    APPL(priority = 1),
    HOME(priority = 2),
    FASH(priority = 3),
    BEAUTY(priority = 4),
    KIDS(priority = 5),
    FOOD(priority = 6),
    PETS(priority = 7),
    SPORT(priority = 8),
}
