package com.example.shoppingassistant.domain.catalog

class Stage21KidsQueryRouter(
    aliases: List<HomeAliasSeedRow> = Stage21KidsPackageLoader.aliasSeedRows,
    browseNodes: List<BrowseNode> = Stage21KidsPackageLoader.browseNodes,
    routingRulesYaml: String = Stage21KidsPackageLoader.routingRulesYaml,
) : QueryRouter {

    private val delegate = Stage21HomeQueryRouter(
        aliases = aliases,
        browseNodes = browseNodes,
        routingRulesYaml = routingRulesYaml,
    )

    override suspend fun route(query: String, locale: String): QueryRoutingResult =
        routeWithCandidates(query = query, locale = locale).result

    fun routeWithCandidates(
        query: String,
        locale: String = "ru-RU",
        topN: Int = 3,
    ): QueryRoutingDebugResult {
        val normalizedQuery = normalize(query)
        val queryTokens = tokenize(normalizedQuery)
        val delegateTopN = topN.coerceAtLeast(8)
        val base = delegate.routeWithCandidates(
            query = query,
            locale = locale,
            topN = delegateTopN,
        )

        val baseTarget = base.result.primaryTargetCode.orEmpty()
        val heuristicCategory = heuristicCategoryFor(
            normalizedQuery = normalizedQuery,
            queryTokens = queryTokens,
        )

        if (!shouldOverride(baseTarget = baseTarget, heuristicCategory = heuristicCategory, queryTokens = queryTokens)) {
            return base.copy(topCandidates = base.topCandidates.take(topN.coerceAtLeast(1)))
        }

        val targetCode = heuristicCategory.orEmpty()
        val result = QueryRoutingResult(
            routeType = QueryRouteType.OPEN_CATEGORY,
            primaryTargetCode = targetCode,
            extractedTokens = queryTokens,
            confidence = maxOf(base.result.confidence, 0.84),
        )
        val candidates = overrideCandidates(
            baseCandidates = base.topCandidates,
            targetCode = targetCode,
            limit = topN.coerceAtLeast(1),
        )
        return QueryRoutingDebugResult(
            result = result,
            topCandidates = candidates,
            wasAmbiguous = base.wasAmbiguous,
        )
    }

    private fun shouldOverride(
        baseTarget: String,
        heuristicCategory: String?,
        queryTokens: List<String>,
    ): Boolean {
        if (heuristicCategory.isNullOrBlank()) return false
        if (baseTarget == heuristicCategory) return false
        if (baseTarget.startsWith("TECH.")) return false
        if (baseTarget == FASH_KIDS) return false
        if (baseTarget == HOME_ROOT && looksLikePetQuery(queryTokens)) return false
        if (baseTarget == KIDS_ROOT || baseTarget == KIDS_BROWSE_ROOT) return true
        if (baseTarget == HOME_ROOT && heuristicCategory.startsWith("KIDS.")) return true
        if (baseTarget.startsWith("KIDS.") && heuristicCategory.startsWith("KIDS.")) return true
        return false
    }

    private fun heuristicCategoryFor(
        normalizedQuery: String,
        queryTokens: List<String>,
    ): String? {
        val hasToken: (String) -> Boolean = { needle ->
            queryTokens.any { token -> tokensMatch(token, normalize(needle)) }
        }
        val hasPhrase: (String) -> Boolean = { phrase ->
            containsPhrase(normalizedQuery, normalize(phrase))
        }

        val strollerSignal = hasToken("коляск") ||
            hasToken("автокресл") ||
            hasToken("автолюльк") ||
            hasToken("isofix") ||
            hasToken("бустер")
        if (strollerSignal) return KIDS_STROLLERS

        val feedingChairSignal = hasPhrase("стульчик для кормления") ||
            (hasToken("стульчик") && hasToken("кормлен"))
        if (feedingChairSignal) return KIDS_NURSERY

        val pelenalMattressSignal = hasPhrase("пеленальный матрасик") ||
            hasPhrase("пеленальный матрас")
        if (pelenalMattressSignal) return KIDS_BABY_GEAR

        val nurserySignal = hasToken("кроватк") ||
            hasToken("комод") ||
            hasToken("пеленальн") ||
            hasToken("манеж") ||
            hasPhrase("ворота безопасности") ||
            hasToken("блокиратор") ||
            hasToken("видеоняня") ||
            hasToken("радионяня") ||
            hasToken("ночник")
        if (nurserySignal) return KIDS_NURSERY

        val babyGearSignal = hasToken("подгуз") ||
            hasToken("памперс") ||
            hasToken("пеленк") ||
            hasToken("салфет") ||
            hasToken("бутылочк") ||
            hasToken("пустышк") ||
            hasToken("молокоотсос") ||
            hasToken("смесь") ||
            hasToken("пюре") ||
            hasToken("каша") ||
            hasToken("ванночк") ||
            hasToken("шампун") ||
            hasToken("прорезывател") ||
            hasToken("подогревател") ||
            hasToken("стерилизатор") ||
            hasToken("аспиратор") ||
            hasToken("слинг") ||
            hasToken("переноск") ||
            hasToken("кенгуру") ||
            hasToken("кокон") ||
            hasToken("гнездышк") ||
            hasToken("бортик") ||
            hasToken("боди") ||
            hasToken("ползунк")
        if (babyGearSignal) return KIDS_BABY_GEAR

        val toysSignal = hasToken("lego") ||
            hasToken("лего") ||
            hasToken("duplo") ||
            hasToken("дупло") ||
            hasToken("конструктор") ||
            hasToken("пазл") ||
            hasToken("настольн") ||
            hasToken("кукла") ||
            hasToken("фигурк") ||
            hasPhrase("мягкая игрушка") ||
            hasToken("игрушк") ||
            hasToken("машинк")
        if (toysSignal) return KIDS_TOYS

        return null
    }

    private fun looksLikePetQuery(queryTokens: List<String>): Boolean {
        val petSignals = listOf("кот", "кошка", "котен", "котён", "собак", "щенок")
        return petSignals.any { signal ->
            queryTokens.any { token -> tokensMatch(token, signal) }
        }
    }

    private fun overrideCandidates(
        baseCandidates: List<QueryRoutingCandidate>,
        targetCode: String,
        limit: Int,
    ): List<QueryRoutingCandidate> {
        val existing = baseCandidates.firstOrNull {
            it.routeType == QueryRouteType.OPEN_CATEGORY && it.targetCode == targetCode
        }
        val boosted = QueryRoutingCandidate(
            targetCode = targetCode,
            routeType = QueryRouteType.OPEN_CATEGORY,
            score = maxOf(existing?.score ?: 0, 95),
            matchedAlias = existing?.matchedAlias ?: "kids-heuristic",
            matchKind = existing?.matchKind ?: AliasMatchKind.TOKEN,
        )
        val tail = baseCandidates
            .filterNot { it.routeType == boosted.routeType && it.targetCode == boosted.targetCode }
            .take((limit - 1).coerceAtLeast(0))
        return listOf(boosted) + tail
    }

    private fun containsPhrase(query: String, phrase: String): Boolean {
        if (phrase.isBlank()) return false
        if (query == phrase) return true
        val escaped = Regex.escape(phrase)
        return Regex("(^|\\s)$escaped(\\s|$)").containsMatchIn(query)
    }

    private fun normalize(input: String): String = Stage21QueryTextNormalizer.normalize(input)

    private fun tokenize(text: String): List<String> = Stage21QueryTextNormalizer.tokenize(text)

    private fun tokensMatch(leftRaw: String, rightRaw: String): Boolean {
        val left = leftRaw.trim()
        val right = rightRaw.trim()
        if (left.isEmpty() || right.isEmpty()) return false
        if (left == right) return true
        if (left.length <= 2 || right.length <= 2) return false
        if (left.startsWith(right) || right.startsWith(left)) return true

        val leftStem = tokenStem(left)
        val rightStem = tokenStem(right)
        if (leftStem.length < 4 || rightStem.length < 4) return false
        return leftStem == rightStem || leftStem.startsWith(rightStem) || rightStem.startsWith(leftStem)
    }

    private fun tokenStem(token: String): String {
        val lower = token.lowercase()
        val suffixes = listOf(
            "иями", "ями", "ами", "ов", "ев", "ей",
            "ого", "ему", "ому", "ыми", "ими",
            "ая", "яя", "ое", "ее", "ые", "ие",
            "ой", "ый", "ий", "ом", "ем", "ам", "ям",
            "ах", "ях", "ы", "и", "а", "я", "е", "о", "у", "ю",
        )
        val matchedSuffix = suffixes.firstOrNull { lower.length > it.length + 2 && lower.endsWith(it) }
        return if (matchedSuffix == null) lower else lower.removeSuffix(matchedSuffix)
    }

    private companion object {
        private const val KIDS_ROOT = "KIDS"
        private const val KIDS_BROWSE_ROOT = "B.KIDS"
        private const val HOME_ROOT = "HOME"
        private const val FASH_KIDS = "FASH.KIDS"
        private const val KIDS_BABY_GEAR = "KIDS.BABY_GEAR"
        private const val KIDS_STROLLERS = "KIDS.STROLLERS_CARSEATS"
        private const val KIDS_NURSERY = "KIDS.NURSERY_FURNITURE"
        private const val KIDS_TOYS = "KIDS.TOYS_GAMES"
    }
}
