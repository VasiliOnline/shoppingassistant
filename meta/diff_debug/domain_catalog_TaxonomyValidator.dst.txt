package com.example.shoppingassistant.domain.catalog

enum class TaxonomyIssueSeverity {
    FAIL,
    WARN,
}

data class CoverageReport(
    val canonicalLeafCode: String,
    val reachableViaBrowse: Boolean,
    val reachableViaAlias: Boolean,
    val examples: List<String>,
)

data class TaxonomyValidationIssue(
    val code: String,
    val message: String,
    val severity: TaxonomyIssueSeverity = TaxonomyIssueSeverity.FAIL,
)

data class TaxonomyValidationReport(
    val issues: List<TaxonomyValidationIssue>,
    val coverage: List<CoverageReport> = emptyList(),
) {
    val failIssues: List<TaxonomyValidationIssue>
        get() = issues.filter { it.severity == TaxonomyIssueSeverity.FAIL }

    val warnIssues: List<TaxonomyValidationIssue>
        get() = issues.filter { it.severity == TaxonomyIssueSeverity.WARN }

    val isValid: Boolean
        get() = failIssues.isEmpty()

    fun summary(maxIssues: Int = 20): String {
        if (issues.isEmpty()) return "Taxonomy validation passed."
        val limit = maxIssues.coerceAtLeast(1)
        val orderedIssues = issues.sortedWith(
            compareBy<TaxonomyValidationIssue>({ if (it.severity == TaxonomyIssueSeverity.FAIL) 0 else 1 }, { it.code }),
        )
        val top = orderedIssues.take(limit)
        val hidden = (orderedIssues.size - top.size).coerceAtLeast(0)
        val unreachable = coverage.filter { !it.reachableViaBrowse && !it.reachableViaAlias }

        return buildString {
            appendLine(
                "Taxonomy validation ${if (isValid) "completed with warnings" else "failed"}: " +
                    "${failIssues.size} fail(s), ${warnIssues.size} warning(s).",
            )
            top.forEachIndexed { index, issue ->
                appendLine("${index + 1}. [${issue.severity}] [${issue.code}] ${issue.message}")
            }
            if (hidden > 0) {
                appendLine("... and $hidden more issue(s).")
            }
            if (unreachable.isNotEmpty()) {
                appendLine("Unreachable canonical leaves: ${unreachable.take(10).joinToString { it.canonicalLeafCode }}")
            }
            append("Hint: verify canonical codes, parent links, browse graph, aliases, google mapping and coverage.")
        }
    }
}

class TaxonomyValidator {
    private val categoryCodeRegex = Regex("^[A-Z][A-Z0-9_]*(\\.[A-Z][A-Z0-9_]*){0,2}$")
    private val browseCodeRegex = Regex("^[A-Z][A-Z0-9_]*(\\.[A-Z0-9][A-Z0-9_]*){0,5}$")
    private val requiredBrowseRoots = setOf(
        "B.TECH",
        "B.APPL",
        "B.HOME",
        "B.FASH",
        "B.BEAUTY",
        "B.KIDS",
        "B.FOOD",
        "B.PETS",
        "B.SPORT",
        "B.AUTO",
    )
    private val genericAliasTerms = setOf(
        "товар",
        "вещь",
        "покупка",
        "магазин",
        "лучшее",
        "новинка",
    )
    private val localeRegex = Regex("^[a-z]{2}-[a-z]{2}$", RegexOption.IGNORE_CASE)

    fun validate(
        categories: List<Category>,
        aliases: List<CategoryAlias>,
        mappings: List<GoogleTaxonomyMapping>,
        browseNodes: List<BrowseNode> = emptyList(),
        aliasEntries: List<AliasEntry> = emptyList(),
    ): TaxonomyValidationReport {
        val issues = mutableListOf<TaxonomyValidationIssue>()
        val categoriesByCode = LinkedHashMap<String, Category>()
        val categoryCodeSeen = HashSet<String>()

        categories.forEach { category ->
            val code = category.code.trim()
            if (code.isBlank()) {
                fail(issues, "CATEGORY_CODE_BLANK", "Category code must not be blank.")
                return@forEach
            }
            if (!categoryCodeRegex.matches(code)) {
                fail(
                    issues,
                    "CATEGORY_CODE_FORMAT",
                    "Category code '$code' has invalid format. Expected UPPER_CASE segments.",
                )
            }
            if (!categoryCodeSeen.add(code)) {
                fail(issues, "CATEGORY_CODE_DUPLICATE", "Category code '$code' is duplicated.")
            }

            categoriesByCode[code] = category

            val expectedSegment = segmentFromCode(code)
            if (expectedSegment != null && category.segment != expectedSegment) {
                fail(
                    issues,
                    "CATEGORY_SEGMENT_MISMATCH",
                    "Category '$code' has segment '${category.segment}', expected '$expectedSegment'.",
                )
            }
        }

        categoriesByCode.values.forEach { category ->
            val code = category.code.trim()
            val parentCode = category.parentCode?.trim()?.takeIf { it.isNotEmpty() }
            if (parentCode == null) return@forEach
            if (parentCode == code) {
                fail(issues, "CATEGORY_PARENT_SELF", "Category '$code' cannot reference itself as parent.")
                return@forEach
            }
            if (!categoriesByCode.containsKey(parentCode)) {
                fail(
                    issues,
                    "CATEGORY_PARENT_MISSING",
                    "Category '$code' points to missing parent '$parentCode'.",
                )
            }
        }

        val cycleReported = HashSet<String>()
        val depthMemo = HashMap<String, Int>()
        categoriesByCode.keys.forEach { code ->
            computeCategoryDepth(
                code = code,
                categoriesByCode = categoriesByCode,
                memo = depthMemo,
                path = LinkedHashSet(),
                cycleReported = cycleReported,
                issues = issues,
            )?.takeIf { it > 3 }?.let { depth ->
                fail(issues, "CATEGORY_DEPTH_EXCEEDED", "Category '$code' has depth $depth (> 3).")
            }
        }

        validateLegacyAliases(aliases, categoriesByCode, issues)
        validateMappings(categoriesByCode, mappings, issues)

        val shouldRunStage2Checks = browseNodes.isNotEmpty() || aliasEntries.isNotEmpty()
        if (!shouldRunStage2Checks) {
            return TaxonomyValidationReport(issues = issues)
        }

        val parentCodes = categoriesByCode.values
            .mapNotNull { it.parentCode?.trim()?.takeIf(String::isNotEmpty) }
            .toSet()
        val leafCodes = categoriesByCode.keys.filter { it !in parentCodes }.toSet()

        val browseByCode = validateBrowseNodes(
            browseNodes = browseNodes,
            categoriesByCode = categoriesByCode,
            leafCodes = leafCodes,
            issues = issues,
        )

        val effectiveAliasEntries = if (aliasEntries.isNotEmpty()) {
            aliasEntries
        } else {
            aliases.map {
                AliasEntry(
                    locale = "ru-RU",
                    term = it.alias,
                    normalizedTerm = normalizeAlias(it.alias),
                    kind = AliasKind.CATEGORY,
                    targetCode = it.categoryCode,
                    weight = 50,
                    source = AliasSource.MANUAL,
                    notes = "Auto-migrated from CategoryAlias",
                )
            }
        }

        validateAliasEntries(
            aliasEntries = effectiveAliasEntries,
            categoriesByCode = categoriesByCode,
            browseByCode = browseByCode,
            leafCodes = leafCodes,
            issues = issues,
        )

        val coverage = buildCoverageReport(
            leafCodes = leafCodes,
            browseNodes = browseByCode.values,
            aliasEntries = effectiveAliasEntries,
            browseByCode = browseByCode,
        )
        coverage
            .filter { !it.reachableViaBrowse && !it.reachableViaAlias }
            .forEach { report ->
                fail(
                    issues,
                    "COVERAGE_LEAF_UNREACHABLE",
                    "Leaf category '${report.canonicalLeafCode}' is unreachable via browse and aliases.",
                )
            }

        return TaxonomyValidationReport(
            issues = issues,
            coverage = coverage,
        )
    }

    private fun validateLegacyAliases(
        aliases: List<CategoryAlias>,
        categoriesByCode: Map<String, Category>,
        issues: MutableList<TaxonomyValidationIssue>,
    ) {
        val seenAliasPairs = HashSet<String>()
        val aliasToCodes = LinkedHashMap<String, MutableSet<String>>()

        aliases.forEach { alias ->
            val normalized = normalizeAlias(alias.alias)
            if (normalized.isBlank()) {
                fail(issues, "ALIAS_BLANK", "Alias '${alias.alias}' is empty after normalization.")
                return@forEach
            }
            val categoryCode = alias.categoryCode.trim()
            if (!categoriesByCode.containsKey(categoryCode)) {
                fail(
                    issues,
                    "ALIAS_CATEGORY_MISSING",
                    "Alias '$normalized' points to unknown category '$categoryCode'.",
                )
            }

            val pairKey = "$normalized|$categoryCode"
            if (!seenAliasPairs.add(pairKey)) {
                fail(
                    issues,
                    "ALIAS_DUPLICATE",
                    "Alias '$normalized' for category '$categoryCode' is duplicated.",
                )
            }

            aliasToCodes.getOrPut(normalized) { linkedSetOf() }.add(categoryCode)
        }

        aliasToCodes.forEach { (normalizedAlias, categoryCodes) ->
            val nonBlankCodes = categoryCodes.filter { it.isNotBlank() }.toSet()
            if (nonBlankCodes.size > 1) {
                fail(
                    issues,
                    "ALIAS_AMBIGUOUS",
                    "Alias '$normalizedAlias' maps to multiple categories: ${nonBlankCodes.sorted().joinToString(", ")}.",
                )
            }
        }
    }

    private fun validateMappings(
        categoriesByCode: Map<String, Category>,
        mappings: List<GoogleTaxonomyMapping>,
        issues: MutableList<TaxonomyValidationIssue>,
    ) {
        val seenCodes = HashSet<String>()
        val mappingByCode = LinkedHashMap<String, GoogleTaxonomyMapping>()

        mappings.forEach { mapping ->
            val canonicalCode = mapping.canonicalCode.trim()
            if (canonicalCode.isBlank()) {
                fail(issues, "MAPPING_CODE_BLANK", "Google mapping canonical code must not be blank.")
                return@forEach
            }
            if (!seenCodes.add(canonicalCode)) {
                fail(issues, "MAPPING_CODE_DUPLICATE", "Google mapping for category '$canonicalCode' is duplicated.")
            }
            if (!categoriesByCode.containsKey(canonicalCode)) {
                fail(issues, "MAPPING_CATEGORY_MISSING", "Google mapping references unknown category '$canonicalCode'.")
            }

            val ids = mapping.googleIds
            val paths = mapping.googlePaths
            if (ids.size != paths.size) {
                fail(
                    issues,
                    "MAPPING_CARDINALITY_MISMATCH",
                    "Category '$canonicalCode' has ${ids.size} googleIds but ${paths.size} googlePaths.",
                )
            }
            if (ids.any { it <= 0L }) {
                fail(issues, "MAPPING_ID_INVALID", "Category '$canonicalCode' has non-positive Google taxonomy id.")
            }
            if (paths.any { it.isBlank() }) {
                fail(issues, "MAPPING_PATH_BLANK", "Category '$canonicalCode' has blank Google taxonomy path.")
            }

            when (mapping.mappingType) {
                GoogleTaxonomyMappingType.SINGLE -> {
                    if (ids.size != 1 || paths.size != 1) {
                        fail(
                            issues,
                            "MAPPING_SINGLE_INVALID",
                            "Category '$canonicalCode' requires exactly 1 id/path for SINGLE mapping.",
                        )
                    }
                }

                GoogleTaxonomyMappingType.MULTI -> {
                    if (ids.size < 2 || paths.size < 2) {
                        fail(
                            issues,
                            "MAPPING_MULTI_INVALID",
                            "Category '$canonicalCode' requires >= 2 id/path values for MULTI mapping.",
                        )
                    }
                }

                GoogleTaxonomyMappingType.NONE -> {
                    if (ids.isNotEmpty() || paths.isNotEmpty()) {
                        fail(
                            issues,
                            "MAPPING_NONE_INVALID",
                            "Category '$canonicalCode' must not contain id/path values for NONE mapping.",
                        )
                    }
                }
            }

            mappingByCode[canonicalCode] = mapping
        }

        categoriesByCode.keys.forEach { categoryCode ->
            if (!mappingByCode.containsKey(categoryCode)) {
                fail(issues, "MAPPING_MISSING", "Category '$categoryCode' has no Google taxonomy mapping.")
            }
        }
    }

    private fun validateBrowseNodes(
        browseNodes: List<BrowseNode>,
        categoriesByCode: Map<String, Category>,
        leafCodes: Set<String>,
        issues: MutableList<TaxonomyValidationIssue>,
    ): Map<String, BrowseNode> {
        val browseByCode = LinkedHashMap<String, BrowseNode>()
        val seenBrowseCodes = HashSet<String>()

        browseNodes.forEach { node ->
            val browseCode = node.browseCode.trim()
            if (browseCode.isBlank()) {
                fail(issues, "BROWSE_CODE_BLANK", "Browse node code must not be blank.")
                return@forEach
            }
            if (!browseCodeRegex.matches(browseCode)) {
                fail(
                    issues,
                    "BROWSE_CODE_FORMAT",
                    "Browse code '$browseCode' has invalid format.",
                )
            }
            if (!seenBrowseCodes.add(browseCode)) {
                fail(issues, "BROWSE_CODE_DUPLICATE", "Browse code '$browseCode' is duplicated.")
            }
            if (node.titleRu.trim().isBlank()) {
                fail(issues, "BROWSE_TITLE_BLANK", "Browse node '$browseCode' has blank titleRu.")
            }
            if (node.order < 0) {
                fail(issues, "BROWSE_ORDER_INVALID", "Browse node '$browseCode' has negative order ${node.order}.")
            }

            val targetCode = node.targetCategoryCode?.trim()?.takeIf(String::isNotEmpty)
            when (node.targetType) {
                BrowseTargetType.CATEGORY -> {
                    if (targetCode == null) {
                        fail(issues, "BROWSE_TARGET_MISSING", "Browse node '$browseCode' has CATEGORY targetType but no targetCategoryCode.")
                    }
                }

                BrowseTargetType.SEARCH_PRESET -> {
                    if (targetCode != null) {
                        fail(
                            issues,
                            "BROWSE_TARGET_INVALID",
                            "Browse node '$browseCode' has SEARCH_PRESET targetType but non-null targetCategoryCode.",
                        )
                    }
                }

                null -> {
                    if (targetCode != null) {
                        fail(
                            issues,
                            "BROWSE_TARGET_TYPE_MISSING",
                            "Browse node '$browseCode' has targetCategoryCode but null targetType.",
                        )
                    }
                }
            }

            if (targetCode != null) {
                if (!categoriesByCode.containsKey(targetCode)) {
                    fail(
                        issues,
                        "BROWSE_TARGET_CATEGORY_MISSING",
                        "Browse node '$browseCode' points to missing category '$targetCode'.",
                    )
                } else if (targetCode !in leafCodes) {
                    fail(
                        issues,
                        "BROWSE_TARGET_NOT_LEAF",
                        "Browse node '$browseCode' points to non-leaf category '$targetCode'.",
                    )
                }
            }

            browseByCode[browseCode] = node
        }

        browseByCode.values.forEach { node ->
            val browseCode = node.browseCode.trim()
            val parentCode = node.parentBrowseCode?.trim()?.takeIf(String::isNotEmpty) ?: return@forEach
            if (parentCode == browseCode) {
                fail(issues, "BROWSE_PARENT_SELF", "Browse node '$browseCode' cannot reference itself as parent.")
                return@forEach
            }
            if (!browseByCode.containsKey(parentCode)) {
                fail(
                    issues,
                    "BROWSE_PARENT_MISSING",
                    "Browse node '$browseCode' points to missing parent '$parentCode'.",
                )
            }
        }

        val duplicateOrdersByParent = browseByCode.values
            .groupBy { normalizeParentCode(it.parentBrowseCode) }
            .mapValues { (_, siblings) ->
                siblings.groupBy { it.order }.filterValues { duplicates -> duplicates.size > 1 }
            }
            .filterValues { it.isNotEmpty() }
        duplicateOrdersByParent.forEach { (parentCode, duplicatesByOrder) ->
            val parentLabel = if (parentCode.isBlank()) "<ROOT>" else parentCode
            duplicatesByOrder.forEach { (order, duplicates) ->
                val duplicateCodes = duplicates.map { it.browseCode }.sorted().joinToString(", ")
                fail(
                    issues,
                    "BROWSE_ORDER_DUPLICATE",
                    "Browse siblings under '$parentLabel' reuse order $order: $duplicateCodes.",
                )
            }
        }

        val hasBrowseNamespace = browseByCode.keys.any { it.startsWith("B.") }
        if (hasBrowseNamespace) {
            val rootNodes = browseByCode.values.filter { normalizeParentCode(it.parentBrowseCode).isBlank() }
            val rootCodes = rootNodes.map { it.browseCode.trim() }.toSet()
            val missingRequiredRoots = requiredBrowseRoots - rootCodes
            missingRequiredRoots.sorted().forEach { rootCode ->
                fail(
                    issues,
                    "BROWSE_ROOT_REQUIRED_MISSING",
                    "Required browse root '$rootCode' is missing.",
                )
            }
            requiredBrowseRoots.forEach { rootCode ->
                val rootNode = browseByCode[rootCode] ?: return@forEach
                if (rootNode.nodeKind != BrowseNodeKind.ROOT) {
                    fail(
                        issues,
                        "BROWSE_ROOT_KIND_INVALID",
                        "Required browse root '$rootCode' must use nodeKind=ROOT.",
                    )
                }
            }
        }

        val browseCycleReported = HashSet<String>()
        val browseDepthMemo = HashMap<String, Int>()
        browseByCode.keys.forEach { browseCode ->
            computeBrowseDepth(
                browseCode = browseCode,
                browseByCode = browseByCode,
                memo = browseDepthMemo,
                path = LinkedHashSet(),
                cycleReported = browseCycleReported,
                issues = issues,
            )
        }

        val childrenPerParent = browseByCode.values
            .groupingBy { it.parentBrowseCode?.trim().orEmpty() }
            .eachCount()
            .filterKeys { it.isNotEmpty() }
        childrenPerParent.forEach { (parentCode, childrenCount) ->
            if (childrenCount > 12) {
                warn(
                    issues,
                    "BROWSE_CHILDREN_LIMIT",
                    "Browse node '$parentCode' has $childrenCount children (>12).",
                )
            }
        }

        return browseByCode
    }

    private fun validateAliasEntries(
        aliasEntries: List<AliasEntry>,
        categoriesByCode: Map<String, Category>,
        browseByCode: Map<String, BrowseNode>,
        leafCodes: Set<String>,
        issues: MutableList<TaxonomyValidationIssue>,
    ) {
        val exactAliasKeySeen = HashSet<String>()
        val localeKindTargetsByTerm = LinkedHashMap<String, MutableSet<String>>()

        aliasEntries.forEach { entry ->
            val locale = entry.locale.trim()
            val localeKey = locale.lowercase()
            val normalized = normalizeAlias(entry.term)
            val normalizedTerm = entry.normalizedTerm.trim()
            val targetCode = entry.targetCode.trim()
            if (locale.isBlank()) {
                fail(issues, "ALIAS_ENTRY_LOCALE_BLANK", "Alias entry term '${entry.term}' has blank locale.")
            }
            if (locale.isNotBlank() && !localeRegex.matches(locale)) {
                fail(
                    issues,
                    "ALIAS_ENTRY_LOCALE_UNSUPPORTED",
                    "Alias entry '${entry.term}' has unsupported locale '$locale'.",
                )
            }
            if (normalized.isBlank()) {
                fail(issues, "ALIAS_ENTRY_TERM_BLANK", "Alias entry term '${entry.term}' is blank after normalization.")
                return@forEach
            }
            if (normalizedTerm.isBlank()) {
                fail(
                    issues,
                    "ALIAS_ENTRY_TERM_NOT_NORMALIZED",
                    "Alias entry '${entry.term}' has blank normalizedTerm.",
                )
            } else if (normalizedTerm != normalized) {
                fail(
                    issues,
                    "ALIAS_ENTRY_TERM_NOT_NORMALIZED",
                    "Alias entry term '${entry.term}' must have normalizedTerm='$normalized'.",
                )
            }
            if (entry.weight !in 0..100) {
                fail(
                    issues,
                    "ALIAS_ENTRY_WEIGHT_INVALID",
                    "Alias entry '$normalizedTerm' has weight ${entry.weight}, expected 0..100.",
                )
            }

            val duplicateKey = "$localeKey|$normalizedTerm|${entry.kind.name}|$targetCode"
            if (!exactAliasKeySeen.add(duplicateKey)) {
                fail(
                    issues,
                    "ALIAS_ENTRY_DUPLICATE",
                    "Alias entry '$normalizedTerm' is duplicated for locale '$locale', kind='${entry.kind}', target='$targetCode'.",
                )
            }

            val collisionKey = "$localeKey|${entry.kind.name}|$normalizedTerm"
            localeKindTargetsByTerm.getOrPut(collisionKey) { linkedSetOf() }.add(targetCode)

            val targetCategoryExists = categoriesByCode.containsKey(targetCode)
            val targetBrowseExists = browseByCode.containsKey(targetCode)
            if (!targetCategoryExists && !targetBrowseExists) {
                fail(
                    issues,
                    "ALIAS_ENTRY_TARGET_MISSING",
                    "Alias entry '$normalizedTerm' points to unknown target '$targetCode'.",
                )
            }

            when (entry.kind) {
                AliasKind.CATEGORY -> {
                    if (!targetCategoryExists) {
                        fail(
                            issues,
                            "ALIAS_ENTRY_TARGET_KIND_MISMATCH",
                            "Category alias '$normalizedTerm' must point to canonical category code.",
                        )
                    } else if (targetCode !in leafCodes) {
                        fail(
                            issues,
                            "ALIAS_ENTRY_CATEGORY_NOT_LEAF",
                            "Category alias '$normalizedTerm' points to non-leaf category '$targetCode'.",
                        )
                    }
                }

                AliasKind.BROWSE -> {
                    if (!targetBrowseExists) {
                        fail(
                            issues,
                            "ALIAS_ENTRY_TARGET_KIND_MISMATCH",
                            "Browse alias '$normalizedTerm' must point to browse code.",
                        )
                    }
                }

                AliasKind.BRAND,
                AliasKind.ATTRIBUTE_HINT,
                -> {
                    // Can point either to category or browse node.
                }
            }

            if (!entry.isBlocked && isGenericAliasTerm(normalizedTerm)) {
                warn(
                    issues,
                    "ALIAS_ENTRY_GENERIC_TERM",
                    "Alias entry '$normalizedTerm' is too generic and may hurt routing quality.",
                )
            }
        }

        localeKindTargetsByTerm.forEach { (key, targets) ->
            if (targets.size > 1) {
                val sortedTargets = targets.sorted().joinToString(", ")
                fail(
                    issues,
                    "ALIAS_ENTRY_COLLISION",
                    "Alias collision for '$key': targets=[$sortedTargets].",
                )
            }
        }

        val stemGroups = aliasEntries
            .asSequence()
            .filterNot { it.isBlocked }
            .groupBy { entry ->
                "${entry.locale.lowercase()}|${aliasStem(entry.normalizedTerm)}"
            }
            .filterKeys { it.isNotBlank() }
        stemGroups.forEach { (_, group) ->
            if (group.size <= 1) return@forEach
            val distinctTargets = group.map { it.targetCode.trim() }.toSet()
            if (distinctTargets.size <= 1) return@forEach
            val distinctWeights = group.map { it.weight }.toSet()
            if (distinctWeights.size <= 1) {
                val terms = group.map { it.term }.sorted().joinToString(", ")
                warn(
                    issues,
                    "ALIAS_ENTRY_WEIGHT_CONFLICT",
                    "Alias conflict without weight split for terms: $terms.",
                )
            }
        }
    }

    private fun buildCoverageReport(
        leafCodes: Set<String>,
        browseNodes: Collection<BrowseNode>,
        aliasEntries: List<AliasEntry>,
        browseByCode: Map<String, BrowseNode>,
    ): List<CoverageReport> {
        val browseHits = LinkedHashMap<String, MutableList<String>>()
        browseNodes.forEach { node ->
            val target = node.targetCategoryCode?.trim()?.takeIf(String::isNotEmpty) ?: return@forEach
            if (target in leafCodes) {
                browseHits.getOrPut(target) { mutableListOf() }.add(node.browseCode)
            }
        }

        val aliasHits = LinkedHashMap<String, MutableList<String>>()
        aliasEntries.forEach { entry ->
            if (entry.isBlocked) return@forEach
            resolveAliasTargets(entry, leafCodes, browseByCode).forEach { leafCode ->
                aliasHits.getOrPut(leafCode) { mutableListOf() }.add(entry.normalizedTerm)
            }
        }

        return leafCodes.map { leafCode ->
            val browseExamples = browseHits[leafCode].orEmpty().take(3).map { "browse:$it" }
            val aliasExamples = aliasHits[leafCode].orEmpty().take(3).map { "alias:$it" }
            CoverageReport(
                canonicalLeafCode = leafCode,
                reachableViaBrowse = browseHits[leafCode].isNullOrEmpty().not(),
                reachableViaAlias = aliasHits[leafCode].isNullOrEmpty().not(),
                examples = (browseExamples + aliasExamples).take(6),
            )
        }
    }

    private fun resolveAliasTargets(
        entry: AliasEntry,
        leafCodes: Set<String>,
        browseByCode: Map<String, BrowseNode>,
    ): List<String> {
        if (entry.isBlocked) return emptyList()
        val targetCode = entry.targetCode.trim()
        if (targetCode in leafCodes) {
            return listOf(targetCode)
        }
        val browseNode = browseByCode[targetCode] ?: return emptyList()
        val leafFromBrowse = browseNode.targetCategoryCode?.trim()?.takeIf { it in leafCodes } ?: return emptyList()
        return listOf(leafFromBrowse)
    }

    private fun computeCategoryDepth(
        code: String,
        categoriesByCode: Map<String, Category>,
        memo: MutableMap<String, Int>,
        path: LinkedHashSet<String>,
        cycleReported: MutableSet<String>,
        issues: MutableList<TaxonomyValidationIssue>,
    ): Int? {
        memo[code]?.let { return it }
        val category = categoriesByCode[code] ?: return null

        if (!path.add(code)) {
            val cycle = (path + code).joinToString(" -> ")
            if (cycleReported.add(cycle)) {
                fail(issues, "CATEGORY_CYCLE", "Cycle detected in category hierarchy: $cycle.")
            }
            return null
        }

        val parentCode = category.parentCode?.trim()?.takeIf(String::isNotEmpty)
        val depth = if (parentCode == null) {
            1
        } else {
            val parentDepth = computeCategoryDepth(
                code = parentCode,
                categoriesByCode = categoriesByCode,
                memo = memo,
                path = path,
                cycleReported = cycleReported,
                issues = issues,
            )
            if (parentDepth == null) null else parentDepth + 1
        }

        path.remove(code)
        if (depth != null) memo[code] = depth
        return depth
    }

    private fun computeBrowseDepth(
        browseCode: String,
        browseByCode: Map<String, BrowseNode>,
        memo: MutableMap<String, Int>,
        path: LinkedHashSet<String>,
        cycleReported: MutableSet<String>,
        issues: MutableList<TaxonomyValidationIssue>,
    ): Int? {
        memo[browseCode]?.let { return it }
        val node = browseByCode[browseCode] ?: return null

        if (!path.add(browseCode)) {
            val cycle = (path + browseCode).joinToString(" -> ")
            if (cycleReported.add(cycle)) {
                fail(issues, "BROWSE_CYCLE", "Cycle detected in browse hierarchy: $cycle.")
            }
            return null
        }

        val parentCode = node.parentBrowseCode?.trim()?.takeIf(String::isNotEmpty)
        val depth = if (parentCode == null) {
            1
        } else {
            val parentDepth = computeBrowseDepth(
                browseCode = parentCode,
                browseByCode = browseByCode,
                memo = memo,
                path = path,
                cycleReported = cycleReported,
                issues = issues,
            )
            if (parentDepth == null) null else parentDepth + 1
        }

        path.remove(browseCode)
        if (depth != null) memo[browseCode] = depth
        return depth
    }

    private fun segmentFromCode(code: String): CategorySegment? = when (code.substringBefore('.')) {
        "TECH" -> CategorySegment.TECH
        "APPL" -> CategorySegment.APPL
        "HOME" -> CategorySegment.HOME
        "FASH" -> CategorySegment.FASH
        "BEAUTY" -> CategorySegment.BEAUTY
        "KIDS" -> CategorySegment.KIDS
        "FOOD" -> CategorySegment.FOOD
        "PETS" -> CategorySegment.PETS
        "SPORT" -> CategorySegment.SPORT
        "AUTO" -> CategorySegment.AUTO
        "SUPP" -> CategorySegment.SUPP
        else -> CategorySegment.OTHER
    }

    private fun normalizeAlias(value: String): String = value
        .trim()
        .lowercase()
        .replace('ё', 'е')
        .replace("[-‐‑‒–—]+".toRegex(), " ")
        .replace("[^\\p{L}\\p{N}\\s]".toRegex(), " ")
        .replace("\\s+".toRegex(), " ")
        .trim()

    private fun aliasStem(value: String): String = value
        .trim()
        .lowercase()
        .replace("[^a-zа-я0-9\\s]".toRegex(), "")
        .split(" ")
        .firstOrNull()
        .orEmpty()
        .removeSuffix("ы")
        .removeSuffix("и")
        .removeSuffix("а")
        .removeSuffix("я")
        .removeSuffix("s")

    private fun isGenericAliasTerm(term: String): Boolean =
        term.length <= 2 || term in genericAliasTerms

    private fun normalizeParentCode(parentBrowseCode: String?): String =
        parentBrowseCode?.trim()?.takeIf(String::isNotEmpty).orEmpty()

    private fun fail(
        issues: MutableList<TaxonomyValidationIssue>,
        code: String,
        message: String,
    ) {
        issues += TaxonomyValidationIssue(
            code = code,
            message = message,
            severity = TaxonomyIssueSeverity.FAIL,
        )
    }

    private fun warn(
        issues: MutableList<TaxonomyValidationIssue>,
        code: String,
        message: String,
    ) {
        issues += TaxonomyValidationIssue(
            code = code,
            message = message,
            severity = TaxonomyIssueSeverity.WARN,
        )
    }
