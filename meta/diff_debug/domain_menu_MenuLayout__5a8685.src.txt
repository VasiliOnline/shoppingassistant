package com.example.shoppingassistant.domain.menu

import kotlinx.serialization.Serializable

@Serializable
enum class MenuActionId {
    DASHBOARD,
    SEARCH,
    POST,
    PROFILE,
    MESSAGES,
    NOTIFICATIONS,
    SUBSCRIPTIONS,
}

@Serializable
enum class MenuEdge {
    RIGHT,
    BOTTOM,
}

@Serializable
data class MenuLayout(
    val right: List<MenuActionId?> = emptyList(),
    val bottom: List<MenuActionId?> = emptyList(),
    val pinned: Set<MenuActionId> = setOf(MenuActionId.DASHBOARD),
    val slotBackgroundAlpha: Float = 0.08f,
    val slotBackgroundBlur: Float = 0f,
    val version: Int = CURRENT_VERSION,
) {
    companion object {
        const val CURRENT_VERSION = 1
    }
}

data class MenuLayoutConfig(
    val rightSlots: Int,
    val bottomSlots: Int,
)

typealias UserPanel = MenuLayout
typealias UserPanelConfig = MenuLayoutConfig

data class MenuSlotRef(
    val edge: MenuEdge,
    val index: Int,
)

data class MenuDropOutcome(
    val layout: MenuLayout,
    val removed: MenuActionId? = null,
)
