package com.example.shoppingassistant.domain.menu

import org.junit.Assert.assertEquals
import org.junit.Assert.assertNull
import org.junit.Test

class UserPanelOpsTest {
    private val config = UserPanelConfig(
        sideSlots = 3,
        bottomSlots = 4,
        supportedActions = listOf(ActionKey.POST, ActionKey.MESSAGES, ActionKey.NOTIFICATIONS),
        supportedModes = listOf(ModeKey.DASHBOARD, ModeKey.SEARCH, ModeKey.SUBSCRIPTIONS, ModeKey.PROFILE),
    )

    @Test
    fun replaceAlways_onOccupiedSlot_removesTargetAndMovesAction() {
        val panel = UserPanel(
            sideActions = sideSlots(ActionKey.POST, ActionKey.MESSAGES, ActionKey.NOTIFICATIONS),
            bottomModes = bottomSlots(ModeKey.DASHBOARD, ModeKey.SEARCH, ModeKey.SUBSCRIPTIONS, ModeKey.PROFILE),
        )

        val result = UserPanelOps.applyActionDrop(
            panel = panel,
            actionKey = ActionKey.POST,
            to = PanelSlotRef(PanelEdge.SIDE, 1),
        )

        assertEquals(ActionKey.POST, result.panel.sideActions[1].key)
        assertNull(result.panel.sideActions[0].key)
        assertEquals(ActionKey.MESSAGES, result.removed)
    }

    @Test
    fun deleteDrop_removesAction() {
        val panel = UserPanel(
            sideActions = sideSlots(ActionKey.POST, ActionKey.MESSAGES, ActionKey.NOTIFICATIONS),
            bottomModes = bottomSlots(ModeKey.DASHBOARD, ModeKey.SEARCH, ModeKey.SUBSCRIPTIONS, ModeKey.PROFILE),
        )

        val result = UserPanelOps.applyActionDrop(
            panel = panel,
            actionKey = ActionKey.MESSAGES,
            to = null,
        )

        assertNull(result.panel.sideActions[1].key)
        assertEquals(ActionKey.MESSAGES, result.removed)
    }

    @Test
    fun normalize_removesDuplicates_andKeepsMinimums() {
        val panel = UserPanel(
            sideActions = sideSlots(ActionKey.POST, ActionKey.POST, null),
            bottomModes = bottomSlots(ModeKey.DASHBOARD, ModeKey.SEARCH, ModeKey.SEARCH, null),
        )

        val normalized = UserPanelOps.normalize(panel, config)

        val sideKeys = normalized.sideActions.mapNotNull { it.key }
        val bottomKeys = normalized.bottomModes.mapNotNull { it.key }
        assertEquals(1, sideKeys.count { it == ActionKey.POST })
        assertEquals(1, bottomKeys.count { it == ModeKey.SEARCH })
        assertEquals(true, bottomKeys.contains(ModeKey.DASHBOARD))
    }

    @Test
    fun validator_flagsSizeMismatch_andDuplicates() {
        val panel = UserPanel(
            sideActions = sideSlots(ActionKey.POST, ActionKey.POST),
            bottomModes = bottomSlots(ModeKey.SEARCH),
        )

        val issues = UserPanelValidator.validate(panel, config)

        val types = issues.map { it.type }.toSet()
        assertEquals(true, types.contains(UserPanelIssueType.SIZE_MISMATCH))
        assertEquals(true, types.contains(UserPanelIssueType.DUPLICATE))
        assertEquals(true, types.contains(UserPanelIssueType.BELOW_MINIMUM))
    }

    @Test
    fun normalize_clearsUnsupportedKeys_andFillsDefaults() {
        val config = UserPanelConfig(
            sideSlots = 2,
            bottomSlots = 2,
            supportedActions = listOf(ActionKey.POST, ActionKey.MESSAGES),
            supportedModes = listOf(ModeKey.DASHBOARD, ModeKey.SEARCH),
        )
        val panel = UserPanel(
            sideActions = sideSlots(ActionKey.POST, ActionKey.NOTIFICATIONS),
            bottomModes = bottomSlots(ModeKey.DASHBOARD, ModeKey.SUBSCRIPTIONS),
        )

        val normalized = UserPanelOps.normalize(panel, config)

        assertEquals(ActionKey.POST, normalized.sideActions[0].key)
        assertEquals(ActionKey.MESSAGES, normalized.sideActions[1].key)
        assertEquals(ModeKey.DASHBOARD, normalized.bottomModes[0].key)
        assertEquals(ModeKey.SEARCH, normalized.bottomModes[1].key)
    }

    @Test
    fun validator_flagsUnsupportedKeys() {
        val config = UserPanelConfig(
            sideSlots = 2,
            bottomSlots = 2,
            supportedActions = listOf(ActionKey.POST),
            supportedModes = listOf(ModeKey.DASHBOARD),
        )
        val panel = UserPanel(
            sideActions = sideSlots(ActionKey.POST, ActionKey.MESSAGES),
            bottomModes = bottomSlots(ModeKey.DASHBOARD, ModeKey.SEARCH),
        )

        val issues = UserPanelValidator.validate(panel, config)

        val unsupportedEdges = issues
            .filter { it.type == UserPanelIssueType.UNSUPPORTED_KEY }
            .mapNotNull { it.edge }
            .toSet()
        assertEquals(setOf(PanelEdge.SIDE, PanelEdge.BOTTOM), unsupportedEdges)
    }

    private fun sideSlots(vararg keys: ActionKey?): List<PanelSlot<ActionKey>> {
        return keys.mapIndexed { index, key ->
            PanelSlot(slotId = "side-$index", key = key)
        }
    }

    private fun bottomSlots(vararg keys: ModeKey?): List<PanelSlot<ModeKey>> {
        return keys.mapIndexed { index, key ->
            PanelSlot(slotId = "bottom-$index", key = key)
        }
    }
}
