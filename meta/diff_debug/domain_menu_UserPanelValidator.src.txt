package com.example.shoppingassistant.domain.menu

enum class UserPanelIssueType {
    SIZE_MISMATCH,
    DUPLICATE,
    BELOW_MINIMUM,
    UNSUPPORTED_KEY,
}

data class UserPanelIssue(
    val type: UserPanelIssueType,
    val edge: PanelEdge? = null,
)

object UserPanelValidator {
    private const val MIN_BOTTOM = 2
    private const val MIN_SIDE = 1

    fun validate(panel: UserPanel, config: UserPanelConfig): List<UserPanelIssue> {
        val issues = mutableListOf<UserPanelIssue>()
        if (panel.sideActions.size != config.sideSlots) {
            issues += UserPanelIssue(UserPanelIssueType.SIZE_MISMATCH, edge = PanelEdge.SIDE)
        }
        if (panel.bottomModes.size != config.bottomSlots) {
            issues += UserPanelIssue(UserPanelIssueType.SIZE_MISMATCH, edge = PanelEdge.BOTTOM)
        }

        if (hasUnsupported(panel.sideActions.mapNotNull { it.key }, config.supportedActions)) {
            issues += UserPanelIssue(UserPanelIssueType.UNSUPPORTED_KEY, edge = PanelEdge.SIDE)
        }
        if (hasUnsupported(panel.bottomModes.mapNotNull { it.key }, config.supportedModes)) {
            issues += UserPanelIssue(UserPanelIssueType.UNSUPPORTED_KEY, edge = PanelEdge.BOTTOM)
        }

        if (hasDuplicates(panel.sideActions.mapNotNull { it.key })) {
            issues += UserPanelIssue(UserPanelIssueType.DUPLICATE, edge = PanelEdge.SIDE)
        }
        if (hasDuplicates(panel.bottomModes.mapNotNull { it.key })) {
            issues += UserPanelIssue(UserPanelIssueType.DUPLICATE, edge = PanelEdge.BOTTOM)
        }

        val minSide = minSlots(config.sideSlots, MIN_SIDE)
        val minBottom = minSlots(config.bottomSlots, MIN_BOTTOM)
        if (panel.sideActions.count { it.key != null } < minSide) {
            issues += UserPanelIssue(UserPanelIssueType.BELOW_MINIMUM, edge = PanelEdge.SIDE)
        }
        if (panel.bottomModes.count { it.key != null } < minBottom) {
            issues += UserPanelIssue(UserPanelIssueType.BELOW_MINIMUM, edge = PanelEdge.BOTTOM)
        }
        return issues
    }

    private fun <T> hasDuplicates(items: List<T>): Boolean {
        val seen = HashSet<T>()
        return items.any { !seen.add(it) }
    }

    private fun <T> hasUnsupported(items: List<T>, supported: List<T>): Boolean {
        if (supported.isEmpty()) return false
        val allowed = supported.toSet()
        return items.any { !allowed.contains(it) }
    }

    private fun minSlots(slots: Int, minRequired: Int): Int {
        if (slots <= 0) return 0
        return if (slots < minRequired) slots else minRequired
    }
}
