package com.example.shoppingassistant.domain.model

/**
 * Доменная модель авторизованного пользователя.
 * Чистый слой: без зависимостей от Android/UI.
 */
data class AuthUser(
    val id: Long,
    val email: String,
    val displayName: String?,
    val phone: String?,
    val avatarUrl: String?,
    val city: String?,
    val emailVerified: Boolean = false,
    val createdAt: Long? = null,
    val photos: List<String> = emptyList(),
)

/**
 * Учётные данные для логина/регистрации.
 */
data class AuthCredentials(
    val email: String,
    val password: String,
)

/**
 * Типы ошибок авторизации (минимальный набор для MVP).
 */
enum class AuthError {
    INVALID_CREDENTIALS,
    EMAIL_ALREADY_EXISTS,
    UNKNOWN,
}

/**
 * Результат операций логина/регистрации.
 */
sealed class AuthResult {

    data class Success(
        val user: AuthUser,
    ) : AuthResult()

    data class Error(
        val error: AuthError,
    ) : AuthResult()
}
