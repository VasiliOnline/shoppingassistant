package com.example.shoppingassistant.domain.model

/**
 * Доменная модель core. Никаких зависимостей от UI/feature.
 */
data class ProductDto(
    val id: String,
    val title: String,
    val brand: String?,
    val model: String?,
    val price: Double?,       // в валютах (а не cents)
    val deliveryTime: Int?,   // дни
    val sellerRating: Double?, // 0..5
    val sellerRatingCount: Int? = null,
    val sellerCountry: String? = null,
    val sellerCity: String? = null,
    val sellerBadges: List<String> = emptyList(),
    val sellerShippingCountries: List<String> = emptyList(),
    val sellerName: String? = null,
    val sellerAvatarUrl: String? = null,
    val sellerType: SellerType? = null,
    val source: OfferSource? = null,
    val sourceName: String? = null,
    val externalUrl: String? = null,
    val imageUrls: List<String> = emptyList(),
    val updatedAt: Long? = null,
    val trustScore: Double? = null,
    val distanceKm: Float? = null,
)

enum class OfferSource {
    EXPRESS,
    EXTERNAL,
}

enum class SellerType {
    PRIVATE,
    COMPANY,
}
