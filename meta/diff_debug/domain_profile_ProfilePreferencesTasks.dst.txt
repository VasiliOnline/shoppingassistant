package com.example.shoppingassistant.domain.profile

interface ProfileSettingsRepository {
    suspend fun getSettings(userId: String? = null): ProfileSettings
    suspend fun saveSettings(userId: String? = null, settings: ProfileSettings)
}

interface ExternalLinksRepository {
    suspend fun getLinks(userId: String): List<ExternalLink>
    suspend fun saveLinks(userId: String, links: List<ExternalLink>)
    suspend fun clearLinks(userId: String)
}

interface ProfileCacheRepository {
    suspend fun getProfile(userId: String): ProfileSnapshot?
    suspend fun saveProfile(profile: ProfileSnapshot)
    suspend fun clearProfile(userId: String)
}

class GetProfileSettingsTask(
    private val repository: ProfileSettingsRepository,
) {
    suspend operator fun invoke(userId: String? = null): ProfileSettings =
        repository.getSettings(userId)
}

class SaveProfileSettingsTask(
    private val repository: ProfileSettingsRepository,
) {
    suspend operator fun invoke(userId: String? = null, settings: ProfileSettings) {
        repository.saveSettings(userId, settings)
    }
}

class GetExternalLinksTask(
    private val repository: ExternalLinksRepository,
) {
    suspend operator fun invoke(userId: String): List<ExternalLink> =
        repository.getLinks(userId)
}

class SaveExternalLinksTask(
    private val repository: ExternalLinksRepository,
) {
    suspend operator fun invoke(userId: String, links: List<ExternalLink>) {
        repository.saveLinks(userId, links)
    }
}

class ClearExternalLinksTask(
    private val repository: ExternalLinksRepository,
) {
    suspend operator fun invoke(userId: String) {
        repository.clearLinks(userId)
    }
}

class GetProfileCacheTask(
    private val repository: ProfileCacheRepository,
) {
    suspend operator fun invoke(userId: String): ProfileSnapshot? =
        repository.getProfile(userId)
}

class SaveProfileCacheTask(
    private val repository: ProfileCacheRepository,
) {
    suspend operator fun invoke(profile: ProfileSnapshot) {
        repository.saveProfile(profile)
    }
}

class ClearProfileCacheTask(
    private val repository: ProfileCacheRepository,
) {
    suspend operator fun invoke(userId: String) {
        repository.clearProfile(userId)
    }
}
