package com.example.shoppingassistant.domain.template

import kotlinx.serialization.Serializable

@Serializable
data class TemplateListMeta(
    val listId: String,
    val title: String,
    val createdAtMillis: Long,
)

@Serializable
data class TemplateListEntry(
    val listId: String,
    val snapshot: TemplateSnapshot,
    val createdAtMillis: Long,
)

interface TemplateListsRepository {
    suspend fun listLists(): List<TemplateListMeta>
    suspend fun createList(title: String): TemplateListMeta
    suspend fun renameList(listId: String, title: String): Boolean
    suspend fun deleteList(listId: String): Boolean
    suspend fun listEntries(listId: String): List<TemplateListEntry>
    suspend fun upsertEntry(listId: String, snapshot: TemplateSnapshot, createdAtMillis: Long = System.currentTimeMillis())
    suspend fun deleteEntry(listId: String, templateId: String)
    suspend fun getSelectedListId(): String?
    suspend fun setSelectedListId(listId: String?)
}
