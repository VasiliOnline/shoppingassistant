package com.example.shoppingassistant.feature.pages.draft.create

import com.example.shoppingassistant.domain.catalog.Category
import com.example.shoppingassistant.domain.catalog.AttributeDef
import com.example.shoppingassistant.domain.ugc.draft.DraftOffer

enum class DraftStep {
    ENTRY,
    MEDIA,
    TITLE_CATEGORY,
    PRICE,
    ATTRIBUTES,
    LOCATION_DELIVERY,
    CONTACTS,
    REVIEW_PUBLISH,
}

enum class DraftSubSheet {
    CATEGORY,
    BRAND,
    CONDITION,
    LOCATION,
    ATTR_VALUE,
    SUMMARY,
}

data class DraftCategoryState(
    val categories: List<Category> = emptyList(),
    val searchQuery: String = "",
    val popular: List<Category> = emptyList(),
)

data class DraftAttributesState(
    val defs: List<AttributeDef> = emptyList(),
    val required: Set<String> = emptySet(),
    val activeAttr: AttributeDef? = null,
    val activeValues: List<String> = emptyList(),
)

data class CreateDraftState(
    val draft: DraftOffer? = null,
    val savedDraft: DraftOffer? = null,
    val duplicateDraft: DraftOffer? = null,
    val step: DraftStep = DraftStep.ENTRY,
    val subSheet: DraftSubSheet? = null,
    val categoryState: DraftCategoryState = DraftCategoryState(),
    val attributesState: DraftAttributesState = DraftAttributesState(),
    val isLoading: Boolean = false,
    val isSaving: Boolean = false,
    val isDirty: Boolean = false,
    val errorMessage: String? = null,
)
