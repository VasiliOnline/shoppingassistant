@Composable
fun PhotoPostFlow(
    visible: Boolean,
    draftId: String? = null,
    origin: DraftInputOrigin? = null,
    onDismiss: () -> Unit,
    onOpenManual: () -> Unit = {},
    modifier: Modifier = Modifier,
    viewModel: CreateDraftViewModel = koinViewModel(),
) {
    if (!visible) return
    val state by viewModel.state.collectAsState()
    val draft = state.draft
    val context = LocalContext.current
    val scope = rememberCoroutineScope()
    val cameraAvailable = remember(context) {
        context.packageManager.hasSystemFeature(PackageManager.FEATURE_CAMERA_ANY)
    }
    val modeStore: DraftCreateModeStore = remember { koinGet(DraftCreateModeStore::class.java) }
    val snackbarHostState = remember { SnackbarHostState() }

    var screen by rememberSaveable { mutableStateOf(PostFlowScreen.ENTRY) }
    var step by rememberSaveable { mutableStateOf(PhotoStep.FRONT) }
    var showExitDialog by rememberSaveable { mutableStateOf(false) }
    var cameraRequested by rememberSaveable { mutableStateOf(false) }
    var permissionState by rememberSaveable { mutableStateOf(CameraPermissionState.Unknown) }
    var pendingSlot by remember { mutableStateOf<PhotoSlotRef?>(null) }
    var coreReadyLogged by rememberSaveable { mutableStateOf(false) }
    var startNewDraft by rememberSaveable { mutableStateOf(false) }

    val importer = remember { DraftMediaImporter(context) }

    val galleryLauncher = rememberLauncherForActivityResult(ActivityResultContracts.GetContent()) { uri ->
        val slot = pendingSlot ?: return@rememberLauncherForActivityResult
        if (uri == null) return@rememberLauncherForActivityResult
        scope.launch {
            val media = importer.importPhoto(uri) ?: return@launch
            viewModel.upsertPhoto(slot.role, slot.index, media)
            FlowMetrics.markPhotoAdded(slot.role.name.lowercase(), "gallery")
        }
    }

    val cameraLauncher = rememberLauncherForActivityResult(ActivityResultContracts.TakePicturePreview()) { bitmap ->
        val slot = pendingSlot ?: return@rememberLauncherForActivityResult
        if (bitmap == null) return@rememberLauncherForActivityResult
        scope.launch {
            val media = importer.importBitmap(bitmap) ?: return@launch
            viewModel.upsertPhoto(slot.role, slot.index, media)
            FlowMetrics.markPhotoAdded(slot.role.name.lowercase(), "camera")
        }
    }

    val permissionLauncher = rememberLauncherForActivityResult(ActivityResultContracts.RequestPermission()) { granted ->
        cameraRequested = true
        permissionState = resolveCameraPermission(context, cameraRequested)
        if (granted) {
            cameraLauncher.launch(null)
        }
    }

    fun closeFlow() {
        if (draft != null) {
            viewModel.closeDraft()
        }
        onDismiss()
    }

    fun openDraftForPhoto() {
        viewModel.openDraft(draftId, DraftInputOrigin.PHOTO)
    }

    fun openAppSettings() {
        val intent = Intent(Settings.ACTION_APPLICATION_DETAILS_SETTINGS).apply {
            data = Uri.fromParts("package", context.packageName, null)
            addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
        }
        context.startActivity(intent)
    }

    fun requestExit() {
        if (draft == null) {
            onDismiss()
            return
        }
        if (state.isDirty) {
            showExitDialog = true
        } else {
            closeFlow()
        }
    }

    fun updateCategoryWithUndo(newCode: String?) {
        val previous = state.draft?.categoryCode
        if (previous == newCode) return
        viewModel.updateCategory(newCode)
        scope.launch {
            val result = snackbarHostState.showSnackbar(
                message = "Категория изменена",
                actionLabel = "Отменить",
            )
            if (result == androidx.compose.material3.SnackbarResult.ActionPerformed) {
                viewModel.updateCategory(previous)
            }
        }
    }

    fun updatePriceTypeWithUndo(type: DraftPriceType) {
        val previous = state.draft?.price
        if (previous?.type == type) return
        viewModel.updatePriceType(type)
        scope.launch {
            val result = snackbarHostState.showSnackbar(
                message = "Цена обновлена",
                actionLabel = "Отменить",
            )
            if (result == androidx.compose.material3.SnackbarResult.ActionPerformed && previous != null) {
                viewModel.updatePriceType(previous.type)
                if (previous.type == DraftPriceType.FIXED) {
                    viewModel.updatePriceInput(previous.rawInput.orEmpty())
                }
            }
        }
    }

    LaunchedEffect(visible, draftId, origin) {
        if (!visible) return@LaunchedEffect
        FlowMetrics.markPostPhotoFlowOpen()
        if (origin != null && origin != DraftInputOrigin.PHOTO) {
            onOpenManual()
            return@LaunchedEffect
        }
        if (draftId != null) {
            viewModel.openDraft(draftId, origin ?: DraftInputOrigin.PHOTO)
            screen = PostFlowScreen.EDITOR
            return@LaunchedEffect
        }
        if (origin == DraftInputOrigin.PHOTO) {
            modeStore.set(DraftCreateMode.PHOTO)
            openDraftForPhoto()
            screen = PostFlowScreen.PHOTO_CAPTURE
            return@LaunchedEffect
        }
        val last = modeStore.get()
        if (last == DraftCreateMode.PHOTO) {
            openDraftForPhoto()
            screen = PostFlowScreen.PHOTO_CAPTURE
        } else {
            screen = PostFlowScreen.ENTRY
        }
    }

    LaunchedEffect(screen) {
        if (screen == PostFlowScreen.PHOTO_CAPTURE) {
            permissionState = resolveCameraPermission(context, cameraRequested)
        }
    }

    LaunchedEffect(startNewDraft) {
        if (!startNewDraft) return@LaunchedEffect
        viewModel.closeDraft()
        delay(200)
        viewModel.openDraft(null, DraftInputOrigin.PHOTO)
        step = PhotoStep.FRONT
        screen = PostFlowScreen.PHOTO_CAPTURE
        startNewDraft = false
    }

    val coreReady = draft?.let { isCoreReady(it) } == true
    LaunchedEffect(draft?.id) {
        coreReadyLogged = false
    }
    LaunchedEffect(coreReady, draft?.id) {
        if (coreReady && !coreReadyLogged) {
            coreReadyLogged = true
            FlowMetrics.markCoreFieldsCompleted()
        }
    }

    BackHandler(enabled = true) {
        when (screen) {
            PostFlowScreen.ENTRY -> requestExit()
            PostFlowScreen.PHOTO_CAPTURE -> requestExit()
            PostFlowScreen.EDITOR -> screen = PostFlowScreen.PHOTO_CAPTURE
            PostFlowScreen.CATEGORY -> screen = PostFlowScreen.EDITOR
            PostFlowScreen.PREVIEW -> screen = PostFlowScreen.EDITOR
            PostFlowScreen.PUBLISHING -> Unit
            PostFlowScreen.RESULT -> requestExit()
        }
    }

    Scaffold(
        snackbarHost = { SnackbarHost(hostState = snackbarHostState) },
        modifier = modifier,
    ) { padding ->
        Surface(
            color = MaterialTheme.colorScheme.surface,
            modifier = Modifier
                .fillMaxSize()
                .padding(padding),
        ) {
            when (screen) {
                PostFlowScreen.ENTRY -> PostEntryScreen(
                    onPhoto = {
                        scope.launch { modeStore.set(DraftCreateMode.PHOTO) }
                        openDraftForPhoto()
                        screen = PostFlowScreen.PHOTO_CAPTURE
                    },
                    onManual = {
                        scope.launch { modeStore.set(DraftCreateMode.MANUAL) }
                        onOpenManual()
                    },
                    onClose = { requestExit() },
                )
                PostFlowScreen.PHOTO_CAPTURE -> PhotoCaptureScreen(
                    step = step,
                    draft = draft,
                    permissionState = permissionState,
                    cameraAvailable = cameraAvailable,
                    onClose = { requestExit() },
                    onStepChange = { step = it },
                    onRequestCamera = { role, index ->
                        pendingSlot = PhotoSlotRef(role, index)
                        cameraRequested = true
                        val permission = resolveCameraPermission(context, cameraRequested)
                        permissionState = permission
                        when (permission) {
                            CameraPermissionState.Granted -> cameraLauncher.launch(null)
                            CameraPermissionState.DeniedPermanent -> openAppSettings()
                            else -> permissionLauncher.launch(Manifest.permission.CAMERA)
                        }
                    },
                    onOpenSettings = { openAppSettings() },
                    onRequestGallery = { role, index ->
                        pendingSlot = PhotoSlotRef(role, index)
                        galleryLauncher.launch("image/*")
                    },
                    onDelete = { role, index, removed ->
                        val slot = PhotoSlotRef(role, index)
                        viewModel.removePhoto(role, index)
                        scope.launch {
                            val result = snackbarHostState.showSnackbar(
                                message = "Фото удалено",
                                actionLabel = "Отменить",
                            )
                            val undo = result == androidx.compose.material3.SnackbarResult.ActionPerformed
                            if (undo) {
                                viewModel.upsertPhoto(slot.role, slot.index, removed)
                            }
                            FlowMetrics.markPhotoDeleted(undo)
                        }
                    },
                    onContinue = { screen = PostFlowScreen.EDITOR },
                )
                PostFlowScreen.EDITOR -> DraftEditorScreen(
                    draft = draft,
                    state = state,
                    onBack = { screen = PostFlowScreen.PHOTO_CAPTURE },
                    onOpenCategory = { screen = PostFlowScreen.CATEGORY },
                    onOpenPreview = {
                        FlowMetrics.markPreviewOpen()
                        screen = PostFlowScreen.PREVIEW
                    },
                    onSaveDraft = {
                        FlowMetrics.markDraftSaved()
                        viewModel.saveDraftNow()
                    },
                    onUpdateTitle = viewModel::updateTitle,
                    onUpdateDescription = viewModel::updateDescription,
                    onUpdatePriceInput = viewModel::updatePriceInput,
                    onUpdatePriceType = { type -> updatePriceTypeWithUndo(type) },
                    onUpdateCategory = { code -> updateCategoryWithUndo(code) },
                    onUpdateCondition = viewModel::updateCondition,
                    onUpdateDelivery = viewModel::updateDelivery,
                    onUpdateContacts = viewModel::updateContacts,
                    onUpdateLocation = viewModel::updateLocation,
                )
                PostFlowScreen.CATEGORY -> CategoryPickerScreen(
                    state = state.categoryState,
                    selectedCode = draft?.categoryCode,
                    onQueryChange = viewModel::setCategorySearch,
                    onPick = {
                        updateCategoryWithUndo(it.code)
                        screen = PostFlowScreen.EDITOR
                    },
                    onBack = { screen = PostFlowScreen.EDITOR },
                )
                PostFlowScreen.PREVIEW -> ListingPreviewScreen(
                    draft = draft,
                    state = state,
                    onBack = { screen = PostFlowScreen.EDITOR },
                    onPublish = {
                        FlowMetrics.markPublishAttemptedDetailed()
                        screen = PostFlowScreen.PUBLISHING
                    },
                    onEdit = { screen = PostFlowScreen.EDITOR },
                    onUpdateContacts = viewModel::updateContacts,
                    onUpdateLocation = viewModel::updateLocation,
                )
                PostFlowScreen.PUBLISHING -> PublishingScreen(
                    draft = draft,
                    onSaveDraft = {
                        FlowMetrics.markDraftSaved()
                        viewModel.saveDraftNow()
                        requestExit()
                    },
                    onExit = { requestExit() },
                    onPublish = {
                        val current = state.draft
                        val validation = DraftPublishValidator.validate(current, state.attributesState.required)
                        if (!validation.isReady) {
                            FlowMetrics.markPublishFailedDetailed("missing_fields")
                            screen = PostFlowScreen.EDITOR
                            return@PublishingScreen
                        }
                        viewModel.publishDraft {
                            FlowMetrics.markPublishSucceededDetailed()
                            screen = PostFlowScreen.RESULT
                        }
                    },
                )
                PostFlowScreen.RESULT -> PublishResultScreen(
                    draft = draft,
                    onOpenListing = { closeFlow() },
                    onShare = { closeFlow() },
                    onCreateAnother = {
                        startNewDraft = true
                    },
                    onClose = { closeFlow() },
                )
            }
        }
    }

    if (showExitDialog) {
        ExitConfirmDialog(
            onSaveExit = {
                showExitDialog = false
                FlowMetrics.markDraftSaved()
                viewModel.saveDraftNow { closeFlow() }
            },
            onDiscard = {
                showExitDialog = false
                FlowMetrics.markDraftDiscarded()
                viewModel.discardDraft { closeFlow() }
            },
            onStay = { showExitDialog = false },
        )
    }
}

@Composable
private fun ModeChoiceCard(
    title: String,
    subtitle: String,
    icon: androidx.compose.ui.graphics.vector.ImageVector,
    primary: Boolean,
    onClick: () -> Unit,
) {
    val shape = RoundedCornerShape(CardRadius)
    Surface(
        tonalElevation = if (primary) 2.dp else 0.dp,
        shape = shape,
        color = if (primary) MaterialTheme.colorScheme.primaryContainer else MaterialTheme.colorScheme.surfaceVariant,
        modifier = Modifier
            .fillMaxWidth()
            .heightIn(min = 88.dp)
            .clip(shape)
            .clickable(onClick = onClick),
    ) {
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 16.dp, vertical = 14.dp),
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.spacedBy(16.dp),
        ) {
            Box(
                modifier = Modifier
                    .size(44.dp)
                    .clip(CircleShape)
                    .background(
                        if (primary) MaterialTheme.colorScheme.primary else MaterialTheme.colorScheme.surface,
                    ),
                contentAlignment = Alignment.Center,
            ) {
                Icon(
                    icon,
                    contentDescription = null,
                    tint = if (primary) MaterialTheme.colorScheme.onPrimary else MaterialTheme.colorScheme.onSurface,
                )
            }
            Column(
                modifier = Modifier.weight(1f),
                verticalArrangement = Arrangement.spacedBy(2.dp),
            ) {
                Text(
                    text = title,
                    style = MaterialTheme.typography.titleSmall,
                    color = if (primary) MaterialTheme.colorScheme.onPrimaryContainer else MaterialTheme.colorScheme.onSurface,
                )
                Text(
                    text = subtitle,
                    style = MaterialTheme.typography.bodySmall,
                    color = if (primary) MaterialTheme.colorScheme.onPrimaryContainer else MaterialTheme.colorScheme.onSurfaceVariant,
                )
            }
            Icon(
                Icons.Outlined.KeyboardArrowRight,
                contentDescription = null,
                tint = if (primary) MaterialTheme.colorScheme.onPrimaryContainer else MaterialTheme.colorScheme.onSurfaceVariant,
            )
        }
    }
}

@Composable
private fun PhotoCaptureScreen(
    step: PhotoStep,
    draft: DraftOffer?,
    permissionState: CameraPermissionState,
    cameraAvailable: Boolean,
    onClose: () -> Unit,
    onStepChange: (PhotoStep) -> Unit,
    onRequestCamera: (DraftPhotoRole, Int) -> Unit,
    onOpenSettings: () -> Unit,
    onRequestGallery: (DraftPhotoRole, Int) -> Unit,
    onDelete: (DraftPhotoRole, Int, DraftMedia) -> Unit,
    onContinue: () -> Unit,
) {
    val media = draft?.media.orEmpty().filter { it.type == DraftMediaType.PHOTO }
    val stepMedia = media.filter { it.role == step.role }
    val slots = remember(stepMedia, step) {
        List(MaxPhotosPerStep) { index -> stepMedia.getOrNull(index) }
    }
    val listState = rememberLazyListState()
    var currentIndex by rememberSaveable(step) { mutableIntStateOf(0) }
    var showHelp by rememberSaveable(step) { mutableStateOf(false) }
    val visibleIndex = remember(listState) { androidx.compose.runtime.derivedStateOf { listState.firstVisibleItemIndex } }

    LaunchedEffect(slots.size) {
        val max = (slots.size - 1).coerceAtLeast(0)
        if (currentIndex > max) currentIndex = max
    }
    LaunchedEffect(visibleIndex.value) {
        currentIndex = visibleIndex.value
    }

    val canProceed = stepMedia.isNotEmpty()
    val nextStep = step.next()

    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(horizontal = 16.dp, vertical = 12.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp),
    ) {
        PhotoCaptureTopBar(
            step = step,
            onHelp = { showHelp = !showHelp },
            onClose = onClose,
        )

        if (showHelp && step.hint.isNotBlank()) {
            InlineInfoCard(text = step.hint)
        }

        SlotDots(count = slots.size, currentIndex = currentIndex)

    LazyRow(
        state = listState,
        horizontalArrangement = Arrangement.spacedBy(12.dp),
        modifier = Modifier
            .fillMaxWidth(),
    ) {
        itemsIndexed(slots) { index, item ->
            PhotoSlotCard(
                media = item,
                index = index,
                modifier = Modifier.fillParentMaxWidth(),
                onClick = {
                    currentIndex = index
                    if (item == null) {
                        if (cameraAvailable) {
                            onRequestCamera(step.role, index)
                        } else {
                            onRequestGallery(step.role, index)
                        }
                    }
                },
            )
        }
    }

        PermissionHint(
            permissionState = permissionState,
            cameraAvailable = cameraAvailable,
            onRequest = { onRequestCamera(step.role, currentIndex) },
            onOpenSettings = onOpenSettings,
            onGallery = { onRequestGallery(step.role, currentIndex) },
        )

        PhotoActionRow(
            canDelete = slots.getOrNull(currentIndex) != null,
            cameraEnabled = cameraAvailable,
            onCamera = { onRequestCamera(step.role, currentIndex) },
            onGallery = { onRequestGallery(step.role, currentIndex) },
            onDelete = {
                val removed = slots.getOrNull(currentIndex) ?: return@PhotoActionRow
                onDelete(step.role, currentIndex, removed)
            },
        )

        Spacer(modifier = Modifier.weight(1f))

        PhotoCaptureBottomBar(
            canProceed = canProceed,
            hasNextStep = nextStep != null,
            onNext = {
                if (nextStep != null) {
                    onStepChange(nextStep)
                } else {
                    onContinue()
                }
            },
            onSkip = {
                if (nextStep != null) {
                    onStepChange(nextStep)
                } else {
                    onContinue()
                }
            },
        )
    }
}

@Composable
private fun PhotoCaptureTopBar(
    step: PhotoStep,
    onHelp: () -> Unit,
    onClose: () -> Unit,
) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .heightIn(min = 56.dp),
        verticalAlignment = Alignment.CenterVertically,
    ) {
        IconButton(onClick = onClose) {
            Icon(Icons.Outlined.Close, contentDescription = "Закрыть")
        }
        Text(
            text = "Шаг ${step.index + 1}/3 — ${step.title}",
            style = MaterialTheme.typography.titleMedium,
            modifier = Modifier.weight(1f),
        )
        IconButton(onClick = onHelp) {

@Composable
private fun DraftEditorScreen(
    draft: DraftOffer?,
    state: com.example.shoppingassistant.feature.pages.draft.create.CreateDraftState,
    onBack: () -> Unit,
    onOpenCategory: () -> Unit,
    onOpenPreview: () -> Unit,
    onSaveDraft: () -> Unit,
    onUpdateTitle: (String) -> Unit,
    onUpdateDescription: (String) -> Unit,
    onUpdatePriceInput: (String) -> Unit,
    onUpdatePriceType: (DraftPriceType) -> Unit,
    onUpdateCategory: (String?) -> Unit,
    onUpdateCondition: (DraftCondition?) -> Unit,
    onUpdateDelivery: (DraftDelivery) -> Unit,
    onUpdateContacts: (DraftContacts) -> Unit,
    onUpdateLocation: (DraftLocation) -> Unit,
) {
    val scroll = rememberScrollState()
    val coreReady = draft?.let { isCoreReady(it) } == true
    var previewExpanded by rememberSaveable { mutableStateOf(false) }
    var showOptionalHints by rememberSaveable { mutableStateOf(true) }
    var showMoreOptional by rememberSaveable { mutableStateOf(false) }
    val expandedSections = remember { mutableStateListOf<DraftSection>() }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .imePadding(),
    ) {
        Column(
            modifier = Modifier
                .fillMaxSize()
                .verticalScroll(scroll)
                .padding(horizontal = 16.dp, vertical = 12.dp)
                .padding(bottom = 120.dp),
            verticalArrangement = Arrangement.spacedBy(16.dp),
        ) {
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .heightIn(min = 56.dp),
                verticalAlignment = Alignment.CenterVertically,
                horizontalArrangement = Arrangement.SpaceBetween,
            ) {
                Row(verticalAlignment = Alignment.CenterVertically) {
                    IconButton(onClick = onBack) {
                        Icon(Icons.AutoMirrored.Outlined.ArrowBack, contentDescription = "Назад")
                    }
                    Text(
                        text = "Черновик",
                        style = MaterialTheme.typography.titleMedium,
                    )
                }
                Text(
                    text = if (state.isSaving) "Сохраняем..." else "Сохранено",
                    style = MaterialTheme.typography.labelMedium,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                )
            }

            CompletionChecklist(draft = draft, coreReady = coreReady)

            if (coreReady) {
                Text(
                    text = "Готово к публикации",
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.primary,
                )
            }

            if (showOptionalHints && coreReady) {
                OptionalImprovementsCard(
                    showMore = showMoreOptional,
                    onShowMore = { showMoreOptional = true },
                    onHide = { showOptionalHints = false },
                    onOpenSection = { section ->
                        if (!expandedSections.contains(section)) expandedSections.add(section)
                    },
                )
            }

            LivePreviewCard(
                draft = draft,
                expanded = previewExpanded,
                coreReady = coreReady,
                onToggle = { previewExpanded = !previewExpanded },
                onOpenPreview = if (coreReady) onOpenPreview else null,
            )

            DraftCoreFields(
                draft = draft,
                categoryState = state.categoryState,
                onUpdateTitle = onUpdateTitle,
                onUpdatePriceInput = onUpdatePriceInput,
                onUpdatePriceType = onUpdatePriceType,
                onUpdateCategory = onUpdateCategory,
                onOpenCategory = onOpenCategory,
            )

            DraftSectionCard(
                title = "Описание",
                status = if (draft?.description.isNullOrBlank()) "не указано" else "добавлено",
                expanded = expandedSections.contains(DraftSection.DESCRIPTION),
                onToggle = { toggleSection(expandedSections, DraftSection.DESCRIPTION) },
            ) {
                OutlinedTextField(
                    value = draft?.description.orEmpty(),
                    onValueChange = onUpdateDescription,
                    label = { Text("Описание") },
                    modifier = Modifier.fillMaxWidth(),
                    minLines = 3,
                )
            }

            DraftSectionCard(
                title = "Состояние",
                status = draft?.condition?.let { conditionLabel(it) } ?: "не указано",
                expanded = expandedSections.contains(DraftSection.CONDITION),
                onToggle = { toggleSection(expandedSections, DraftSection.CONDITION) },
            ) {
                ConditionChips(
                    selected = draft?.condition,
                    onSelect = onUpdateCondition,
                )
            }

            DraftSectionCard(
                title = "Доставка/Самовывоз",
                status = deliveryStatus(draft?.delivery),
                expanded = expandedSections.contains(DraftSection.DELIVERY),
                onToggle = { toggleSection(expandedSections, DraftSection.DELIVERY) },
            ) {
                val delivery = draft?.delivery ?: DraftDelivery()
                ToggleRow(
                    label = "Самовывоз",
                    checked = delivery.pickup,
                    onCheckedChange = { onUpdateDelivery(delivery.copy(pickup = it)) },
                )
                ToggleRow(
                    label = "Доставка",
                    checked = delivery.delivery,
                    onCheckedChange = { onUpdateDelivery(delivery.copy(delivery = it)) },
                )
            }

            DraftSectionCard(
                title = "Контакты",
                status = contactsStatus(draft?.contacts),
                expanded = expandedSections.contains(DraftSection.CONTACTS),
                onToggle = { toggleSection(expandedSections, DraftSection.CONTACTS) },
            ) {
                val contacts = draft?.contacts ?: DraftContacts()
                ToggleRow(
                    label = "Показывать телефон",
                    checked = contacts.phoneEnabled,
                    onCheckedChange = { onUpdateContacts(contacts.copy(phoneEnabled = it)) },
                )
                Text(
                    text = "Чат доступен всегда.",
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                )
            }

            DraftSectionCard(
                title = "Гео",
                status = geoStatus(draft?.location),
                expanded = expandedSections.contains(DraftSection.GEO),
                onToggle = { toggleSection(expandedSections, DraftSection.GEO) },
            ) {
            val location = draft?.location ?: DraftLocation()
            val areaValue = location.area.orEmpty()
            val showArea = !location.publicLabel.isNullOrBlank()
            ToggleRow(
                label = "Показывать район",
                checked = showArea,
                onCheckedChange = { checked ->
                    val resolved = areaValue.ifBlank { location.city.orEmpty() }
                    val label = if (checked) resolved.takeIf { it.isNotBlank() } else null
                    onUpdateLocation(location.copy(publicLabel = label))
                },
            )
            OutlinedTextField(
                value = areaValue,
                onValueChange = { value ->
                    val resolved = value.ifBlank { location.city.orEmpty() }
                    val label = if (showArea) resolved.takeIf { it.isNotBlank() } else location.publicLabel
                    onUpdateLocation(location.copy(area = value, publicLabel = label))
                },
                    label = { Text("Район или город") },
                    modifier = Modifier.fillMaxWidth(),
                )
            }

            DraftSectionCard(
                title = "Импорт по ссылке",
                status = "не указано",
                expanded = expandedSections.contains(DraftSection.IMPORT),
                onToggle = { toggleSection(expandedSections, DraftSection.IMPORT) },
            ) {
                Text(
                    text = "Можно добавить данные по ссылке позже.",
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                )
                OutlinedButton(onClick = {}, enabled = false) {
                    Text("Добавить ссылку")
                }
            }
        }

        Surface(
            tonalElevation = 3.dp,
            modifier = Modifier
                .align(Alignment.BottomCenter)
                .fillMaxWidth(),
        ) {
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = 16.dp, vertical = 12.dp),
                verticalArrangement = Arrangement.spacedBy(8.dp),
            ) {
                HorizontalDivider()
                StickyActionBar(
                    coreReady = coreReady,
                    missingCore = coreMissingLabels(draft),
                    onPreview = onOpenPreview,
                    onSave = onSaveDraft,
                )
            }
        }
    }
}

@Composable
private fun CompletionChecklist(
    draft: DraftOffer?,
    coreReady: Boolean,
) {
    val items = listOf(
        "Фото" to (draft?.media?.isNotEmpty() == true),
        "Заголовок" to (!draft?.title.isNullOrBlank()),
        "Цена" to (draft?.let { DraftPublishValidator.isPriceReady(it) } == true),
        "Категория" to (!draft?.categoryCode.isNullOrBlank()),
    )
    val tint by animateColorAsState(
        targetValue = if (coreReady) MaterialTheme.colorScheme.primary.copy(alpha = 0.08f) else Color.Transparent,
        label = "readyTint",
    )
    Surface(
        shape = RoundedCornerShape(CardRadius),
        color = tint,
        tonalElevation = if (coreReady) 2.dp else 0.dp,
    ) {
        Column(
            modifier = Modifier.padding(12.dp),
            verticalArrangement = Arrangement.spacedBy(8.dp),
        ) {
            Text(
                text = "Чтобы опубликовать, нужно:",
                style = MaterialTheme.typography.bodySmall,
                color = MaterialTheme.colorScheme.onSurfaceVariant,
            )
            Row(
                horizontalArrangement = Arrangement.spacedBy(8.dp),
                verticalAlignment = Alignment.CenterVertically,
            ) {
                items.forEachIndexed { index, (label, ok) ->
                    Row(
                        verticalAlignment = Alignment.CenterVertically,
                        horizontalArrangement = Arrangement.spacedBy(4.dp),
                    ) {
                        Icon(
                            imageVector = if (ok) Icons.Outlined.CheckCircle else Icons.Outlined.Check,
                            contentDescription = null,
                            tint = if (ok) MaterialTheme.colorScheme.primary else MaterialTheme.colorScheme.onSurfaceVariant,
                            modifier = Modifier.size(16.dp),
                        )
                        Text(
                            text = label,
                            style = MaterialTheme.typography.labelSmall,
                            color = if (ok) MaterialTheme.colorScheme.onSurface else MaterialTheme.colorScheme.onSurfaceVariant,
                        )
                    }
                    if (index != items.lastIndex) {
                        Text(
                            text = " / ",
                            style = MaterialTheme.typography.labelSmall,
                            color = MaterialTheme.colorScheme.onSurfaceVariant,
                        )
                    }
                }
            }
        }
    }
}

@Composable
private fun OptionalImprovementsCard(
    showMore: Boolean,
    onShowMore: () -> Unit,
    onHide: () -> Unit,
    onOpenSection: (DraftSection) -> Unit,
) {
    val suggestions = listOf(
        OptionalSuggestion("description", "Добавьте описание", DraftSection.DESCRIPTION),
        OptionalSuggestion("condition", "Укажите состояние", DraftSection.CONDITION),
        OptionalSuggestion("delivery", "Добавьте доставку", DraftSection.DELIVERY),
    )
    val visible = if (showMore) suggestions else suggestions.take(2)
    Surface(
        shape = RoundedCornerShape(CardRadius),
        color = MaterialTheme.colorScheme.surfaceVariant,
    ) {
        Column(
            modifier = Modifier.padding(12.dp),
            verticalArrangement = Arrangement.spacedBy(8.dp),
        ) {
            Text(
                text = "Можно улучшить (не обязательно)",
                style = MaterialTheme.typography.bodySmall,
                color = MaterialTheme.colorScheme.onSurfaceVariant,
            )
            visible.forEach { suggestion ->
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically,
                ) {
                    Text(
                        text = suggestion.label,
                        style = MaterialTheme.typography.bodyMedium,
                    )
                    TextButton(onClick = { onOpenSection(suggestion.section) }) {
                        Text("Добавить")
                    }
                }
            }
            if (!showMore) {
                TextButton(onClick = onShowMore) {
                    Text("Еще 1 идея")
                }
            }
            TextButton(onClick = onHide) {
                Text("Скрыть подсказки")
            }
        }
    }
}

                            text = "Превью карточки",

                            .fillMaxWidth()
                            .clickable(enabled = onOpenPreview != null) { onOpenPreview?.invoke() },

                }
            }
        }
    }

@Composable
private fun DraftCoreFields(
    draft: DraftOffer?,
    categoryState: DraftCategoryState,
    onUpdateTitle: (String) -> Unit,
    onUpdatePriceInput: (String) -> Unit,
    onUpdatePriceType: (DraftPriceType) -> Unit,
    onUpdateCategory: (String?) -> Unit,
    onOpenCategory: () -> Unit,
) {
    Column(verticalArrangement = Arrangement.spacedBy(12.dp)) {
        OutlinedTextField(
            value = draft?.title.orEmpty(),
            onValueChange = onUpdateTitle,
            label = { Text("Заголовок") },
            placeholder = { Text("Например: iPhone 13 128GB") },
            modifier = Modifier.fillMaxWidth(),
        )

        OutlinedTextField(
            value = draft?.price?.rawInput.orEmpty(),
            onValueChange = onUpdatePriceInput,
            label = { Text("Цена") },
            placeholder = { Text("0") },
            modifier = Modifier.fillMaxWidth(),
            keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
        )
        Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
            AssistChip(
                label = "Договорная",
                selected = draft?.price?.type == DraftPriceType.NEGOTIABLE,
                onClick = { onUpdatePriceType(DraftPriceType.NEGOTIABLE) },
            )
            AssistChip(
                label = "Бесплатно",
                selected = draft?.price?.type == DraftPriceType.FREE,
                onClick = { onUpdatePriceType(DraftPriceType.FREE) },
            )
        }

        Column(verticalArrangement = Arrangement.spacedBy(6.dp)) {
            OutlinedTextField(
                value = categoryLabel(draft?.categoryCode, categoryState.categories),
                onValueChange = {},
                label = { Text("Категория") },
                modifier = Modifier.fillMaxWidth(),
                readOnly = true,
                trailingIcon = {
                    IconButton(onClick = onOpenCategory) {
                        Icon(Icons.Outlined.KeyboardArrowRight, contentDescription = null)
                    }
                },
            )
            if (draft?.categoryCode.isNullOrBlank()) {
                Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                    categoryState.popular.take(3).forEach { category ->
                        SuggestChip(
                            label = category.title ?: category.code,
                            onClick = { onUpdateCategory(category.code) },
                        )
                    }
                }
            }
            TextButton(onClick = onOpenCategory) {
                Text("Выбрать из списка")
            }
        }
    }
}

@Composable
private fun CategoryPickerScreen(
    state: DraftCategoryState,
    selectedCode: String?,
    onQueryChange: (String) -> Unit,
    onPick: (Category) -> Unit,
    onBack: () -> Unit,
) {
    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(horizontal = 16.dp, vertical = 12.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp),
    ) {
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .heightIn(min = 56.dp),
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.SpaceBetween,
        ) {
            Row(verticalAlignment = Alignment.CenterVertically) {
                IconButton(onClick = onBack) {
                    Icon(Icons.AutoMirrored.Outlined.ArrowBack, contentDescription = "Назад")
                }
                Text("Категория", style = MaterialTheme.typography.titleMedium)
            }
            Spacer(modifier = Modifier.width(24.dp))
        }

        OutlinedTextField(
            value = state.searchQuery,
            onValueChange = onQueryChange,
            label = { Text("Поиск категории") },
            modifier = Modifier.fillMaxWidth(),
        )

        if (state.popular.isNotEmpty()) {
            Text(
                text = "Популярные",
                style = MaterialTheme.typography.bodySmall,
                color = MaterialTheme.colorScheme.onSurfaceVariant,
            )
            Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                state.popular.forEach { cat ->
                    AssistChip(
                        label = cat.title ?: cat.code,
                        selected = cat.code == selectedCode,
                        onClick = { onPick(cat) },
                    )
                }
            }
        }

        val filtered = filterCategories(state.categories, state.searchQuery)
        val tree = buildCategoryTree(filtered)
        LazyColumn(
            verticalArrangement = Arrangement.spacedBy(6.dp),
            modifier = Modifier.fillMaxSize(),
        ) {
            items(tree) { node ->
                CategoryNodeRow(node, selectedCode, onPick, 0)
            }
        }
    }
}

    node: CategoryNode,

    indent: Int,
) {
    Column(modifier = Modifier.fillMaxWidth()) {
        val selected = node.category.code == selectedCode
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .clickable { onPick(node.category) }
                .padding(start = (indent * 12).dp, top = 6.dp, bottom = 6.dp),
            verticalAlignment = Alignment.CenterVertically,
        ) {
            Text(
                text = node.category.title ?: node.category.code,
                style = MaterialTheme.typography.bodyMedium,
                color = if (selected) MaterialTheme.colorScheme.primary else MaterialTheme.colorScheme.onSurface,
                modifier = Modifier.weight(1f),
            )
            if (selected) {
                Icon(
                    Icons.Outlined.CheckCircle,
                    contentDescription = null,
                    tint = MaterialTheme.colorScheme.primary,
                )
            }
        }
        node.children.forEach { child ->
            CategoryNodeRow(child, selectedCode, onPick, indent + 1)
        }
    }
}

@Composable

    val media = draft?.media.orEmpty().filter { it.type == DraftMediaType.PHOTO }
    if (media.isEmpty()) return
    val listState = rememberLazyListState()
    LazyRow(
        state = listState,
        horizontalArrangement = Arrangement.spacedBy(12.dp),
        modifier = Modifier
            .fillMaxWidth()
            .heightIn(min = 220.dp),
    ) {
        items(media) { item ->
            val model = item.localUri ?: item.remoteUrl
            AsyncImage(
                model = model,
                contentDescription = null,
                contentScale = ContentScale.Crop,
                modifier = Modifier
                    .widthIn(min = 280.dp)
                    .fillMaxHeight()
                    .clip(RoundedCornerShape(CardRadius)),
            )
        }
    }
}

@Composable
private fun PublishingScreen(
    draft: DraftOffer?,
    onSaveDraft: () -> Unit,
    onExit: () -> Unit,
    onPublish: () -> Unit,
) {
    var step by rememberSaveable { mutableIntStateOf(0) }
    val errorMessage = draft?.publishError?.takeIf { it.isNotBlank() }

    LaunchedEffect(errorMessage) {
        if (!errorMessage.isNullOrBlank()) return@LaunchedEffect
        step = 0
        delay(500)
        step = 1
        onPublish()
    }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(horizontal = 16.dp, vertical = 12.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp),
    ) {
        Text(
            text = if (errorMessage == null) "Публикуем" else "Не удалось опубликовать",
            style = MaterialTheme.typography.titleMedium,
        )
        if (errorMessage == null) {
            PublishingStepItem(
                label = "Загружаем фото...",
                active = step >= 0,
            )
            PublishingStepItem(
                label = "Публикуем...",
                active = step >= 1,
            )
        } else {
            Text(
                text = errorMessage,
                style = MaterialTheme.typography.bodySmall,
                color = MaterialTheme.colorScheme.onSurfaceVariant,
            )
            Button(
                onClick = onPublish,
                modifier = Modifier
                    .fillMaxWidth()
                    .heightIn(min = 48.dp),
                shape = RoundedCornerShape(ControlRadius),
            ) {
                Text("Повторить")
            }
        }
        Spacer(modifier = Modifier.weight(1f))
        TextButton(onClick = onSaveDraft) { Text("Сохранить черновик") }
        TextButton(onClick = onExit) { Text("Выйти") }
    }
}

private fun isCoreReady(draft: DraftOffer): Boolean {
    val hasMedia = draft.media.any { it.type == DraftMediaType.PHOTO }
    val hasTitle = !draft.title.isNullOrBlank()
    val hasPrice = DraftPublishValidator.isPriceReady(draft)
    val hasCategory = !draft.categoryCode.isNullOrBlank()
    return hasMedia && hasTitle && hasPrice && hasCategory
}

private fun coreMissingLabels(draft: DraftOffer?): List<String> {
    if (draft == null) return emptyList()
    val missing = mutableListOf<String>()
    if (draft.media.none { it.type == DraftMediaType.PHOTO }) missing += "фото"
    if (draft.title.isNullOrBlank()) missing += "заголовок"
    if (!DraftPublishValidator.isPriceReady(draft)) missing += "цена"
    if (draft.categoryCode.isNullOrBlank()) missing += "категория"
    return missing
}

private fun conditionLabel(condition: DraftCondition): String = when (condition) {
    DraftCondition.NEW -> "Новое"
    DraftCondition.LIKE_NEW -> "Как новое"
    DraftCondition.GOOD -> "Хорошее"
    DraftCondition.USED -> "Б/У"
    DraftCondition.FOR_PARTS -> "На запчасти"
}

private fun deliveryStatus(delivery: DraftDelivery?): String {
    val safe = delivery ?: return "не указано"
    val parts = mutableListOf<String>()
    if (safe.pickup) parts += "самовывоз"
    if (safe.delivery) parts += "доставка"
    return if (parts.isEmpty()) "не указано" else parts.joinToString(", ")
}

private fun contactsStatus(contacts: DraftContacts?): String {
    val safe = contacts ?: return "не указано"
    return if (safe.phoneEnabled) "телефон" else "чат"
}

private fun geoStatus(location: DraftLocation?): String {
    val safe = location ?: return "не указано"
    return safe.publicLabel?.takeIf { it.isNotBlank() } ?: "не указано"
}

private fun toggleSection(list: MutableList<DraftSection>, section: DraftSection) {
    if (list.contains(section)) {
        list.remove(section)
    } else {
        list.add(section)
    }
}

private fun categoryLabel(code: String?, categories: List<Category>): String {
    if (code.isNullOrBlank()) return ""
    return categories.firstOrNull { it.code == code }?.title ?: code
}

private fun formatDraftPrice(draft: DraftOffer): String = when (draft.price.type) {
    DraftPriceType.FIXED -> draft.price.amountMajor?.let {
        val currency = draft.price.currency ?: "USD"
        "%.0f %s".format(it, currency)
    } ?: "—"
    DraftPriceType.NEGOTIABLE -> "Договорная"
    DraftPriceType.FREE -> "Бесплатно"
    DraftPriceType.EXCHANGE -> "Обмен"
}

private fun filterCategories(categories: List<Category>, query: String): List<Category> {
    val trimmed = query.trim().lowercase()
    if (trimmed.isBlank()) return categories
    return categories.filter {
        it.code.lowercase().contains(trimmed) ||
            (it.title?.lowercase()?.contains(trimmed) == true)
    }
}

private data class CategoryNode(
    val category: Category,
    val children: List<CategoryNode> = emptyList(),
)

private fun buildCategoryTree(categories: List<Category>): List<CategoryNode> {
    val grouped = categories.groupBy { it.parentCode }
    fun build(parent: String?): List<CategoryNode> =
        grouped[parent].orEmpty().sortedBy { it.title ?: it.code }.map { cat ->
            CategoryNode(cat, build(cat.code))
        }
    return build(null)
}

private fun resolveCameraPermission(context: Context, requested: Boolean): CameraPermissionState {
    val granted = ContextCompat.checkSelfPermission(
        context,
        Manifest.permission.CAMERA,
    ) == android.content.pm.PackageManager.PERMISSION_GRANTED
    if (granted) return CameraPermissionState.Granted
    val activity = context.findActivity()
    val shouldExplain = activity?.let {
        ActivityCompat.shouldShowRequestPermissionRationale(it, Manifest.permission.CAMERA)
    } == true
    return if (!requested) {
        CameraPermissionState.Unknown
    } else if (shouldExplain) {
        CameraPermissionState.DeniedTemporary
    } else {
        CameraPermissionState.DeniedPermanent
    }
}

private fun Context.findActivity(): Activity? = when (this) {
    is Activity -> this
    is ContextWrapper -> baseContext.findActivity()
    else -> null
}
