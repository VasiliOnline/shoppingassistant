import com.example.shoppingassistant.feature.pages.useroffers.UserOffersTab

import java.util.Locale
import com.example.shoppingassistant.ui.theme.AppTheme as UiAppTheme

@Composable
fun AppRoot(modifier: Modifier = Modifier) {
    val nav = rememberNavController()
    val settingsViewModel: AppSettingsViewModel = koinViewModel()
    val settings by settingsViewModel.settings.collectAsState()
    var createDraftVisible by rememberSaveable { mutableStateOf(false) }
    var createDraftOrigin by rememberSaveable { mutableStateOf<DraftInputOrigin?>(null) }
    var createDraftId by rememberSaveable { mutableStateOf<String?>(null) }

    LaunchedEffect(settings.languageCode) {
        val raw = settings.languageCode.trim()
        val tags = if(raw.isBlank()) "" else raw.lowercase(Locale.ROOT)
        val locales = if (tags.isBlank()) {
            LocaleListCompat.getEmptyLocaleList()
        } else {
            LocaleListCompat.forLanguageTags(tags)
        }
        AppCompatDelegate.setApplicationLocales(locales)
    }

    CompositionLocalProvider(LocalProfileSettings provides settings) {
        UiAppTheme(themePreference = settings.theme) {
            NavHost(
                navController = nav,
                startDestination = AppRoutes.MainRoute,
                modifier = modifier,
            ) {
                composable(
                    route = AppRoutes.MainRoute,
                    arguments = listOf(
                        navArgument(AppRoutes.ArgOpenSearchHub) { defaultValue = "0" },
                    ),
                ) { entry ->
                    val openSearchHub = entry.arguments?.getString(AppRoutes.ArgOpenSearchHub) == "1"
                    MainPage(navController = nav, openSearchHubOnStart = openSearchHub)
                }
                composable(AppRoutes.Ingest) { IngestPage() }
                composable(AppRoutes.FeedCategories) { entry ->
                    val parentEntry = remember(entry) { nav.getBackStackEntry(AppRoutes.MainRoute) }
                    val mainViewModel: com.example.shoppingassistant.feature.pages.main.state.MainPageViewModel =
                        koinViewModel(viewModelStoreOwner = parentEntry)
                    FeedCategoriesPage(
                        viewModel = mainViewModel,
                        onBack = { nav.popBackStack() },
                        onClose = {
                            nav.navigate(AppRoutes.Main) {
                                launchSingleTop = true
                                popUpTo(AppRoutes.MainRoute) { inclusive = false }
                            }
                        },
                    )
                }
                composable(
                    route = "${AppRoutes.MyItemsRoute}?${AppRoutes.ArgMyItemsTab}={${AppRoutes.ArgMyItemsTab}}",
                    arguments = listOf(
                        navArgument(AppRoutes.ArgMyItemsTab) { defaultValue = "" },
                    ),
                ) { entry ->
                    val rawTab = entry.arguments?.getString(AppRoutes.ArgMyItemsTab).orEmpty()
                    val initialTab = runCatching { UserOffersTab.valueOf(rawTab) }
                        .getOrNull() ?: UserOffersTab.ACTIVE
                    UserOffersPage(
                        initialTab = initialTab,
                        callbacks = com.example.shoppingassistant.feature.pages.useroffers.UserOffersCallbacks(
                            onCreate = { method ->
                                createDraftOrigin = method.toDraftOrigin()
                                createDraftId = null
                                if (createDraftOrigin != null) {
                                    createDraftVisible = true
                                }
                            },
                            onOpen = { offerId ->
                                createDraftOrigin = null
                                createDraftId = offerId
                                createDraftVisible = true
                            },
                            onFindOffers = {
                                val payload = ResultsPayload(
                                    mode = ResultsMode.Categories,
                                    origin = ResultsOrigin.Category,
                                )
                                nav.navigate(ResultsRoutes.build(payload))
                            },
                            onBack = { nav.popBackStack() },
                        ),
                    )
                }
                composable(
                    route = "${ResultsRoutes.Route}?${ResultsRoutes.ArgPayload}={${ResultsRoutes.ArgPayload}}",
                    arguments = listOf(
                        navArgument(ResultsRoutes.ArgPayload) { defaultValue = "" },
                    ),
                ) { entry ->
                    val payload =
                        ResultsRoutes.parse(entry.arguments?.getString(ResultsRoutes.ArgPayload))
                    ResultsPage(
                        payload = payload,
                        navController = nav,
                        onOpenDraft = { draftPayload ->
                            createDraftOrigin = draftPayload.source.toDraftOrigin()
                            createDraftId = null
                            createDraftVisible = true
                        },
                        onEditQuery = {
                            nav.navigate(AppRoutes.mainWithSearchHub(expanded = true)) {
                                launchSingleTop = true
                                popUpTo(AppRoutes.MainRoute) { inclusive = false }
                            }
                        },
                    )
                }
                composable(
                    route = "${DraftRoutes.Route}?${DraftRoutes.ArgPayload}={${DraftRoutes.ArgPayload}}",
                    arguments = listOf(
                        navArgument(DraftRoutes.ArgPayload) { defaultValue = "" },
                    ),
                ) { entry ->
                    val payload =
                        DraftRoutes.parse(entry.arguments?.getString(DraftRoutes.ArgPayload))
                    if (payload == null) {
                        nav.popBackStack()
                    } else if (payload.mode == DraftMode.Create) {
                        LaunchedEffect(payload) {
                            createDraftOrigin = payload.source.toDraftOrigin()
                            createDraftId = null
                            createDraftVisible = true
                            nav.popBackStack()
                        }
                    } else {
                        DraftPage(payload = payload, navController = nav)
                    }
                }

                composable(AppRoutes.TrackedItems) {
                    TrackedItemsPage(
                        onBack = { nav.popBackStack() },
                        onOpenTrack = { trackId ->
                            if (trackId.isNotBlank()) {
                                nav.navigate(AppRoutes.trackedItemsTop10(trackId))
                            }
                        },
                        onEditTrack = { trackId ->
                            if (trackId.isNotBlank()) {
                                nav.navigate(AppRoutes.trackedItemsEdit(trackId))
                            }
                        },
                        onOpenEvents = { trackId ->
                            if (trackId.isNotBlank()) {
                                nav.navigate(AppRoutes.trackedItemsEvents(trackId))
                            }
                        },
                    )
                }
                composable(
                    route = AppRoutes.TrackedItemsTop10,
                    arguments = listOf(
                        navArgument(AppRoutes.ArgTrackId) { defaultValue = "" },
                    ),
                ) { entry ->
                    val trackId = entry.arguments?.getString(AppRoutes.ArgTrackId).orEmpty()
                    if (trackId.isBlank()) {
                        nav.popBackStack()
                    } else {
                        TrackTop10Page(
                            trackId = trackId,
                            onBack = { nav.popBackStack() },
                            onEditTrack = { nav.navigate(AppRoutes.trackedItemsEdit(trackId)) },
                        )
                    }
                }
                composable(
                    route = AppRoutes.TrackedItemsEvents,
                    arguments = listOf(
                        navArgument(AppRoutes.ArgTrackId) { defaultValue = "" },
                    ),
                ) { entry ->
                    val trackId = entry.arguments?.getString(AppRoutes.ArgTrackId).orEmpty()
                    if (trackId.isBlank()) {
                        nav.popBackStack()
                    } else {
                        TrackEventsPage(
                            trackId = trackId,
                            onBack = { nav.popBackStack() },
                        )
                    }
                }
                composable(
                    route = AppRoutes.TrackedItemsEdit,
                    arguments = listOf(
                        navArgument(AppRoutes.ArgTrackId) { defaultValue = "" },
                    ),
                ) { entry ->
                    val trackId = entry.arguments?.getString(AppRoutes.ArgTrackId).orEmpty()
                    TrackEditPage(
                        trackId = trackId,
                        onBack = { nav.popBackStack() },
                        onOpenTrack = { targetId ->
                            if (targetId.isNotBlank()) {
                                nav.navigate(AppRoutes.trackedItemsTop10(targetId))
                            }
                        },
                    )
                }
                composable(AppRoutes.Profile) {
                    ProfilePage(
                        onMyItemsClick = { navigateSafely(nav, AppRoutes.myItems()) },
                        onTrackedItemsClick = { navigateSafely(nav, AppRoutes.TrackedItems) },
                        onClose = { nav.popBackStack() },
                    )
                }

                composable(
                    route = AppRoutes.ChatRoute,
                    arguments = listOf(
                        navArgument(AppRoutes.ArgOfferId) { defaultValue = "" },
                        navArgument(AppRoutes.ArgSellerName) { defaultValue = "" },
                        navArgument(AppRoutes.ArgOfferTitle) { defaultValue = "" },
                        navArgument(AppRoutes.ArgPrice) { defaultValue = "" },
                        navArgument(AppRoutes.ArgStatus) { defaultValue = "" },
                        navArgument(AppRoutes.ArgExternalUrl) { defaultValue = "" },
                        navArgument(AppRoutes.ArgSourceName) { defaultValue = "" },
                        navArgument(AppRoutes.ArgQuerySessionId) { defaultValue = "" },
                        navArgument(AppRoutes.ArgPosition) { defaultValue = "" },
                    ),
                ) { entry ->
                    val offerId = entry.arguments?.getString(AppRoutes.ArgOfferId).orEmpty().ifBlank { null }
                    val sellerName =
                        entry.arguments?.getString(AppRoutes.ArgSellerName).orEmpty().ifBlank { "Продавец" }
                    val offerTitle =
                        entry.arguments?.getString(AppRoutes.ArgOfferTitle).orEmpty().ifBlank { "Предложение" }
                    val priceStr = entry.arguments?.getString(AppRoutes.ArgPrice).orEmpty()
                    val status =
                        entry.arguments?.getString(AppRoutes.ArgStatus).orEmpty().takeIf { it.isNotBlank() }
                    val externalUrl =
                        entry.arguments?.getString(AppRoutes.ArgExternalUrl).orEmpty().takeIf { it.isNotBlank() }
                    val sourceName =
                        entry.arguments?.getString(AppRoutes.ArgSourceName).orEmpty().takeIf { it.isNotBlank() }
                    val querySessionId =
                        entry.arguments?.getString(AppRoutes.ArgQuerySessionId).orEmpty().takeIf { it.isNotBlank() }
                    val position =
                        entry.arguments?.getString(AppRoutes.ArgPosition)?.toIntOrNull()

                    ChatPage(
                        props = ChatProps(
                            offerId = offerId,
                            sellerName = sellerName,
                            sellerStatus = status,
                            offerTitle = offerTitle,
                            offerPrice = priceStr.toDoubleOrNull(),
                            externalUrl = externalUrl,
                            sourceName = sourceName,
                            searchSessionId = querySessionId,
                            offerPosition = position,
                            onBack = { nav.popBackStack() },
                        ),
                    )
                }
            }



            val usePhotoPostFlow = createDraftId == null &&
                (createDraftOrigin == null || createDraftOrigin == DraftInputOrigin.PHOTO)

            if (createDraftVisible && usePhotoPostFlow) {
                PhotoPostFlow(
                    visible = true,
                    draftId = createDraftId,
                    origin = createDraftOrigin,
                    onDismiss = {
                        createDraftVisible = false
                        createDraftOrigin = null
                        createDraftId = null
                    },
                    onOpenManual = {
                        createDraftOrigin = DraftInputOrigin.TEXT
                        createDraftId = null
                    },
                )
            }

            if (createDraftVisible && !usePhotoPostFlow) {
                CreateDraftSheet(
                    visible = true,
                    draftId = createDraftId,
                    origin = createDraftOrigin,
                    onDismiss = {
                        createDraftVisible = false
                        createDraftOrigin = null
                        createDraftId = null
                    },
                )
            }
        }
    }
}


private fun UserOfferCreationMethod.toDraftOrigin(): DraftInputOrigin? = when (this) {
    UserOfferCreationMethod.PHOTO -> DraftInputOrigin.PHOTO
    UserOfferCreationMethod.LINK -> DraftInputOrigin.LINK
    UserOfferCreationMethod.VOICE -> DraftInputOrigin.VOICE
    UserOfferCreationMethod.JSON_FILE -> null
}

private fun DraftSource.toDraftOrigin(): DraftInputOrigin = when (this) {
    DraftSource.Photo -> DraftInputOrigin.PHOTO
    DraftSource.Link -> DraftInputOrigin.LINK
    DraftSource.Voice -> DraftInputOrigin.VOICE
    DraftSource.Text -> DraftInputOrigin.TEXT
}

private fun navigateSafely(nav: NavHostController, route: String) {
    val current = nav.currentDestination?.route
    if (current == route) return

    nav.navigate(route) {
        launchSingleTop = true
        restoreState = true
        popUpTo(nav.graph.startDestinationId) { saveState = true }
    }
}
