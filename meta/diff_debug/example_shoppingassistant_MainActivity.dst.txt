package com.example.shoppingassistant

import android.os.Bundle
import androidx.activity.ComponentActivity
import androidx.activity.compose.setContent

/**
 * Main entry point of the Android application.
 * Initializes the Compose content and draws the root of the app.
 */
import org.koin.android.ext.koin.androidContext
import org.koin.core.context.startKoin
import org.koin.core.context.GlobalContext
import com.example.shoppingassistant.core.di.coreModule
import com.example.shoppingassistant.core.di.domainModule
import com.example.shoppingassistant.core.di.rankModule
import com.example.shoppingassistant.feature.pages.di.mainPageModule
import com.example.shoppingassistant.feature.pages.di.subscriptionsModule

class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        // Initialize Koin dependency injection only once. If another Koin application
        // has already been started (e.g. during a previous activity lifecycle), skip
        // initialization to avoid KoinApplicationAlreadyStartedException.
        if (GlobalContext.getOrNull() == null) {
            startKoin {
                // Provide Android context for Koin so it can resolve resources and services.
                androidContext(this@MainActivity)
                // Register all DI modules needed for the app. The order doesn't matter.
                modules(
                    coreModule,
                    domainModule,
                    rankModule,
                    mainPageModule,
                    subscriptionsModule,
                )
            }
        }
        setContent {
            // Render the root composable which handles theme and navigation
            AppRoot()
        }
    }
}
