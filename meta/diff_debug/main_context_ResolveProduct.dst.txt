package com.example.shoppingassistant.feature.pages.main.context

import java.util.Locale
import com.example.shoppingassistant.feature.pages.model.Product

/**
 * Очень лёгкий парсер из текста подсказки → Product(brand, model).
 * Нужен, чтобы сразу после клика по подсказке иметь «конкретный товар»
 * для загрузки атрибутов из БД.
 */
fun resolveProduct(text: String?): Product? {
    if (text.isNullOrBlank()) return null
    val t = text.trim().lowercase(Locale.getDefault())

    // Мини-словарь — при желании расширишь/заменишь реальным поиском по БД.
    return when {
        t.startsWith("iphone 17 pro") -> Product(brand = "Apple",   model = "iPhone 17 Pro")
        t.startsWith("iphone 17")     -> Product(brand = "Apple",   model = "iPhone 17")
        t.startsWith("iphone 16 pro") -> Product(brand = "Apple",   model = "iPhone 16 Pro")
        t.startsWith("iphone 16")     -> Product(brand = "Apple",   model = "iPhone 16")
        t.startsWith("iphone 15 pro") -> Product(brand = "Apple",   model = "iPhone 15 Pro")
        t.startsWith("iphone 15")     -> Product(brand = "Apple",   model = "iPhone 15")

        t.startsWith("galaxy s24 ultra") -> Product(brand = "Samsung", model = "Galaxy S24 Ultra")
        t.startsWith("galaxy s24")       -> Product(brand = "Samsung", model = "Galaxy S24")

        t.startsWith("pixel 9 pro")   -> Product(brand = "Google",  model = "Pixel 9 Pro")
        t.startsWith("pixel 9")       -> Product(brand = "Google",  model = "Pixel 9")

        else -> null
    }
}
