@Deprecated("Заменено Template-флоу (InputRow → TemplateHintsBlock). Не использовать.")
@Composable
fun LinkDraftBlock(
    draft: LinkDraft,
    onCategoryChange: (String) -> Unit,
    onCreate: () -> Unit,
    onClear: () -> Unit,
    modifier: Modifier = Modifier,
) {
    val missingCategory = draft.category.isNullOrBlank()
    val domain = remember(draft.url) { resolveDomain(draft.url) }
    val categoryState = remember(draft.category) { mutableStateOf(draft.category.orEmpty()) }

    Column(
        modifier = modifier
            .fillMaxWidth()
            .background(
                color = MaterialTheme.colorScheme.surfaceVariant.copy(alpha = 0.6f),
                shape = RoundedCornerShape(16.dp),
            )
            .border(
                width = 1.dp,
                color = MaterialTheme.colorScheme.outline.copy(alpha = 0.6f),
                shape = RoundedCornerShape(16.dp),
            )
            .padding(14.dp),
        verticalArrangement = Arrangement.spacedBy(10.dp),
    ) {
        Row(
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.SpaceBetween,
            modifier = Modifier.fillMaxWidth(),
        ) {
            Column {
                Text(
                    text = "Новое предложение по ссылке",
                    style = MaterialTheme.typography.titleMedium,
                    fontWeight = FontWeight.SemiBold,
                    color = MaterialTheme.colorScheme.onSurface,
                )
                Text(
                    text = "Заполните обязательные поля и подтвердите создание",
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                )
            }
            IconButton(onClick = onClear) {
                Icon(
                    imageVector = Icons.Outlined.Close,
                    contentDescription = "Очистить",
                    tint = MaterialTheme.colorScheme.onSurfaceVariant,
                )
            }
        }

        Row(
            horizontalArrangement = Arrangement.spacedBy(12.dp),
            verticalAlignment = Alignment.Top,
        ) {
            Box(
                modifier = Modifier
                    .size(88.dp)
                    .background(
                        color = MaterialTheme.colorScheme.surface,
                        shape = RoundedCornerShape(12.dp),
                    ),
                contentAlignment = Alignment.Center,
            ) {
                AsyncImage(
                    model = draft.thumbUrl,
                    contentDescription = null,
                    modifier = Modifier
                        .fillMaxSize()
                        .background(MaterialTheme.colorScheme.surface, RoundedCornerShape(12.dp)),
                    contentScale = ContentScale.Crop,
                )
            }

            Column(
                modifier = Modifier.weight(1f),
                verticalArrangement = Arrangement.spacedBy(6.dp),
            ) {
                Text(
                    text = listOfNotNull(draft.brand, draft.model).joinToString(" ").ifBlank { draft.raw.title ?: "Без названия" },
                    style = MaterialTheme.typography.titleSmall,
                    color = MaterialTheme.colorScheme.onSurface,
                    maxLines = 2,
                    overflow = TextOverflow.Ellipsis,
                )
                Text(
                    text = buildPriceLine(draft),
                    style = MaterialTheme.typography.bodyMedium,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                )

                if (missingCategory) {
                    Row(
                        verticalAlignment = Alignment.CenterVertically,
                        horizontalArrangement = Arrangement.spacedBy(6.dp),
                    ) {
                        Icon(
                            imageVector = Icons.Outlined.WarningAmber,
                            contentDescription = null,
                            tint = MaterialTheme.colorScheme.error,
                        )
                        Text(
                            text = "Категория обязательна для отслеживания",
                            style = MaterialTheme.typography.bodySmall,
                            color = MaterialTheme.colorScheme.error,
                        )
                    }
                }

                OutlinedTextField(
                    value = categoryState.value,
                    onValueChange = {
                        categoryState.value = it
                        onCategoryChange(it)
                    },
                    label = { Text("Категория") },
                    singleLine = true,
                    modifier = Modifier.fillMaxWidth(),
                )
            }
        }

        if (draft.attributes.isNotEmpty()) {
            Column(
                verticalArrangement = Arrangement.spacedBy(4.dp),
            ) {
                draft.attributes.take(4).forEach { (k, v) ->
                    Text(
                        text = "$k: $v",
                        style = MaterialTheme.typography.bodySmall,
                        color = MaterialTheme.colorScheme.onSurfaceVariant,
                    )
                }
                if (draft.attributes.size > 4) {
                    Text(
                        text = "Ещё ${draft.attributes.size - 4} атрибутов",
                        style = MaterialTheme.typography.labelSmall,
                        color = MaterialTheme.colorScheme.onSurfaceVariant,
                    )
                }
            }
        }

        Button(
            onClick = onCreate,
            enabled = draft.canTrackPrice && categoryState.value.isNotBlank(),
            modifier = Modifier.fillMaxWidth(),
        ) {
            Row(
                verticalAlignment = Alignment.CenterVertically,
                horizontalArrangement = Arrangement.Center,
            ) {
                Icon(
                    imageVector = Icons.Outlined.Public,
                    contentDescription = null,
                    tint = MaterialTheme.colorScheme.onPrimary,
                )
                Spacer(Modifier.width(8.dp))
                Text(
                    text = "Создать и отслеживать предложение из ${platformTitle(draft.source, domain)}",
                    maxLines = 2,
                    overflow = TextOverflow.Ellipsis,
                )
            }
        }

        if (!draft.canTrackPrice) {
            Text(
                text = "По этой ссылке не удалось найти цену или валюту — отслеживание недоступно.",
                style = MaterialTheme.typography.bodySmall,
                color = MaterialTheme.colorScheme.error,
            )
        }
    }
}

private fun buildPriceLine(draft: LinkDraft): String {
    val price = draft.price
    val currency = draft.currency
    return if (price != null && !currency.isNullOrBlank()) {
        "Цена: $price $currency"
    } else {
        "Цена не определена"
    }
}

private fun platformTitle(source: SourceType, domain: String?): String {
    return when (source) {
        SourceType.AVITO -> domain ?: "avito.ru"
        SourceType.UNKNOWN -> domain ?: "Источник"
    }
}

private fun resolveDomain(url: String): String? =
    runCatching { URI(url).host?.removePrefix("www.") }.getOrNull()
