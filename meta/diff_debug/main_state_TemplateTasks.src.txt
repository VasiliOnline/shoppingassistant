package com.example.shoppingassistant.feature.pages.main.state

import com.example.shoppingassistant.core.data.link.LinkTemplateRaw
import com.example.shoppingassistant.domain.catalog.RequiredIfRule
import com.example.shoppingassistant.domain.catalog.constraints.CatalogConstraints
import com.example.shoppingassistant.domain.ingest.SourceType
import com.example.shoppingassistant.domain.template.TemplateAnchorType

/**
 * Источники значений атрибутов шаблона.
 */
enum class ValueSource { ParsedFromText, UserSelected, FromSuggestion }

/**
 * Единое представление значения атрибута шаблона.
 */
data class TemplateAttribute(
    val code: String,
    val canonicalValue: String?,
    val source: ValueSource,
)

/**
 * Режимы работы шаблона товара на главном экране.
 */
enum class TemplateMode {
    /** Поиск или подписка по выбранной подсказке. */
    SearchOrSubscribe,

    /** Создание предложения, полученного из внешней ссылки. */
    OfferFromLink,

    /** Экспресс-объявление, собранное с фото пользователя. */
    ExpressFromPhoto,

    /** Режим для голоса (опциональный, пока не используется). */
    OfferFromVoice,
}

/**
 * Метаданные привязки к внешней ссылке.
 */
data class TemplateLinkMeta(
    val url: String,
    val sourceType: SourceType?,
    val domainName: String? = null,
    val canonicalUrl: String? = null,
    val listingId: String? = null,
)

/**
 * UI-шаблон товара: единый источник правды для поля ввода, фильтров и флоу по ссылке.
 */
data class UiTemplate(
    val inputText: String,
    val anchorType: TemplateAnchorType? = null,
    val anchorId: String? = null,
    val categoryCode: String? = null,
    val lockedTitle: String? = null,
    val isLocked: Boolean = false,
    val attributes: Map<String, TemplateAttribute> = emptyMap(),
    val photoUrls: List<String> = emptyList(),
    val primaryPhotoUrl: String? = null,
    val linkMeta: TemplateLinkMeta? = null,
    val linkTemplate: LinkTemplateRaw? = null,
    val mode: TemplateMode = TemplateMode.SearchOrSubscribe,
    val requiredKeys: Set<String> = emptySet(),
    val errorKeys: Set<String> = emptySet(),
    val errorMessages: Map<String, String> = emptyMap(),
) {
    companion object {
        val Empty = UiTemplate(inputText = "")
    }

    /**
    * Удобное отображение в Map для FilterTemplateBlock.
    */
    fun asSelectedFilters(): Map<String, String> =
        attributes.mapNotNull { (key, attr) ->
            attr.canonicalValue?.takeIf { it.isNotBlank() }?.let { key to it }
        }.toMap()
}

data class ValidationResult(
    val isValid: Boolean,
    val missingKeys: Set<String> = emptySet(),
    val errorMessages: Map<String, String> = emptyMap(),
)

/**
 * Категориальный словарь атрибутов и токенов для быстрого парсинга текста.
 */
data class CategoryDictionary(
    val categoryCode: String,
    val attributes: Map<String, AttributeDict>,
    val tokenIndex: Map<String, List<TokenIndexEntry>>,
    val requiredIfRules: List<RequiredIfRule> = emptyList(),
    val constraints: List<CatalogConstraints> = emptyList(),
)

data class AttributeDict(
    val code: String,
    val isRequiredForOffer: Boolean,
    val isRequiredForSearch: Boolean,
    val isRequiredForExpress: Boolean,
    val tokenToCanonical: Map<String, String>,
    val canonicalValues: Set<String>,
)

data class TokenIndexEntry(
    val attributeCode: String,
    val canonical: String,
)

/**
 * Движок синхронизации поля ввода и фильтров.
 */
interface TemplateEngine {
    fun onInputTextChanged(
        prev: UiTemplate,
        newText: String,
        dict: CategoryDictionary,
    ): UiTemplate

    fun onAttributeChanged(
        prev: UiTemplate,
        code: String,
        newValue: String?,
        dict: CategoryDictionary,
    ): UiTemplate

    fun attachLinkTemplate(
        prev: UiTemplate,
        linkTemplate: LinkTemplateRaw,
        dict: CategoryDictionary,
    ): UiTemplate

    fun validateForOfferFromLink(
        template: UiTemplate,
        dict: CategoryDictionary,
    ): ValidationResult
}
