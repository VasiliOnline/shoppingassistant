package com.example.shoppingassistant.feature.pages.main.suggest

import com.example.shoppingassistant.core.data.BrandModelRules
import com.example.shoppingassistant.core.data.suggest.ProductSuggestRepository
import com.example.shoppingassistant.core.data.suggest.ProductSuggestCandidate
import com.example.shoppingassistant.domain.catalog.CatalogRepository
import com.example.shoppingassistant.domain.catalog.Category
import com.example.shoppingassistant.domain.catalog.CategoryAlias
import com.example.shoppingassistant.domain.catalog.CategoryAliasRepository
import com.example.shoppingassistant.domain.template.TemplateHistoryRepository
import com.example.shoppingassistant.domain.template.presets.TemplatePresetSource
import com.example.shoppingassistant.domain.template.presets.TemplatePresetsRepository
import com.example.shoppingassistant.domain.template.status.TemplateStatusContext
import com.example.shoppingassistant.domain.template.status.TemplateStatus
import com.example.shoppingassistant.domain.template.status.TemplateStatusResolver
import java.util.Locale

class MainSuggestEngineImpl(
    private val catalogRepository: CatalogRepository,
    private val categoryAliasRepository: CategoryAliasRepository,
    private val productSuggestRepository: ProductSuggestRepository,
    private val historyRepository: TemplateHistoryRepository,
    private val presetsRepository: TemplatePresetsRepository,
    private val statusResolver: TemplateStatusResolver,
) : MainSuggestEngine {

    @Volatile private var cachedCategories: List<Category>? = null
    @Volatile private var cachedAliases: List<CategoryAlias>? = null
    private val breadcrumbSeparator = " → "

    override suspend fun suggest(queryText: String): List<MainSuggestItem> {
        val normalized = queryText.replace("\\s+".toRegex(), " ").trim()
        val qLower = normalized.lowercase(Locale.getDefault())

        val categories = cachedCategories
            ?: runCatching { catalogRepository.listCategories() }.getOrElse { emptyList() }
                .also { cachedCategories = it }

        val aliases = cachedAliases
            ?: runCatching { categoryAliasRepository.listAliases() }.getOrElse { emptyList() }
                .also { cachedAliases = it }

        val breadcrumbByCode = buildBreadcrumbIndex(categories)

        // Чтобы не было "рандомных" подсказок на 1 букву (как на твоём скрине с "e").
        if (normalized.isNotBlank() && normalized.length < 2) {
            return listOf(
                InfoSuggest(text = "Введите ещё пару букв, чтобы появились подсказки.")
            )
        }

        if (normalized.isBlank()) {
            val history = runCatching { historyRepository.listRecent(limit = 12) }.getOrElse { emptyList() }
            val recentItems = mutableListOf<HistoryTemplateSuggest>()
            val continueItems = mutableListOf<HistoryTemplateSuggest>()

            history.forEach { entry ->
                val snapshot = entry.snapshot
                val attrsByKey = snapshot.data.attrs.associate { it.key to it.value }
                val categoryCode = snapshot.data.categoryCode?.takeIf { it.isNotBlank() }

                val brand = attrsByKey["brand"]
                val model = attrsByKey["model"]

                val profile = categoryCode?.let {
                    runCatching { catalogRepository.getCategoryProfile(it) }.getOrNull()
                }
                val constraints = if (categoryCode == null) {
                    emptyList()
                } else {
                    runCatching { catalogRepository.listConstraints(categoryCode, brand, model) }
                        .getOrElse { emptyList() }
                }

                val status = statusResolver.resolve(
                    TemplateStatusContext(
                        template = snapshot.data,
                        profile = profile,
                        constraints = constraints,
                        hasPhotos = false,
                    )
                )

                val title = when (snapshot.data.anchorType.name) {
                    "CATEGORY" -> snapshot.data.categoryCode?.let { breadcrumbByCode[it] } ?: snapshot.data.anchorId
                    "PRODUCT" -> {
                        val heading = listOfNotNull(
                            attrsByKey["brand"]?.takeIf { it.isNotBlank() },
                            attrsByKey["model"]?.takeIf { it.isNotBlank() },
                        ).joinToString(" ")
                        heading.ifBlank { snapshot.data.anchorId }
                    }
                    else -> snapshot.data.anchorId
                }

                val item = HistoryTemplateSuggest(
                    entry = entry,
                    text = title,
                    caption = summarizeSnapshot(snapshot.data, breadcrumbByCode),
                    status = status.status,
                    firstErrorKey = status.firstErrorKey,
                )
                if (status.status == TemplateStatus.DRAFT) {
                    continueItems += item
                } else {
                    recentItems += item
                }
            }

            val popular = runCatching {
                presetsRepository.listPresets(source = TemplatePresetSource.POPULAR, limit = 12)
            }.getOrElse { emptyList() }
                .filter { it.snapshot.categoryCode?.isNotBlank() == true }

            val categoryItems = categories
                .sortedBy { (it.title ?: it.code).lowercase(Locale.getDefault()) }
                .map { category ->
                    val breadcrumb = breadcrumbByCode[category.code] ?: category.title ?: category.code
                    CategoryAnchorSuggest(
                        categoryCode = category.code,
                        breadcrumb = breadcrumb,
                        count = null,
                        matchedAlias = null,
                    )
                }
                .take(20)

            val popularItems = popular.map { preset ->
                val attrsByKey = preset.snapshot.attrs.associate { it.key to it.value }
                val heading = preset.title?.takeIf { it.isNotBlank() }
                    ?: listOfNotNull(
                        attrsByKey["brand"]?.takeIf { it.isNotBlank() },
                        attrsByKey["model"]?.takeIf { it.isNotBlank() }
                            ?: attrsByKey["model_line"]?.takeIf { it.isNotBlank() },
                    ).joinToString(" ").ifBlank { preset.snapshot.anchorId }

                val caption = preset.snapshot.categoryCode?.let { breadcrumbByCode[it] }
                PresetTemplateSuggest(
                    preset = preset,
                    text = heading,
                    caption = caption,
                )
            }

            val items = buildList {
                if (continueItems.isNotEmpty()) {
                    add(SectionHeaderSuggest(text = "Вы продолжали"))
                    addAll(continueItems)
                }
                if (recentItems.isNotEmpty()) {
                    add(SectionHeaderSuggest(text = "Недавнее"))
                    addAll(recentItems)
                }
                if (popularItems.isNotEmpty()) {
                    add(SectionHeaderSuggest(text = "Популярные"))
                    addAll(popularItems)
                }
                if (categoryItems.isNotEmpty()) {
                    addAll(categoryItems)
                }
            }

            return dropEmptyHeaders(dedupeByLabel(items))
                .distinctBy { it.stableId }
        }

        val out = mutableListOf<MainSuggestItem>()

        // 1) Ищем кандидатов по исходному запросу
        var productCandidates = runCatching {
            productSuggestRepository.search(normalized, limit = 6)
        }.getOrElse { emptyList() }

        // 2) Если ничего не нашли — пробуем безопасный typo-fix для iPhone 145 -> iPhone 14 и т.п.
        var appliedFix: String? = null
        if (productCandidates.isEmpty()) {
            val typoFix = iphoneTypoFix(normalized)
            if (typoFix != null) {
                val fixedCandidates = runCatching {
                    productSuggestRepository.search(typoFix, limit = 6)
                }.getOrElse { emptyList() }

                if (fixedCandidates.isNotEmpty()) {
                    productCandidates = fixedCandidates
                    appliedFix = typoFix
                }
            }
        }

        // 3) Дедуп: убираем повторяющиеся “одинаковые” подсказки даже если у них разные id
        val distinctCandidates = productCandidates.distinctBy { c ->
            val brand = c.brand?.trim()
                ?.takeIf { it.isNotBlank() }
                ?.lowercase(Locale.getDefault())
            val model = c.model?.trim()
                ?.takeIf { it.isNotBlank() }
                ?.lowercase(Locale.getDefault())
            val key = listOfNotNull(brand, model).joinToString("|").trim()
            if (key.isNotBlank()) key else c.title.trim().lowercase(Locale.getDefault())
        }

        // 4) "Confident" - если точное совпадение по title или по brand+model
        val confident = distinctCandidates.any { c ->
            if (c.title.equals(normalized, ignoreCase = true)) return@any true
            val displayTitle = displayTitleFor(c)
            displayTitle.isNotBlank() && displayTitle.equals(normalized, ignoreCase = true)
        }

        val candidateTitles = distinctCandidates
            .map { displayTitleFor(it).ifBlank { it.title } }
            .filter { it.isNotBlank() }

        // 5) TextFix показываем ТОЛЬКО если он реально подтверждён кандидатами
        val fix = appliedFix ?: textFix(normalized)?.takeIf { suggested ->
            candidateTitles.any { it.equals(suggested, ignoreCase = true) }
        }

        if (fix != null && !fix.equals(normalized, ignoreCase = true)) {
            out += TextFixSuggest(
                fixedText = fix,
                caption = "Возможно вы имели в виду…",
            )
        }

        val autoTemplate = autoTemplateSuggest(
            raw = normalized,
            queryLower = qLower,
            categories = categories,
            breadcrumbByCode = breadcrumbByCode,
            aliases = aliases,
            confident = confident,
        )
        if (autoTemplate != null) {
            out += SectionHeaderSuggest(text = "Собран шаблон")
            out += autoTemplate
        }

        distinctCandidates.take(3).forEach { c ->
            val displayTitle = displayTitleFor(c).ifBlank { c.title }

            out += ProductAnchorSuggest(
                productId = c.id,
                text = displayTitle,
                caption = null,
                brand = c.brand,
                model = c.model,
                categoryCode = c.categoryCode,
            )
        }

        if (!confident) {
            out += InfoSuggest(
                text = "Не нашли товар по названию? Выберите категорию ниже — откроется шаблон с атрибутами.",
            )
            val categoryAnchors = suggestCategories(
                queryLower = qLower,
                categories = categories,
                breadcrumbByCode = breadcrumbByCode,
                aliases = aliases,
                limit = 6,
            )
            out += categoryAnchors
        }

        return dropEmptyHeaders(dedupeByLabel(out)).distinctBy { it.stableId }
    }

    private fun textFix(raw: String): String? {
        if (raw.isBlank()) return null
        val q = BrandModelRules.fromRaw(raw)
        val suggested = listOf(q.brand, q.model)
            .joinToString(" ")
            .replace("\\s+".toRegex(), " ")
            .trim()
        return suggested.takeIf { it.isNotBlank() && !it.equals(raw, ignoreCase = true) }
    }

    private data class CategoryHint(
        val categoryCode: String,
        val breadcrumb: String,
        val matchedAlias: String?,
    )

    private fun autoTemplateSuggest(
        raw: String,
        queryLower: String,
        categories: List<Category>,
        breadcrumbByCode: Map<String, String>,
        aliases: List<CategoryAlias>,
        confident: Boolean,
    ): AutoTemplateSuggest? {
        val normalized = raw.trim().replace("\\s+".toRegex(), " ")
        if (normalized.isBlank()) return null
        if (confident) return null

        val tokens = normalized.split(" ").filter { it.isNotBlank() }
        val hasPrefix = normalized.startsWith("iphone", ignoreCase = true) ||
            normalized.startsWith("galaxy", ignoreCase = true)
        if (tokens.size < 2 && !hasPrefix) return null
        if (tokens.size > 5) return null

        val parsed = BrandModelRules.fromRaw(normalized)
        val brand = parsed.brand.trim()
        val model = parsed.model.trim()
        if (!isBrandCandidate(brand) || !isModelCandidate(model)) return null

        val hint = pickCategoryHint(queryLower, categories, breadcrumbByCode, aliases)
        val modelLine = extractModelLine(model)
        val reasons = buildAutoTemplateReasons(
            raw = normalized,
            brand = brand,
            model = model,
            modelLine = modelLine,
            matchedAlias = hint?.matchedAlias,
            categoryBreadcrumb = hint?.breadcrumb,
        )

        val display = listOf(brand, model)
            .joinToString(" ")
            .replace("\\s+".toRegex(), " ")
            .trim()

        return AutoTemplateSuggest(
            text = display,
            caption = "Собран шаблон",
            reasons = reasons,
            brand = brand,
            model = model,
            modelLine = modelLine,
            categoryCode = hint?.categoryCode,
            categoryBreadcrumb = hint?.breadcrumb,
        )
    }

    private fun isBrandCandidate(brand: String): Boolean {
        val trimmed = brand.trim()
        if (trimmed.length < 2) return false
        if (trimmed.all { it.isDigit() }) return false
        return true
    }

    private fun isModelCandidate(model: String): Boolean {
        val trimmed = model.trim()
        if (trimmed.isBlank()) return false
        if (trimmed.length < 2 && trimmed.none { it.isDigit() }) return false
        return true
    }

    private fun extractModelLine(model: String): String? {
        val parts = model.split(" ").map { it.trim() }.filter { it.isNotBlank() }
        if (parts.size < 2) return null
        val line = parts.firstOrNull { part -> part.any { it.isLetter() } } ?: return null
        if (parts.drop(1).none { part -> part.any { it.isDigit() } }) return null
        return line
    }

    private fun pickCategoryHint(
        queryLower: String,
        categories: List<Category>,
        breadcrumbByCode: Map<String, String>,
        aliases: List<CategoryAlias>,
    ): CategoryHint? {
        if (queryLower.isBlank()) return null
        val byCode = categories.associateBy { it.code }

        val aliasMatch = aliases
            .filter { a -> a.alias.trim().isNotBlank() && queryLower.contains(a.alias.trim().lowercase()) }
            .maxByOrNull { it.alias.trim().length }

        if (aliasMatch != null) {
            val breadcrumb = breadcrumbByCode[aliasMatch.categoryCode]
                ?: byCode[aliasMatch.categoryCode]?.title
                ?: aliasMatch.categoryCode
            return CategoryHint(
                categoryCode = aliasMatch.categoryCode,
                breadcrumb = breadcrumb,
                matchedAlias = aliasMatch.alias.trim(),
            )
        }

        val exact = categories.firstOrNull { category ->
            val title = (category.title ?: category.code).trim().lowercase()
            title == queryLower
        } ?: return null

        val breadcrumb = breadcrumbByCode[exact.code] ?: exact.title ?: exact.code
        return CategoryHint(
            categoryCode = exact.code,
            breadcrumb = breadcrumb,
            matchedAlias = null,
        )
    }

    private fun buildAutoTemplateReasons(
        raw: String,
        brand: String,
        model: String,
        modelLine: String?,
        matchedAlias: String?,
        categoryBreadcrumb: String?,
    ): List<String> {
        val reasons = mutableListOf<String>()
        val firstToken = raw.split(" ").firstOrNull()?.trim().orEmpty()
        when {
            firstToken.equals(brand, ignoreCase = true) -> reasons += "$firstToken -> бренд"
            raw.startsWith("iphone", ignoreCase = true) && brand.equals("Apple", ignoreCase = true) ->
                reasons += "iPhone -> бренд Apple"
            raw.startsWith("galaxy", ignoreCase = true) && brand.equals("Samsung", ignoreCase = true) ->
                reasons += "Galaxy -> бренд Samsung"
            brand.isNotBlank() -> reasons += "$brand -> бренд"
        }

        val modelToken = raw.removePrefix(firstToken).trim().ifBlank { model }
        if (modelToken.isNotBlank()) {
            reasons += "$modelToken -> модель"
        } else if (model.isNotBlank()) {
            reasons += "$model -> модель"
        }

        if (!modelLine.isNullOrBlank()) {
            reasons += "$modelLine -> линейка"
        }

        if (!matchedAlias.isNullOrBlank()) {
            val breadcrumb = categoryBreadcrumb?.takeIf { it.isNotBlank() }
            val label = breadcrumb?.let { "$matchedAlias -> категория $it" } ?: "$matchedAlias -> категория"
            reasons += label
        }
        return reasons
    }

    /**
     * Аккуратный фикс для кейса iPhone 145 -> iPhone 14.
     * Возвращаем строку ТОЛЬКО если она реально отличается от raw.
     */
    private fun iphoneTypoFix(raw: String): String? {
        if (raw.isBlank()) return null

        val q = BrandModelRules.fromRaw(raw)
        val model = q.model ?: return null

        val m = Regex("(?i)\\biphone\\s*(\\d{3,})\\b").find(model) ?: return null
        val digits = m.groupValues[1]
        if (digits.length < 3) return null

        val truncated = digits.dropLast(1)
        if (truncated.length < 2) return null

        val newModel = model.replace(m.value, "iPhone $truncated", ignoreCase = true)

        val brand = q.brand ?: "Apple" // iPhone -> Apple (в рамках UX-подсказок)
        val suggested = listOfNotNull(brand, newModel)
            .joinToString(" ")
            .replace("\\s+".toRegex(), " ")
            .trim()

        return suggested.takeIf { it.isNotBlank() && !it.equals(raw, ignoreCase = true) }
    }

    private fun suggestCategories(
        queryLower: String,
        categories: List<Category>,
        breadcrumbByCode: Map<String, String>,
        aliases: List<CategoryAlias>,
        limit: Int,
    ): List<CategoryAnchorSuggest> {
        if (queryLower.isBlank()) return emptyList()
        val byCode = categories.associateBy { it.code }
        val allCodes = categories.map { it.code }

        val aliasMatches = aliases
            .filter { a -> a.alias.trim().isNotBlank() && queryLower.contains(a.alias.trim().lowercase()) }
            .map { a -> a.categoryCode to a.alias }
            .distinctBy { it.first }
            .toMap()

        // Для очень коротких запросов не показываем "общие" категории,
        // иначе появляются нерелевантные пункты. Исключение - явный алиас.
        if (queryLower.length < 2 && aliasMatches.isEmpty()) return emptyList()

        data class Scored(val code: String, val score: Int, val matchedAlias: String?)

        fun titleOf(code: String): String =
            byCode[code]?.title ?: code

        fun scoreCategory(code: String): Scored? {
            val title = titleOf(code).lowercase()
            val matchedAlias = aliasMatches[code]
            val score = when {
                matchedAlias != null -> 100
                title == queryLower -> 90
                title.contains(queryLower) -> 75
                else -> 0
            }
            if (score == 0) return null
            return Scored(code, score, matchedAlias)
        }

        val aliasOnlyCodes = aliasMatches.keys.filterNot { it in allCodes }
        val candidates = allCodes + aliasOnlyCodes
        val scored = candidates.mapNotNull { scoreCategory(it) }
            .sortedByDescending { it.score }
            .take(limit)

        val picked = scored

        return picked.map { s ->
            val breadcrumb = breadcrumbByCode[s.code] ?: titleOf(s.code)
            CategoryAnchorSuggest(
                categoryCode = s.code,
                breadcrumb = breadcrumb,
                count = null,
                matchedAlias = s.matchedAlias,
            )
        }
    }

    private fun buildBreadcrumbIndex(categories: List<Category>): Map<String, String> {
        val byCode = categories.associateBy { it.code }
        fun titleOf(code: String): String = byCode[code]?.title ?: code

        fun breadcrumb(code: String): String {
            val path = ArrayList<String>()
            var cur: Category? = byCode[code]
            val seen = HashSet<String>()
            while (cur != null && seen.add(cur.code)) {
                val t = cur.title?.takeIf { it.isNotBlank() } ?: cur.code
                path.add(t)
                cur = cur.parentCode?.let { byCode[it] }
            }
            return path.asReversed().joinToString(breadcrumbSeparator)
        }

        return categories.associate { it.code to breadcrumb(it.code) }
            .mapValues { (code, bc) -> if (bc.isBlank()) titleOf(code) else bc }
    }

    private fun summarizeSnapshot(
        data: com.example.shoppingassistant.domain.template.TemplateSnapshotData,
        breadcrumbByCode: Map<String, String>,
    ): String? {
        val parts = mutableListOf<String>()
        val anchorText = if (data.anchorType.name == "CATEGORY") {
            data.categoryCode?.let { breadcrumbByCode[it] } ?: data.anchorId
        } else data.anchorId

        if (anchorText.isNotBlank()) parts += anchorText

        val attrs = data.attrs
            .sortedBy { it.key.lowercase() }
            .take(2)
            .map { "${it.key}=${it.value}" }
        if (attrs.isNotEmpty()) parts += attrs.joinToString(" · ")

        data.freeText?.takeIf { it.isNotBlank() }?.let { parts += "Запрос: $it" }

        return parts.drop(1).joinToString(" · ").takeIf { it.isNotBlank() }
    }

    private fun displayTitleFor(c: ProductSuggestCandidate): String {
        return listOfNotNull(
            c.brand?.takeIf { it.isNotBlank() },
            c.model?.takeIf { it.isNotBlank() },
        ).joinToString(" ").replace("\\s+".toRegex(), " ").trim()
    }

    private fun dedupeByLabel(items: List<MainSuggestItem>): List<MainSuggestItem> {
        val seen = HashSet<String>()
        val out = ArrayList<MainSuggestItem>(items.size)
        items.forEach { item ->
            val label = when (item) {
                is ProductAnchorSuggest -> item.text
                is PresetTemplateSuggest -> item.text
                is HistoryTemplateSuggest -> item.text
                is AutoTemplateSuggest -> item.text
                is CategoryAnchorSuggest -> item.breadcrumb
                is TextFixSuggest -> item.fixedText
                else -> null
            }?.trim().orEmpty()
            if (label.isNotBlank()) {
                val key = label.lowercase(Locale.getDefault())
                if (!seen.add(key)) return@forEach
            }
            out.add(item)
        }
        return out
    }

    private fun dropEmptyHeaders(items: List<MainSuggestItem>): List<MainSuggestItem> {
        val out = ArrayList<MainSuggestItem>(items.size)
        var pendingHeader: SectionHeaderSuggest? = null
        items.forEach { item ->
            if (item is SectionHeaderSuggest) {
                pendingHeader = item
                return@forEach
            }
            if (pendingHeader != null) {
                out.add(pendingHeader!!)
                pendingHeader = null
            }
            out.add(item)
        }
        return out
    }
}
