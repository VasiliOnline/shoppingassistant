@Composable
fun CompactTemplatePanel(
    filledCount: Int,
    totalCount: Int,
    foundCount: Int?,
    onEdit: () -> Unit,
    onShowOffers: () -> Unit,
    isAuthorized: Boolean,
    onCreateExpress: () -> Unit,
    onLogin: () -> Unit,
    onSubscribe: () -> Unit,
    contentStartPadding: Dp = 0.dp,
    modifier: Modifier = Modifier,
) {
    Column(
        modifier = modifier.fillMaxWidth(),
        verticalArrangement = Arrangement.spacedBy(8.dp),
    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(start = contentStartPadding),
            verticalArrangement = Arrangement.spacedBy(8.dp),
        ) {
            CompactEditRow(
                filledCount = filledCount,
                totalCount = totalCount,
                onEdit = onEdit,
            )
            CompactActionButton(
                title = "Показать предложения",
                subtitle = "Найдено ${foundCount ?: 0}",
                icon = Icons.Outlined.Search,
                primary = true,
                enabled = (foundCount ?: 0) > 0,
                onClick = onShowOffers,
            )
            CompactActionButton(
                title = "Создать экспресс-объявление",
                icon = Icons.Outlined.Edit,
                primary = false,
                enabled = true,
                onClick = if (isAuthorized) onCreateExpress else onLogin,
            )
            CompactActionButton(
                title = "Отслеживать изменения",
                icon = Icons.Outlined.NotificationsNone,
                primary = false,
                enabled = true,
                onClick = onSubscribe,
            )
        }
    }
}

@Composable
fun CompactInputPhotoSlot(
    photoUrls: List<String>,
    onClick: () -> Unit,
    modifier: Modifier = Modifier,
    size: Dp = 56.dp,
) {
    val photos = remember(photoUrls) { photoUrls.filter { it.isNotBlank() } }
    val primary = photos.firstOrNull()
    val count = photos.size
    val shape = RoundedCornerShape(12.dp)
    Box(
        modifier = modifier
            .size(size)
            .clip(shape)
            .border(1.dp, MaterialTheme.colorScheme.outline.copy(alpha = 0.45f), shape)
            .clickable { onClick() },
        contentAlignment = Alignment.Center,
    ) {
        if (primary != null) {
            AsyncImage(
                model = primary,
                contentDescription = null,
                contentScale = ContentScale.Crop,
                modifier = Modifier.fillMaxSize(),
            )
        } else {
            Row(
                verticalAlignment = Alignment.CenterVertically,
                horizontalArrangement = Arrangement.spacedBy(4.dp),
            ) {
                Text(
                    text = "Add photo",
                    style = MaterialTheme.typography.labelSmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                    maxLines = 1,
                    overflow = TextOverflow.Ellipsis,
                )
                Icon(
                    imageVector = Icons.Outlined.PhotoCamera,
                    contentDescription = null,
                    tint = MaterialTheme.colorScheme.onSurfaceVariant,
                    modifier = Modifier.size(14.dp),
                )
            }
        }
        if (count > 0) {
            Surface(
                shape = CircleShape,
                color = MaterialTheme.colorScheme.primary,
                modifier = Modifier
                    .align(Alignment.BottomEnd)
                    .padding(4.dp),
            ) {
                Text(
                    text = count.toString(),
                    style = MaterialTheme.typography.labelSmall,
                    color = MaterialTheme.colorScheme.onPrimary,
                    modifier = Modifier.padding(horizontal = 6.dp, vertical = 2.dp),
                )
            }
        }
    }
}

@Composable
private fun CompactEditRow(
    filledCount: Int,
    totalCount: Int,
    onEdit: () -> Unit,
) {
    val iconTint = MaterialTheme.colorScheme.onSurfaceVariant
    Row(
        modifier = Modifier.fillMaxWidth(),
        verticalAlignment = Alignment.CenterVertically,
        horizontalArrangement = Arrangement.spacedBy(8.dp),
    ) {
        Surface(
            shape = RoundedCornerShape(14.dp),
            tonalElevation = 0.dp,
            shadowElevation = 0.dp,
            color = Color.Transparent,
            modifier = Modifier
                .clickable { onEdit() }
                .border(1.dp, MaterialTheme.colorScheme.outline.copy(alpha = 0.45f), RoundedCornerShape(14.dp)),
        ) {
            Row(
                modifier = Modifier.padding(horizontal = 12.dp, vertical = 6.dp),
                verticalAlignment = Alignment.CenterVertically,
                horizontalArrangement = Arrangement.spacedBy(8.dp),
            ) {
                Icon(
                    imageVector = Icons.Outlined.Edit,
                    contentDescription = null,
                    tint = iconTint,
                    modifier = Modifier.size(18.dp),
                )
                Text(
                    text = "Редактировать шаблон",
                    style = MaterialTheme.typography.bodyMedium,
                    color = MaterialTheme.colorScheme.onSurface,
                    maxLines = 1,
                    overflow = TextOverflow.Ellipsis,
                )
                if (totalCount > 0) {
                    Surface(
                        shape = RoundedCornerShape(10.dp),
                        color = MaterialTheme.colorScheme.surfaceVariant,
                        tonalElevation = 0.dp,
                        shadowElevation = 0.dp,
                    ) {
                        Text(
                            text = "$filledCount/$totalCount",
                            style = MaterialTheme.typography.labelSmall,
                            color = MaterialTheme.colorScheme.onSurfaceVariant,
                            modifier = Modifier.padding(horizontal = 8.dp, vertical = 2.dp),
                            maxLines = 1,
                        )
                    }
                }
            }
        }
    }
}

@Composable
private fun CompactActionButton(
    title: String,
    icon: ImageVector,
    primary: Boolean,
    enabled: Boolean,
    subtitle: String? = null,
    onClick: () -> Unit,
    modifier: Modifier = Modifier,
) {
    val colors = MaterialTheme.colorScheme
    val shape = RoundedCornerShape(14.dp)
    val background = if (primary) {
        colors.primary.copy(alpha = 0.18f)
    } else {
        colors.surfaceVariant
    }
    val titleColor = if (primary) colors.onSurface else colors.onSurface
    val iconTint = if (primary) colors.primary else colors.onSurfaceVariant
    Surface(
        shape = shape,
        color = background,
        tonalElevation = 0.dp,
        shadowElevation = 0.dp,
        modifier = modifier
            .fillMaxWidth()
            .heightIn(min = 46.dp)
            .then(if (!primary) Modifier.border(1.dp, colors.outline.copy(alpha = 0.35f), shape) else Modifier)
            .alpha(if (enabled) 1f else 0.5f)
            .clickable(enabled = enabled) { onClick() },
    ) {
        Row(
            modifier = Modifier.padding(horizontal = 12.dp, vertical = 10.dp),
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.spacedBy(10.dp),
        ) {
            Icon(
                imageVector = icon,
                contentDescription = null,
                tint = iconTint,
                modifier = Modifier.size(20.dp),
            )
            Column(
                modifier = Modifier.weight(1f),
                verticalArrangement = Arrangement.spacedBy(2.dp),
            ) {
                Text(
                    text = title,
                    style = MaterialTheme.typography.bodyMedium.copy(fontWeight = FontWeight.SemiBold),
                    color = titleColor,
                    maxLines = 1,
                    overflow = TextOverflow.Ellipsis,
                )
                subtitle?.let {
                    Text(
                        text = it,
                        style = MaterialTheme.typography.labelSmall,
                        color = colors.onSurfaceVariant,
                        maxLines = 1,
                        overflow = TextOverflow.Ellipsis,
                    )
                }
            }
        }
    }
}

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun TemplateSubscriptionSheet(
    visible: Boolean,
    templateTitle: String,
    onConfirm: () -> Unit,
    onDismiss: () -> Unit,
) {
    if (!visible) return
    val sheetState = rememberModalBottomSheetState(skipPartiallyExpanded = true)
    ModalBottomSheet(
        onDismissRequest = onDismiss,
        sheetState = sheetState,
    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 20.dp, vertical = 12.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp),
            horizontalAlignment = Alignment.CenterHorizontally,
        ) {
            Text(
                text = "Подписка на шаблон",
                style = MaterialTheme.typography.titleMedium,
                color = MaterialTheme.colorScheme.onSurface,
            )
            Text(
                text = templateTitle.ifBlank { "Шаблон товара" },
                style = MaterialTheme.typography.bodyMedium,
                color = MaterialTheme.colorScheme.onSurfaceVariant,
                textAlign = TextAlign.Center,
            )
            Button(
                onClick = onConfirm,
                modifier = Modifier.fillMaxWidth(),
            ) {
                Text("Подписаться")
            }
            TextButton(onClick = onDismiss) {
                Text("Отмена")
            }
            Spacer(Modifier.height(6.dp))
        }
    }
}
