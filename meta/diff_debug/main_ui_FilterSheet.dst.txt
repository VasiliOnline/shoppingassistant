@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun FilterSheet(
    visible: Boolean,
    stage: FilterStage,
    defs: List<AttributeDef>,
    current: AttributeDef?,
    selected: Map<String, String>,
    attrFacetCounts: Map<String, Int>,
    attrFacetCountsLoading: Set<String>,
    facetCounts: Map<String, Int>?,
    facetCountsLoading: Boolean,
    onDismiss: () -> Unit,
    onPickAttr: (AttributeDef) -> Unit,
    onPickValue: (AttributeDef, String) -> Unit,
    onClearAttr: ((AttributeDef) -> Unit)? = null,
) {
    if (!visible) return
    val sheetState = rememberModalBottomSheetState(skipPartiallyExpanded = true)
    ModalBottomSheet(onDismissRequest = onDismiss, sheetState = sheetState) {
        when (stage) {
            FilterStage.ATTRS -> {
                Text(
                    text = "Выберите атрибут",
                    style = MaterialTheme.typography.titleMedium,
                    color = MaterialTheme.colorScheme.onSurface,
                    modifier = Modifier.padding(16.dp),
                )
                val iconTint = MaterialTheme.colorScheme.onSurfaceVariant
                LazyColumn {
                    items(defs) { def ->
                        val locked = def.locked
                        val value = selected[def.key]?.takeIf { it.isNotBlank() }
                        val displayValue = if (def.key == "category") categoryValueTail(value) else value
                        val normalizedValue = value?.trim()?.lowercase()
                        val count = attrFacetCounts[def.key]
                        val isCountLoading = def.key in attrFacetCountsLoading
                        val rowColor = if (locked) {
                            MaterialTheme.colorScheme.surfaceVariant.copy(alpha = 0.6f)
                        } else {
                            Color.Transparent
                        }
                        Surface(
                            shape = RoundedCornerShape(12.dp),
                            color = rowColor,
                            tonalElevation = 0.dp,
                            shadowElevation = 0.dp,
                            modifier = Modifier
                                .fillMaxWidth()
                                .padding(horizontal = 8.dp, vertical = 2.dp),
                        ) {
                            ListItem(
                                leadingContent = { Icon(pickAttrIcon(def.key, def.title), null, tint = iconTint) },
                                headlineContent = { Text(def.title) },
                                supportingContent = {
                                    displayValue?.let {
                                        Text(
                                            text = it,
                                            style = MaterialTheme.typography.bodySmall,
                                            color = MaterialTheme.colorScheme.onSurfaceVariant,
                                        )
                                    }
                                },
                                trailingContent = {
                                    Row(
                                        verticalAlignment = Alignment.CenterVertically,
                                        horizontalArrangement = Arrangement.spacedBy(8.dp),
                                    ) {
                                        if (normalizedValue != null) {
                                            val countText = when {
                                                isCountLoading -> "..."
                                                count != null -> count.toString()
                                                else -> "0"
                                            }
                                            Text(
                                                text = countText,
                                                style = MaterialTheme.typography.labelMedium,
                                                color = MaterialTheme.colorScheme.onSurfaceVariant,
                                                textAlign = TextAlign.End,
                                                modifier = Modifier.widthIn(min = 32.dp),
                                            )
                                        }
                                        if (!locked) {
                                            Icon(Icons.Outlined.ChevronRight, null, tint = iconTint)
                                        }
                                    }
                                },
                                modifier = Modifier
                                    .alpha(if (locked) 0.7f else 1f)
                                    .clickable(enabled = !locked) { onPickAttr(def) }
                            )
                        }
                    }
                }
            }
            FilterStage.VALUES -> {
                val manualValue = remember(current?.key) { mutableStateOf("") }
                val iconTint = MaterialTheme.colorScheme.onSurfaceVariant
                val isDictionaryComplete = current?.isDictionaryComplete != false
                val selectedValue = current?.key?.let { selected[it] }

                Row(
                    verticalAlignment = Alignment.CenterVertically,
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(start = 4.dp, top = 12.dp, end = 16.dp, bottom = 8.dp)
                ) {
                    IconButton(onClick = onDismiss) {
                        Icon(Icons.AutoMirrored.Outlined.ArrowBack, null, tint = iconTint)
                    }
                    Text(
                        text = current?.title.orEmpty(),
                        style = MaterialTheme.typography.titleMedium,
                        color = MaterialTheme.colorScheme.onSurface,
                    )
                    Spacer(modifier = Modifier.weight(1f))
                    if (selectedValue != null && onClearAttr != null) {
                        TextButton(onClick = { current?.let(onClearAttr) }) {
                            Text("Сбросить")
                        }
                    }
                }
                if (!isDictionaryComplete) {
                    Text(
                        text = "Справочник неполный",
                        style = MaterialTheme.typography.bodySmall,
                        color = MaterialTheme.colorScheme.onSurfaceVariant,
                        modifier = Modifier.padding(horizontal = 16.dp, vertical = 4.dp),
                    )
                }
                val selectionOnly = current?.selectionOnly == true
                val allowedValues = current?.allowedValues.orEmpty()
                val recommended = allowedValues
                    .filter { it.rank > 0 }
                    .sortedWith(compareByDescending<ValueDef> { it.rank }
                        .thenBy { it.canonical.lowercase() })
                val recommendedSet = recommended.map { it.canonical.lowercase() }.toSet()
                val allAllowed = if (recommended.isEmpty()) {
                    allowedValues
                } else {
                    allowedValues.filterNot { it.canonical.lowercase() in recommendedSet }
                }
                val opts = if (allowedValues.isNotEmpty()) {
                    allowedValues.map { it.canonical }
                } else {
                    current?.options.orEmpty()
                }
                val showManual = !selectionOnly
                val counts = facetCounts
                val showCountPlaceholder = facetCountsLoading && counts == null

                fun normalizeValueKey(value: String): String = value.trim().lowercase()

                fun resolveCount(value: String): Int? {
                    val map = counts ?: return null
                    return map[normalizeValueKey(value)] ?: 0
                }

                @Composable
                fun valueRow(opt: String) {
                    val count = resolveCount(opt)
                    val disabledByCount = count != null && count == 0
                    val disabledByConstraint = current?.forbiddenValues?.any { it.equals(opt, ignoreCase = true) } == true
                    val enabled = !disabledByCount && !disabledByConstraint
                    val countText = when {
                        count != null -> count.toString()
                        showCountPlaceholder -> "..."
                        else -> "0"
                    }
                    val displayOpt = if (current?.key == "category") categoryValueTail(opt) ?: opt else opt
                    ListItem(
                        headlineContent = { Text(displayOpt) },
                        trailingContent = {
                            Text(
                                text = countText,
                                style = MaterialTheme.typography.labelMedium,
                                color = MaterialTheme.colorScheme.onSurfaceVariant,
                                textAlign = TextAlign.End,
                                modifier = Modifier.widthIn(min = 32.dp),
                            )
                        },
                        modifier = Modifier
                            .alpha(if (enabled) 1f else 0.45f)
                            .clickable(enabled = enabled) { current?.let { def -> onPickValue(def, opt) } }
                    )
                }
                if (opts.isEmpty()) {
                    if (selectionOnly) {
                        Text(
                            text = "Для этого атрибута доступны только значения из списка, но список пока пуст.",
                            style = MaterialTheme.typography.bodyMedium,
                            color = MaterialTheme.colorScheme.onSurfaceVariant,
                            modifier = Modifier.padding(horizontal = 16.dp, vertical = 12.dp),
                        )
                    } else {
                        OutlinedTextField(
                            value = manualValue.value,
                            onValueChange = { manualValue.value = it },
                            label = { Text("Введите ${current?.title ?: "значение"}") },
                            modifier = Modifier
                                .fillMaxWidth()
                                .padding(horizontal = 16.dp, vertical = 8.dp),
                        )
                        Button(
                            onClick = {
                                val trimmed = manualValue.value.trim()
                                if (trimmed.isNotEmpty()) {
                                    current?.let { def -> onPickValue(def, trimmed) }
                                }
                            },
                            enabled = manualValue.value.isNotBlank(),
                            modifier = Modifier
                                .fillMaxWidth()
                                .padding(horizontal = 16.dp, vertical = 8.dp),
                        ) {
                            Text("Сохранить")
                        }
                    }
                } else {
                    LazyColumn {
                        if (recommended.isNotEmpty()) {
                            item {
                                Text(
                                    text = "Рекомендуемые",
                                    style = MaterialTheme.typography.labelMedium,
                                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                                    modifier = Modifier.padding(horizontal = 16.dp, vertical = 6.dp),
                                )
                            }
                            items(recommended.map { it.canonical }) { opt ->
                                valueRow(opt)
                            }
                        }
                        if (allAllowed.isNotEmpty()) {
                            item {
                                Text(
                                    text = "Все значения",
                                    style = MaterialTheme.typography.labelMedium,
                                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                                    modifier = Modifier.padding(horizontal = 16.dp, vertical = 6.dp),
                                )
                            }
                            items(allAllowed.map { it.canonical }) { opt ->
                                valueRow(opt)
                            }
                        }
                        if (allowedValues.isEmpty()) {
                            items(opts) { opt ->
                                valueRow(opt)
                            }
                        }
                        if (showManual) {
                            item {
                                OutlinedTextField(
                                    value = manualValue.value,
                                    onValueChange = { manualValue.value = it },
                                    label = { Text("Другое значение") },
                                    modifier = Modifier
                                        .fillMaxWidth()
                                        .padding(horizontal = 16.dp, vertical = 8.dp),
                                )
                                Button(
                                    onClick = {
                                        val trimmed = manualValue.value.trim()
                                        if (trimmed.isNotEmpty()) {
                                            current?.let { def -> onPickValue(def, trimmed) }
                                        }
                                    },
                                    enabled = manualValue.value.isNotBlank(),
                                    modifier = Modifier
                                        .fillMaxWidth()
                                        .padding(horizontal = 16.dp, vertical = 8.dp),
                                ) {
                                    Text("Сохранить другое")
                                }
                            }
                        }
                    }
                }
            }
        }
    }
}

private fun pickAttrIcon(key: String, title: String) = when {
    "price" in key.lowercase() || "цена" in title.lowercase() -> Icons.Outlined.LocalOffer
    "currency" in key.lowercase() || "валют" in title.lowercase() -> Icons.Outlined.Language
    "brand" in key.lowercase() || "бренд" in title.lowercase() -> Icons.Outlined.Star
    "model" in key.lowercase() || "модель" in title.lowercase() -> Icons.AutoMirrored.Outlined.Label
    "color" in key.lowercase() || "цвет" in title.lowercase() -> Icons.Outlined.ColorLens
    "mem" in key.lowercase() || "память" in title.lowercase() || "gb" in title.lowercase() -> Icons.Outlined.Memory
    "size" in key.lowercase() || "размер" in title.lowercase() || "диагонал" in title.lowercase() -> Icons.Outlined.Straighten
    "state" in key.lowercase() || "состояние" in title.lowercase() -> Icons.Outlined.Schedule
    "link" in key.lowercase() || "url" in title.lowercase() -> Icons.Outlined.Link
    "photo" in key.lowercase() || "image" in title.lowercase() -> Icons.Outlined.Settings
    else -> Icons.Outlined.Tune
}

private val CategoryValueSplitRegex = Regex("\\s*(?:→|/|\\u001A|->)\\s*")

private fun categoryValueTail(value: String?): String? {
    val raw = value?.trim().orEmpty()
    if (raw.isBlank()) return value
    val parts = raw.split(CategoryValueSplitRegex).map { it.trim() }.filter { it.isNotBlank() }
    return parts.lastOrNull() ?: value
}
