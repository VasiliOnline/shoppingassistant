private enum class OffersLayoutMode {
    Horizontal,
    Vertical
}

private enum class OffersBadge {
    None,
    BestPrice,
    BestRating,
    Nearest
}

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun OffersBottomSheet(
    visible: Boolean,
    query: NormalizedQuery?,
    onDismiss: () -> Unit,
    prefetchedItems: List<ExplainedItem> = emptyList(),
) {
    if (!visible || query == null) return

    val searchOffers: SearchOffersUseCase = koinGet(SearchOffersUseCase::class.java)
    val createAlert: CreateOfferPriceAlertUseCase = koinGet(CreateOfferPriceAlertUseCase::class.java)
    val scope = rememberCoroutineScope()
    val criteria = remember(query) {
        val locale = Locale.getDefault()
        OfferSearchCriteria(
            brand = query.brand,
            model = query.model,
            attributes = query.attributes,
            limit = 20,
            userCountry = locale.country.takeIf { it.isNotBlank() },
            userLanguage = locale.language.takeIf { it.isNotBlank() },
        )
    }

    var items by remember(query) { mutableStateOf<List<ExplainedItem>>(prefetchedItems) }
    var isLoading by remember(query) { mutableStateOf(true) }
    var minimized by remember { mutableStateOf(false) }
    var selectedItem by remember { mutableStateOf<ExplainedItem?>(null) }
    var alertMessage by remember { mutableStateOf<String?>(null) }
    var alertIsError by remember { mutableStateOf(false) }
    val userCountryLabel = criteria.userCountry

    LaunchedEffect(criteria, prefetchedItems) {
        if (prefetchedItems.isNotEmpty()) {
            items = prefetchedItems
            isLoading = false
            return@LaunchedEffect
        }
        isLoading = true
        items = runCatching { searchOffers(criteria) }.getOrElse { emptyList() }
        isLoading = false
    }

    LaunchedEffect(selectedItem?.dto?.id) {
        alertMessage = null
        alertIsError = false
    }

    var layoutMode by remember { mutableStateOf(OffersLayoutMode.Horizontal) }
    var badge by remember { mutableStateOf(OffersBadge.None) }

    val sheetState = rememberModalBottomSheetState(skipPartiallyExpanded = true)

    val displayedItems = remember(items, badge) {
        when (badge) {
            OffersBadge.None -> items
            OffersBadge.BestPrice -> items.sortedWith(
                compareBy<ExplainedItem> { it.dto.price == null }.thenBy { it.dto.price }
            )
            OffersBadge.BestRating -> items.sortedWith(
                compareByDescending<ExplainedItem> { it.dto.sellerRating ?: 0.0 }
            )
            OffersBadge.Nearest -> items.sortedWith(
                compareBy<ExplainedItem> { it.dto.deliveryTime == null }
                    .thenBy { it.dto.deliveryTime ?: Int.MAX_VALUE }
            )
        }
    }

    if (visible && !minimized) {
        ModalBottomSheet(
            onDismissRequest = { minimized = true },
            sheetState = sheetState,
            containerColor = MaterialTheme.colorScheme.surface.copy(alpha = 0.92f),
        ) {
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = 16.dp, vertical = 12.dp),
            ) {
                Row(
                    modifier = Modifier
                        .fillMaxWidth(),
                    verticalAlignment = Alignment.CenterVertically,
                    horizontalArrangement = Arrangement.SpaceBetween,
                ) {
                    Icon(
                        imageVector = Icons.Outlined.Close,
                        contentDescription = "Закрыть",
                        modifier = Modifier
                            .clickable { onDismiss() }
                            .padding(4.dp),
                        tint = MaterialTheme.colorScheme.onSurfaceVariant,
                    )
                    Column(
                        horizontalAlignment = Alignment.CenterHorizontally,
                        modifier = Modifier.weight(1f),
                    ) {
                        Box(
                            modifier = Modifier
                                .padding(bottom = 6.dp)
                                .width(40.dp)
                                .height(4.dp)
                                .clip(RoundedCornerShape(50))
                                .background(MaterialTheme.colorScheme.onSurfaceVariant.copy(alpha = 0.5f))
                        )
                        Text(
                            text = listOfNotNull(query.brand, query.model)
                                .joinToString(" ") { it.ifBlank { null } ?: "" }
                                .ifBlank { "Предложение" },
                            style = MaterialTheme.typography.titleMedium,
                        )
                    }
                    TextButton(onClick = { minimized = true }) {
                        Text(text = "Свернуть")
                    }
                }

                Spacer(Modifier.height(8.dp))
                HorizontalDivider()
                Spacer(Modifier.height(8.dp))

                BadgeMenu(
                    active = badge,
                    onSelect = { badge = it },
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(bottom = 10.dp),
                )

                Row(
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(bottom = 8.dp),
                    horizontalArrangement = Arrangement.End,
                    verticalAlignment = Alignment.CenterVertically,
                ) {
                    FilterChip(
                        selected = layoutMode == OffersLayoutMode.Horizontal,
                        onClick = { layoutMode = OffersLayoutMode.Horizontal },
                        label = {
                    Icon(
                        imageVector = Icons.Outlined.ViewWeek,
                        contentDescription = "Горизонтальные карточки",
                    )
                        },
                        colors = FilterChipDefaults.filterChipColors(
                            selectedContainerColor = MaterialTheme.colorScheme
                                .primary.copy(alpha = 0.18f),
                        ),
                    )
                    Spacer(Modifier.width(8.dp))
                    FilterChip(
                        selected = layoutMode == OffersLayoutMode.Vertical,
                        onClick = { layoutMode = OffersLayoutMode.Vertical },
                        label = {
                    Icon(
                        imageVector = Icons.Outlined.ViewStream,
                        contentDescription = "Вертикальные карточки",
                    )
                        },
                        colors = FilterChipDefaults.filterChipColors(
                            selectedContainerColor = MaterialTheme.colorScheme
                                .primary.copy(alpha = 0.18f),
                        ),
                    )
                }

                when {
                    isLoading -> OffersSkeleton(layout = layoutMode)
                    displayedItems.isEmpty() -> {
                        Text(
                            text = "Подходящих предложений не найдено.",
                            style = MaterialTheme.typography.bodyMedium,
                            color = MaterialTheme.colorScheme.onSurfaceVariant,
                            modifier = Modifier.padding(vertical = 24.dp),
                        )
                    }

                    else -> {
                        when (layoutMode) {
                            OffersLayoutMode.Horizontal -> {
                                OffersHorizontalList(
                                    items = displayedItems,
                                    userCountry = userCountryLabel,
                                    modifier = Modifier.fillMaxWidth(),
                                    onSelect = { selectedItem = it },
                                )
                            }

                            OffersLayoutMode.Vertical -> {
                                OffersVerticalList(
                                    items = displayedItems,
                                    userCountry = userCountryLabel,
                                    modifier = Modifier.fillMaxWidth(),
                                    onSelect = { selectedItem = it },
                                )
                            }
                        }
                    }
                }

                selectedItem?.let { item ->
                    Spacer(Modifier.height(12.dp))
                    OfferDetailsPanel(
                        item = item,
                        onClose = { selectedItem = null },
                        alertMessage = alertMessage,
                        alertIsError = alertIsError,
                        onAlertRequest = { percent ->
                            scope.launch {
                                when (val result = createAlert(item.dto.id, percent)) {
                                    OfferAlertResult.Success -> {
                                        alertMessage = "Подписка оформлена"
                                        alertIsError = false
                                    }

                                    OfferAlertResult.Unauthorized -> {
                                        alertMessage = "Нужна авторизация, чтобы оформить подписку"
                                        alertIsError = true
                                    }

                                    is OfferAlertResult.Error -> {
                                        alertMessage = "Не удалось оформить подписку: ${result.reason}"
                                        alertIsError = true
                                    }
                                }
                            }
                        },
                    )
                }

                Spacer(Modifier.height(12.dp))
            }
        }
    }

    if (visible && minimized) {
        CollapsedTab(onRestore = { minimized = false }, onClose = onDismiss)
    }
}

@Composable
private fun OffersSkeleton(
    layout: OffersLayoutMode,
    modifier: Modifier = Modifier,
) {
    val shape = RoundedCornerShape(20.dp)
    val baseColor = MaterialTheme.colorScheme.surfaceVariant.copy(alpha = 0.6f)

    when (layout) {
        OffersLayoutMode.Horizontal -> {
            LazyRow(
                modifier = modifier
                    .fillMaxWidth()
                    .padding(vertical = 4.dp),
                horizontalArrangement = Arrangement.spacedBy(12.dp),
                contentPadding = PaddingValues(horizontal = 4.dp),
            ) {
                items(3) {
                    Box(
                        modifier = Modifier
                            .width(260.dp)
                            .height(150.dp)
                            .clip(shape)
                            .background(baseColor),
                    )
                }
            }
        }

        OffersLayoutMode.Vertical -> {
            Column(
                modifier = modifier
                    .fillMaxWidth()
                    .padding(vertical = 4.dp),
                verticalArrangement = Arrangement.spacedBy(8.dp),
            ) {
                repeat(3) {
                    Box(
                        modifier = Modifier
                            .fillMaxWidth()
                            .height(120.dp)
                            .clip(shape)
                            .background(baseColor),
                    )
                }
            }
        }
    }
}

@Composable
private fun OffersHorizontalList(
    items: List<ExplainedItem>,
    userCountry: String?,
    modifier: Modifier = Modifier,
    onSelect: (ExplainedItem) -> Unit,
) {
    LazyRow(
        modifier = modifier.padding(vertical = 4.dp),
        horizontalArrangement = Arrangement.spacedBy(12.dp),
        contentPadding = PaddingValues(horizontal = 4.dp, vertical = 4.dp),
    ) {
        items(items = items, key = { it.dto.id }) { item ->
            OfferCardHorizontal(item = item, userCountry = userCountry, onClick = { onSelect(item) })
        }
    }
}

@Composable
private fun OfferCardHorizontal(
    item: ExplainedItem,
    userCountry: String?,
    modifier: Modifier = Modifier,
    onClick: () -> Unit = {},
) {
    val dto = item.dto
    var expanded by remember { mutableStateOf(false) }
    val contentSpacing = if (expanded) 8.dp else 6.dp
    val cardMinHeight = if (expanded) 360.dp else 300.dp
    val gradient = Brush.linearGradient(
        listOf(
            MaterialTheme.colorScheme.primary.copy(alpha = 0.16f),
            MaterialTheme.colorScheme.secondary.copy(alpha = 0.12f),
        )
    )

    Card(
        modifier = modifier
            .width(260.dp)
            .heightIn(min = cardMinHeight)
            .clickable(onClick = onClick),
        shape = RoundedCornerShape(20.dp),
        colors = CardDefaults.cardColors(containerColor = Color.Transparent),
    ) {
        BoxWithConstraints(
            modifier = Modifier
                .background(gradient)
                .padding(12.dp),
        ) {
            val sliderHeight =
                if (maxHeight.value.isFinite()) maxHeight * 0.5f else cardMinHeight * 0.5f
            Column(
                verticalArrangement = Arrangement.spacedBy(contentSpacing),
            ) {
                SliderStub(
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(sliderHeight),
                )

                Text(
                    text = dto.title,
                    style = MaterialTheme.typography.titleMedium,
                    maxLines = 2,
                )

                Row(
                    verticalAlignment = Alignment.CenterVertically,
                    horizontalArrangement = Arrangement.spacedBy(contentSpacing),
                    modifier = Modifier.fillMaxWidth(),
                ) {
                            ProfileRow(
                                name = dto.brand ?: dto.model ?: "Продавец",
                                contentColor = MaterialTheme.colorScheme.onSurfaceVariant,
                            )
                    Box(
                        modifier = Modifier.weight(1f),
                        contentAlignment = Alignment.CenterEnd,
                    ) {
                        PriceRow(
                            price = dto.price,
                            contentColor = MaterialTheme.colorScheme.onSurface,
                            emphasize = true,
                        )
                    }
                }

                DeliveryRow(
                    deliveryAvailable = dto.deliveryTime != null,
                    userCountry = userCountry,
                    contentColor = MaterialTheme.colorScheme.onSurfaceVariant,
                )

                LocationRow(
                    country = dto.sellerCountry,
                    city = dto.sellerCity,
                    contentColor = MaterialTheme.colorScheme.onSurfaceVariant,
                )

                StarsRatingRow(
                    rating = dto.sellerRating,
                    reviews = null,
                    contentColor = MaterialTheme.colorScheme.onSurfaceVariant,
                )

                SellerBadgesRow(
                    sellerBadges = dto.sellerBadges,
                    shippingCountries = dto.sellerShippingCountries,
                    userCountry = userCountry,
                )

                if (expanded) {
                    OfferTabs(item = item)
                }

                CardActions(expanded = expanded, onToggleExpand = { expanded = !expanded })
            }
        }
    }
}

@Composable
private fun OffersVerticalList(
    items: List<ExplainedItem>,
    userCountry: String?,
    modifier: Modifier = Modifier,
    onSelect: (ExplainedItem) -> Unit,
) {
    LazyColumn(
        modifier = modifier.padding(vertical = 4.dp),
        verticalArrangement = Arrangement.spacedBy(10.dp),
        contentPadding = PaddingValues(bottom = 16.dp),
    ) {
        itemsIndexed(items = items, key = { _, item -> item.dto.id }) { index, item ->
            OfferCardVertical(
                item = item,
                isTop = index == 0,
                userCountry = userCountry,
                onClick = { onSelect(item) },
            )
        }
    }
}

@Composable
private fun PriceRow(
    price: Double?,
    contentColor: Color,
    modifier: Modifier = Modifier,
    emphasize: Boolean = false,
) {
    Row(
        modifier = modifier,
        verticalAlignment = Alignment.CenterVertically,
        horizontalArrangement = Arrangement.spacedBy(6.dp),
    ) {
        Icon(
            imageVector = Icons.Outlined.AttachMoney,
            contentDescription = "Стоимость",
            tint = contentColor,
        )
        Text(
            text = price?.let { "%.0f".format(it) } ?: "—",
            style = if (emphasize) MaterialTheme.typography.titleLarge else MaterialTheme.typography.titleMedium,
            color = contentColor,
        )
    }
}

@Composable
private fun DeliveryRow(
    deliveryAvailable: Boolean,
    userCountry: String?,
    contentColor: Color,
    modifier: Modifier = Modifier,
) {
    Row(
        modifier = modifier,
        verticalAlignment = Alignment.CenterVertically,
        horizontalArrangement = Arrangement.spacedBy(6.dp),
    ) {
        Icon(
            imageVector = Icons.Outlined.LocalShipping,
            contentDescription = "Доставка",
            tint = contentColor,
        )
        Text(
            text = if (deliveryAvailable) {
                "Доставка в ${userCountry ?: "вашу страну"}"
            } else {
                "Нет доставки по выбранной стране"
            },
            style = MaterialTheme.typography.bodyMedium,
            color = contentColor,
        )
        if (deliveryAvailable) {
            Icon(
                imageVector = Icons.Outlined.Check,
                contentDescription = "Доступно",
                tint = contentColor,
            )
        }
    }
}

) {
    Box(
        modifier = modifier
            .clip(RoundedCornerShape(14.dp))
            .background(MaterialTheme.colorScheme.surfaceVariant.copy(alpha = 0.6f)),
        contentAlignment = Alignment.Center,
    ) {
        Image(
            imageVector = Icons.Outlined.Image,
            contentDescription = "Слайдер изображений",
            alpha = 0.75f,
        )
    }
}

@Composable
private fun CardActions(
    expanded: Boolean,
    onToggleExpand: () -> Unit,
    modifier: Modifier = Modifier,
    contentColor: Color = MaterialTheme.colorScheme.onSurface,

@Composable
private fun StarsRatingRow(
    rating: Double?,
    reviews: Int?,
    contentColor: Color,
) {
    val hasRating = rating != null
    val value = (rating ?: 0.0).coerceIn(0.0, 5.0)
    val fullStars = value.toInt()
    val hasHalf = (value - fullStars) in 0.25..0.74
    val totalStars = 5

    Row(
        verticalAlignment = Alignment.CenterVertically,
        horizontalArrangement = Arrangement.spacedBy(6.dp),
    ) {
        Row(horizontalArrangement = Arrangement.spacedBy(2.dp)) {
            repeat(fullStars) {
                Icon(
                    imageVector = Icons.Rounded.Star,
                    contentDescription = null,
                    tint = MaterialTheme.colorScheme.primary,
                )
            }
            if (hasHalf) {
                Icon(
                    imageVector = Icons.AutoMirrored.Rounded.StarHalf,
                    contentDescription = null,
                    tint = MaterialTheme.colorScheme.primary,
                )
            }
            repeat(totalStars - fullStars - if (hasHalf) 1 else 0) {
                Icon(
                    imageVector = Icons.Rounded.StarOutline,
                    contentDescription = null,
                    tint = MaterialTheme.colorScheme.onSurfaceVariant,
                )
            }
        }
        Text(
            text = if (hasRating) "%.1f/5.0".format(value) else "Нет рейтинга",
            style = MaterialTheme.typography.bodyMedium,
            color = contentColor,
        )
        if (hasRating && reviews != null) {
            Text(
                text = "• $reviews отзывов",
                style = MaterialTheme.typography.bodySmall,
                color = contentColor,
            )
        }
    }
}

@Composable
private fun SellerBadgesRow(
    sellerBadges: List<String>,
    shippingCountries: List<String>,
    userCountry: String?,
    modifier: Modifier = Modifier,
) {
    val badges = buildList {
        if (userCountry != null && shippingCountries.any { it.equals(userCountry, ignoreCase = true) }) {
            add("Доставка в $userCountry")
        }
        addAll(sellerBadges)
    }
    if (badges.isEmpty()) return

    val scroll = rememberScrollState()
    var autoScroll by remember { mutableStateOf(true) }
    val scope = rememberCoroutineScope()

    LaunchedEffect(badges, autoScroll) {
        while (autoScroll && badges.size > 1) {
            val maxScroll = (scroll.maxValue.takeIf { it > 0 } ?: 0)
            if (maxScroll > 0) {
                val target = if (scroll.value >= maxScroll) 0 else scroll.value + 120
                scroll.animateScrollTo(target, animationSpec = tween(durationMillis = 1800))
            }
            delay(900)
        }
    }

    Row(
        modifier = modifier
            .horizontalScroll(scroll)
            .pointerInput(badges) {
                detectHorizontalDragGestures(
                    onDragStart = { autoScroll = false },
                    onDragEnd = { autoScroll = true },
                    onDragCancel = { autoScroll = true },
                ) { _, delta ->
                    scope.launch { scroll.scrollBy(-delta) }
                }
            },
        horizontalArrangement = Arrangement.spacedBy(8.dp),
        verticalAlignment = Alignment.CenterVertically,
    ) {
        badges.forEach { badge ->
            Text(
                text = badge.replace('_', ' '),
                style = MaterialTheme.typography.labelMedium,
                color = MaterialTheme.colorScheme.onSurface,
                modifier = Modifier
                    .background(
                        color = MaterialTheme.colorScheme.primary.copy(alpha = 0.15f),
                        shape = RoundedCornerShape(10.dp),
                    )
                    .padding(horizontal = 8.dp, vertical = 4.dp),
            )
        }
    }
}

@Composable

    label: String,
    icon: ImageVector,
    selected: Boolean,
    onClick: () -> Unit,
) {
    val bg = if (selected) {
        MaterialTheme.colorScheme.primary.copy(alpha = 0.18f)
    } else {
        MaterialTheme.colorScheme.surface.copy(alpha = 0.65f)
    }
    Row(
        verticalAlignment = Alignment.CenterVertically,
        modifier = Modifier
            .background(bg, RoundedCornerShape(12.dp))
            .clip(RoundedCornerShape(12.dp))
            .clickable(onClick = onClick)
            .padding(horizontal = 12.dp, vertical = 8.dp),
        horizontalArrangement = Arrangement.spacedBy(8.dp),
    ) {
        Icon(
            imageVector = icon,
            contentDescription = null,
            tint = MaterialTheme.colorScheme.primary,
        )
        Text(
            text = label,
            style = MaterialTheme.typography.bodyMedium,
            color = MaterialTheme.colorScheme.onSurface,
        )
    }
}
