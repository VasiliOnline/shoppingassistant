@Composable
fun PhotoBox(
    photoUrl: String?,
    placeholderCategory: String?,
    templateMode: TemplateMode,
    linkMeta: TemplateLinkMeta?,
    width: Dp = 170.dp,
    height: Dp = 220.dp,
    onAddPhoto: () -> Unit,
    onOpenSource: (() -> Unit)? = null,
    onDetachSource: (() -> Unit)? = null,
    onRemovePhoto: (() -> Unit)? = null,
    onChangePhoto: (() -> Unit)? = null,
    modifier: Modifier = Modifier,
) {
    val contentPadding = if (photoUrl.isNullOrBlank()) 0.dp else 6.dp
    Surface(
        modifier = modifier
            .size(width = width, height = height)
            .background(Color.Transparent),
        shape = RoundedCornerShape(16.dp),
        tonalElevation = 0.dp,
        shadowElevation = 0.dp,
        // Временно убираем фон, блок оставляем прозрачным
        color = Color.Transparent,
    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(contentPadding),
            verticalArrangement = Arrangement.spacedBy(8.dp),
            horizontalAlignment = Alignment.CenterHorizontally,
        ) {
            if (photoUrl.isNullOrBlank()) {
                PlaceholderBox(
                    placeholderCategory = placeholderCategory,
                    onAddPhoto = onAddPhoto,
                    templateMode = templateMode,
                    linkMeta = linkMeta,
                )
            } else {
                LoadedPhotoBox(
                    photoUrl = photoUrl,
                    templateMode = templateMode,
                    linkMeta = linkMeta,
                    onOpenSource = onOpenSource,
                    onDetachSource = onDetachSource,
                    onRemovePhoto = onRemovePhoto,
                    onChangePhoto = onChangePhoto,
                )
            }
        }
    }
}

@Composable
@Suppress("UNUSED_PARAMETER")
private fun PlaceholderBox(
    placeholderCategory: String?,
    onAddPhoto: () -> Unit,
    templateMode: TemplateMode,
    linkMeta: TemplateLinkMeta?,
) {
    val placeholderLabel = "Добавить фото"

    Box(
        modifier = Modifier
            .fillMaxWidth()
            .height(160.dp)
            .background(
                color = MaterialTheme.colorScheme.surface.copy(alpha = 0.85f),
                shape = RoundedCornerShape(12.dp),
            )
            .clickable { onAddPhoto() },
        contentAlignment = Alignment.Center,
    ) {
        Column(
            horizontalAlignment = Alignment.CenterHorizontally,
            verticalArrangement = Arrangement.spacedBy(6.dp),
        ) {
            Icon(
                imageVector = Icons.Outlined.PhotoCamera,
                contentDescription = null,
                tint = MaterialTheme.colorScheme.onSurfaceVariant.copy(alpha = 0.7f),
                modifier = Modifier.size(32.dp),
            )
            Text(
                text = placeholderLabel,
                style = MaterialTheme.typography.bodyMedium,
                color = MaterialTheme.colorScheme.onSurfaceVariant.copy(alpha = 0.75f),
                textAlign = TextAlign.Center,
            )
        }
    }
}

@Composable
private fun LoadedPhotoBox(
    photoUrl: String,
    templateMode: TemplateMode,
    linkMeta: TemplateLinkMeta?,
    onOpenSource: (() -> Unit)?,
    onDetachSource: (() -> Unit)?,
    onRemovePhoto: (() -> Unit)? = null,
    onChangePhoto: (() -> Unit)? = null,
) {
    Column(
        modifier = Modifier.fillMaxWidth(),
        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = Arrangement.spacedBy(6.dp),
    ) {
        Surface(
            shape = RoundedCornerShape(12.dp),
            tonalElevation = 0.dp,
            shadowElevation = 0.dp,
            color = Color.Transparent,
        ) {
            AsyncImage(
                model = photoUrl,
                contentDescription = null,
                modifier = Modifier
                    .size(width = 140.dp, height = 140.dp)
                    .clickable { onChangePhoto?.invoke() },
            )
        }

        val sourceLabel = linkMeta?.domainName
            ?: linkMeta?.sourceType?.name?.lowercase()
            ?: when (templateMode) {
                TemplateMode.ExpressFromPhoto -> "Фото пользователя"
                TemplateMode.OfferFromLink -> "Ссылка"
                else -> null
            }
        sourceLabel?.let { label ->
            SourceBadge(
                text = "Источник: $label",
                onClick = onOpenSource,
            )
        }

        if (linkMeta != null && onDetachSource != null) {
            OutlinedButton(onClick = onDetachSource) {
                Text("Отвязать от ссылки")
            }
        }
        if (onChangePhoto != null) {
            OutlinedButton(onClick = onChangePhoto, contentPadding = PaddingValues(vertical = 6.dp, horizontal = 10.dp)) {
                Text("Сменить фото")
            }
        }
        if (onRemovePhoto != null) {
            OutlinedButton(onClick = onRemovePhoto, contentPadding = PaddingValues(vertical = 6.dp, horizontal = 10.dp)) {
                Text("Удалить фото")
            }
        }
        if (linkMeta != null && onOpenSource != null) {
            Button(onClick = onOpenSource, contentPadding = PaddingValues(horizontal = 12.dp, vertical = 6.dp)) {
                Icon(Icons.Outlined.Link, contentDescription = null, modifier = Modifier.padding(end = 6.dp))
                Text("Открыть объявление")
            }
        }
    }
}

@Composable
private fun SourceBadge(
    text: String,
    onClick: (() -> Unit)?,
) {
    Surface(
        shape = RoundedCornerShape(10.dp),
        color = MaterialTheme.colorScheme.surface.copy(alpha = 0.9f),
        tonalElevation = 0.dp,
        shadowElevation = 0.dp,
    ) {
        Row(
            modifier = Modifier
                .padding(horizontal = 10.dp, vertical = 6.dp)
                .let { base -> if (onClick != null) base.clickable { onClick() } else base },
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.spacedBy(8.dp),
        ) {
            Icon(
                imageVector = if (onClick != null) Icons.Outlined.Link else Icons.Outlined.BrokenImage,
                contentDescription = null,
                tint = MaterialTheme.colorScheme.onSurfaceVariant,
            )
            Text(
                text = text,
                style = MaterialTheme.typography.bodySmall.copy(fontWeight = FontWeight.SemiBold),
                color = MaterialTheme.colorScheme.onSurface,
            )
        }
    }
}
