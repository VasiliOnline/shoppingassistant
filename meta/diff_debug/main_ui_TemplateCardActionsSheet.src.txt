@Composable
fun TemplateCardActionsSheet(
    visible: Boolean,
    title: String,
    card: TemplateCardUi?,
    anchorBounds: Rect?,
    foundCount: Int?,
    missingFields: List<String>,
    onShowOffers: () -> Unit,
    onCreateExpress: () -> Unit,
    onSubscribe: () -> Unit,
    onDismiss: () -> Unit,
) {
    if (!visible) return
    Dialog(
        onDismissRequest = onDismiss,
        properties = DialogProperties(usePlatformDefaultWidth = false),
    ) {
        val view = LocalView.current
        SideEffect {
            val window = (view.parent as? DialogWindowProvider)?.window
            window?.setDimAmount(0f)
            window?.setBackgroundDrawable(ColorDrawable(android.graphics.Color.TRANSPARENT))
        }
        val colors = MaterialTheme.colorScheme
        BoxWithConstraints(modifier = Modifier.fillMaxSize()) {
            val density = LocalDensity.current
            val containerHeightPx = constraints.maxHeight.toFloat()
            val containerWidthPx = constraints.maxWidth.toFloat()
            val containerCenterY = containerHeightPx / 2f
            val cardScale = 1f
            val anchorCenterY = anchorBounds?.center?.y ?: containerCenterY
            val showAbove = anchorCenterY > containerCenterY
            var menuHeightPx by remember { mutableIntStateOf(0) }
            var menuWidthPx by remember { mutableIntStateOf(0) }
            val gapPx = with(density) { 16.dp.toPx() }
            val scaledTop = anchorBounds?.let {
                it.top - (it.height * (cardScale - 1f) / 2f)
            } ?: anchorCenterY
            val scaledBottom = anchorBounds?.let {
                it.bottom + (it.height * (cardScale - 1f) / 2f)
            } ?: anchorCenterY
            val rawMenuY = if (anchorBounds == null) {
                containerCenterY - menuHeightPx / 2f
            } else if (showAbove) {
                scaledTop - menuHeightPx - gapPx
            } else {
                scaledBottom + gapPx
            }
            val clampedMenuY = rawMenuY.coerceIn(0f, (containerHeightPx - menuHeightPx).coerceAtLeast(0f))
            val menuLeftPx = ((containerWidthPx - menuWidthPx) / 2f).coerceAtLeast(0f)

            Box(modifier = Modifier.fillMaxSize()) {
                val dismissInteraction = remember { MutableInteractionSource() }
                Box(
                    modifier = Modifier
                        .fillMaxSize()
                        .clickable(
                            interactionSource = dismissInteraction,
                            indication = null,
                        ) { onDismiss() },
                )
                if (anchorBounds != null && menuWidthPx > 0 && menuHeightPx > 0) {
                    val paddingPx = with(density) { 10.dp.toPx() }
                    val menuTopPx = clampedMenuY
                    val menuBottomPx = clampedMenuY + menuHeightPx
                    val menuRightPx = menuLeftPx + menuWidthPx
                    val left = min(anchorBounds.left, menuLeftPx) - paddingPx
                    val right = max(anchorBounds.right, menuRightPx) + paddingPx
                    val top = min(anchorBounds.top, menuTopPx) - paddingPx
                    val bottom = max(anchorBounds.bottom, menuBottomPx) + paddingPx
                    val clampedLeft = left.coerceIn(0f, containerWidthPx)
                    val clampedRight = right.coerceIn(0f, containerWidthPx)
                    val clampedTop = top.coerceIn(0f, containerHeightPx)
                    val clampedBottom = bottom.coerceIn(0f, containerHeightPx)
                    val widthPx = (clampedRight - clampedLeft).coerceAtLeast(0f)
                    val heightPx = (clampedBottom - clampedTop).coerceAtLeast(0f)
                    Box(
                        modifier = Modifier
                            .offset { IntOffset(clampedLeft.roundToInt(), clampedTop.roundToInt()) }
                            .size(
                                width = with(density) { widthPx.toDp() },
                                height = with(density) { heightPx.toDp() },
                            )
                            .background(
                                colors.surface.copy(alpha = 0.16f),
                                RoundedCornerShape(22.dp),
                            ),
                    )
                }
                if (card != null && anchorBounds != null) {
                    val widthDp = with(density) { anchorBounds.width.toDp() }
                    val heightDp = with(density) { anchorBounds.height.toDp() }
                    val offsetXPx = anchorBounds.left - (anchorBounds.width * (cardScale - 1f) / 2f)
                    val offsetYPx = anchorBounds.top - (anchorBounds.height * (cardScale - 1f) / 2f)
                    val consumeInteraction = remember { MutableInteractionSource() }
                    Box(
                        modifier = Modifier
                            .offset {
                                IntOffset(
                                    offsetXPx.roundToInt(),
                                    offsetYPx.roundToInt(),
                                )
                            }
                            .size(widthDp, heightDp)
                            .graphicsLayer {
                                scaleX = cardScale
                                scaleY = cardScale
                            }
                            .clickable(
                                interactionSource = consumeInteraction,
                                indication = null,
                            ) { },
                    ) {
                        TemplateCardPreview(card = card, modifier = Modifier.fillMaxSize())
                    }
                }
                Column(
                    modifier = Modifier
                        .clickable(
                            interactionSource = remember { MutableInteractionSource() },
                            indication = null,
                        ) { }
                        .align(Alignment.TopCenter)
                        .offset { IntOffset(0, clampedMenuY.roundToInt()) }
                        .fillMaxWidth(0.82f)
                        .padding(horizontal = 16.dp, vertical = 12.dp)
                        .onSizeChanged {
                            menuHeightPx = it.height
                            menuWidthPx = it.width
                        },
                    verticalArrangement = Arrangement.spacedBy(10.dp),
                    horizontalAlignment = Alignment.CenterHorizontally,
                ) {
                    Text(
                        text = title.ifBlank { "Шаблон товара" },
                        style = MaterialTheme.typography.titleSmall,
                        color = colors.onSurface,
                        textAlign = TextAlign.Center,
                    )

                    ActionButton(
                        text = "Показать предложения",
                        icon = Icons.Outlined.Search,
                        enabled = (foundCount ?: 0) > 0,
                        onClick = onShowOffers,
                    )
                    Text(
                        text = "В базе: ${foundCount ?: 0} предложений",
                        style = MaterialTheme.typography.bodySmall,
                        color = colors.onSurfaceVariant,
                        textAlign = TextAlign.Center,
                    )

                    ActionButton(
                        text = "Разместить объявление",
                        icon = Icons.Outlined.FlashOn,
                        enabled = true,
                        isActive = missingFields.isEmpty(),
                        onClick = onCreateExpress,
                    )

                    ActionButton(
                        text = "Отслеживать товар",
                        icon = Icons.Outlined.NotificationsNone,
                        onClick = onSubscribe,
                    )

                    TextButton(onClick = onDismiss) {
                        Text("Закрыть")
                    }
                }
            }
        }
    }
}

@Composable
private fun ActionButton(
    text: String,
    icon: ImageVector,
    onClick: () -> Unit,
    modifier: Modifier = Modifier,
    enabled: Boolean = true,
    isActive: Boolean = enabled,
) {
    val colors = MaterialTheme.colorScheme
    val containerColor = if (isActive) {
        colors.primary.copy(alpha = 0.72f)
    } else {
        colors.primary.copy(alpha = 0.32f)
    }
    val contentColor = if (isActive) {
        Color.White
    } else {
        colors.onSurfaceVariant
    }
    Button(
        onClick = onClick,
        enabled = enabled,
        shape = RoundedCornerShape(16.dp),
        colors = ButtonDefaults.buttonColors(
            containerColor = containerColor,
            contentColor = contentColor,
            disabledContainerColor = containerColor,
            disabledContentColor = contentColor,
        ),
        contentPadding = PaddingValues(horizontal = 16.dp, vertical = 10.dp),
        modifier = modifier.fillMaxWidth(0.76f),
    ) {
        Icon(
            imageVector = icon,
            contentDescription = null,
            modifier = Modifier.size(16.dp),
        )
        Spacer(Modifier.size(8.dp))
        Text(text)
    }
}
