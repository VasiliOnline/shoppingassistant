@Composable
fun TemplateListsPanel(
    lists: List<TemplateListMeta>,
    activeListId: String,
    cards: List<TemplateCardUi>,
    selectedCardId: String?,
    onListClick: (String) -> Unit,
    onActiveListClick: (TemplateListMeta) -> Unit,
    onCreateListClick: () -> Unit,
    onCardClick: (TemplateCardUi, Rect) -> Unit,
    onCardEdit: (TemplateCardUi) -> Unit,
    onCardDelete: (TemplateCardUi) -> Unit,
    modifier: Modifier = Modifier,
) {
    Column(
        modifier = modifier,
        verticalArrangement = Arrangement.spacedBy(8.dp),
    ) {
        Row(
            modifier = Modifier.fillMaxWidth(),
            verticalAlignment = Alignment.CenterVertically,
        ) {
            LazyRow(
                modifier = Modifier.weight(1f),
                horizontalArrangement = Arrangement.spacedBy(8.dp),
                verticalAlignment = Alignment.Bottom,
            ) {
                item {
                    TemplateListChip(
                        title = "Недавние",
                        active = activeListId == RECENT_TEMPLATES_LIST_ID,
                        onClick = { onListClick(RECENT_TEMPLATES_LIST_ID) },
                    )
                }
                items(lists) { list ->
                    val isActive = activeListId == list.listId
                    TemplateListChip(
                        title = list.title,
                        active = isActive,
                        onClick = {
                            if (isActive) onActiveListClick(list) else onListClick(list.listId)
                        },
                    )
                }
                item {
                    NewListIconButton(onClick = onCreateListClick)
                }
            }
        }

        if (cards.isEmpty()) {
            Text(
                text = "Нет сохранённых шаблонов",
                style = MaterialTheme.typography.bodyMedium,
                color = MaterialTheme.colorScheme.onSurfaceVariant,
                modifier = Modifier.padding(start = 4.dp),
            )
            return
        }

        VerticalTemplateCards(
            cards = cards,
            selectedCardId = selectedCardId,
            onCardClick = onCardClick,
            onCardEdit = onCardEdit,
            onCardDelete = onCardDelete,
        )
    }
}

@Composable
private fun TemplateListChip(
    title: String,
    active: Boolean,
    onClick: () -> Unit,
) {
    val colors = MaterialTheme.colorScheme
    val textColor = if (active) colors.onSurface else colors.onSurfaceVariant
    val underlineColor = if (active) Color.White else Color.Transparent
    val density = LocalDensity.current
    var textWidthPx by remember { mutableIntStateOf(0) }
    val underlineWidth = with(density) {
        if (textWidthPx > 0) textWidthPx.toDp() else 18.dp
    }
    Column(
        modifier = Modifier
            .clickable { onClick() }
            .padding(vertical = 4.dp),
        horizontalAlignment = Alignment.CenterHorizontally,
    ) {
        Text(
            text = title,
            style = MaterialTheme.typography.labelLarge,
            color = textColor,
            maxLines = 1,
            overflow = TextOverflow.Ellipsis,
            modifier = Modifier.onGloballyPositioned { textWidthPx = it.size.width },
        )
        Box(
            modifier = Modifier
                .padding(top = 4.dp)
                .heightIn(min = 2.dp)
                .width(underlineWidth)
                .background(underlineColor, RoundedCornerShape(2.dp)),
        )
    }
}

@Composable
private fun VerticalTemplateCards(
    cards: List<TemplateCardUi>,
    selectedCardId: String?,
    onCardClick: (TemplateCardUi, Rect) -> Unit,
    onCardEdit: (TemplateCardUi) -> Unit,
    onCardDelete: (TemplateCardUi) -> Unit,
) {
    LazyColumn(
        modifier = Modifier
            .fillMaxWidth()
            .heightIn(max = 320.dp),
        verticalArrangement = Arrangement.spacedBy(10.dp),
    ) {
        items(cards) { card ->
            val selected = selectedCardId == card.templateId
            TemplateMiniCard(
                card = card,
                compact = false,
                selected = selected,
                onClick = { bounds -> onCardClick(card, bounds) },
                onEdit = { onCardEdit(card) },
                onDelete = { onCardDelete(card) },
            )
        }
    }
}

@Composable
fun TemplateCardPreview(
    card: TemplateCardUi,
    modifier: Modifier = Modifier,
) {
    val shape = RoundedCornerShape(14.dp)
    Surface(
        shape = shape,
        color = Color.Transparent,
        tonalElevation = 0.dp,
        shadowElevation = 0.dp,
        modifier = modifier,
    ) {
        TemplateCardBody(
            card = card,
            compact = false,
            selected = false,
            onEdit = null,
            onDelete = null,
            modifier = Modifier.fillMaxWidth(),
        )
    }
}

@Composable
private fun TemplateMiniCard(
    card: TemplateCardUi,
    compact: Boolean,
    selected: Boolean,
    onClick: (Rect) -> Unit,
    onEdit: () -> Unit,
    onDelete: () -> Unit,
    modifier: Modifier = Modifier,
) {
    val colors = MaterialTheme.colorScheme
    val shape = RoundedCornerShape(14.dp)
    var fingerprintTrigger by remember { mutableIntStateOf(0) }
    var cardBounds by remember { mutableStateOf<Rect?>(null) }
    Surface(
        shape = shape,
        color = Color.Transparent,
        tonalElevation = 0.dp,
        shadowElevation = 0.dp,
        modifier = modifier
            .fillMaxWidth()
            .onGloballyPositioned { cardBounds = it.boundsInRoot() }
            .clickable {
                fingerprintTrigger += 1
                onClick(cardBounds ?: Rect(0f, 0f, 0f, 0f))
            },
    ) {
        Box {
            TemplateCardBody(
                card = card,
                compact = compact,
                selected = selected,
                onEdit = onEdit,
                onDelete = onDelete,
                modifier = Modifier.fillMaxWidth(),
            )
            FingerprintPulse(
                trigger = fingerprintTrigger,
                modifier = Modifier
                    .align(Alignment.Center)
                    .size(width = 30.dp, height = 24.dp)
                    .zIndex(1f),
            )
        }
    }
}

@Composable
private fun TemplateCardBody(
    card: TemplateCardUi,
    compact: Boolean,
    selected: Boolean,
    onEdit: (() -> Unit)?,
    onDelete: (() -> Unit)?,
    modifier: Modifier = Modifier,
) {
    val colors = MaterialTheme.colorScheme
    val shape = RoundedCornerShape(14.dp)
    val highlight = if (selected) {
        colors.primary.copy(alpha = 0.08f)
    } else {
        Color.Transparent
    }
    Row(
        modifier = modifier
            .height(IntrinsicSize.Min)
            .background(highlight, shape)
            .let { base ->
                if (selected) {
                    base.border(1.dp, colors.primary.copy(alpha = 0.2f), shape)
                } else {
                    base
                }
            }
            .padding(vertical = 2.dp),
        verticalAlignment = Alignment.CenterVertically,
    ) {
        Box(
            modifier = Modifier
                .fillMaxHeight()
                .aspectRatio(1f)
                .border(1.dp, colors.outline.copy(alpha = 0.4f), RoundedCornerShape(12.dp)),
            contentAlignment = Alignment.Center,
        ) {
            Icon(
                imageVector = Icons.Outlined.Image,
                contentDescription = null,
                tint = colors.onSurfaceVariant,
                modifier = Modifier.size(20.dp),
            )
        }
        Column(
            modifier = Modifier
                .weight(1f)
                .padding(start = 12.dp, top = 10.dp, bottom = 10.dp, end = 8.dp),
            verticalArrangement = Arrangement.spacedBy(4.dp),
        ) {
            Text(
                text = card.title,
                style = MaterialTheme.typography.bodyMedium,
                color = colors.onSurface,
                maxLines = 1,
                overflow = TextOverflow.Ellipsis,
            )
            Text(
                text = formatCardDate(card.createdAtMillis),
                style = MaterialTheme.typography.labelSmall,
                color = colors.onSurfaceVariant,
            )
            if (selected && onEdit != null) {
                Surface(
                    shape = RoundedCornerShape(10.dp),
                    color = Color.Transparent,
                    tonalElevation = 0.dp,
                    shadowElevation = 0.dp,
                    modifier = Modifier
                        .border(1.dp, colors.outline.copy(alpha = 0.4f), RoundedCornerShape(10.dp))
                        .clickable { onEdit() },
                ) {
                    Text(
                        text = "Редактировать",
                        style = MaterialTheme.typography.labelMedium,
                        color = colors.onSurface,
                        modifier = Modifier.padding(horizontal = 10.dp, vertical = 4.dp),
                    )
                }
            }
        }
        if (selected && onDelete != null) {
            Icon(
                imageVector = Icons.Outlined.Delete,
                contentDescription = "Удалить",
                tint = colors.onSurfaceVariant,
                modifier = Modifier
                    .size(32.dp)
                    .padding(end = 12.dp)
                    .clickable { onDelete() },
            )
        } else {
            Spacer(modifier = Modifier.width(12.dp))
        }
    }
}

@Composable
private fun NewListIconButton(
    onClick: () -> Unit,
) {
    val interaction = remember { MutableInteractionSource() }
    val pressed by interaction.collectIsPressedAsState()
    val alpha by remember(pressed) { derivedStateOf { if (pressed) 0.6f else 1f } }
    Column(
        modifier = Modifier.padding(vertical = 4.dp),
        horizontalAlignment = Alignment.CenterHorizontally,
    ) {
        Icon(
            imageVector = Icons.Outlined.Add,
            contentDescription = "Создать список",
            tint = Color.White.copy(alpha = alpha),
            modifier = Modifier
                .size(20.dp)
                .clickable(interactionSource = interaction, indication = null) { onClick() },
        )
        Box(
            modifier = Modifier
                .padding(top = 6.dp)
                .heightIn(min = 2.dp)
                .width(18.dp),
        )
    }
}

@Composable
private fun FingerprintPulse(
    trigger: Int,
    modifier: Modifier = Modifier,
) {
    if (trigger == 0) return
    val colors = MaterialTheme.colorScheme
    val alphaAnim = remember { androidx.compose.animation.core.Animatable(0f) }
    var visible by remember { mutableStateOf(false) }

    LaunchedEffect(trigger) {
        if (trigger == 0) return@LaunchedEffect
        visible = true
        alphaAnim.snapTo(0.18f)
        alphaAnim.animateTo(
            0.55f,
            animationSpec = androidx.compose.animation.core.tween(durationMillis = 700),
        )
        alphaAnim.animateTo(
            0.75f,
            animationSpec = androidx.compose.animation.core.tween(durationMillis = 600),
        )
        alphaAnim.animateTo(
            0.12f,
            animationSpec = androidx.compose.animation.core.tween(durationMillis = 700),
        )
        visible = false
    }

    if (!visible) return

    Icon(
        imageVector = Icons.Outlined.Fingerprint,
        contentDescription = null,
        tint = colors.primary.copy(alpha = alphaAnim.value),
        modifier = modifier,
    )
}

private fun formatCardDate(timestamp: Long): String {
    val formatter = SimpleDateFormat("d MMM, HH:mm", Locale.getDefault())
    return formatter.format(Date(timestamp))
}
