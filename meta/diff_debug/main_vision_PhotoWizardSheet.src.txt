@Composable
fun PhotoWizardSheet(
    state: PhotoWizardState,
    onDismiss: () -> Unit,
    onStepChange: (PhotoWizardStep) -> Unit,
    onAddPhoto: (VisionPhotoRole) -> Unit,
    onAddPhotoFromCamera: (VisionPhotoRole) -> Unit,
    onAddPhotoFromGallery: (VisionPhotoRole) -> Unit,
    onRemovePhoto: (VisionPhotoRole) -> Unit,
    onRecognize: () -> Unit,
) {
    if (!state.visible) return
    val isRu = Locale.getDefault().language.lowercase().startsWith("ru")
    val (title, hint) = remember(state.step, isRu) { stepCopy(step = state.step, isRu = isRu) }
    var showHelp by rememberSaveable(state.step) { mutableStateOf(false) }

    val slots = remember(state.step, state) {
        when (state.step) {
            PhotoWizardStep.FRONT -> listOf(state.frontSlot())
            PhotoWizardStep.OPTIONAL -> state.optionalAppearanceSlots()
            PhotoWizardStep.SPECS -> state.techSlots
        }
    }
    var currentIndex by rememberSaveable(state.step) { mutableIntStateOf(0) }
    LaunchedEffect(slots) {
        val last = (slots.size - 1).coerceAtLeast(0)
        if (currentIndex > last) currentIndex = last
    }
    val currentSlot = slots.getOrNull(currentIndex)

    val canProceed = state.hasFront()
    val nextStep = when (state.step) {
        PhotoWizardStep.FRONT -> PhotoWizardStep.OPTIONAL
        PhotoWizardStep.OPTIONAL -> PhotoWizardStep.SPECS
        PhotoWizardStep.SPECS -> null
    }
    val primaryLabel = when {
        nextStep != null -> if (isRu) "Далее" else "Next"
        state.isRecognizing -> if (isRu) "Распознаем..." else "Recognizing..."
        else -> if (isRu) "Распознать" else "Recognize"
    }
    val skipLabel = if (isRu) "Пропустить дополнительные фото" else "Skip extra photos"

    Surface(
        color = MaterialTheme.colorScheme.surface,
        modifier = Modifier.fillMaxSize(),
    ) {
        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(horizontal = 16.dp, vertical = 12.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp),
        ) {
            PhotoWizardTopBar(
                title = if (isRu) "Шаг ${stepIndex(state.step) + 1}/3 — $title" else "Step ${stepIndex(state.step) + 1}/3 — $title",
                helpActive = showHelp,
                onHelpClick = { showHelp = !showHelp },
                onClose = onDismiss,
            )

            if (showHelp) {
                InlineHelpCard(text = hint)
            }

            if (slots.isNotEmpty()) {
                SlotDots(
                    count = slots.size,
                    currentIndex = currentIndex,
                )
            }

            PhotoSlotPager(
                slots = slots,
                currentIndex = currentIndex,
                onIndexChange = { currentIndex = it },
                onSlotClick = { slot -> onAddPhoto(slot.role) },
            )

            PhotoActionRow(
                hasPhoto = currentSlot?.uri.isNullOrBlank().not(),
                isRu = isRu,
                onCamera = { currentSlot?.let { onAddPhotoFromCamera(it.role) } },
                onGallery = { currentSlot?.let { onAddPhotoFromGallery(it.role) } },
                onDelete = { currentSlot?.let { onRemovePhoto(it.role) } },
            )

            state.errorMessage?.let { msg ->
                Text(
                    text = msg,
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.error,
                    modifier = Modifier.fillMaxWidth(),
                )
            }

            if (!canProceed && state.errorMessage.isNullOrBlank()) {
                Text(
                    text = if (isRu) "Добавьте фото" else "Add a photo",
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                )
            }

            Spacer(Modifier.weight(1f))

            PhotoWizardBottomBar(
                primaryLabel = primaryLabel,
                primaryEnabled = canProceed && !state.isRecognizing,
                showSkip = canProceed && nextStep != null,
                skipLabel = skipLabel,
                onPrimary = {
                    val target = nextStep
                    if (target != null) {
                        onStepChange(target)
                    } else {
                        onRecognize()
                    }
                },
                onSkip = {
                    nextStep?.let { onStepChange(it) }
                },
            )
        }
    }
}

@Composable
private fun PhotoWizardTopBar(
    title: String,
    helpActive: Boolean,
    onHelpClick: () -> Unit,
    onClose: () -> Unit,
) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .heightIn(min = 56.dp),
        verticalAlignment = Alignment.CenterVertically,
    ) {
        IconButton(onClick = onClose) {
            Icon(Icons.Filled.Close, contentDescription = null)
        }
        Text(
            text = title,
            style = MaterialTheme.typography.titleMedium,
            textAlign = TextAlign.Center,
            modifier = Modifier.weight(1f),
        )
        IconButton(onClick = onHelpClick) {
            Icon(
                imageVector = Icons.Outlined.HelpOutline,
                contentDescription = null,
                tint = if (helpActive) MaterialTheme.colorScheme.primary else MaterialTheme.colorScheme.onSurfaceVariant,
            )
        }
    }
}

@Composable
private fun InlineHelpCard(
    text: String,
) {
    Surface(
        shape = RoundedCornerShape(16.dp),
        color = MaterialTheme.colorScheme.surfaceVariant,
        modifier = Modifier.fillMaxWidth(),
    ) {
        Text(
            text = text,
            style = MaterialTheme.typography.bodySmall,
            color = MaterialTheme.colorScheme.onSurfaceVariant,
            modifier = Modifier.padding(horizontal = 12.dp, vertical = 10.dp),
        )
    }
}

@Composable
private fun PhotoSlotPager(
    slots: List<PhotoSlot>,
    currentIndex: Int,
    onIndexChange: (Int) -> Unit,
    onSlotClick: (PhotoSlot) -> Unit,
) {
    if (slots.isEmpty()) return
    val pager = rememberLazyListState(initialFirstVisibleItemIndex = currentIndex.coerceIn(0, slots.lastIndex))
    val currentIdx by remember(pager, slots) {
        derivedStateOf { pager.firstVisibleItemIndex.coerceIn(0, slots.lastIndex) }
    }

    LaunchedEffect(currentIdx) {
        if (currentIdx != currentIndex) {
            onIndexChange(currentIdx)
        }
    }

    LaunchedEffect(currentIndex, slots) {
        val safeIndex = currentIndex.coerceIn(0, slots.lastIndex)
        if (pager.firstVisibleItemIndex != safeIndex) {
            pager.scrollToItem(safeIndex)
        }
    }

    LazyRow(
        state = pager,
        userScrollEnabled = slots.size > 1,
        modifier = Modifier.fillMaxWidth(),
    ) {
        items(slots.size) { i ->
            val slot = slots[i]
            Box(
                modifier = Modifier
                    .fillParentMaxWidth()
                    .padding(end = 12.dp),
            ) {
                PhotoSlotCard(
                    slot = slot,
                    onClick = { onSlotClick(slot) },
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(280.dp),
                )
            }
        }
    }
}

@Composable
private fun PhotoSlotCard(
    slot: PhotoSlot,
    onClick: () -> Unit,
    modifier: Modifier = Modifier,
) {
    val shape = RoundedCornerShape(24.dp)
    val isEmpty = slot.uri.isNullOrBlank()
    val isRu = Locale.getDefault().language.lowercase().startsWith("ru")
    val placeholderTitle = if (isRu) "Добавьте фото" else "Add a photo"
    val placeholderSubtitle = if (isRu) "Камера или галерея" else "Camera or gallery"

    Box(
        modifier = modifier
            .clip(shape)
            .background(MaterialTheme.colorScheme.surfaceContainerLow)
            .clickable { onClick() },
        contentAlignment = Alignment.Center,
    ) {
        if (isEmpty) {
            Column(
                horizontalAlignment = Alignment.CenterHorizontally,
                verticalArrangement = Arrangement.spacedBy(8.dp),
                modifier = Modifier.padding(horizontal = 16.dp),
            ) {
                Icon(
                    imageVector = Icons.Outlined.PhotoCamera,
                    contentDescription = null,
                    tint = MaterialTheme.colorScheme.onSurfaceVariant,
                    modifier = Modifier.size(36.dp),
                )
                Text(
                    text = placeholderTitle,
                    style = MaterialTheme.typography.titleSmall.copy(fontWeight = FontWeight.SemiBold),
                    textAlign = TextAlign.Center,
                )
                Text(
                    text = placeholderSubtitle,
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                    textAlign = TextAlign.Center,
                )
            }
        } else {
            AsyncImage(
                model = slot.uri,
                contentDescription = null,
                contentScale = ContentScale.Crop,
                modifier = Modifier.fillMaxSize(),
            )
        }
    }
}

@Composable
private fun PhotoActionRow(
    hasPhoto: Boolean,
    isRu: Boolean,
    onCamera: () -> Unit,
    onGallery: () -> Unit,
    onDelete: () -> Unit,
) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .padding(horizontal = 4.dp),
        horizontalArrangement = Arrangement.SpaceAround,
        verticalAlignment = Alignment.CenterVertically,
    ) {
        PhotoActionButton(
            icon = Icons.Outlined.PhotoCamera,
            label = if (isRu) "Камера" else "Camera",
            enabled = true,
            onClick = onCamera,
        )
        PhotoActionButton(
            icon = Icons.Outlined.Image,
            label = if (isRu) "Галерея" else "Gallery",
            enabled = true,
            onClick = onGallery,
        )
        PhotoActionButton(
            icon = Icons.Outlined.Delete,
            label = if (isRu) "Удалить" else "Delete",
            enabled = hasPhoto,
            onClick = onDelete,
        )
    }
}

@Composable
private fun PhotoActionButton(
    icon: androidx.compose.ui.graphics.vector.ImageVector,
    label: String,
    enabled: Boolean,
    onClick: () -> Unit,
) {
    Column(
        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = Arrangement.spacedBy(4.dp),
        modifier = Modifier.widthIn(min = 72.dp),
    ) {
        IconButton(
            onClick = onClick,
            enabled = enabled,
            modifier = Modifier.sizeIn(minWidth = 48.dp, minHeight = 48.dp),
        ) {
            Icon(
                imageVector = icon,
                contentDescription = null,
            )
        }
        Text(
            text = label,
            style = MaterialTheme.typography.labelSmall,
            color = if (enabled) MaterialTheme.colorScheme.onSurface else MaterialTheme.colorScheme.onSurfaceVariant,
            textAlign = TextAlign.Center,
        )
    }
}

@Composable
private fun PhotoWizardBottomBar(
    primaryLabel: String,
    primaryEnabled: Boolean,
    showSkip: Boolean,
    skipLabel: String,
    onPrimary: () -> Unit,
    onSkip: () -> Unit,
) {
    Column(
        modifier = Modifier.fillMaxWidth(),
        verticalArrangement = Arrangement.spacedBy(6.dp),
    ) {
        Button(
            onClick = onPrimary,
            enabled = primaryEnabled,
            modifier = Modifier.fillMaxWidth(),
        ) {
            Text(primaryLabel)
        }
        if (showSkip) {
            TextButton(
                onClick = onSkip,
                modifier = Modifier.fillMaxWidth(),
            ) {
                Text(skipLabel)
            }
        }
    }
}

@Composable
private fun SlotDots(
    count: Int,
    currentIndex: Int,
) {
    Row(
        horizontalArrangement = Arrangement.spacedBy(8.dp, Alignment.CenterHorizontally),
        verticalAlignment = Alignment.CenterVertically,
        modifier = Modifier.fillMaxWidth(),
    ) {
        repeat(count) { i ->
            val active = i == currentIndex
            Box(
                modifier = Modifier
                    .size(if (active) 10.dp else 8.dp)
                    .clip(CircleShape)
                    .background(
                        if (active) MaterialTheme.colorScheme.primary
                        else MaterialTheme.colorScheme.outlineVariant,
                    )
            )
        }
    }
}

private fun stepCopy(step: PhotoWizardStep, isRu: Boolean): Pair<String, String> {
    val title = when (step) {
        PhotoWizardStep.FRONT ->
            if (isRu) "Лицевая сторона" else "Front side"

        PhotoWizardStep.OPTIONAL ->
            if (isRu) "Детали и комплект" else "Details and kit"

        PhotoWizardStep.SPECS ->
            if (isRu) "Дефекты и этикетка" else "Defects and label"
    }

    val hint = when (step) {
        PhotoWizardStep.FRONT ->
            if (isRu) "Снимите лицевую сторону, чтобы мы быстрее подготовили объявление."
            else "Capture the front side to speed up the listing."

        PhotoWizardStep.OPTIONAL ->
            if (isRu) "Покажите комплект и детали. Эти фото помогут покупателям лучше понять состояние."
            else "Show accessories and details. It helps buyers understand the condition."

        PhotoWizardStep.SPECS ->
            if (isRu) "Если есть дефекты — снимите их. Можно добавить этикетку или штрихкод."
            else "If there are defects, capture them. You can add a label or barcode."
    }

    return title to hint
}

private fun stepIndex(step: PhotoWizardStep): Int = when (step) {
    PhotoWizardStep.FRONT -> 0
    PhotoWizardStep.OPTIONAL -> 1
    PhotoWizardStep.SPECS -> 2
}
