private enum class OffersLayoutMode { Horizontal, Vertical }
private enum class ContentMode { List, Details }

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun OffersSheetTask(props: OffersSheetProps) {
    val (visible, query, prefetchedItems, onDismiss) = props
    if (!visible || query == null) return

    val searchOffers: SearchOffersUseCase = koinGet(SearchOffersUseCase::class.java)
    val profileSettings = LocalProfileSettings.current
    val context = LocalContext.current
    val clipboard = LocalClipboardManager.current

    val baseCriteria = remember(query) {
        val locale = Locale.getDefault()
        OfferSearchCriteria(
            brand = query.brand.ifBlank { null },
            model = query.model.ifBlank { null },
            attributes = query.attributes,
            limit = 20,
            userCountry = locale.country.takeIf { it.isNotBlank() },
            userLanguage = locale.language.takeIf { it.isNotBlank() },
        )
    }

    var items by remember(query) { mutableStateOf(prefetchedItems) }
    var isLoading by remember(query) { mutableStateOf(prefetchedItems.isEmpty()) }
    var isLoadingMore by remember(query) { mutableStateOf(false) }
    var currentLimit by remember(query) { mutableStateOf(20) }
    var errorMessage by remember(query) { mutableStateOf<String?>(null) }
    var minimized by remember { mutableStateOf(false) }
    var layoutMode by remember { mutableStateOf(OffersLayoutMode.Horizontal) }
    var cardStyle by remember { mutableStateOf(OfferCardStyle.Tiles) }
    var contentMode by remember { mutableStateOf(ContentMode.List) }
    var selectedOffer by remember { mutableStateOf<ExplainedItem?>(null) }
    var badge by remember { mutableStateOf(OffersBadge.None) }
    var savedOfferIds by rememberSaveable { mutableStateOf(setOf<String>()) }
    val hiddenIds = remember(query) { mutableStateListOf<String>() }
    val scope = rememberCoroutineScope()
    var undo by remember { mutableStateOf<UndoState?>(null) }

    LaunchedEffect(visible) {
        if (!visible) {
            contentMode = ContentMode.List
            selectedOffer = null
        }
    }

    LaunchedEffect(query, prefetchedItems) {
        currentLimit = 20
        errorMessage = null

        if (prefetchedItems.isNotEmpty()) {
            items = prefetchedItems
            isLoading = false
            return@LaunchedEffect
        }

        isLoading = true
        val result = runCatching { searchOffers(baseCriteria.copy(limit = currentLimit)) }
        if (result.isSuccess) {
            items = result.getOrElse { emptyList() }
        } else {
            items = emptyList()
            errorMessage = "Не удалось загрузить предложения"
        }
        isLoading = false
    }

    val filteredItems = remember(items, profileSettings.hideUndeliverable, baseCriteria.userCountry) {
        if (!profileSettings.hideUndeliverable) {
            items
        } else {
            items.filter { isDeliverableToUser(it.dto.sellerShippingCountries, baseCriteria.userCountry) }
        }
    }

    val displayedItems = remember(filteredItems, badge) {
        when (badge) {
            OffersBadge.None -> filteredItems
            OffersBadge.BestPrice -> filteredItems.sortedWith(
                compareBy<ExplainedItem> { it.dto.price == null }.thenBy { it.dto.price },
            )
            OffersBadge.BestRating -> filteredItems.sortedWith(
                compareByDescending<ExplainedItem> { it.dto.sellerRating ?: 0.0 },
            )
            OffersBadge.Nearest -> filteredItems.sortedWith(
                compareBy<ExplainedItem> { it.dto.deliveryTime == null }
                    .thenBy { it.dto.deliveryTime ?: Int.MAX_VALUE },
            )
        }
    }

    val visibleItems by remember(displayedItems, hiddenIds) {
        derivedStateOf { displayedItems.filterNot { hiddenIds.contains(it.dto.id) } }
    }
    val nominations = remember(visibleItems) { determineOfferNominations(visibleItems) }

    val canLoadMore = !isLoading &&
        !isLoadingMore &&
        items.size >= currentLimit
    val showLoadMore = items.isNotEmpty() && (canLoadMore || isLoadingMore)

    val loadMore: () -> Unit = loadMore@{
        if (!canLoadMore) return@loadMore
        val nextLimit = currentLimit + 20
        isLoadingMore = true
        errorMessage = null

        scope.launch {
            val result = runCatching { searchOffers(baseCriteria.copy(limit = nextLimit)) }
            if (result.isSuccess) {
                items = result.getOrElse { emptyList() }
                currentLimit = nextLimit
            } else {
                errorMessage = "Не удалось загрузить ещё предложения"
            }
            isLoadingMore = false
        }
    }

    val retryLoad: () -> Unit = retry@{
        if (isLoading) return@retry
        isLoading = true
        errorMessage = null
        scope.launch {
            val result = runCatching { searchOffers(baseCriteria.copy(limit = currentLimit)) }
            if (result.isSuccess) {
                items = result.getOrElse { emptyList() }
            } else {
                errorMessage = "Не удалось загрузить предложения"
            }
            isLoading = false
        }
    }

    fun toggleSavedOffer(offerId: String) {
        savedOfferIds = if (savedOfferIds.contains(offerId)) {
            savedOfferIds - offerId
        } else {
            savedOfferIds + offerId
        }
    }

    fun handleToggleSave(item: ExplainedItem) {
        toggleSavedOffer(item.dto.id)
        props.onAction(OfferCardMenuAction.Favorite, item)
    }

    fun shareExternal(url: String) {
        runCatching {
            val intent = Intent(Intent.ACTION_SEND).apply {
                type = "text/plain"
                putExtra(Intent.EXTRA_TEXT, url)
            }
            val chooser = Intent.createChooser(intent, "Share link")
            context.startActivity(chooser)
        }.onFailure {
            Toast.makeText(context, "Could not share the link.", Toast.LENGTH_SHORT).show()
        }
    }

    fun copyLink(url: String) {
        clipboard.setText(AnnotatedString(url))
        Toast.makeText(context, "Link copied.", Toast.LENGTH_SHORT).show()
    }

    val showingDetails = contentMode == ContentMode.Details && selectedOffer != null
    val screenState = when {
        isLoading && visibleItems.isEmpty() -> ScreenState.Loading(LoadingPhase.INITIAL)
        isLoading -> ScreenState.Loading(LoadingPhase.REFRESHING)
        errorMessage != null && visibleItems.isEmpty() && !showingDetails -> ScreenState.Error(
            message = errorMessage ?: stringResource(R.string.state_error_title),
            primaryAction = StateAction(
                label = stringResource(R.string.state_error_retry),
                onAction = retryLoad,
                type = StateActionType.RETRY,
            ),
        )
        visibleItems.isEmpty() && !showingDetails -> ScreenState.Empty(
            reason = EmptyReason.NO_RESULTS,
            message = "Подходящих предложений не найдено. Попробуйте изменить критерии поиска.",
            primaryAction = StateAction(
                label = stringResource(R.string.state_empty_action_default),
                onAction = onDismiss,
                type = StateActionType.CHANGE_QUERY,
            ),
        )
        else -> ScreenState.Content()
    }

    val handleAction: (OfferCardMenuAction, ExplainedItem) -> Unit = { action, item ->
        when (action) {
            OfferCardMenuAction.OpenExpressDetails -> {
                contentMode = ContentMode.Details
                selectedOffer = item
            }
            else -> props.onAction(action, item)
        }
    }

    fun hideItem(item: ExplainedItem, progress: Float, result: SwipeResult) {
        val id = item.dto.id
        when (result) {
            SwipeResult.Favorite -> handleToggleSave(item)
            SwipeResult.Hide -> {
                if (hiddenIds.contains(id)) return
                hiddenIds.add(id)
                val totalMillis = (2_000L + (progress.coerceIn(0f, 1f) * 3_000L)).toLong()
                val start = System.currentTimeMillis()
                undo = UndoState(id = id, remainingMillis = totalMillis)
                scope.launch {
                    while (true) {
                        val elapsed = System.currentTimeMillis() - start
                        val left = totalMillis - elapsed
                        if (left <= 0 || undo?.id != id) break
                        undo = undo?.copy(remainingMillis = left)
                        delay(200)
                    }
                    if (undo?.id == id && System.currentTimeMillis() - start >= totalMillis) {
                        undo = null
                    }
                }
            }
        }
    }

    fun handleOverflow(action: OfferOverflowAction, item: ExplainedItem) {
        val url = item.dto.externalUrl
        when (action) {
            OfferOverflowAction.Share -> {
                if (!url.isNullOrBlank()) shareExternal(url)
                props.onAction(OfferCardMenuAction.Share, item)
            }
            OfferOverflowAction.Hide -> {
                hideItem(item, 1f, SwipeResult.Hide)
                props.onAction(OfferCardMenuAction.Hide, item)
            }
            OfferOverflowAction.Report -> {
                Toast.makeText(context, "Thanks, we will review this offer.", Toast.LENGTH_SHORT).show()
                props.onAction(OfferCardMenuAction.Report, item)
            }
            OfferOverflowAction.CopyLink -> {
                if (!url.isNullOrBlank()) copyLink(url)
                props.onAction(OfferCardMenuAction.CopyLink, item)
            }
        }
    }

    if (!minimized) {
        val sheetState = rememberModalBottomSheetState(skipPartiallyExpanded = true)

        ModalBottomSheet(
            onDismissRequest = { minimized = true },
            sheetState = sheetState,
            containerColor = MaterialTheme.colorScheme.surface.copy(alpha = 0.99f),
            // Плотный полупрозрачный scrim даёт эффект затемнённого/слегка размытого фона
            scrimColor = MaterialTheme.colorScheme.background.copy(alpha = 0.80f),
        ) {
            Box(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = 16.dp, vertical = 12.dp),
            ) {
                Column(
                    modifier = Modifier.fillMaxWidth(),
                ) {
                    SheetHeader(title = "Предложения", onDismiss = onDismiss)
                    Spacer(Modifier.height(8.dp))
                    HorizontalDivider()
                    Spacer(Modifier.height(8.dp))

                    OffersBadgeMenu(
                        active = badge,
                        onSelect = { badge = it },
                        modifier = Modifier
                            .fillMaxWidth()
                            .padding(bottom = 10.dp),
                    )

                    OffersLayoutToggle(
                        layoutMode = layoutMode == OffersLayoutMode.Horizontal,
                        onHorizontal = { layoutMode = OffersLayoutMode.Horizontal },
                        onVertical = { layoutMode = OffersLayoutMode.Vertical },
                    )

                    OfferCardStyleToggle(
                        style = cardStyle,
                        onFlat = { cardStyle = OfferCardStyle.Flat },
                        onTiles = { cardStyle = OfferCardStyle.Tiles },
                    )

                    StateHost(
                        state = screenState,
                        contentPadding = PaddingValues(),
                        screenName = "offers_sheet",
                        loadingContent = { _ ->
                            OffersSkeleton(
                                horizontal = layoutMode == OffersLayoutMode.Horizontal,
                                cardStyle = cardStyle,
                            )
                        },
                        modifier = Modifier.fillMaxWidth(),
                    ) { innerPadding ->
                        Box(modifier = Modifier.padding(bottom = innerPadding.calculateBottomPadding())) {
                            if (contentMode == ContentMode.Details && selectedOffer != null) {
                                OfferDetailsContent(
                                    item = selectedOffer!!,
                                    onBack = {
                                        contentMode = ContentMode.List
                                        selectedOffer = null
                                    },
                                    onAction = { action ->
                                        selectedOffer?.let { props.onAction(action, it) }
                                    },
                                )
                            } else if (layoutMode == OffersLayoutMode.Horizontal) {
                                OffersListHorizontal(
                                    items = visibleItems,
                                    onSelect = {
                                        contentMode = ContentMode.Details
                                        selectedOffer = it
                                    },
                                    onAction = handleAction,
                                    onHide = ::hideItem,
                                    nominations = nominations,
                                    activeBadge = badge,
                                    cardStyle = cardStyle,
                                    savedIds = savedOfferIds,
                                    onToggleSave = ::handleToggleSave,
                                    onOverflow = { item, action -> handleOverflow(action, item) },
                                    showLoadMore = showLoadMore,
                                    isLoadingMore = isLoadingMore,
                                    onLoadMore = loadMore,
                                )
                            } else {
                                OffersListVertical(
                                    items = visibleItems,
                                    onSelect = {
                                        contentMode = ContentMode.Details
                                        selectedOffer = it
                                    },
                                    onAction = handleAction,
                                    onHide = ::hideItem,
                                    nominations = nominations,
                                    activeBadge = badge,
                                    cardStyle = cardStyle,
                                    savedIds = savedOfferIds,
                                    onToggleSave = ::handleToggleSave,
                                    onOverflow = { item, action -> handleOverflow(action, item) },
                                    photoPeekEnabled = profileSettings.photoPeekEnabled,
                                    showLoadMore = showLoadMore,
                                    isLoadingMore = isLoadingMore,
                                    onLoadMore = loadMore,
                                )
                            }
                        }
                    }

                    if (contentMode == ContentMode.List && errorMessage != null && visibleItems.isNotEmpty()) {
                        OffersInlineError(
                            message = errorMessage ?: "Не удалось загрузить предложения",
                            onRetry = loadMore,
                        )
                    }

                    // Небольшой нижний отступ, чтобы контент не прилипал к системной жестовой панели
                    Spacer(Modifier.height(12.dp))
                    undo?.let { state ->
                        UndoBar(
                            state = state,
                            onCancel = {
                                hiddenIds.remove(state.id)
                                undo = null
                            },
                        )
                    }
                }
            }
        }
    }

    if (minimized) {
        Box(
            modifier = Modifier
                .fillMaxSize()
                .padding(horizontal = 12.dp, vertical = 12.dp),
            contentAlignment = Alignment.BottomCenter,
        ) {
            CollapsedTab(
                onRestore = { minimized = false },
                onClose = onDismiss,
            )
        }
    }
}

private data class UndoState(
    val id: String,
    val remainingMillis: Long,
)

@Composable
private fun OffersInlineError(
    message: String,
    onRetry: () -> Unit,
) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .padding(vertical = 8.dp),
        horizontalArrangement = Arrangement.SpaceBetween,
        verticalAlignment = Alignment.CenterVertically,
    ) {
        Text(
            text = message,
            style = MaterialTheme.typography.bodySmall,
            color = MaterialTheme.colorScheme.onSurfaceVariant,
            modifier = Modifier.weight(1f),
        )
        Spacer(Modifier.width(12.dp))
        Text(
            text = "Повторить",
            style = MaterialTheme.typography.labelLarge,
            color = MaterialTheme.colorScheme.primary,
            modifier = Modifier.clickable { onRetry() },
        )
    }
}

@Composable
private fun UndoBar(
    state: UndoState,
    onCancel: () -> Unit,
) {
    Box(
        modifier = Modifier
            .fillMaxWidth()
            .padding(vertical = 8.dp, horizontal = 4.dp)
            .background(
                color = MaterialTheme.colorScheme.surfaceVariant.copy(alpha = 0.98f),
                shape = RoundedCornerShape(14.dp),
            )
            .padding(horizontal = 16.dp, vertical = 14.dp),
        contentAlignment = Alignment.Center,
    ) {
        Row(
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.SpaceBetween,
            modifier = Modifier.fillMaxWidth(),
        ) {
            val seconds = ceil(state.remainingMillis / 1000f).toInt().coerceAtLeast(0)
            Text(
                text = "Карточка скрыта. Отменить? (${seconds}s)",
                style = MaterialTheme.typography.titleSmall.copy(fontWeight = FontWeight.SemiBold),
                color = MaterialTheme.colorScheme.onSurface,
                modifier = Modifier.weight(1f),
            )
            Spacer(Modifier.width(10.dp))
            Text(
                text = "Отменить",
                style = MaterialTheme.typography.titleSmall.copy(fontWeight = FontWeight.SemiBold),
                color = MaterialTheme.colorScheme.primary,
                modifier = Modifier.clickable { onCancel() },
            )
        }
    }
}

@OptIn(ExperimentalLayoutApi::class)
@Composable
private fun OfferDetailsContent(
    item: ExplainedItem,
    onBack: () -> Unit,
    onAction: (OfferCardMenuAction) -> Unit,
) {
    val dto = item.dto
    val offerImage = rememberOfferImageTask()

    Column(
        modifier = Modifier
            .fillMaxWidth()
            .padding(vertical = 8.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp),
    ) {
        Row(
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.spacedBy(8.dp),
            modifier = Modifier.fillMaxWidth(),
        ) {
            IconButton(onClick = onBack) {
                Icon(
                    imageVector = Icons.AutoMirrored.Outlined.ArrowBack,
                    contentDescription = "Назад к списку",
                )
            }
            Text(
                text = "Предложение",
                style = MaterialTheme.typography.titleMedium,
                color = MaterialTheme.colorScheme.onSurface,
            )
        }

        offerImage.Render(
            imageUrl = dto.imageUrls.firstOrNull(),
            placeholderKey = dto.id,
            placeholderLabel = dto.sourceName ?: dto.sellerName ?: dto.title,
            contentDescription = "Фото предложения",
            modifier = Modifier
                .fillMaxWidth()
                .height(220.dp),
            shape = RoundedCornerShape(16.dp),
            contentScale = ContentScale.Crop,
        )

        Column(
            verticalArrangement = Arrangement.spacedBy(6.dp),
            modifier = Modifier.fillMaxWidth(),
        ) {
            val priceText = formatPriceCompact(dto.price)
            val priceA11y = if (dto.price == null) "Цена не указана" else "Цена $priceText"
            Text(
                text = dto.title,
                style = MaterialTheme.typography.titleLarge,
                color = MaterialTheme.colorScheme.onSurface,
            )
            Text(
                text = priceText,
                style = MaterialTheme.typography.titleMedium.copy(fontWeight = FontWeight.SemiBold),
                color = MaterialTheme.colorScheme.primary,
                modifier = Modifier.semantics { contentDescription = priceA11y },
            )
        }

        val ratingA11y = dto.sellerRating?.let {
            val value = String.format(Locale("ru", "RU"), "%.1f", it)
            "Рейтинг $value, отзывов ${dto.sellerRatingCount ?: 0}"
        } ?: "Рейтинг отсутствует"

        Row(
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.spacedBy(8.dp),
            modifier = Modifier.semantics { contentDescription = ratingA11y },
        ) {
            Icon(
                imageVector = Icons.Rounded.Star,
                contentDescription = null,
                tint = MaterialTheme.colorScheme.tertiary,
            )
            Text(
                text = dto.sellerRating?.let { String.format(Locale("ru", "RU"), "%.1f", it) } ?: "—",
                style = MaterialTheme.typography.bodyMedium,
                color = MaterialTheme.colorScheme.onSurface,
            )
            Text(
                text = "(${dto.sellerRatingCount ?: 0})",
                style = MaterialTheme.typography.bodySmall,
                color = MaterialTheme.colorScheme.onSurfaceVariant,
            )
        }

        if (dto.sellerBadges.isNotEmpty()) {
            FlowRow(
                horizontalArrangement = Arrangement.spacedBy(8.dp),
                verticalArrangement = Arrangement.spacedBy(8.dp),
            ) {
                dto.sellerBadges.forEach { badge ->
                    Text(
                        text = badge.replace('_', ' '),
                        style = MaterialTheme.typography.labelMedium,
                        color = MaterialTheme.colorScheme.onSurface,
                        modifier = Modifier
                            .background(
                                color = MaterialTheme.colorScheme.surfaceVariant.copy(alpha = 0.7f),
                                shape = RoundedCornerShape(50),
                            )
                            .padding(horizontal = 10.dp, vertical = 6.dp),
                    )
                }
            }
        }

        Row(
            horizontalArrangement = Arrangement.spacedBy(12.dp),
            verticalAlignment = Alignment.CenterVertically,
        ) {
            Button(
                onClick = { onAction(OfferCardMenuAction.OpenChat) },
                modifier = Modifier.weight(1f),
            ) {
                Icon(
                    imageVector = Icons.Outlined.Chat,
                    contentDescription = null,
                    modifier = Modifier.size(18.dp),
                )
                Spacer(Modifier.width(6.dp))
                Text("Написать")
            }
            OutlinedButton(
                onClick = onBack,
                modifier = Modifier.weight(1f),
            ) {
                Text("Назад к списку")
            }
        }
    }
}

private fun formatPriceCompact(price: Double?): String =
    if (price == null) "Цена не указана"
    else String.format(Locale("ru", "RU"), "%,.0f ₽", price)
