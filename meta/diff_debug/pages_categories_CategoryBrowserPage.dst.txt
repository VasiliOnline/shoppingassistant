@Composable
fun CategoryBrowserPage(
    onBack: () -> Unit,
    onClose: () -> Unit,
    onSelectCategory: (Category, String) -> Unit,
    modifier: Modifier = Modifier,
    applySafeInsets: Boolean = true,
    extraBottomPadding: Dp = LayoutDefaults.ContentBottomSpacing,
    isModal: Boolean = false,
) {
    val catalogRepository: CatalogRepository = remember { koinGet(CatalogRepository::class.java) }
    val countsTask: GetCategoryOfferCountsTask = remember { koinGet(GetCategoryOfferCountsTask::class.java) }
    var categories by remember { mutableStateOf<List<Category>>(emptyList()) }
    var pathCodes by remember { mutableStateOf<List<String>>(emptyList()) }
    var countsByCode by remember { mutableStateOf<Map<String, Int>?>(null) }

    LaunchedEffect(Unit) {
        categories = runCatching { catalogRepository.listCategories() }.getOrElse { emptyList() }
    }

    LaunchedEffect(categories) {
        if (categories.isEmpty()) return@LaunchedEffect
        countsByCode = runCatching { countsTask(categories.map { it.code }) }
            .getOrElse { emptyMap() }
    }

    val categoriesByParent = remember(categories) { categories.groupBy { it.parentCode } }
    val categoriesByCode = remember(categories) { categories.associateBy { it.code } }
    val descendantsByCode = remember(categoriesByParent) { buildDescendantsIndex(categoriesByParent) }
    val countsWithDescendants = remember(categories, countsByCode, descendantsByCode) {
        val counts = countsByCode ?: return@remember emptyMap()
        val out = LinkedHashMap<String, Int>(categories.size)
        categories.forEach { category ->
            val descendants = descendantsByCode[category.code].orEmpty()
            val sum = (counts[category.code] ?: 0) +
                descendants.sumOf { code -> counts[code] ?: 0 }
            out[category.code] = sum
        }
        out
    }

    val popularRank = remember { popularCategoryRank() }
    val currentParent = pathCodes.lastOrNull()
    val list = categoriesByParent[currentParent].orEmpty()
        .sortedWith(compareBy<Category> { popularRank[it.code] ?: Int.MAX_VALUE }
            .thenBy { it.title ?: it.code })

    ScreenRoot(
        modifier = modifier
            .fillMaxSize()
            .background(MaterialTheme.colorScheme.background),
        applySafeInsets = applySafeInsets,
        extraBottomPadding = extraBottomPadding,
    ) { contentPadding ->
        Column(
            modifier = Modifier.fillMaxSize(),
        ) {
            val canNavigateUp = pathCodes.isNotEmpty()
            val handleBack: (() -> Unit)? = if (canNavigateUp || !isModal) {
                {
                    if (pathCodes.isNotEmpty()) {
                        pathCodes = pathCodes.dropLast(1)
                    } else {
                        onBack()
                    }
                }
            } else null
            val handleClose: (() -> Unit)? = if (isModal && !canNavigateUp) onClose else null
            AppTopBar(
                title = "Категории",
                onBack = handleBack,
                onClose = handleClose,
                applySafeInsets = !applySafeInsets,
            )

            LazyColumn(
                modifier = Modifier.weight(1f),
                verticalArrangement = Arrangement.spacedBy(LayoutDefaults.SectionSpacing),
                contentPadding = PaddingValues(
                    start = LayoutDefaults.HorizontalPadding,
                    end = LayoutDefaults.HorizontalPadding,
                    top = LayoutDefaults.SectionSpacing,
                    bottom = contentPadding.calculateBottomPadding(),
                ),
            ) {
                items(list, key = { it.code }) { category ->
                    val hasChildren = categoriesByParent[category.code].orEmpty().isNotEmpty()
                    val isPopular = popularRank.containsKey(category.code)
                    val count = countsWithDescendants[category.code]
                    val breadcrumb = remember(category, pathCodes, categoriesByCode) {
                        val codes = pathCodes + category.code
                        codes.joinToString(" / ") { code ->
                            categoriesByCode[code]?.title ?: code
                        }
                    }
                    CategoryCard(
                        category = category,
                        count = count,
                        isPopular = isPopular,
                        hasChildren = hasChildren,
                        onClick = {
                            if (hasChildren) {
                                pathCodes = pathCodes + category.code
                            } else {
                                onSelectCategory(category, breadcrumb)
                            }
                        },
                    )
                }
            }
        }
    }
}

@Composable
private fun CategoryCard(
    category: Category,
    count: Int?,
    isPopular: Boolean,
    hasChildren: Boolean,
    onClick: () -> Unit,
) {
    Surface(
        shape = RoundedCornerShape(24.dp),
        color = MaterialTheme.colorScheme.surface,
        modifier = Modifier
            .fillMaxWidth()
            .clickable(onClick = onClick),
    ) {
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 12.dp, vertical = 12.dp),
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.spacedBy(12.dp),
        ) {
            CategoryCover(
                category = category,
                modifier = Modifier.size(width = 88.dp, height = 64.dp),
            )
            Column(
                modifier = Modifier.weight(1f),
                verticalArrangement = Arrangement.spacedBy(4.dp),
            ) {
                Text(
                    text = category.title ?: category.code,
                    style = MaterialTheme.typography.bodyLarge.copy(fontWeight = FontWeight.Medium),
                    maxLines = 1,
                    overflow = TextOverflow.Ellipsis,
                )
                Text(
                    text = count?.let { "$it объявлений" } ?: "— объявлений",
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                    maxLines = 1,
                    overflow = TextOverflow.Ellipsis,
                )
                if (isPopular) {
                    PopularBadge()
                }
            }
            if (hasChildren) {
                Icon(
                    imageVector = Icons.Outlined.KeyboardArrowRight,
                    contentDescription = "Подкатегории",
                    tint = MaterialTheme.colorScheme.onSurfaceVariant,
                    modifier = Modifier.size(22.dp),
                )
            }
        }
    }
}

@Composable
private fun CategoryCover(
    category: Category,
    modifier: Modifier = Modifier,
) {
    val colors = MaterialTheme.colorScheme
    val brush = coverBrushForSegment(
        category.segment,
        colors.primaryContainer,
        colors.secondaryContainer,
        colors.tertiaryContainer,
    )
    val coverUrl = categoryCoverUrl(category)

    Surface(
        shape = RoundedCornerShape(18.dp),
        color = MaterialTheme.colorScheme.surfaceVariant,
        modifier = modifier,
    ) {
        Box(
            modifier = Modifier.fillMaxSize(),
            contentAlignment = Alignment.Center,
        ) {
            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .background(brush),
            )
            coverUrl?.let { url ->
                AsyncImage(
                    model = url,
                    contentDescription = null,
                    contentScale = ContentScale.Crop,
                    modifier = Modifier.fillMaxSize(),
                )
                Box(
                    modifier = Modifier
                        .fillMaxSize()
                        .background(colors.surfaceVariant.copy(alpha = 0.22f)),
                )
            }
            Icon(
                imageVector = Icons.Outlined.Label,
                contentDescription = null,
                tint = if (coverUrl == null) {
                    colors.onPrimaryContainer
                } else {
                    colors.onSurface.copy(alpha = 0.85f)
                },
                modifier = Modifier.size(24.dp),
            )
        }
    }
}

@Composable
private fun PopularBadge() {
    Surface(
        shape = RoundedCornerShape(10.dp),
        color = MaterialTheme.colorScheme.primaryContainer,
        modifier = Modifier.heightIn(min = 20.dp),
    ) {
        Text(
            text = "Популярное",
            style = MaterialTheme.typography.labelSmall,
            color = MaterialTheme.colorScheme.onPrimaryContainer,
            maxLines = 1,
            overflow = TextOverflow.Ellipsis,
            modifier = Modifier.padding(horizontal = 8.dp, vertical = 2.dp),
        )
    }
}

private fun coverBrushForSegment(
    segment: CategorySegment,
    primary: androidx.compose.ui.graphics.Color,
    secondary: androidx.compose.ui.graphics.Color,
    tertiary: androidx.compose.ui.graphics.Color,
): Brush {
    val colors = when (segment) {
        CategorySegment.TECH -> listOf(primary, secondary)
        CategorySegment.FOOD -> listOf(secondary, tertiary)
        CategorySegment.FASH -> listOf(tertiary, primary)
        CategorySegment.KIDS -> listOf(primary, tertiary)
        CategorySegment.HOME -> listOf(secondary, primary)
        CategorySegment.SUPP -> listOf(tertiary, secondary)
        CategorySegment.OTHER -> listOf(primary, secondary)
    }
    return Brush.linearGradient(colors)
}

private fun popularCategoryRank(): Map<String, Int> {
    val popular = listOf(
        "TECH",
        "FOOD",
        "FASH",
        "HOME",
        "KIDS",
        "SUPP",
        "TECH.SMARTPHONE",
        "FOOD.FASTFOOD",
        "FOOD.ROLLS_SUSHI",
        "FOOD.PIZZA",
    )
    return popular.withIndex().associate { it.value to it.index }
}

private fun buildDescendantsIndex(
    categoriesByParent: Map<String?, List<Category>>,
): Map<String, Set<String>> {
    val result = mutableMapOf<String, Set<String>>()

    fun collect(code: String): Set<String> {
        val cached = result[code]
        if (cached != null) return cached
        val children = categoriesByParent[code].orEmpty().map { it.code }
        val all = buildSet {
            children.forEach { child ->
                add(child)
                addAll(collect(child))
            }
        }
        result[code] = all
        return all
    }

    categoriesByParent.values.flatten().forEach { category ->
        collect(category.code)
    }

    return result
}
