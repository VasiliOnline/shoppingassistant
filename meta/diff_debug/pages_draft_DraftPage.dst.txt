@Composable
fun DraftPage(
    payload: DraftPayload,
    navController: NavHostController? = null,
    providedViewModel: MainPageViewModel? = null,
    onBack: (() -> Unit)? = null,
    onOpenResults: ((ResultsPayload) -> Unit)? = null,
    onOpenMyItems: (() -> Unit)? = null,
    onWizardDone: (() -> Unit)? = null,
    modifier: Modifier = Modifier,
    applySafeInsets: Boolean = true,
    extraBottomPadding: Dp = LayoutDefaults.ContentBottomSpacing,
) {
    val viewModel: MainPageViewModel = providedViewModel ?: run {
        val safeNav = requireNotNull(navController) { "navController is required when viewModel is not provided" }
        val navBackStackEntry by safeNav.currentBackStackEntryAsState()
        val parentEntry = remember(navBackStackEntry) { safeNav.getBackStackEntry(AppRoutes.Main) }
        koinViewModel(viewModelStoreOwner = parentEntry)
    }
    val context = LocalContext.current
    val focusManager = LocalFocusManager.current
    val state by viewModel.state.collectAsState()
    val sourceRegistry: SourceRegistry = remember { koinGet(SourceRegistry::class.java) }
    val urlNormalizer: UrlNormalizer = remember { koinGet(UrlNormalizer::class.java) }
    val templateIdTask: TemplateIdTask = remember { koinGet(TemplateIdTask::class.java) }
    val createdStore: UserOffersCreatedStore = remember { koinGet(UserOffersCreatedStore::class.java) }

    var initialized by rememberSaveable { mutableStateOf(false) }
    var linkSheetVisible by remember { mutableStateOf(false) }
    var linkSheetState by remember { mutableStateOf<LinkSheetState>(LinkSheetState.Input) }
    var linkInput by remember { mutableStateOf(LinkInputState()) }
    var preferredPhotoInput by rememberSaveable { mutableStateOf(payload.photoInput) }
    val linkStepsTemplate = remember {
        listOf(
            LinkLoadingStep("Шаг 1 - получаем страницу по ссылке:", false),
            LinkLoadingStep("Шаг 2 - парсим цену и валюту:", false),
            LinkLoadingStep("Шаг 3 - определяем бренд и модель:", false),
        )
    }

    var voiceState by remember { mutableStateOf(VoiceWizardState()) }
    val permissionLauncher = rememberLauncherForActivityResult(
        ActivityResultContracts.RequestPermission()
    ) { granted ->
        voiceState = voiceState.copy(
            permission = if (granted) VoicePermissionState.Granted else VoicePermissionState.Denied,
        )
    }

    val recognizer = remember {
        AndroidSpeechSegmentRecognizer(
            context = context,
            onPartial = { partial ->
                voiceState = voiceState.copy(transcript = partial)
            },
            onFinal = { final ->
                val segment = VoiceSegment(
                    id = System.currentTimeMillis(),
                    text = final,
                    createdAtMs = System.currentTimeMillis(),
                )
                val transcript = listOfNotNull(voiceState.transcript, final)
                    .joinToString(" ")
                    .replace("\\s+".toRegex(), " ")
                    .trim()
                voiceState = voiceState.copy(
                    stage = VoiceWizardStage.Review,
                    segments = voiceState.segments + segment,
                    transcript = transcript,
                    recordingSeconds = 0,
                    eligibility = if (transcript.isNotBlank()) VoiceWizardEligibility.Ready else VoiceWizardEligibility.MissingRequired,
                    errorMessage = null,
                )
            },
            onError = {
                voiceState = voiceState.copy(
                    stage = VoiceWizardStage.Review,
                    errorMessage = "Не удалось распознать речь. Попробуйте ещё раз.",
                    eligibility = VoiceWizardEligibility.MissingRequired,
                )
            },
        )
    }

    DisposableEffect(Unit) {
        onDispose { recognizer.release() }
    }

    LaunchedEffect(Unit) {
        val granted = ContextCompat.checkSelfPermission(
            context,
            Manifest.permission.RECORD_AUDIO,
        ) == PackageManager.PERMISSION_GRANTED
        voiceState = voiceState.copy(
            permission = if (granted) VoicePermissionState.Granted else VoicePermissionState.Unknown,
        )
    }

    LaunchedEffect(voiceState.stage) {
        if (voiceState.stage != VoiceWizardStage.Recording) return@LaunchedEffect
        while (isActive && voiceState.stage == VoiceWizardStage.Recording) {
            delay(1000)
            voiceState = voiceState.copy(recordingSeconds = voiceState.recordingSeconds + 1)
        }
    }

    LaunchedEffect(state.createOfferMessage) {
        state.createOfferMessage?.let { msg ->
            Toast.makeText(context, msg, Toast.LENGTH_SHORT).show()
        }
    }

    val supportedSourcesHint = remember {
        val names = sourceRegistry.all()
            .filter { it.rolloutEnabled && it.capabilities.canIngest }
            .map { it.displayName }
            .sorted()
        if (names.isEmpty()) {
            "Поддерживаемые площадки пока недоступны."
        } else {
            "Поддерживаемые площадки: ${names.joinToString(", ")}."
        }
    }

    fun validateUrl(raw: String): Boolean =
        urlNormalizer.normalize(raw).host != null

    fun startLinkLoading() {
        val url = linkInput.url.trim()
        if (!validateUrl(url)) {
            linkInput = linkInput.copy(isValid = false)
            linkSheetState = LinkSheetState.Error("Некорректная ссылка. Проверьте домен и формат.")
            return
        }
        var steps = linkStepsTemplate
        linkSheetState = LinkSheetState.Loading(steps)
        viewModel.loadLinkTemplate(url) { result ->
            result.onSuccess { template ->
                when (template.ingestStatus) {
                    IngestStatus.OK -> {
                        steps = steps.map { it.copy(done = true) }
                        linkSheetState = LinkSheetState.Loading(steps)
                        linkSheetVisible = false
                        linkSheetState = LinkSheetState.Input
                    }
                    IngestStatus.TEMP_BLOCKED -> {
                        linkSheetState = LinkSheetState.Error("Слишком много запросов. Попробуйте позже.")
                    }
                    IngestStatus.NETWORK_ERROR -> {
                        linkSheetState = LinkSheetState.Error("Ошибка сети. Проверьте подключение.")
                    }
                    IngestStatus.UNSUPPORTED -> {
                        linkSheetState = LinkSheetState.Error("Эта площадка пока не поддерживается.")
                    }
                    IngestStatus.PARSE_ERROR -> {
                        linkSheetState = LinkSheetState.Error("Не удалось распознать данные по ссылке.")
                    }
                }
            }
            result.onFailure {
                linkSheetState = LinkSheetState.Error("Не удалось загрузить ссылку. Попробуйте снова.")
            }
        }
    }

    LaunchedEffect(payload) {
        if (initialized) return@LaunchedEffect
        initialized = true
        viewModel.resetTemplate()
        if (payload.query != null || payload.categoryCode != null || payload.queryText.isNotBlank()) {
            val snapshot = buildSnapshotFromQuery(
                query = payload.query,
                queryText = payload.queryText,
                categoryCode = payload.categoryCode,
                templateIdTask = templateIdTask,
            )
            if (snapshot != null) {
                viewModel.applyTemplateSnapshot(snapshot)
            } else if (payload.queryText.isNotBlank()) {
                viewModel.onQueryChange(payload.queryText)
            }
        }
        when (payload.source) {
            DraftSource.Photo -> {
                val targetMode = if (payload.mode == DraftMode.Search) {
                    TemplateMode.SearchOrSubscribe
                } else {
                    TemplateMode.ExpressFromPhoto
                }
                viewModel.applyTemplateMode(targetMode)
                viewModel.openPhotoWizard(step = PhotoWizardStep.FRONT)
                if (payload.photoInput != DraftPhotoInput.None) {
                    viewModel.requestPhotoSlot(com.example.shoppingassistant.domain.vision.VisionPhotoRole.FRONT)
                }
            }
            DraftSource.Link -> {
                viewModel.applyTemplateMode(TemplateMode.OfferFromLink)
                if (payload.linkUrl.isNotBlank()) {
                    val isValid = validateUrl(payload.linkUrl)
                    linkInput = linkInput.copy(url = payload.linkUrl, isValid = isValid)
                    linkSheetVisible = true
                    linkSheetState = if (isValid) LinkSheetState.Input else {
                        LinkSheetState.Error("Некорректная ссылка. Проверьте домен и формат.")
                    }
                    if (isValid) {
                        startLinkLoading()
                    }
                } else {
                    linkSheetVisible = true
                    linkSheetState = LinkSheetState.Input
                }
            }
            DraftSource.Voice -> {
                val targetMode = if (payload.mode == DraftMode.Search) {
                    TemplateMode.SearchOrSubscribe
                } else {
                    TemplateMode.OfferFromVoice
                }
                viewModel.applyTemplateMode(targetMode)
                voiceState = voiceState.copy(visible = true)
            }
            DraftSource.Text -> {
                viewModel.applyTemplateMode(TemplateMode.ExpressFromPhoto)
            }
        }
    }

    fun onPublish() {
        FlowMetrics.markPublishTapped()
        if (state.template.mode == TemplateMode.OfferFromLink) {
            viewModel.submitTrackedOffer()
            onWizardDone?.invoke()
            return
        }
        val missing = buildDraftMissingFields(state)
        if (missing.isNotEmpty()) {
            Toast.makeText(context, "Заполните: ${missing.first()}", Toast.LENGTH_SHORT).show()
            return
        }
        Toast.makeText(context, "Экспресс-оффер: вызов use-case (заглушка)", Toast.LENGTH_SHORT).show()
        onWizardDone?.invoke()
    }

    fun resolveQueryFromTemplate(): com.example.shoppingassistant.domain.model.NormalizedQuery? {
        val filters = state.template.asSelectedFilters()
        val filteredAttrs = filters.filterKeys { it != "category" }
        return buildQuery(state.template, state.product)
            ?: BrandModelRules.fromRaw(state.template.inputText, filteredAttrs)
    }

    fun toResults() {
        val query = resolveQueryFromTemplate()
        viewModel.recordSearchHistory(
            queryText = state.template.inputText,
            query = query,
            categoryCode = state.template.categoryCode,
        )
        val payload = ResultsPayload(
            query = query,
            queryText = state.template.inputText,
            categoryCode = state.template.categoryCode,
            mode = com.example.shoppingassistant.feature.pages.results.ResultsMode.Offers,
            origin = ResultsOrigin.Suggestion,
        )
        FlowMetrics.startResults(taps = 1)
        onOpenResults?.invoke(payload) ?: navController?.navigate(ResultsRoutes.build(payload))
    }

    fun saveDraft() {
        val title = state.template.lockedTitle
            ?.takeIf { it.isNotBlank() }
            ?: state.template.inputText.trim()
        if (title.isBlank()) {
            Toast.makeText(context, "Добавьте описание для черновика", Toast.LENGTH_SHORT).show()
            return
        }
        val filters = state.template.asSelectedFilters()
        val priceMajor = filters["price"]?.replace(",", ".")?.toDoubleOrNull()
        val currency = filters["currency"]?.takeIf { it.length == 3 } ?: "USD"
        val cover = state.template.primaryPhotoUrl ?: state.template.photoUrls.firstOrNull()
        val snapshotMode = when (state.template.mode) {
            TemplateMode.ExpressFromPhoto -> TemplateSnapshotMode.EXPRESS
            TemplateMode.OfferFromLink,
            TemplateMode.OfferFromVoice,
            -> TemplateSnapshotMode.OFFER
            else -> TemplateSnapshotMode.OFFER
        }
        val snapshot = buildSnapshotFromQuery(
            query = resolveQueryFromTemplate(),
            queryText = title,
            categoryCode = state.template.categoryCode,
            templateIdTask = templateIdTask,
            mode = snapshotMode,
        )
        val now = System.currentTimeMillis()
        val draftId = snapshot?.templateId?.let { "draft-$it" } ?: "draft-$now"
        createdStore.add(
            UserOfferCardUi(
                id = draftId,
                title = title,
                category = state.template.categoryCode ?: state.product?.categoryCode,
                priceMajor = priceMajor,
                currency = currency,
                status = UserOfferStatus.DRAFT,
                publicationStatus = UserOfferPublicationStatus.PUBLISHED,
                coverUrl = cover,
                publishedAtMillis = null,
                updatedAtMillis = now,
                sourceUpdatedAtMillis = now,
            )
        )
        FlowMetrics.markDraftSaved()
        Toast.makeText(context, "Черновик сохранен", Toast.LENGTH_SHORT).show()
        if (onWizardDone != null) {
            onWizardDone()
        } else if (onOpenMyItems != null) {
            onOpenMyItems()
        } else {
            navController?.navigate(AppRoutes.myItems()) { launchSingleTop = true }
        }
    }

    val status = draftStatusLabel(state, payload, voiceState)
    val progress = draftProgress(state)
    val missingFields = remember(state) { buildDraftMissingFields(state) }
    val resolvedQuery = resolveQueryFromTemplate()
    val canPublish = payload.mode != DraftMode.Search && missingFields.isEmpty()
    val canSearch = resolvedQuery != null || !state.template.categoryCode.isNullOrBlank()
    val canTrack = state.template.isLocked
    val canSaveDraft = state.template.lockedTitle?.isNotBlank() == true ||
        state.template.inputText.isNotBlank() ||
        progress.done > 0
    val actionsVisible = payload.source == DraftSource.Text || canSaveDraft
    LaunchedEffect(actionsVisible) {
        if (actionsVisible) {
            FlowMetrics.markDraftReady()
        }
    }
    val hasOverlay = linkSheetVisible ||
        voiceState.visible ||
        state.photoWizard.visible ||
        state.categoryFallback.visible
    BackHandler(enabled = hasOverlay) {
        when {
            linkSheetVisible -> linkSheetVisible = false
            voiceState.visible -> voiceState = voiceState.copy(visible = false)
            state.photoWizard.visible -> {
                val previous = when (state.photoWizard.step) {
                    PhotoWizardStep.FRONT -> null
                    PhotoWizardStep.OPTIONAL -> PhotoWizardStep.FRONT
                    PhotoWizardStep.SPECS -> PhotoWizardStep.OPTIONAL
                }
                if (previous == null) {
                    viewModel.closePhotoWizard()
                } else {
                    viewModel.setPhotoWizardStep(previous)
                }
            }
            state.categoryFallback.visible -> viewModel.dismissCategoryFallback()
        }
    }

    ScreenRoot(
        modifier = modifier
            .fillMaxSize()
            .background(MaterialTheme.colorScheme.background),
        applySafeInsets = applySafeInsets,
        extraBottomPadding = extraBottomPadding,
    ) { contentPadding ->
        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(
                    horizontal = LayoutDefaults.HorizontalPadding,
                    vertical = LayoutDefaults.SectionSpacing,
                )
                .padding(bottom = contentPadding.calculateBottomPadding()),
            verticalArrangement = Arrangement.spacedBy(LayoutDefaults.SectionSpacing),
        ) {
        Row(
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.spacedBy(8.dp),
        ) {
            IconButton(onClick = { onBack?.invoke() ?: navController?.popBackStack() }) {
                Icon(Icons.AutoMirrored.Outlined.ArrowBack, contentDescription = "Назад")
            }
            Text(
                text = "Черновик",
                style = MaterialTheme.typography.titleLarge,
            )
        }

        DraftSourceHeader(payload.source, status)

        DraftProgressCard(
            state = state,
            progress = progress,
        )

        if (progress.total > 0) {
            Text(
                text = "Готово ${progress.done} из ${progress.total} обязательных",
                style = MaterialTheme.typography.bodySmall,
                color = MaterialTheme.colorScheme.onSurfaceVariant,
            )
        }

        if (actionsVisible) {
            Column(verticalArrangement = Arrangement.spacedBy(10.dp)) {
                if (payload.mode == DraftMode.Search) {
                    if (canSearch) {
                        Button(
                            onClick = { toResults() },
                            modifier = Modifier.fillMaxWidth(),
                        ) {
                            Text("Найти предложения")
                        }
                    }
                } else if (canPublish) {
                    Button(
                        onClick = { onPublish() },
                        modifier = Modifier.fillMaxWidth(),
                    ) {
                        Text("Опубликовать")
                    }
                }
                if (payload.mode != DraftMode.Search && canSearch) {
                    OutlinedButton(onClick = { toResults() }, modifier = Modifier.fillMaxWidth()) {
                        Text("Найти предложения")
                    }
                }
                if (canTrack) {
                    OutlinedButton(
                        onClick = {
                            FlowMetrics.markTrackClicked("draft")
                            viewModel.subscribeToCurrentTemplate()
                            Toast.makeText(context, "Отслеживание включено", Toast.LENGTH_SHORT).show()
                        },
                        modifier = Modifier.fillMaxWidth(),
                    ) {
                        Text("Отслеживать")
                    }
                }
                if (payload.mode != DraftMode.Search && canSaveDraft) {
                    OutlinedButton(
                        onClick = { saveDraft() },
                        modifier = Modifier.fillMaxWidth(),
                    ) {
                        Text("Сохранить черновик")
                    }
                }
                FilledTonalButton(
                    onClick = {
                        focusManager.clearFocus(force = true)
                        viewModel.openAttributesSheet()
                    },
                    modifier = Modifier.fillMaxWidth(),
                ) {
                    Text("Редактировать")
                }
            }
        }
    }

    LinkSheet(
        visible = linkSheetVisible,
        state = linkSheetState,
        input = linkInput,
        supportedSourcesHint = supportedSourcesHint,
        onUrlChange = { value -> linkInput = linkInput.copy(url = value, isValid = validateUrl(value)) },
        onSubmit = { startLinkLoading() },
        onRetry = {
            linkSheetState = LinkSheetState.Input
            linkInput = linkInput.copy(url = "", isValid = false)
        },
        onDismiss = { linkSheetVisible = false },
    )

    VoiceWizardSheet(
        state = voiceState,
        onDismiss = { voiceState = voiceState.copy(visible = false) },
        onStartRecording = {
            if (voiceState.permission != VoicePermissionState.Granted) {
                permissionLauncher.launch(Manifest.permission.RECORD_AUDIO)
                return@VoiceWizardSheet
            }
            voiceState = voiceState.copy(stage = VoiceWizardStage.Recording)
            recognizer.start()
        },
        onPauseRecording = {
            recognizer.stop()
            voiceState = voiceState.copy(stage = VoiceWizardStage.Review)
        },
        onResumeRecording = {
            voiceState = voiceState.copy(stage = VoiceWizardStage.Recording)
            recognizer.start()
        },
        onStopRecording = {
            recognizer.stop()
        },
        onCancelWizard = {
            recognizer.cancel()
            voiceState = VoiceWizardState()
        },
        onUndoLastSegment = {
            voiceState = voiceState.copy(segments = emptyList(), transcript = "")
        },
        onRewriteAll = {
            voiceState = voiceState.copy(segments = emptyList(), transcript = "")
        },
        onTranscriptChange = { value -> voiceState = voiceState.copy(transcript = value) },
        onCreateTemplate = {
            val text = voiceState.transcript.trim()
            if (text.isBlank()) {
                Toast.makeText(context, "Добавьте текст запроса", Toast.LENGTH_SHORT).show()
                return@VoiceWizardSheet
            }
            viewModel.onQueryChange(text)
            val targetMode = if (payload.mode == DraftMode.Search) {
                TemplateMode.SearchOrSubscribe
            } else {
                TemplateMode.OfferFromVoice
            }
            viewModel.applyTemplateMode(targetMode)
            voiceState = voiceState.copy(visible = false)
        },
        onRefineWithGpt = { },
    )

    PhotoWizardSheet(
        state = state.photoWizard,
        onDismiss = { viewModel.closePhotoWizard() },
        onStepChange = { step -> viewModel.setPhotoWizardStep(step) },
        onAddPhoto = { role -> viewModel.requestPhotoSlot(role) },
        onAddPhotoFromCamera = { role ->
            preferredPhotoInput = DraftPhotoInput.Camera
            viewModel.requestPhotoSlot(role)
        },
        onAddPhotoFromGallery = { role ->
            preferredPhotoInput = DraftPhotoInput.Gallery
            viewModel.requestPhotoSlot(role)
        },
        onRemovePhoto = { role -> viewModel.removeWizardPhoto(role) },
        onRecognize = { handleWizardRecognize(context, state, viewModel) },
    )

    CategoryFallbackSheet(
        state = state.categoryFallback,
        onPick = { candidate -> viewModel.applyCategoryCandidate(candidate) },
        onDismiss = { viewModel.dismissCategoryFallback() },
    )

    PhotoSourcesDialog(
        state = state,
        viewModel = viewModel,
        preferredSource = preferredPhotoInput,
        onPreferredConsumed = { preferredPhotoInput = DraftPhotoInput.None },
    )
}
}

@Composable
private fun DraftSourceHeader(
    source: DraftSource,
    status: String?,
) {
    val icon = when (source) {
        DraftSource.Photo -> Icons.Outlined.CameraAlt
        DraftSource.Link -> Icons.Outlined.Link
        DraftSource.Voice -> Icons.Outlined.Mic
        DraftSource.Text -> Icons.Outlined.Image
    }
    Row(
        modifier = Modifier

private data class DraftProgress(
    val done: Int,
    val total: Int,
    val items: List<Pair<String, Boolean>>,
)

@Composable
private fun DraftProgressCard(
    state: MainPageState,
    progress: DraftProgress,
) {
    val fraction = if (progress.total == 0) 0f else progress.done.toFloat() / progress.total.toFloat()
    var visibleCount by remember(progress.items) { mutableIntStateOf(0) }
    LaunchedEffect(progress.items) {
        visibleCount = 0
        progress.items.forEachIndexed { index, _ ->
            delay(120)
            visibleCount = index + 1
        }
    }
    val fillColor by animateColorAsState(
        targetValue = MaterialTheme.colorScheme.primary.copy(alpha = 0.18f + 0.4f * fraction),
        label = "draft-fill-color",
    )
    val bg = Brush.verticalGradient(
        colors = listOf(
            fillColor,
            MaterialTheme.colorScheme.surfaceVariant.copy(alpha = 0.6f),
        ),
    )

    Surface(
        shape = RoundedCornerShape(20.dp),
        tonalElevation = 0.dp,
        shadowElevation = 0.dp,
        color = Color.Transparent,
    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .background(bg, RoundedCornerShape(20.dp))
                .padding(horizontal = 16.dp, vertical = 14.dp),
            verticalArrangement = Arrangement.spacedBy(8.dp),
        ) {
            Text(
                text = "Заполнение карточки",
                style = MaterialTheme.typography.titleMedium,
            )
            progress.items.take(visibleCount).take(6).forEach { (label, done) ->
                Text(
                    text = if (done) "$label - готово" else "$label - не распознано",
                    style = MaterialTheme.typography.bodySmall,
                    color = if (done) MaterialTheme.colorScheme.onSurface else MaterialTheme.colorScheme.onSurfaceVariant,
                )
            }
        }
    }
}

private fun draftProgress(state: MainPageState): DraftProgress {
    val filters = state.template.asSelectedFilters()
    val items = mutableListOf<Pair<String, Boolean>>()

    val hasPhoto = !state.template.primaryPhotoUrl.isNullOrBlank() || state.template.photoUrls.isNotEmpty()
    items += "Фотография" to hasPhoto

    val hasCategory = !state.template.categoryCode.isNullOrBlank()
    items += "Категория" to hasCategory

    val required = state.template.requiredKeys
    required.forEach { key ->
        val label = state.attributeDefs.firstOrNull { it.key == key }?.title ?: key
        items += label to (filters[key]?.isNotBlank() == true)
    }

    val done = items.count { it.second }
    return DraftProgress(done = done, total = items.size, items = items)
}

private fun buildDraftMissingFields(state: MainPageState): List<String> {
    val filters = state.template.asSelectedFilters()
    val missing = mutableListOf<String>()

    val hasPhoto = !state.template.primaryPhotoUrl.isNullOrBlank() || state.template.photoUrls.isNotEmpty()
    if (!hasPhoto) missing += "Фотография"

    if (state.template.categoryCode.isNullOrBlank()) {
        missing += "Категория"
    }

    state.template.requiredKeys.forEach { key ->
        if (filters[key].isNullOrBlank()) {
            missing += (state.attributeDefs.firstOrNull { it.key == key }?.title ?: key)
        }
    }
    return missing
}

private fun draftStatusLabel(
    state: MainPageState,
    payload: DraftPayload,
    voiceState: VoiceWizardState,
): String? {
    return when (payload.source) {
        DraftSource.Photo -> when {
            state.photoWizard.isRecognizing -> "Распознаём фото..."
            state.photoWizard.errorMessage != null -> state.photoWizard.errorMessage
            else -> "Подготовьте фото для распознавания"
        }
        DraftSource.Link -> when (state.template.linkMeta?.url?.isNotBlank()) {
            true -> "Ссылка обработана, проверяем детали"
            false -> "Вставьте ссылку на объявление"
            null -> "Вставьте ссылку на объявление"
        }
        DraftSource.Voice -> when (voiceState.stage) {
            VoiceWizardStage.Recording -> "Слушаю описание товара..."
            VoiceWizardStage.Review -> "Проверьте распознанный текст"
            else -> "Готовы записать описание"
        }
        DraftSource.Text -> "Заполняем черновик по тексту"
    }
}

private fun handleWizardRecognize(
    context: Context,
    state: MainPageState,
    viewModel: MainPageViewModel,
) {
    val wizard = state.photoWizard
    val inputs = wizard.allSlots().mapNotNull { slot ->
        val base64 = slot.base64
            ?: slot.uri?.let { uri -> loadPhotoPayload(context, uri)?.base64 }
        if (base64.isNullOrBlank()) return@mapNotNull null
        com.example.shoppingassistant.domain.vision.VisionPhotoInput(role = slot.role, base64 = base64)
    }
    if (inputs.isEmpty()) {
        viewModel.setWizardError("Нет фото для распознавания.")
        return
    }
    viewModel.recognizeWizardPhotos(inputs)
}

@Composable
private fun PhotoSourcesDialog(
    state: MainPageState,
    viewModel: MainPageViewModel,
    preferredSource: DraftPhotoInput = DraftPhotoInput.None,
    onPreferredConsumed: () -> Unit = {},
) {
    val context = LocalContext.current
    var dialogVisible by remember { mutableStateOf(false) }
    var autoLaunchedRole by remember { mutableStateOf<com.example.shoppingassistant.domain.vision.VisionPhotoRole?>(null) }

    val galleryLauncher = rememberLauncherForActivityResult(ActivityResultContracts.GetContent()) { uri ->
        val role = state.photoWizard.pendingRole
        if (uri == null || role == null) {
            viewModel.clearPendingPhotoRole()
            return@rememberLauncherForActivityResult
        }
        val payload = loadPhotoPayload(context, uri.toString()) ?: run {
            viewModel.setWizardError("Не удалось прочитать фото. Попробуйте другое.")
            viewModel.clearPendingPhotoRole()
            return@rememberLauncherForActivityResult
        }
        viewModel.addWizardPhoto(
            role = role,
            uri = payload.previewUri,
            base64 = payload.base64,
            hash = payload.hash,
            qualityOk = payload.qualityOk,
        )
    }

    val cameraLauncher = rememberLauncherForActivityResult(ActivityResultContracts.TakePicturePreview()) { bitmap ->
        val role = state.photoWizard.pendingRole
        if (bitmap == null || role == null) {
            viewModel.clearPendingPhotoRole()
            return@rememberLauncherForActivityResult
        }
        val payload = bitmapToPayload(bitmap) ?: run {
            viewModel.setWizardError("Не удалось получить фото с камеры.")
            viewModel.clearPendingPhotoRole()
            return@rememberLauncherForActivityResult
        }
        viewModel.addWizardPhoto(
            role = role,
            uri = payload.previewUri,
            base64 = payload.base64,
            hash = payload.hash,
            qualityOk = payload.qualityOk,
        )
    }

    LaunchedEffect(state.photoWizard.pendingRole, preferredSource) {
        val role = state.photoWizard.pendingRole
        if (role == null) {
            dialogVisible = false
            autoLaunchedRole = null
            return@LaunchedEffect
        }
        if (preferredSource != DraftPhotoInput.None && autoLaunchedRole != role) {
            autoLaunchedRole = role
            when (preferredSource) {
                DraftPhotoInput.Camera -> cameraLauncher.launch(null)
                DraftPhotoInput.Gallery -> galleryLauncher.launch("image/*")
                DraftPhotoInput.None -> Unit
            }
            onPreferredConsumed()
            return@LaunchedEffect
        }
        dialogVisible = preferredSource == DraftPhotoInput.None && autoLaunchedRole != role
    }

    if (dialogVisible && state.photoWizard.pendingRole != null) {
        AlertDialog(
            onDismissRequest = {
                dialogVisible = false
                viewModel.clearPendingPhotoRole()
            },
            title = { Text("Добавить фото") },
            text = { Text("Выберите источник") },
            confirmButton = {
                TextButton(
                    onClick = {
                        dialogVisible = false
                        cameraLauncher.launch(null)
                    }
                ) {
                    Text("Камера")
                }
            },
            dismissButton = {
                TextButton(
                    onClick = {
                        dialogVisible = false
                        galleryLauncher.launch("image/*")
                    }
                ) {
                    Text("Галерея")
                }
            },
        )
    }
}

private data class PhotoPayload(
    val previewUri: String,
    val base64: String,
    val hash: String,
    val qualityOk: Boolean,
)

private fun loadPhotoPayload(context: Context, uriString: String): PhotoPayload? {
    val uri = runCatching { uriString.toUri() }.getOrNull() ?: return null
    val stream = runCatching { context.contentResolver.openInputStream(uri) }.getOrNull()
        ?: return null
    stream.use {
        val bytes = it.readBytes()
        if (bytes.isEmpty()) return null
        val base64 = Base64.encodeToString(bytes, Base64.NO_WRAP)
        val hash = hashBytes(bytes)
        val (width, height) = decodeImageSize(bytes)
        val qualityOk = isQualityOk(width, height)
        return PhotoPayload(
            previewUri = uriString,
            base64 = base64,
            hash = hash,
            qualityOk = qualityOk,
        )
    }
}

private fun bitmapToPayload(bitmap: Bitmap): PhotoPayload? {
    val bytes = runCatching {
        ByteArrayOutputStream().use { stream ->
            val ok = bitmap.compress(Bitmap.CompressFormat.JPEG, 92, stream)
            if (!ok) return@use null
            stream.toByteArray()
        }
    }.getOrNull() ?: return null
    val base64 = Base64.encodeToString(bytes, Base64.NO_WRAP)
    val hash = hashBytes(bytes)
    val qualityOk = isQualityOk(bitmap.width, bitmap.height)
    return PhotoPayload(
        previewUri = "data:image/jpeg;base64,$base64",
        base64 = base64,
        hash = hash,
        qualityOk = qualityOk,
    )
}

private fun decodeImageSize(bytes: ByteArray): Pair<Int, Int> {
    val opts = BitmapFactory.Options().apply { inJustDecodeBounds = true }
    BitmapFactory.decodeByteArray(bytes, 0, bytes.size, opts)
    return opts.outWidth to opts.outHeight
}

private fun isQualityOk(width: Int, height: Int): Boolean {
    if (width <= 0 || height <= 0) return false
    val minSide = minOf(width, height)
    return minSide >= 480
}

private fun hashBytes(bytes: ByteArray): String {
    val digest = MessageDigest.getInstance("SHA-256").digest(bytes)
    return digest.joinToString("") { "%02x".format(it) }
}
