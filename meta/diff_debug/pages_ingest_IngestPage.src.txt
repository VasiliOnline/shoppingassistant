/**
 * Простой тестовый экран: вводим URL, выбираем источник (пока Avito) и
 * смотрим ingest-результат.
 */
@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun IngestPage(
    modifier: Modifier = Modifier,
    applySafeInsets: Boolean = true,
    extraBottomPadding: Dp = LayoutDefaults.ContentBottomSpacing,
) {
    val loadRawOffer: LoadRawOfferTask = remember { koinGet(LoadRawOfferTask::class.java) }
    val json = remember {
        Json {
            prettyPrint = true
            encodeDefaults = false
        }
    }
    var state by remember { mutableStateOf(IngestUiState()) }
    var sourceMenuExpanded by remember { mutableStateOf(false) }
    val scope = rememberCoroutineScope()
    val scrollState = rememberScrollState()

    fun triggerLoad() {
        val url = state.url.trim()
        if (url.isEmpty()) {
            state = state.copy(error = "Введите URL объявления")
            return
        }

        scope.launch {
            state = state.copy(loading = true, error = null)
            val result = runCatching { loadRawOffer(state.source, url) }
            state = result.fold(
                onSuccess = { ingest -> state.copy(loading = false, result = ingest) },
                onFailure = { err -> state.copy(loading = false, error = err.message ?: "Не удалось загрузить") },
            )
        }
    }

    val showInputError = state.error != null && state.url.isBlank()
    val hasResult = state.result != null
    val screenState = when {
        state.loading && !hasResult -> ScreenState.Loading(LoadingPhase.INITIAL)
        state.loading -> ScreenState.Loading(LoadingPhase.REFRESHING)
        state.error != null && !showInputError -> ScreenState.Error(
            message = state.error.orEmpty(),
            primaryAction = StateAction(
                label = stringResource(R.string.state_error_retry),
                onAction = ::triggerLoad,
                type = StateActionType.RETRY,
            ),
        )
        else -> ScreenState.Content()
    }

    ScreenRoot(
        modifier = modifier
            .fillMaxSize()
            .background(MaterialTheme.colorScheme.surface),
        applySafeInsets = applySafeInsets,
        extraBottomPadding = extraBottomPadding,
    ) { contentPadding ->
        StateHost(
            state = screenState,
            contentPadding = contentPadding,
            screenName = "ingest",
            modifier = Modifier.fillMaxSize(),
        ) { innerPadding ->
            Column(
                modifier = Modifier
                    .fillMaxSize()
                    .verticalScroll(scrollState)
                    .padding(horizontal = LayoutDefaults.HorizontalPadding, vertical = LayoutDefaults.SectionSpacing)
                    .padding(bottom = innerPadding.calculateBottomPadding()),
                verticalArrangement = Arrangement.spacedBy(LayoutDefaults.SectionSpacing),
            ) {
                Text(
                    text = "Ingest-парсер (Avito)",
                    style = MaterialTheme.typography.headlineSmall,
                    fontWeight = FontWeight.Bold,
                )

                ExposedDropdownMenuBox(
                    expanded = sourceMenuExpanded,
                    onExpandedChange = { sourceMenuExpanded = it },
                ) {
                    OutlinedTextField(
                        value = state.source.name,
                        onValueChange = {},
                        readOnly = true,
                        label = { Text("Источник") },
                        trailingIcon = { ExposedDropdownMenuDefaults.TrailingIcon(expanded = sourceMenuExpanded) },
                        modifier = Modifier
                            .menuAnchor()
                            .fillMaxWidth(),
                        colors = outlinedTextFieldColors(
                            disabledBorderColor = MaterialTheme.colorScheme.outline,
                        ),
                        enabled = true,
                    )
                    ExposedDropdownMenu(
                        expanded = sourceMenuExpanded,
                        onDismissRequest = { sourceMenuExpanded = false },
                    ) {
                        DropdownMenuItem(
                            text = { Text("AVITO") },
                            onClick = {
                                state = state.copy(source = SourceType.AVITO)
                                sourceMenuExpanded = false
                            },
                        )
                    }
                }

                OutlinedTextField(
                    value = state.url,
                    onValueChange = { state = state.copy(url = it, error = null) },
                    label = { Text("URL объявления") },
                    placeholder = { Text("https://www.avito.ru/...") },
                    singleLine = true,
                    isError = showInputError,
                    modifier = Modifier.fillMaxWidth(),
                    keyboardOptions = KeyboardOptions(
                        keyboardType = KeyboardType.Uri,
                        imeAction = ImeAction.Done,
                    ),
                )
                if (showInputError) {
                    Text(
                        text = state.error.orEmpty(),
                        style = MaterialTheme.typography.bodySmall,
                        color = MaterialTheme.colorScheme.error,
                    )
                }

                Row(
                    verticalAlignment = Alignment.CenterVertically,
                    horizontalArrangement = Arrangement.spacedBy(12.dp),
                ) {
                    Button(
                        onClick = { triggerLoad() },
                        enabled = !state.loading,
                    ) {
                        Text(text = "Загрузить")
                    }
                    if (state.result != null) {
                        Text(
                            text = "Получено",
                            style = MaterialTheme.typography.labelMedium,
                            color = MaterialTheme.colorScheme.primary,
                        )
                    }
                }

                state.result?.let { ingest ->
                    val pretty = remember(ingest) { json.encodeToString(ingest) }
                    Column(
                        verticalArrangement = Arrangement.spacedBy(8.dp),
                    ) {
                        Text(
                            text = "IngestResult",
                            style = MaterialTheme.typography.titleMedium,
                        )
                        Card(
                            modifier = Modifier.fillMaxWidth(),
                            shape = MaterialTheme.shapes.medium,
                        ) {
                            SelectionContainer {
                                Text(
                                    text = pretty,
                                    modifier = Modifier
                                        .fillMaxWidth()
                                        .padding(12.dp),
                                    style = MaterialTheme.typography.bodyMedium,
                                    fontFamily = FontFamily.Monospace,
                                )
                            }
                        }
                    }
                }
            }
        }
    }
}
