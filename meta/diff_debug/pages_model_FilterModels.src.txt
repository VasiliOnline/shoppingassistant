package com.example.shoppingassistant.feature.pages.model

/** Описание атрибута (ключ, заголовок, набор значений). */
data class AttributeDef(
    val key: String,
    val title: String,
    val options: List<String> = emptyList(),
    val allowedValues: List<ValueDef> = emptyList(),
    val observedValues: List<String> = emptyList(),
    val requiredForSearch: Boolean = false,
    val requiredForOffer: Boolean = false,
    val requiredForExpress: Boolean = false,
    val facetEnabled: Boolean = false,
    val multiValued: Boolean = false,
    /** Если true — пользователь должен выбирать только из options (без ручного ввода). */
    val selectionOnly: Boolean = false,
    /** Если true — атрибут статичен (нельзя менять/удалять, пока якорь не очищен). */
    val locked: Boolean = false,
    /** Если false — значения взяты из наблюдений/БД, справочник неполный. */
    val isDictionaryComplete: Boolean = true,
    /** Значения, запрещенные constraints (показываем, но блокируем выбор). */
    val forbiddenValues: List<String> = emptyList(),
)

/** Допустимое значение атрибута с синонимами для парсинга. */
data class ValueDef(
    val canonical: String,
    val synonyms: List<String> = emptyList(),
    val rank: Int = 0,
)

/** Связка фрагмента текста в query с атрибутом. */
data class BoundSegment(
    val key: String,
    val value: String,
    val range: IntRange,
)

/** Стадия нижнего листа выбора фильтров. */
enum class FilterStage { ATTRS, VALUES }
