class ProfileViewModel(
    private val getCurrentUser: GetCurrentUserUseCase,
    private val getProfileSettings: GetProfileSettingsTask,
    private val profileSettingsStore: ProfileSettingsStore,
    private val getExternalLinks: GetExternalLinksTask,
    private val saveExternalLinks: SaveExternalLinksTask,
    private val getProfileCache: GetProfileCacheTask,
    private val saveProfileCache: SaveProfileCacheTask,
    private val loginUserUseCase: LoginUserUseCase,
    private val registerUserUseCase: RegisterUserUseCase,
    private val logoutUseCase: LogoutUseCase,
    private val changePasswordUseCase: ChangePasswordUseCase,
    private val updateProfileTask: UpdateProfileTask,
    private val requestEmailChangeTask: RequestEmailChangeTask,
    private val confirmEmailChangeTask: ConfirmEmailChangeTask,
    private val deleteAccountTask: DeleteAccountTask,
    private val updatePhotosTask: UpdateProfilePhotosTask,
    private val uploadPhotoUseCase: UploadPhotoUseCase,
    private val authRepository: RemoteAuthRepository,
    private val analyticsLogger: ProfileAnalyticsLogger,
    private val debugAuthStore: DebugAuthStore,
) : ViewModel() {

    private val _uiState = MutableStateFlow(
        ProfileStateTaskUiState(
            isLoading = true,
            error = null,
            state = ProfileState.NotAuthorized,
            isStale = false,
            banner = null,
        ),
    )
    val uiState: StateFlow<ProfileStateTaskUiState> = _uiState.asStateFlow()

    private var currentUserId: String? = null

    private fun logEvent(name: String, attributes: Map<String, String> = emptyMap()) {
        analyticsLogger.log(ProfileAnalyticsEvent(name = name, attributes = attributes))
    }

    private fun logAuthResult(base: String, result: AuthResult) {
        when (result) {
            is AuthResult.Success -> logEvent("${base}_success")
            is AuthResult.Error -> logEvent(
                "${base}_failed",
                mapOf("error" to result.error.name),
            )
        }
    }

    init {
        reload()
    }

    fun reload() {
        viewModelScope.launch { loadProfile() }
    }

    fun logout() {
        viewModelScope.launch {
            if (BuildConfig.DEBUG && debugAuthStore.user.value != null) {
                debugAuthStore.logout()
                currentUserId = null
                _uiState.value = ProfileStateTaskUiState(
                    isLoading = false,
                    error = null,
                    state = ProfileState.NotAuthorized,
                    isStale = false,
                    banner = null,
                )
                return@launch
            }
            runCatching { logoutUseCase() }
                .onSuccess { logEvent("profile_logout") }
                .onFailure { throwable ->
                    logEvent(
                        "profile_logout_failed",
                        mapOf("error" to throwable.javaClass.simpleName),
                    )
                }
            currentUserId = null
            reload()
        }
    }

    suspend fun login(email: String, password: String): AuthResult {
        val result = loginUserUseCase(email.trim(), password)
        logAuthResult("profile_login", result)
        if (result is AuthResult.Success) {
            debugAuthStore.logout()
        }
        return result
    }

    suspend fun register(email: String, password: String, displayName: String?): AuthResult {
        val result = registerUserUseCase(email.trim(), password, displayName)
        logAuthResult("profile_register", result)
        if (result is AuthResult.Success) {
            debugAuthStore.logout()
        }
        return result
    }

    suspend fun updatePhone(newPhone: String) {
        runCatching { updateProfileTask(ProfileUpdatePayload(phone = newPhone.trim())) }
            .onSuccess { logEvent("profile_phone_updated") }
            .onFailure { throwable ->
                logEvent(
                    "profile_phone_update_failed",
                    mapOf("error" to throwable.javaClass.simpleName),
                )
                throw throwable
            }
        reload()
    }

    suspend fun changePassword(oldPassword: String, newPassword: String): AuthResult {
        val result = changePasswordUseCase(oldPassword, newPassword)
        if (result is AuthResult.Success) {
            reload()
        }
        logAuthResult("profile_password_change", result)
        return result
    }

    suspend fun requestEmailChange(email: String) {
        runCatching { requestEmailChangeTask(email.trim()) }
            .onSuccess { logEvent("profile_email_change_requested") }
            .onFailure { throwable ->
                logEvent(
                    "profile_email_change_request_failed",
                    mapOf("error" to throwable.javaClass.simpleName),
                )
                throw throwable
            }
    }

    suspend fun confirmEmailChange(token: String): AuthResult {
        val result = confirmEmailChangeTask(token.trim())
        if (result is AuthResult.Success) {
            reload()
        }
        logAuthResult("profile_email_change_confirm", result)
        return result
    }

    suspend fun deleteAccount(): Boolean {
        val success = deleteAccountTask()
        if (success) {
            reload()
            logEvent("profile_account_delete_requested")
        } else {
            logEvent("profile_account_delete_failed")
        }
        return success
    }

    suspend fun startPasswordReset(email: String): String? =
        runCatching { authRepository.startPasswordReset(email) }
            .onSuccess { logEvent("profile_password_reset_requested") }
            .onFailure { throwable ->
                logEvent(
                    "profile_password_reset_request_failed",
                    mapOf("error" to throwable.javaClass.simpleName),
                )
            }
            .getOrThrow()

    suspend fun resetPassword(resetToken: String, newPassword: String) {
        runCatching { authRepository.resetPassword(resetToken = resetToken, newPassword = newPassword) }
            .onSuccess { logEvent("profile_password_reset_completed") }
            .onFailure { throwable ->
                logEvent(
                    "profile_password_reset_failed",
                    mapOf("error" to throwable.javaClass.simpleName),
                )
                throw throwable
            }
    }

    suspend fun uploadPhoto(bytes: ByteArray, filename: String, contentType: String?): String =
        runCatching { uploadPhotoUseCase(bytes, filename, contentType) }
            .onFailure { throwable ->
                logEvent(
                    "profile_photo_upload_failed",
                    mapOf("error" to throwable.javaClass.simpleName),
                )
            }
            .getOrThrow()

    suspend fun updatePhotos(photos: List<String>) {
        runCatching { updatePhotosTask(photos) }
            .onSuccess { logEvent("profile_photos_saved") }
            .onFailure { throwable ->
                logEvent(
                    "profile_photos_save_failed",
                    mapOf("error" to throwable.javaClass.simpleName),
                )
                throw throwable
            }
        reload()
    }

    fun updateSettings(settings: ProfileSettings) {
        viewModelScope.launch {
            val userId = currentUserId
            runCatching {
                profileSettingsStore.update(userId, settings)
                updateState { authorized ->
                    authorized.copy(settings = settings)
                }
                saveSnapshotFromState()
            }
                .onSuccess { logEvent("profile_settings_saved") }
                .onFailure { throwable ->
                    logEvent(
                        "profile_settings_save_failed",
                        mapOf("error" to throwable.javaClass.simpleName),
                    )
                }
        }
    }

    fun updateLinks(links: List<ExternalLink>) {
        viewModelScope.launch {
            val userId = currentUserId ?: return@launch
            runCatching {
                saveExternalLinks(userId, links)
                updateState { authorized ->
                    authorized.copy(externalLinks = links)
                }
                saveSnapshotFromState()
            }
                .onSuccess { logEvent("profile_links_saved") }
                .onFailure { throwable ->
                    logEvent(
                        "profile_links_save_failed",
                        mapOf("error" to throwable.javaClass.simpleName),
                    )
                }
        }
    }

    fun debugLogin() {
        if (!BuildConfig.DEBUG) return
        debugAuthStore.login()
        logEvent("debug_login_click")
        reload()
    }

    private suspend fun loadProfile() {
        val debugUser = debugAuthStore.user.value
        if (BuildConfig.DEBUG && debugUser != null) {
            currentUserId = debugUser.id
            _uiState.value = ProfileStateTaskUiState(
                isLoading = false,
                error = null,
                state = debugUser.toAuthorizedState(),
                isStale = false,
                banner = null,
            )
            return
        }
        val cached = runCatching { getProfileCache() }.getOrNull()
        if (cached != null) {
            _uiState.value = _uiState.value.copy(
                isLoading = false,
                error = null,
                state = cached.toAuthorizedState(),
                isStale = false,
                banner = null,
            )
        } else {
            _uiState.value = _uiState.value.copy(isLoading = true, error = null)
        }

        runCatching { getCurrentUser() }.onSuccess { user ->
            if (user == null) {
                currentUserId = null
                _uiState.value = ProfileStateTaskUiState(
                    isLoading = false,
                    error = null,
                    state = ProfileState.NotAuthorized,
                    isStale = false,
                    banner = null,
                )
                return@onSuccess
            }

            val userId = user.id.toString()
            currentUserId = userId

            val settings = runCatching { getProfileSettings(userId) }
                .getOrElse { ProfileSettings() }
            profileSettingsStore.load(userId)

            val links = runCatching { getExternalLinks(userId) }
                .getOrElse { emptyList() }

            val snapshot = user.toSnapshot(settings, links)
            runCatching { saveProfileCache(snapshot) }

            _uiState.value = ProfileStateTaskUiState(
                isLoading = false,
                error = null,
                state = snapshot.toAuthorizedState(),
                isStale = false,
                banner = null,
            )
        }.onFailure { throwable ->
            val kind = when (throwable) {
                is IOException -> ProfileErrorKind.NETWORK
                else -> ProfileErrorKind.UNKNOWN
            }
            logEvent(
                "profile_load_failed",
                mapOf(
                    "kind" to kind.name,
                    "error" to throwable.javaClass.simpleName,
                ),
            )
            if (kind == ProfileErrorKind.NETWORK && cached != null) {
                _uiState.value = ProfileStateTaskUiState(
                    isLoading = false,
                    error = null,
                    state = cached.toAuthorizedState(),
                    isStale = true,
                    banner = ProfileBanner.OFFLINE,
                )
            } else {
                _uiState.value = ProfileStateTaskUiState(
                    isLoading = false,
                    error = ProfileErrorUi(
                        kind = kind,
                        messageResId = when (kind) {
                            ProfileErrorKind.NETWORK -> R.string.profile_error_network_body
                            ProfileErrorKind.UNKNOWN -> R.string.profile_error_unknown_body
                        },
                    ),
                    state = ProfileState.NotAuthorized,
                    isStale = false,
                    banner = null,
                )
            }
        }
    }

    private fun updateState(block: (ProfileState.Authorized) -> ProfileState.Authorized) {
        val current = _uiState.value.state as? ProfileState.Authorized ?: return
        _uiState.value = _uiState.value.copy(state = block(current))
    }

    private suspend fun saveSnapshotFromState() {
        val current = _uiState.value.state as? ProfileState.Authorized ?: return
        val snapshot = current.toSnapshot()
        runCatching { saveProfileCache(snapshot) }
    }
}

    ProfileState.Authorized(
        id = id,
        email = email,
        emails = listOf(
            ProfileEmail(
                address = email,
                isPrimary = true,
                isVerified = emailVerified,
            ),
        ),
        displayName = displayName,
        avatarUrl = avatarUrl,
        city = city,
        phone = phone,
        registeredAt = registeredAt,
        emailVerified = emailVerified,
        emailVerifiedAt = emailVerifiedAt,
        phoneVerifiedAt = phoneVerifiedAt,
        trustScore = trustScore,
        dealsCount = dealsCount,
        favoritesCount = favoritesCount,
        alertsCount = alertsCount,
        photos = photos,
        settings = settings,
        externalLinks = externalLinks,
    )

private fun AuthUser.toSnapshot(
    settings: ProfileSettings,
    links: List<ExternalLink>,
): ProfileSnapshot {
    val sanitizedPhotos = if (photos.isNotEmpty()) photos else listOfNotNull(avatarUrl)
    return ProfileSnapshot(
        id = id.toString(),
        email = email,
        displayName = displayName ?: email,
        avatarUrl = avatarUrl,
        city = city,
        phone = phone,
        registeredAt = createdAt,
        emailVerified = emailVerified,
        emailVerifiedAt = emailVerifiedAt,
        phoneVerifiedAt = phoneVerifiedAt,
        photos = sanitizedPhotos,
        settings = settings,
        externalLinks = links,
    )
}
