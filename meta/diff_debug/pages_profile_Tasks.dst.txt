/**
 * Общие модели и манифест задач страницы профиля.
 */
/**
 * Базовое состояние экрана профиля.
 *
 * Authorized содержит все данные, которые нужны UI, включая дату регистрации
 * и список фото/ссылок.
 */
sealed class ProfileState {
    data object NotAuthorized : ProfileState()

    data class Authorized(
        val id: String,
        val email: String,
        val emails: List<ProfileEmail> = emptyList(),
        val displayName: String,
        val avatarUrl: String?,
        val city: String?,
        val phone: String?,
        val registeredAt: Long?,
        val trustScore: Int,
        val dealsCount: Int,
        val favoritesCount: Int,
        val alertsCount: Int,
        val photos: List<String> = emptyList(),
        val settings: ProfileSettings = ProfileSettings(),
        val externalLinks: List<ExternalLink> = emptyList(),
    ) : ProfileState()
}

data class ProfileEmail(
    val address: String,
    val isPrimary: Boolean = false,
    val isVerified: Boolean = false,
)

enum class ProfileTaskId {
    SHEET,
    STATE,
    ERROR,
    UNAUTHORIZED,
    AUTHORIZED,
    LOGIN,
    REGISTER,
    FORGOT_PASSWORD,
    PERSONAL_DATA,
    SETTINGS,
}

data class ProfileTaskDescriptor(
    val id: ProfileTaskId,
    val path: String,
    val description: String,
)
