@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun SubscriptionsPage(

    onRegisterClick: () -> Unit = {},
    onGoToMain: () -> Unit = {},
    onOpenOffer: ((Long) -> Unit)? = null,
    onPrivacyPolicy: (() -> Unit)? = null,
    onBack: (() -> Unit)? = null,

    val state by viewModel.state.collectAsState()
    var showSettings by remember { mutableStateOf(false) }

            primaryAction = StateAction(
                label = "Войти",
                onAction = onLoginClick,
                type = StateActionType.LOGIN,
            ),
            secondaryAction = StateAction(
                label = "Создать аккаунт",
                onAction = onRegisterClick,
                type = StateActionType.CREATE,
            ),
            icon = Icons.Outlined.Lock,
        )
        is SubscriptionsAuthState.Error -> ScreenState.Error(
            title = "Не удалось проверить авторизацию",
            message = authState.message,
            primaryAction = StateAction(
                label = stringResource(R.string.state_error_retry),
                onAction = viewModel::retryAuth,
                type = StateActionType.RETRY,
            ),
            icon = Icons.Outlined.Lock,
        )
        SubscriptionsAuthState.Authorized -> ScreenState.Content()
    }
    val snackbarHostState = remember { SnackbarHostState() }
    LaunchedEffect(state.actionMessage) {
        val msg = state.actionMessage ?: return@LaunchedEffect
        snackbarHostState.showSnackbar(message = msg, duration = SnackbarDuration.Short)
        viewModel.consumeActionMessage()
    }

    ScreenRoot(
        modifier = modifier.fillMaxSize(),
        applySafeInsets = applySafeInsets,
        extraBottomPadding = extraBottomPadding,
    ) { contentPadding ->

            state = authScreenState,
            contentPadding = contentPadding,
            screenName = "subscriptions_auth",
            modifier = Modifier.fillMaxSize(),
        ) {
            if (authState != SubscriptionsAuthState.Authorized) return@StateHost

                snackbarHost = { SnackbarHost(snackbarHostState) },

                    if (state.mode == SubscriptionsMode.CREATE) {

                            title = { Text("Новая подписка") },
                            navigationIcon = {
                                IconButton(onClick = viewModel::requestCloseCreate) {
                                    Icon(Icons.Outlined.Close, contentDescription = "Закрыть")
                                }
                            },
                            windowInsets = WindowInsets(0, 0, 0, 0),

@Composable
private fun NotificationsTab(
    state: ListLoadState<SubscriptionsInboxNotification>,
    canLoadMore: Boolean,
    isLoadingMore: Boolean,
    filter: NotificationsFilter,
    onRetry: () -> Unit,
    onLoadMore: () -> Unit,
    onMarkRead: (Long) -> Unit,
    onMarkAllRead: () -> Unit,
    onFilterChange: (NotificationsFilter) -> Unit,
    onOpenOffer: ((Long) -> Unit)?,
    lastSyncAt: Long?,
    onOpenSettings: () -> Unit,
    onLoginClick: () -> Unit,
    contentPadding: PaddingValues,
) {
    val screenState: ScreenState = when (state) {
        ListLoadState.Loading -> ScreenState.Loading(LoadingPhase.INITIAL)
        ListLoadState.Unauthorized -> ScreenState.Empty(
            reason = EmptyReason.OTHER,
            title = "Нужна авторизация",
            message = "Войдите, чтобы видеть уведомления.",
            primaryAction = StateAction(
                label = "Войти",
                onAction = onLoginClick,
                type = StateActionType.LOGIN,
            ),
            icon = Icons.Outlined.Lock,
        )
        ListLoadState.Empty -> ScreenState.Empty(
            reason = EmptyReason.NO_MESSAGES,
            title = "Уведомлений пока нет",
            message = "Когда сработает условие подписки - здесь появятся уведомления.",
            primaryAction = StateAction(
                label = stringResource(R.string.state_empty_action_default),
                onAction = onRetry,
                type = StateActionType.RETRY,
            ),
            icon = Icons.Outlined.NotificationsNone,
        )
        is ListLoadState.Error -> {
            val lastSync = lastSyncAt?.let { "Последняя синхронизация: ${formatTime(it)}" }
            ScreenState.Error(
                message = listOfNotNull(state.message, lastSync).joinToString("\n"),
                title = "Не удалось загрузить уведомления",
                primaryAction = StateAction(
                    label = stringResource(R.string.state_error_retry),
                    onAction = onRetry,
                    type = StateActionType.RETRY,
                ),
                icon = Icons.Outlined.NotificationsNone,
            )
        }
        is ListLoadState.Content -> ScreenState.Content()
    }

    StateHost(
        state = screenState,
        contentPadding = contentPadding,
        screenName = "subscriptions_notifications",
        modifier = Modifier.fillMaxSize(),
    ) { paddingValues ->
        if (state !is ListLoadState.Content) return@StateHost
        val items = state.items
        val unreadCount = items.count { !it.isRead }

        val filtered = when (filter) {
            NotificationsFilter.ALL -> items
            NotificationsFilter.UNREAD -> items.filter { !it.isRead }
        }
        val listItems = remember(filtered) { buildNotificationsList(filtered) }

        Box(modifier = Modifier.padding(bottom = paddingValues.calculateBottomPadding())) {
            LazyColumn(
                modifier = Modifier.fillMaxSize(),
                verticalArrangement = Arrangement.spacedBy(LayoutDefaults.SectionSpacing),
                contentPadding = PaddingValues(vertical = LayoutDefaults.SectionSpacing),
            ) {
                item {
                    Column(verticalArrangement = Arrangement.spacedBy(8.dp)) {
                        Row(
                            modifier = Modifier.fillMaxWidth(),
                            verticalAlignment = Alignment.CenterVertically,
                            horizontalArrangement = Arrangement.SpaceBetween,
                        ) {
                            Text(
                                text = "Уведомления",
                                style = MaterialTheme.typography.titleMedium,
                                fontWeight = FontWeight.SemiBold,
                            )
                            Row(horizontalArrangement = Arrangement.spacedBy(6.dp)) {
                                TextButton(
                                    onClick = onMarkAllRead,
                                    enabled = unreadCount > 0,
                                ) {
                                    Text("Прочитать всё")
                                }
                                TextButton(onClick = onOpenSettings) {
                                    Icon(Icons.Outlined.Settings, contentDescription = null)
                                    Spacer(Modifier.width(4.dp))
                                    Text("Настройки")
                                }
                            }
                        }
                        Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                            FilterChip(
                                selected = filter == NotificationsFilter.ALL,
                                onClick = { onFilterChange(NotificationsFilter.ALL) },
                                label = { Text("Все") }
                            )
                            FilterChip(
                                selected = filter == NotificationsFilter.UNREAD,
                                onClick = { onFilterChange(NotificationsFilter.UNREAD) },
                                label = { Text("Непрочитанные") }
                            )
                        }
                        lastSyncAt?.let { ts ->
                            Text(
                                text = "Обновлено: ${formatTime(ts)}",
                                style = MaterialTheme.typography.labelSmall,
                                color = MaterialTheme.colorScheme.onSurfaceVariant,
                            )
                        }
                    }
                }

                items(listItems) { model ->
                    when (model) {
                        is NotificationsListItem.Header -> {
                            Text(
                                text = model.title,
                                style = MaterialTheme.typography.labelLarge,
                                color = MaterialTheme.colorScheme.onSurfaceVariant,
                                modifier = Modifier.padding(top = 6.dp, bottom = 2.dp),
                            )
                        }

                        is NotificationsListItem.Item -> {
                            val n = model.notification
                            val bg =
                                if (n.isRead) MaterialTheme.colorScheme.surfaceVariant
                                else MaterialTheme.colorScheme.primary.copy(alpha = 0.14f)

                            Column(
                                modifier = Modifier
                                    .fillMaxWidth()
                                    .background(bg, MaterialTheme.shapes.medium)
                                    .clickable { onMarkRead(n.id) }
                                    .padding(12.dp)
                            ) {
                                Row(
                                    modifier = Modifier.fillMaxWidth(),
                                    horizontalArrangement = Arrangement.SpaceBetween,
                                    verticalAlignment = Alignment.CenterVertically,
                                ) {
                                    Text(
                                        text = n.message,
                                        style = MaterialTheme.typography.bodyMedium,
                                        modifier = Modifier.weight(1f),
                                        maxLines = 2,
                                        overflow = TextOverflow.Ellipsis,
                                    )
                                    if (!n.isRead) {
                                        Text(
                                            text = "Новые",
                                            style = MaterialTheme.typography.labelSmall,
                                            color = MaterialTheme.colorScheme.primary,
                                        )
                                    }
                                }
                                buildInboxDetails(n)?.let { details ->
                                    Text(
                                        text = details,
                                        style = MaterialTheme.typography.bodySmall,
                                        color = MaterialTheme.colorScheme.onSurfaceVariant,
                                    )
                                }
                                Row(
                                    modifier = Modifier.fillMaxWidth(),
                                    horizontalArrangement = Arrangement.SpaceBetween,
                                    verticalAlignment = Alignment.CenterVertically,
                                ) {
                                    Text(
                                        text = formatTime(n.createdAt),
                                        style = MaterialTheme.typography.labelSmall,
                                        color = MaterialTheme.colorScheme.onSurfaceVariant,
                                    )
                                    if (onOpenOffer != null) {
                                        TextButton(
                                            onClick = {
                                                onMarkRead(n.id)
                                                onOpenOffer(n.offerId)
                                            }
                                        ) {
                                            Text("Открыть")
                                        }
                                    }
                                }
                            }
                        }
                    }
                }

                if (isLoadingMore) {
                    item {
                        Box(Modifier.fillMaxWidth(), contentAlignment = Alignment.Center) {
                            CircularProgressIndicator(modifier = Modifier.size(24.dp))
                        }
                    }
                } else if (canLoadMore) {
                    item {
                        OutlinedButton(
                            onClick = onLoadMore,
                            modifier = Modifier.fillMaxWidth(),
                        ) {
                            Text("Показать ещё")
                        }
                    }
                }
            }
        }
    }
}

private sealed interface NotificationsListItem {
    data class Header(val title: String) : NotificationsListItem
    data class Item(
        val notification: SubscriptionsInboxNotification,
    ) : NotificationsListItem
}

private sealed interface SubscriptionsListItem {
    data class Header(val title: String) : SubscriptionsListItem
    data class Item(val subscription: TemplateSubscription) : SubscriptionsListItem
}

private enum class NotificationDayGroup { TODAY, YESTERDAY, EARLIER }

private fun buildNotificationsList(
    items: List<SubscriptionsInboxNotification>,
): List<NotificationsListItem> {
    if (items.isEmpty()) return emptyList()

    val out = ArrayList<NotificationsListItem>(items.size + 3)
    var lastGroup: NotificationDayGroup? = null
    for (n in items) {
        val group = notificationDayGroup(n.createdAt)
        if (group != lastGroup) {
            out += NotificationsListItem.Header(groupTitle(group))
            lastGroup = group
        }
        out += NotificationsListItem.Item(n)
    }
    return out
}

private fun applySubscriptionsSearch(
    items: List<TemplateSubscription>,
    query: String,
): List<TemplateSubscription> {
    val needle = query.trim().lowercase()
    if (needle.isBlank()) return items
    return items.filter { s ->
        val data = s.snapshot.data
        val attrText = data.attrs.joinToString(" ") { "${it.key} ${it.value}" }
        val haystack = listOfNotNull(
            data.freeText,
            data.anchorId,
            data.categoryCode,
            attrText,
        ).joinToString(" ").lowercase()
        haystack.contains(needle)
    }
}

private fun buildSubscriptionsListItems(
    items: List<TemplateSubscription>,
    groupBy: SubscriptionsGroupBy,
): List<SubscriptionsListItem> {
    if (groupBy == SubscriptionsGroupBy.NONE) {
        return items.map { SubscriptionsListItem.Item(it) }
    }

    val grouped = LinkedHashMap<String, MutableList<TemplateSubscription>>()
    items.forEach { s ->
        val key = when (groupBy) {
            SubscriptionsGroupBy.CATEGORY -> s.snapshot.data.categoryCode
                ?.takeIf { it.isNotBlank() }
                ?.replace(".", " • ")
                ?: "Другое"
            SubscriptionsGroupBy.BRAND -> s.snapshot.data.attrs
                .firstOrNull { it.key.equals("brand", ignoreCase = true) }
                ?.value
                ?.takeIf { it.isNotBlank() }
                ?: "Другое"
            SubscriptionsGroupBy.NONE -> "Все"
        }
        grouped.getOrPut(key) { mutableListOf() }.add(s)
    }

    val out = ArrayList<SubscriptionsListItem>(items.size + grouped.size)
    grouped.forEach { (key, group) ->
        out += SubscriptionsListItem.Header(key)
        out += group.map { SubscriptionsListItem.Item(it) }
    }
    return out
}

private fun groupTitle(group: NotificationDayGroup): String =
    when (group) {
        NotificationDayGroup.TODAY -> "Сегодня"
        NotificationDayGroup.YESTERDAY -> "Вчера"
        NotificationDayGroup.EARLIER -> "Ранее"
    }

private fun notificationDayGroup(ms: Long): NotificationDayGroup {
    val now = Calendar.getInstance()
    val todayY = now.get(Calendar.YEAR)
    val todayD = now.get(Calendar.DAY_OF_YEAR)

    val cal = Calendar.getInstance().apply { timeInMillis = ms }
    val y = cal.get(Calendar.YEAR)
    val d = cal.get(Calendar.DAY_OF_YEAR)

    return when {
        y == todayY && d == todayD -> NotificationDayGroup.TODAY
        y == todayY && d == todayD - 1 -> NotificationDayGroup.YESTERDAY
        y == todayY - 1 && todayD == 1 && d == cal.getActualMaximum(Calendar.DAY_OF_YEAR) -> NotificationDayGroup.YESTERDAY
        else -> NotificationDayGroup.EARLIER
    }
}

@Composable
private fun SubscriptionsTabContent(
    state: SubscriptionsState,
    onEdit: (TemplateSubscription) -> Unit,
    onDelete: (String) -> Unit,
    onActiveChange: (templateId: String, active: Boolean) -> Unit,
    onRetry: () -> Unit,
    onCreateSubscription: () -> Unit,
    onSearchChange: (String) -> Unit,
    onGroupChange: (SubscriptionsGroupBy) -> Unit,
    onLoginClick: () -> Unit,
    contentPadding: PaddingValues,
) {
    Column(Modifier.fillMaxSize()) {
        val showControls = state.subscriptions is ListLoadState.Content || state.subscriptions is ListLoadState.Empty
        if (showControls) {
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = 12.dp, vertical = 10.dp),
                verticalArrangement = Arrangement.spacedBy(8.dp),
            ) {
                TextField(
                    value = state.searchQuery,
                    onValueChange = onSearchChange,
                    leadingIcon = { Icon(Icons.Outlined.Search, contentDescription = null) },
                    placeholder = { Text("Поиск по названию, бренду, категории") },
                    singleLine = true,
                    modifier = Modifier.fillMaxWidth(),
                )
                Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                    FilterChip(
                        selected = state.subscriptionsGroupBy == SubscriptionsGroupBy.NONE,
                        onClick = { onGroupChange(SubscriptionsGroupBy.NONE) },
                        label = { Text("Без групп") },
                    )
                    FilterChip(
                        selected = state.subscriptionsGroupBy == SubscriptionsGroupBy.CATEGORY,
                        onClick = { onGroupChange(SubscriptionsGroupBy.CATEGORY) },
                        label = { Text("Категории") },
                    )
                    FilterChip(
                        selected = state.subscriptionsGroupBy == SubscriptionsGroupBy.BRAND,
                        onClick = { onGroupChange(SubscriptionsGroupBy.BRAND) },
                        label = { Text("Бренды") },
                    )
                }
                state.subscriptionsLastSyncAt?.let { ts ->
                    Text(
                        text = "Обновлено: ${formatTime(ts)}",
                        style = MaterialTheme.typography.labelSmall,
                        color = MaterialTheme.colorScheme.onSurfaceVariant,
                    )
                }
            }
        }

        val subsState = state.subscriptions as? ListLoadState.Content
        val sorted = subsState?.items
            ?.sortedWith(
                compareByDescending<TemplateSubscription> { it.isActive }
                    .thenByDescending { it.updatedAtMillis }
            )
            .orEmpty()
        val filtered = if (subsState != null) applySubscriptionsSearch(sorted, state.searchQuery) else emptyList()
        val screenState: ScreenState = when (val current = state.subscriptions) {
            ListLoadState.Loading -> ScreenState.Loading(LoadingPhase.INITIAL)
            ListLoadState.Unauthorized -> ScreenState.Empty(
                reason = EmptyReason.OTHER,
                title = "Нужна авторизация",
                message = "Войдите, чтобы видеть подписки.",
                primaryAction = StateAction(
                    label = "Войти",
                    onAction = onLoginClick,
                    type = StateActionType.LOGIN,
                ),
                icon = Icons.Outlined.Lock,
            )
            ListLoadState.Empty -> {
                val primaryAction = StateAction(
                    label = "Создать подписку",
                    onAction = onCreateSubscription,
                    type = StateActionType.CREATE,
                )
                ScreenState.Empty(
                    reason = EmptyReason.NO_SUBSCRIPTIONS,
                    title = "Подписок пока нет",
                    message = "Создайте подписку на категорию или товар.\nНапример: \"iPhone 16 256GB\" или \"Nike Air Max\".",
                    primaryAction = primaryAction,
                    icon = Icons.Outlined.StarOutline,
                )
            }
            is ListLoadState.Error -> {
                val lastSync = state.subscriptionsLastSyncAt?.let { "Последняя синхронизация: ${formatTime(it)}" }
                ScreenState.Error(
                    message = listOfNotNull(current.message, lastSync).joinToString("\n"),
                    title = "Не удалось загрузить подписки",
                    primaryAction = StateAction(
                        label = stringResource(R.string.state_error_retry),
                        onAction = onRetry,
                        type = StateActionType.RETRY,
                    ),
                    icon = Icons.Outlined.StarOutline,
                )
            }
            is ListLoadState.Content -> {
                if (filtered.isEmpty() && state.searchQuery.isNotBlank()) {
                    ScreenState.Empty(
                        reason = EmptyReason.FILTERS_TOO_STRICT,
                        title = "Ничего не найдено",
                        message = "Попробуйте изменить запрос или сбросить фильтр.",
                        primaryAction = StateAction(
                            label = "Сбросить фильтр",
                            onAction = { onSearchChange("") },
                            type = StateActionType.RESET_FILTERS,
                        ),
                        icon = Icons.Outlined.Search,
                    )
                } else {
                    ScreenState.Content()
                }
            }
        }

        StateHost(
            state = screenState,
            contentPadding = contentPadding,
            screenName = "subscriptions_list",
            modifier = Modifier.fillMaxSize(),
        ) { paddingValues ->
            if (state.subscriptions !is ListLoadState.Content) return@StateHost
            val listItems = buildSubscriptionsListItems(filtered, state.subscriptionsGroupBy)

            Box(modifier = Modifier.padding(bottom = paddingValues.calculateBottomPadding())) {
                LazyColumn(
                    modifier = Modifier.fillMaxSize(),
                    verticalArrangement = Arrangement.spacedBy(LayoutDefaults.SectionSpacing),
                    contentPadding = PaddingValues(vertical = LayoutDefaults.SectionSpacing),
                ) {
                    items(listItems) { item ->
                        when (item) {
                            is SubscriptionsListItem.Header -> Text(
                                text = item.title,
                                style = MaterialTheme.typography.labelLarge,
                                color = MaterialTheme.colorScheme.onSurfaceVariant,
                                modifier = Modifier.padding(top = 6.dp, bottom = 2.dp),
                            )
                            is SubscriptionsListItem.Item -> {
                                val s = item.subscription
                                val hint = buildAutoPauseHint(s)
                                SubscriptionCard(
                                    title = buildTemplateTitle(s.snapshot),
                                    subtitle = buildSummary(s),
                                    hint = hint,
                                    active = s.isActive,
                                    onActiveChange = { onActiveChange(s.templateId, it) },
                                    onEdit = { onEdit(s) },
                                    onDelete = { onDelete(s.templateId) }
                                )
                            }
                        }
                    }
                }
            }
        }
    }
}

private fun buildTemplateTitle(snapshot: TemplateSnapshot): String {
    val data = snapshot.data
    val freeText = data.freeText?.trim().orEmpty()
    if (freeText.isNotBlank()) return freeText
    return when (data.anchorType) {
        com.example.shoppingassistant.domain.template.TemplateAnchorType.PRODUCT -> data.anchorId
        com.example.shoppingassistant.domain.template.TemplateAnchorType.CATEGORY -> data.categoryCode ?: data.anchorId
    }.ifBlank { data.anchorId }
}

private fun buildTemplateSummary(snapshot: TemplateSnapshot): String {
    val data = snapshot.data
    val parts = mutableListOf<String>()
    data.categoryCode
        ?.takeIf { it.isNotBlank() }
        ?.let { parts += "Категория: ${it.replace(".", " • ")}" }
    val attrs = data.attrs
        .mapNotNull { it.value.takeIf { v -> v.isNotBlank() } }
        .distinct()
        .take(3)
    if (attrs.isNotEmpty()) {
        parts += "Фильтры: ${attrs.joinToString(", ")}"
    }
    return parts.joinToString(" • ")
}

private fun buildAutoPauseHint(s: TemplateSubscription): String? {
    if (!s.isActive) return null
    return if (s.minAlertIntervalMinutes <= 30) {
        "Слишком часто? Увеличьте интервал или поставьте на паузу."
    } else null
}

private fun buildInboxDetails(n: SubscriptionsInboxNotification): String? {
    val baseline = formatMoney(n.baselinePriceMinor, n.currency)
    val current = formatMoney(n.currentPriceMinor, n.currency)
    val drop = n.dropPercent.takeIf { it > 0.0 }?.let { "−${it.toInt()}%" }
    return listOfNotNull(
        "Было: $baseline",
        "Сейчас: $current",
        drop,
    ).joinToString(" • ").takeIf { it.isNotBlank() }
}

private fun formatMoney(minor: Long, currency: String): String {
    val major = minor / 100.0
    return "${"%.2f".format(major)} $currency"
}

private fun buildSummary(s: TemplateSubscription): String {
    val templateSummary = buildTemplateSummary(s.snapshot)

    val rules = mutableListOf<String>()
    val hasNewOffer = s.triggers.any { it.type == TemplateTriggerType.NEW_OFFER_APPEARED }
    if (hasNewOffer) rules += "Новый оффер"

    val max = s.triggers.firstOrNull { it.type == TemplateTriggerType.MAX_PRICE }
    if (max?.moneyMinor != null) {
        val major = max.moneyMinor!! / 100.0
        rules += "Цена ≤ ${"%.2f".format(major)} ${max.currency ?: ""}".trim()
    }

    val drop = s.triggers.firstOrNull { it.type == TemplateTriggerType.PRICE_DROPPED_PERCENT }?.numberValue
    if (drop != null) rules += "Падение ≥ $drop%"

    rules += "Интервал: ${s.minAlertIntervalMinutes} мин"
    rules += "Обновлено: ${formatDate(s.updatedAtMillis)}"

    val lines = mutableListOf<String>()
    if (templateSummary.isNotBlank()) lines += templateSummary
    lines += rules.joinToString(" • ")
    return lines.joinToString("\n")
}

private fun formatDate(ms: Long): String {
    val df = SimpleDateFormat("dd.MM.yyyy", Locale.getDefault())
    return df.format(Date(ms))
}

private fun formatTime(ms: Long): String {
    val df = SimpleDateFormat("dd.MM • HH:mm", Locale.getDefault())
    return df.format(Date(ms))
}
