private const val MAX_EXTRA_KEYS = 5

@Composable
@OptIn(ExperimentalMaterial3Api::class)
fun TrackEditPage(
    trackId: String,
    onBack: () -> Unit,
    onOpenTrack: (String) -> Unit = {},
    viewModel: TrackEditViewModel = koinViewModel(),
) {
    val state by viewModel.state.collectAsState()
    val snackbarHostState = remember { SnackbarHostState() }
    val scope = rememberCoroutineScope()
    val trackRepository: TrackRepository = remember { koinGet(TrackRepository::class.java) }
    val updateTrackTarget: UpdateTrackTargetTask = remember { koinGet(UpdateTrackTargetTask::class.java) }
    val catalogRepository: CatalogRepository = remember { koinGet(CatalogRepository::class.java) }
    val getProfileCache: GetProfileCacheTask = remember { koinGet(GetProfileCacheTask::class.java) }
    val leafCategories = rememberLeafCategories(catalogRepository)
    var targetDraft by remember { mutableStateOf<TrackTargetDraft?>(null) }
    var showTargetEditor by remember { mutableStateOf(false) }
    var targetDedupConflictTrackId by remember { mutableStateOf<String?>(null) }
    var countryCode by remember { mutableStateOf("") }

    LaunchedEffect(getProfileCache) {
        val profileCountry = runCatching { getProfileCache()?.settings?.countryCode }.getOrNull().orEmpty()
        countryCode = profileCountry.ifBlank { java.util.Locale.getDefault().country }
    }

    LaunchedEffect(trackId) {
        viewModel.load(trackId)
        val track = runCatching { trackRepository.getTrack(trackId) }.getOrNull()
        if (track != null) {
            targetDraft = track.toTargetDraft()
        }
    }

    LaunchedEffect(state.uiMessage) {
        val message = state.uiMessage ?: return@LaunchedEffect
        val result = snackbarHostState.showSnackbar(message.text, actionLabel = message.actionLabel)
        if (result == SnackbarResult.ActionPerformed) {
            when (message.action) {
                UiAction.RetrySave -> viewModel.apply()
                UiAction.RetryLoad -> viewModel.retryLoad()
                null -> Unit
            }
        }
        viewModel.consumeUiMessage()
    }

    LaunchedEffect(state.closeRequested) {
        if (state.closeRequested) {
            viewModel.consumeCloseRequest()
            onBack()
        }
    }

    LaunchedEffect(state.navigateToTrackId) {
        val track = state.navigateToTrackId ?: return@LaunchedEffect
        viewModel.consumeNavigateToTrack()
        onOpenTrack(track)
    }

    BackHandler {
        viewModel.onBackPressed()
    }

    Scaffold(
        topBar = {
            TopAppBar(
                title = {
                    Row(verticalAlignment = Alignment.CenterVertically) {
                        Text(
                            text = "Фильтры",
                            maxLines = 1,
                            overflow = TextOverflow.Ellipsis,
                        )
                        if (state.hasUnsavedChanges) {
                            Text(
                                text = " • Изменено",
                                style = MaterialTheme.typography.bodySmall,
                                color = MaterialTheme.colorScheme.primary,
                                modifier = Modifier.padding(start = 6.dp),
                            )
                        }
                    }
                },
                navigationIcon = {
                    IconButton(onClick = { viewModel.onBackPressed() }) {
                        Icon(Icons.AutoMirrored.Outlined.ArrowBack, contentDescription = "Назад")
                    }
                },
                actions = {
                    if (state.activeCount > 0) {
                        TextButton(onClick = { viewModel.resetAll() }) {
                            Text("Сбросить")
                        }
                    }
                },
            )
        },
        snackbarHost = { SnackbarHost(snackbarHostState) },
        bottomBar = {
            TrackEditApplyBar(
                isSaving = state.save is SaveState.Saving,
                applyEnabled = state.hasUnsavedChanges && state.save !is SaveState.Saving,
                onCancel = { viewModel.onBackPressed() },
                onApply = { viewModel.apply() },
            )
        },
    ) { padding ->
        val filters = state.edited
        when (val load = state.load) {
            LoadState.Loading -> {
                Column(
                    modifier = Modifier
                        .fillMaxSize()
                        .padding(padding)
                        .padding(16.dp),
                    verticalArrangement = Arrangement.spacedBy(12.dp),
                ) {
                    Text("Загрузка фильтров…", style = MaterialTheme.typography.bodyMedium)
                }
            }
            is LoadState.Error -> {
                Column(
                    modifier = Modifier
                        .fillMaxSize()
                        .padding(padding)
                        .padding(16.dp),
                    verticalArrangement = Arrangement.spacedBy(12.dp),
                ) {
                    Text("Не удалось загрузить", style = MaterialTheme.typography.titleMedium)
                    Text(load.message, style = MaterialTheme.typography.bodySmall)
                    if (load.canRetry) {
                        Button(onClick = { viewModel.retryLoad() }) {
                            Text("Повторить")
                        }
                    }
                }
            }
            LoadState.Content -> {
                LazyColumn(
                    modifier = Modifier
                        .fillMaxSize()
                        .padding(padding)
                        .padding(horizontal = 16.dp, vertical = 12.dp),
                    verticalArrangement = Arrangement.spacedBy(12.dp),
                ) {
                    item {
                        SummaryBlock(
                            title = state.trackTitle.ifBlank { "Трек" },
                            activeCount = state.activeCount,
                        )
                    }
                    item {
                        SectionHeader("Цель трека")
                    }
                    item {
                        TrackEditRow(
                            title = "Цель",
                            value = targetDraft.toTargetLabel(),
                            onClick = { showTargetEditor = true },
                        )
                    }
                    item { SectionHeader("Основные фильтры") }
                    item {
                        TrackEditRow(
                            title = "Регион",
                            value = filters.region ?: "Любой",
                            onClick = { viewModel.openSheet(TrackFilterKey.REGION) },
                        )
                    }
                    item {
                        TrackEditRow(
                            title = "Доставка",
                            value = filters.delivery ?: "Любая",
                            onClick = { viewModel.openSheet(TrackFilterKey.DELIVERY) },
                        )
                    }
                    item {
                        TrackEditRow(
                            title = "Состояние",
                            value = filters.condition ?: "Любое",
                            onClick = { viewModel.openSheet(TrackFilterKey.CONDITION) },
                        )
                    }
                    item { SectionHeader("Продавец") }
                    item {
                        TrackEditRow(
                            title = "Продавец",
                            value = filters.seller ?: "Любой",
                            onClick = { viewModel.openSheet(TrackFilterKey.SELLER) },
                        )
                    }
                    item { SectionHeader("Дополнительные фильтры") }
                    if (filters.extra.isEmpty()) {
                        item {
                            Text(
                                text = "Дополнительные фильтры не заданы",
                                style = MaterialTheme.typography.bodySmall,
                                color = MaterialTheme.colorScheme.onSurfaceVariant,
                            )
                        }
                    } else {
                        items(filters.extra.entries.sortedBy { it.key.lowercase() }) { entry ->
                            TrackEditRow(
                                title = entry.key,
                                value = entry.value,
                                onClick = { viewModel.openSheet(TrackFilterKey.EXTRA_VALUE, extraKey = entry.key) },
                            )
                        }
                    }
                    item {
                        if (filters.extra.size < MAX_EXTRA_KEYS) {
                            AddExtraRow(onClick = { viewModel.openSheet(TrackFilterKey.EXTRA_KEY) })
                        } else {
                            Text(
                                text = "Можно добавить не более $MAX_EXTRA_KEYS фильтров",
                                style = MaterialTheme.typography.bodySmall,
                                color = MaterialTheme.colorScheme.onSurfaceVariant,
                            )
                        }
                    }
                }
            }
        }
    }

    when (val dialog = state.blockingDialog) {
        BlockingDialog.ConfirmDiscard -> {
            AlertDialog(
                onDismissRequest = { viewModel.cancelDiscard() },
                title = { Text("Не сохранять изменения?") },
                text = { Text("Вы уверены, что хотите выйти без сохранения?") },
                confirmButton = {
                    TextButton(onClick = { viewModel.confirmDiscard() }) {
                        Text("Не сохранять")
                    }
                },
                dismissButton = {
                    TextButton(onClick = { viewModel.cancelDiscard() }) {
                        Text("Отмена")
                    }
                },
            )
        }
        is BlockingDialog.DedupConflict -> {
            AlertDialog(
                onDismissRequest = { viewModel.cancelDiscard() },
                title = { Text("Такой трек уже существует") },
                text = { Text(dialog.message) },
                confirmButton = {
                    TextButton(onClick = { viewModel.navigateToTrack(dialog.existingTrackId) }) {
                        Text("Открыть")
                    }
                },
                dismissButton = {
                    TextButton(onClick = { viewModel.cancelDiscard() }) {
                        Text("Остаться")
                    }
                },
            )
        }
        null -> Unit
    }

    state.sheetState?.let { sheet ->
        TrackFilterOptionsSheet(
            sheet = sheet,
            onDismiss = { viewModel.closeSheet() },
            onQueryChange = viewModel::onSheetQueryChanged,
            onClear = viewModel::clearSelection,
            onRetry = viewModel::retryOptions,
            onOptionSelected = viewModel::onOptionSelected,
            onCustomInputChange = viewModel::onCustomInputChanged,
            onCustomApply = viewModel::applyCustomInput,
        )
    }

    val initialDraft = targetDraft
    if (showTargetEditor && initialDraft != null) {
        TrackTargetAttributesWizardSheet(
            mode = TargetWizardMode.EDIT,
            entrypoint = "track.edit_target",
            initial = initialDraft,
            categories = leafCategories,
            catalogRepository = catalogRepository,
            countryCode = countryCode,
            onDismiss = { showTargetEditor = false },
            onSave = { saveResult ->
                showTargetEditor = false
                scope.launch {
                    val result = runCatching {
                        updateTrackTarget(
                            trackId = trackId,
                            input = TrackTargetUpdateInput(
                                type = saveResult.draft.type,
                                targetSpec = TrackTargetSpec(
                                    categoryCode = saveResult.draft.categoryCode,
                                    attributes = saveResult.draft.attributes,
                                    attributesMulti = saveResult.draft.attributesMulti,
                                    attributesRange = saveResult.draft.attributesRange,
                                    matchKey = saveResult.draft.matchKeyOrNull(),
                                    queryText = saveResult.draft.queryText,
                                    schemaVersion = saveResult.draft.schemaVersion,
                                    taxonomyVersion = saveResult.draft.taxonomyVersion,
                                    locale = saveResult.draft.locale,
                                    unboundTokens = saveResult.draft.unboundTokens,
                                ),
                                title = saveResult.title,
                            ),
                        )
                    }.getOrElse {
                        snackbarHostState.showSnackbar(it.message ?: "Не удалось обновить цель")
                        return@launch
                    }
                    when (result) {
                        is TrackTargetUpdateResult.Updated -> {
                            targetDraft = result.track.toTargetDraft()
                            FlowMetrics.markEvent(
                                "Tracks.TargetUpdated",
                                "trackId=$trackId dedupKey=${saveResult.dedupKey.orEmpty()}",
                            )
                            viewModel.load(trackId)
                            snackbarHostState.showSnackbar("Цель обновлена")
                        }
                        is TrackTargetUpdateResult.AlreadyExists -> {
                            targetDedupConflictTrackId = result.track.id
                            FlowMetrics.markEvent(
                                "Tracks.DedupConflict",
                                "requestedDedupKey=${saveResult.dedupKey.orEmpty()} existingTrackId=${result.track.id}",
                            )
                        }
                        is TrackTargetUpdateResult.InvalidInput -> {
                            snackbarHostState.showSnackbar(result.reason.ifBlank { "Некорректная цель" })
                        }
                        TrackTargetUpdateResult.NotFound -> {
                            snackbarHostState.showSnackbar("Трек не найден")
                        }
                    }
                }
            },
        )
    }

    targetDedupConflictTrackId?.let { existingTrackId ->
        AlertDialog(
            onDismissRequest = { targetDedupConflictTrackId = null },
            title = { Text("Такой трек уже существует") },
            text = { Text("Изменение цели создаёт дубликат существующего трека.") },
            confirmButton = {
                TextButton(
                    onClick = {
                        targetDedupConflictTrackId = null
                        onOpenTrack(existingTrackId)
                    },
                ) {
                    Text("Открыть существующий")
                }
            },
            dismissButton = {
                TextButton(onClick = { targetDedupConflictTrackId = null }) {
                    Text("Отменить изменение")
                }
            },
        )
    }
}

            modifier = Modifier.padding(14.dp),

                imageVector = Icons.AutoMirrored.Outlined.KeyboardArrowRight,

@Composable
@OptIn(ExperimentalMaterial3Api::class)
private fun TrackFilterOptionsSheet(
    sheet: SheetOptionsState,
    onDismiss: () -> Unit,
    onQueryChange: (String) -> Unit,
    onClear: () -> Unit,
    onRetry: () -> Unit,
    onOptionSelected: (TrackFilterOption) -> Unit,
    onCustomInputChange: (String) -> Unit,
    onCustomApply: () -> Unit,
) {
    val sheetState = rememberModalBottomSheetState(skipPartiallyExpanded = true)
    ModalBottomSheet(onDismissRequest = onDismiss, sheetState = sheetState) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 16.dp, vertical = 12.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp),
        ) {
            Text(text = sheetTitle(sheet), style = MaterialTheme.typography.titleMedium)
            if (sheet.searchEnabled) {
                TextField(
                    value = sheet.query,
                    onValueChange = onQueryChange,
                    placeholder = { Text("Поиск") },
                    modifier = Modifier.fillMaxWidth(),
                )
            }
            Row(
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically,
                modifier = Modifier.fillMaxWidth(),
            ) {
                if (sheet.allowEmpty) {
                    TextButton(onClick = onClear) { Text("Очистить") }
                }
                if (sheet.isLoading) {
                    CircularProgressIndicator(
                        strokeWidth = 2.dp,
                        modifier = Modifier.sizeIn(minWidth = 18.dp, minHeight = 18.dp),
                    )
                }
            }

            if (sheet.errorMessage != null) {
                Text(
                    text = sheet.errorMessage,
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.error,
                )
                Button(onClick = onRetry, modifier = Modifier.height(44.dp)) { Text("Повторить") }
            } else {
                OptionsSection(
                    title = "Рекомендуем",
                    options = sheet.options.filter { it.isSuggested },
                    selectedId = sheet.selectedId,
                    onOptionSelected = onOptionSelected,
                )
                OptionsSection(
                    title = "Недавние",
                    options = sheet.options.filter { it.isRecent && !it.isSuggested },
                    selectedId = sheet.selectedId,
                    onOptionSelected = onOptionSelected,
                )
                OptionsSection(
                    title = "Все варианты",
                    options = sheet.options.filter { !it.isSuggested && !it.isRecent },
                    selectedId = sheet.selectedId,
                    onOptionSelected = onOptionSelected,
                )
            }

            if (sheet.allowCustom) {
                Column(verticalArrangement = Arrangement.spacedBy(8.dp)) {
                    Text(text = "Свой вариант", style = MaterialTheme.typography.titleSmall)
                    TextField(
                        value = sheet.customInput,
                        onValueChange = onCustomInputChange,
                        placeholder = { Text("Введите значение") },
                        modifier = Modifier.fillMaxWidth(),
                    )
                    Button(
                        onClick = onCustomApply,
                        enabled = sheet.customInput.trim().isNotBlank(),
                        modifier = Modifier.align(Alignment.End),
                    ) {
                        Text("Применить")
                    }
                }
            }
        }
    }
}

@Composable
private fun OptionsSection(
    title: String,
    options: List<TrackFilterOption>,
    selectedId: String?,
    onOptionSelected: (TrackFilterOption) -> Unit,
) {
    if (options.isEmpty()) return
    Column(verticalArrangement = Arrangement.spacedBy(8.dp)) {
        Text(text = title, style = MaterialTheme.typography.titleSmall)
        options.forEach { option ->
            val isSelected = option.id == selectedId
            val enabled = !option.isDisabled
            Surface(
                shape = MaterialTheme.shapes.medium,
                color = if (isSelected) MaterialTheme.colorScheme.primary.copy(alpha = 0.12f)
                else MaterialTheme.colorScheme.surfaceVariant.copy(alpha = 0.4f),
                modifier = Modifier
                    .fillMaxWidth()
                    .clickable(enabled = enabled) { onOptionSelected(option) },
            ) {
                Column(
                    modifier = Modifier.padding(horizontal = 12.dp, vertical = 10.dp),
                    verticalArrangement = Arrangement.spacedBy(2.dp),
                ) {
                    Row(horizontalArrangement = Arrangement.spacedBy(6.dp), verticalAlignment = Alignment.CenterVertically) {
                        Text(text = option.label, style = MaterialTheme.typography.bodyMedium)
                        option.count?.let {
                            Text(
                                text = "• $it",
                                style = MaterialTheme.typography.bodySmall,
                                color = MaterialTheme.colorScheme.onSurfaceVariant,
                            )
                        }
                    }
                    val reason = option.disabledReason
                    if (option.isDisabled && !reason.isNullOrBlank()) {
                        Text(
                            text = reason,
                            style = MaterialTheme.typography.bodySmall,
                            color = MaterialTheme.colorScheme.error,
                        )
                    }
                }
            }
        }
    }
}

private fun sheetTitle(sheet: SheetOptionsState): String {
    return when (sheet.key) {
        TrackFilterKey.REGION -> "Регион"
        TrackFilterKey.DELIVERY -> "Доставка"
        TrackFilterKey.CONDITION -> "Состояние"
        TrackFilterKey.SELLER -> "Продавец"
        TrackFilterKey.EXTRA_KEY -> "Дополнительный параметр"
        TrackFilterKey.EXTRA_VALUE -> sheet.extraKey?.let { "Значение: $it" } ?: "Значение"
    }
}

private fun Track.toTargetDraft(): TrackTargetDraft {
    val resolvedMatchKey = target.spec?.matchKey
    val (brand, model) = TrackMatchKeyFactory.parse(resolvedMatchKey) ?: ("" to "")
    val normalizedType = when (type) {
        TrackType.CATEGORY -> TrackType.CATEGORY
        else -> TrackType.PRODUCT
    }
    return TrackTargetDraft(
        type = normalizedType,
        brand = brand,
        model = model,
        categoryCode = target.spec?.categoryCode ?: categoryCode,
        attributes = target.spec?.attributes.orEmpty(),
        attributesMulti = target.spec?.attributesMulti.orEmpty(),
        attributesRange = target.spec?.attributesRange.orEmpty(),
        queryText = target.spec?.queryText,
        schemaVersion = target.spec?.schemaVersion ?: 1,
        taxonomyVersion = target.spec?.taxonomyVersion,
        locale = target.spec?.locale,
        unboundTokens = target.spec?.unboundTokens.orEmpty(),
    )
}

private fun TrackTargetDraft?.toTargetLabel(): String {
    val draft = this ?: return "Не выбрано"
    return when (draft.type) {
        TrackType.CATEGORY -> draft.categoryCode?.takeIf { it.isNotBlank() } ?: "Категория"
        TrackType.PRODUCT -> {
            val brandModel = listOf(draft.brand, draft.model).joinToString(" ").trim()
            when {
                brandModel.isNotBlank() -> brandModel
                !draft.categoryCode.isNullOrBlank() -> "Товар в ${draft.categoryCode}"
                else -> "Товар"
            }
        }
        else -> "Цель"
    }
}
