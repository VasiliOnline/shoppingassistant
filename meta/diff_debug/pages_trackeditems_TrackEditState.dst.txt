package com.example.shoppingassistant.feature.pages.trackeditems

import com.example.shoppingassistant.domain.tracks.TrackFilters

enum class TrackEditSheet {
    Region,
    Delivery,
    Condition,
    Seller,
    Extra,
}

data class TrackEditState(
    val trackId: String = "",
    val isLoading: Boolean = true,
    val isSaving: Boolean = false,
    val trackTitle: String = "",
    val original: TrackFilters = TrackFilters(),
    val edited: TrackFilters = TrackFilters(),
    val activeCount: Int = 0,
    val errorMessage: String? = null,
    val confirmDiscardVisible: Boolean = false,
    val openSheet: TrackEditSheet? = null,
    val extraDraftKey: String = "",
    val extraDraftValue: String = "",
    val extraDraftOriginalKey: String? = null,
    val actionMessage: String? = null,
    val actionLabel: String? = null,
    val actionTrackId: String? = null,
    val closeScreen: Boolean = false,
) {
    val hasUnsavedChanges: Boolean
        get() = edited != original
}
