enum class TargetWizardMode { CREATE, EDIT }

data class TrackTargetDraft(
    val type: TrackType = TrackType.PRODUCT,
    val brand: String = "",
    val model: String = "",
    val categoryCode: String? = null,
    val attributes: Map<String, String> = emptyMap(),
    val sourceLabel: String? = null,
    val autoFilledAttributes: Set<String> = emptySet(),
) {
    fun matchKeyOrNull(): String? = TrackMatchKeyFactory.fromBrandModel(brand, model)

    fun dedupKeyOrNull(): String? = when (type) {
        TrackType.PRODUCT -> matchKeyOrNull()
            ?: categoryCode?.trim()?.uppercase()?.takeIf { it.isNotBlank() }
        TrackType.CATEGORY -> categoryCode?.trim()?.uppercase()?.takeIf { it.isNotBlank() }
        else -> null
    }
}

data class TrackTargetWizardResult(
    val draft: TrackTargetDraft,
    val title: String,
    val dedupKey: String?,
)

private enum class SchemaStatus { LOADING, OK, EMPTY, FAILED }

private enum class AttributeControl {
    CHIPS,
    DROPDOWN,
    INPUT,
}

private data class AttributeFieldSchema(
    val code: String,
    val title: String,
    val required: Boolean,
    val values: List<String>,
    val control: AttributeControl,
    val order: Int,
)

private data class ValidationResult(
    val brandError: String? = null,
    val modelError: String? = null,
    val categoryError: String? = null,
    val attributeErrors: Map<String, String> = emptyMap(),
) {
    val isValid: Boolean
        get() = brandError == null &&
            modelError == null &&
            categoryError == null &&
            attributeErrors.isEmpty()
}

private fun TrackTargetDraft.withType(type: TrackType): TrackTargetDraft = when (type) {
    TrackType.PRODUCT -> copy(type = TrackType.PRODUCT)
    TrackType.CATEGORY -> copy(type = TrackType.CATEGORY, brand = "", model = "")
    else -> copy(type = type)
}

@Composable
fun rememberLeafCategories(catalogRepository: CatalogRepository): List<Category> {
    var categories by remember { mutableStateOf<List<Category>>(emptyList()) }
    LaunchedEffect(catalogRepository) {
        categories = runCatching { catalogRepository.listCategories() }.getOrElse { emptyList() }
    }
    return remember(categories) {
        val parents = categories.mapNotNull { it.parentCode }.toSet()
        categories
            .filter { !it.code.isNullOrBlank() && !parents.contains(it.code) }
            .sortedBy { (it.title ?: it.code).lowercase() }
    }
}

fun TrackTargetDraft.isValid(profile: CategoryProfile?): Boolean {
    val baseValid = when (type) {
        TrackType.PRODUCT, TrackType.CATEGORY -> !categoryCode.isNullOrBlank()
        else -> false
    }
    if (!baseValid) return false
    return missingRequiredAttributeCodes(profile).isEmpty()
}

fun TrackTargetDraft.requiredAttributeCodes(profile: CategoryProfile?): Set<String> {
    profile ?: return emptySet()
    if (profile.attributes.isEmpty()) return emptySet()

    val required = LinkedHashSet<String>()
    val requiredByCategory = profile.categoryAttributes
        .asSequence()
        .filter { attr -> attr.isRequiredForCategory }
        .map { attr -> normalizeDraftAttributeCode(attr.attributeCode) }
        .filter { code -> code.isNotEmpty() }
        .toSet()

    profile.attributes
        .asSequence()
        .forEach { def ->
            val normalizedCode = normalizeDraftAttributeCode(def.code)
            if (normalizedCode.isEmpty()) return@forEach
            if (def.requiredForSearch || normalizedCode in requiredByCategory) {
                required += def.code.trim()
            }
        }

    val normalizedValues = normalizedDraftAttributesForValidation()
    profile.requiredIfRules.forEach { rule ->
        val requiredCode = rule.requiredAttributeCode.trim()
        if (requiredCode.isBlank()) return@forEach
        val shouldRequire = rule.whenAll.isNotEmpty() &&
            rule.whenAll.all { condition ->
                val currentValue = normalizedValues[normalizeDraftAttributeCode(condition.attributeCode)]
                matchesDraftCondition(condition, currentValue)
            }
        if (shouldRequire) {
            required += requiredCode
        }
    }

    return required
}

fun TrackTargetDraft.missingRequiredAttributeCodes(profile: CategoryProfile?): Set<String> {
    val requiredCodes = requiredAttributeCodes(profile)
    if (requiredCodes.isEmpty()) return emptySet()

    val normalizedValues = normalizedDraftAttributesForValidation()
    return requiredCodes
        .filter { code ->
            val normalizedCode = normalizeDraftAttributeCode(code)
            normalizedValues[normalizedCode].isNullOrBlank()
        }
        .toSet()
}

private fun normalizeDraftAttributeCode(rawCode: String): String =
    rawCode.trim().lowercase()

private fun TrackTargetDraft.normalizedDraftAttributesForValidation(): Map<String, String> {
    val normalized = LinkedHashMap<String, String>()
    attributes.forEach { (key, value) ->
        val normalizedCode = normalizeDraftAttributeCode(key)
        val normalizedValue = value.trim()
        if (normalizedCode.isNotEmpty() && normalizedValue.isNotEmpty()) {
            normalized[normalizedCode] = normalizedValue
        }
    }
    val normalizedBrand = TrackMatchKeyFactory.normalizePart(brand)?.trim()?.takeIf { it.isNotEmpty() }
    if (normalizedBrand != null) {
        normalized.putIfAbsent("brand", normalizedBrand)
    }
    val normalizedModel = TrackMatchKeyFactory.normalizePart(model)?.trim()?.takeIf { it.isNotEmpty() }
    if (normalizedModel != null) {
        normalized.putIfAbsent("model", normalizedModel)
    }
    return normalized
}

private fun matchesDraftCondition(
    condition: AttributeCondition,
    currentValue: String?,
): Boolean {
    if (currentValue.isNullOrBlank()) return false
    return when (condition.op) {
        AttributeConditionOp.EQUALS_ANY ->
            condition.values.any { value -> currentValue.equals(value, ignoreCase = true) }
        AttributeConditionOp.STARTS_WITH_ANY ->
            condition.values.any { value -> currentValue.startsWith(value, ignoreCase = true) }
    }
}

fun TrackTargetDraft.previewTitle(categories: List<Category>): String {
    val resolvedCategoryTitle = categoryCode
        ?.trim()
        ?.takeIf { it.isNotBlank() }
        ?.let { code -> categories.firstOrNull { it.code == code }?.title ?: code }
    return when (type) {
        TrackType.PRODUCT -> {
            val brandModel = listOf(brand, model)
                .joinToString(" ")
                .trim()
                .takeIf { it.isNotBlank() }
            if (brandModel != null) {
                brandModel
            } else {
                val attributePreview = attributes.entries
                    .asSequence()
                    .map { (key, value) -> key.trim() to value.trim() }
                    .filter { (key, value) -> key.isNotBlank() && value.isNotBlank() }
                    .filterNot { (key, _) ->
                        key.equals("brand", ignoreCase = true) ||
                            key.equals("model", ignoreCase = true)
                    }
                    .sortedBy { (key, _) -> key.lowercase() }
                    .take(2)
                    .joinToString(", ") { (key, value) -> "$key=$value" }
                    .takeIf { it.isNotBlank() }
                when {
                    !resolvedCategoryTitle.isNullOrBlank() && !attributePreview.isNullOrBlank() ->
                        "$resolvedCategoryTitle · $attributePreview"
                    !resolvedCategoryTitle.isNullOrBlank() -> resolvedCategoryTitle
                    else -> "Товар"
                }
            }
        }
        TrackType.CATEGORY -> {
            if (resolvedCategoryTitle.isNullOrBlank()) {
                "Категория"
            } else {
                resolvedCategoryTitle
            }
        }
        else -> "Цель"
    }
}

@OptIn(ExperimentalMaterial3Api::class, ExperimentalLayoutApi::class)
@Composable
fun TrackTargetAttributesWizardSheet(
    mode: TargetWizardMode,
    entrypoint: String,
    initial: TrackTargetDraft,
    categories: List<Category>,
    catalogRepository: CatalogRepository,
    countryCode: String,
    onDismiss: () -> Unit,
    onSave: (TrackTargetWizardResult) -> Unit,
) {
    val sheetState = rememberModalBottomSheetState(skipPartiallyExpanded = true)
    val autoFilledInitial = remember(mode, initial) {
        if (mode == TargetWizardMode.CREATE && initial.autoFilledAttributes.isEmpty()) {
            initial.attributes.keys
        } else {
            initial.autoFilledAttributes
        }
    }
    var draft by remember(initial, autoFilledInitial) {
        mutableStateOf(initial.copy(autoFilledAttributes = autoFilledInitial))
    }
    var showCategoryPicker by remember { mutableStateOf(false) }
    var schemaStatus by remember { mutableStateOf(SchemaStatus.EMPTY) }
    var schemaFields by remember { mutableStateOf(minimalFallbackSchema()) }
    var categoryProfile by remember { mutableStateOf<CategoryProfile?>(null) }
    var schemaErrorMessage by remember { mutableStateOf<String?>(null) }
    var showOptionalAttributes by remember { mutableStateOf(false) }
    var showUnknownMap by remember { mutableStateOf(false) }
    var saveAttempted by remember { mutableStateOf(false) }
    var touchedFields by remember { mutableStateOf<Set<String>>(emptySet()) }
    var schemaVersion by remember { mutableStateOf(0) }
    var closeLogged by remember { mutableStateOf(false) }

    fun closeWithReason(reason: String, action: () -> Unit) {
        if (!closeLogged) {
            TargetWizardAnalytics.closed(reason)
            closeLogged = true
        }
        action()
    }

    val selectedCategoryTitle = remember(draft.categoryCode, categories) {
        categories.firstOrNull { it.code == draft.categoryCode }?.title ?: draft.categoryCode.orEmpty()
    }

    LaunchedEffect(mode, entrypoint) {
        TargetWizardAnalytics.opened(mode, entrypoint)
    }

    LaunchedEffect(draft.categoryCode, schemaVersion) {
        val code = draft.categoryCode?.trim().orEmpty()
        if (code.isBlank()) {
            schemaStatus = SchemaStatus.EMPTY
            schemaFields = minimalFallbackSchema()
            categoryProfile = null
            schemaErrorMessage = null
            return@LaunchedEffect
        }

        schemaStatus = SchemaStatus.LOADING
        schemaErrorMessage = null
        val profile = runCatching { catalogRepository.getCategoryProfile(code) }
            .onFailure {
                schemaStatus = SchemaStatus.FAILED
                schemaFields = minimalFallbackSchema()
                categoryProfile = null
                schemaErrorMessage = "Не удалось загрузить параметры категории"
                TargetWizardAnalytics.schemaLoadFailed(code, it::class.simpleName ?: "unknown")
            }
            .getOrNull()

        if (profile == null) {
            if (schemaStatus != SchemaStatus.FAILED) {
                schemaStatus = SchemaStatus.EMPTY
                schemaFields = minimalFallbackSchema()
                categoryProfile = null
            }
            return@LaunchedEffect
        }

        categoryProfile = profile
        val fromSchema = buildSchemaFields(profile)
        if (fromSchema.isEmpty()) {
            schemaStatus = SchemaStatus.EMPTY
            schemaFields = minimalFallbackSchema()
        } else {
            schemaStatus = SchemaStatus.OK
            schemaFields = fromSchema
        }
    }

    val validation = remember(draft, schemaFields, categoryProfile) {
        validateDraft(draft, schemaFields, categoryProfile)
    }

    val knownSchemaCodes = remember(schemaFields) { schemaFields.map { it.code }.toSet() }
    val unknownMapEntries = remember(draft.attributes, knownSchemaCodes) {
        draft.attributes.entries
            .filter { !knownSchemaCodes.contains(it.key) }
            .sortedBy { it.key.lowercase() }
    }

    val requiredFieldCodes = remember(draft, categoryProfile, schemaFields) {
        val dynamicRequired = draft.requiredAttributeCodes(categoryProfile)
            .map(::normalizeDraftAttributeCode)
            .toSet()
        if (dynamicRequired.isNotEmpty()) {
            dynamicRequired
        } else {
            schemaFields.filter { it.required }.map { normalizeDraftAttributeCode(it.code) }.toSet()
        }
    }
    val requiredFields = remember(schemaFields, requiredFieldCodes) {
        schemaFields.filter { field -> normalizeDraftAttributeCode(field.code) in requiredFieldCodes }
    }
    val optionalFields = remember(schemaFields, requiredFieldCodes) {
        schemaFields.filterNot { field -> normalizeDraftAttributeCode(field.code) in requiredFieldCodes }
    }
    val visibleOptionalFields = remember(optionalFields, showOptionalAttributes) {
        if (showOptionalAttributes) optionalFields else optionalFields.take(6)
    }
    val visibleFields = remember(requiredFields, visibleOptionalFields) {
        requiredFields + visibleOptionalFields
    }

    ModalBottomSheet(
        onDismissRequest = { closeWithReason("back") { onDismiss() } },
        sheetState = sheetState,
    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(16.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp),
        ) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                verticalAlignment = Alignment.CenterVertically,
                horizontalArrangement = Arrangement.SpaceBetween,
            ) {
                Text(
                    text = "Параметры отслеживания",
                    style = MaterialTheme.typography.titleMedium,
                )
                Row(verticalAlignment = Alignment.CenterVertically) {
                    TextButton(
                        onClick = {
                            TargetWizardAnalytics.saveTapped(validation.isValid)
                            saveAttempted = true
                            if (!validation.isValid) return@TextButton

                            val normalized = normalizeDraftForSave(draft)
                            closeWithReason("save") {
                                onSave(
                                    TrackTargetWizardResult(
                                        draft = normalized,
                                        title = normalized.previewTitle(categories),
                                        dedupKey = normalized.dedupKeyOrNull(),
                                    ),
                                )
                            }
                        },
                        enabled = validation.isValid,
                    ) {
                        Text(if (mode == TargetWizardMode.CREATE) "Готово" else "Сохранить")
                    }
                    IconButton(onClick = { closeWithReason("cancel") { onDismiss() } }) {
                        Icon(Icons.Outlined.Close, contentDescription = "Закрыть")
                    }
                }
            }

            Text("Цель", style = MaterialTheme.typography.titleSmall)
            Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                TargetTypeChip(
                    label = "Товар",
                    selected = draft.type == TrackType.PRODUCT,
                    onClick = {
                        draft = draft.withType(TrackType.PRODUCT).copy(sourceLabel = null)
                        touchedFields = touchedFields + "category"
                        TargetWizardAnalytics.targetChanged(draft)
                    },
                )
                TargetTypeChip(
                    label = "Категория",
                    selected = draft.type == TrackType.CATEGORY,
                    onClick = {
                        draft = draft.withType(TrackType.CATEGORY).copy(sourceLabel = null)
                        touchedFields = touchedFields + "category"
                        TargetWizardAnalytics.targetChanged(draft)
                    },
                )
            }

            if (draft.type == TrackType.PRODUCT) {
                OutlinedTextField(
                    value = draft.brand,
                    onValueChange = {
                        draft = draft.copy(brand = it, sourceLabel = null)
                        touchedFields = touchedFields + "brand"
                        TargetWizardAnalytics.targetChanged(draft)
                    },
                    label = { Text("Бренд") },
                    supportingText = {
                        val showError = saveAttempted || touchedFields.contains("brand")
                        if (showError && validation.brandError != null) {
                            Text(validation.brandError ?: "", color = MaterialTheme.colorScheme.error)
                        }
                    },
                    isError = (saveAttempted || touchedFields.contains("brand")) && validation.brandError != null,
                    singleLine = true,
                    modifier = Modifier.fillMaxWidth(),
                )
                OutlinedTextField(
                    value = draft.model,
                    onValueChange = {
                        draft = draft.copy(model = it, sourceLabel = null)
                        touchedFields = touchedFields + "model"
                        TargetWizardAnalytics.targetChanged(draft)
                    },
                    label = { Text("Модель") },
                    supportingText = {
                        val showError = saveAttempted || touchedFields.contains("model")
                        if (showError && validation.modelError != null) {
                            Text(validation.modelError ?: "", color = MaterialTheme.colorScheme.error)
                        }
                    },
                    isError = (saveAttempted || touchedFields.contains("model")) && validation.modelError != null,
                    singleLine = true,
                    modifier = Modifier.fillMaxWidth(),
                )
            }

            CategoryPickerRow(
                categoryTitle = selectedCategoryTitle,
                isError = (saveAttempted || touchedFields.contains("category")) && validation.categoryError != null,
                errorMessage = validation.categoryError,
                onClick = {
                    touchedFields = touchedFields + "category"
                    showCategoryPicker = true
                },
            )

            TargetPreviewCard(
                title = draft.previewTitle(categories),
                type = draft.type,
                countryCode = countryCode,
                sourceLabel = draft.sourceLabel,
            )

            Text("Атрибуты", style = MaterialTheme.typography.titleSmall)

            when (schemaStatus) {
                SchemaStatus.LOADING -> {
                    Text(
                        text = "Загружаем параметры категории…",
                        style = MaterialTheme.typography.bodySmall,
                        color = MaterialTheme.colorScheme.onSurfaceVariant,
                    )
                }
                SchemaStatus.FAILED -> {
                    SchemaErrorBanner(
                        message = schemaErrorMessage ?: "Не удалось загрузить параметры категории",
                        onRetry = { schemaVersion += 1 },
                    )
                }
                SchemaStatus.EMPTY -> {
                    Text(
                        text = "Схема категории недоступна. Показаны общие атрибуты.",
                        style = MaterialTheme.typography.bodySmall,
                        color = MaterialTheme.colorScheme.onSurfaceVariant,
                    )
                }
                SchemaStatus.OK -> Unit
            }

            if (visibleFields.isEmpty()) {
                Text(
                    text = "Атрибуты не заданы для этой категории.",
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                )
            } else {
                LazyColumn(
                    modifier = Modifier
                        .fillMaxWidth()
                        .heightIn(max = 340.dp),
                    verticalArrangement = Arrangement.spacedBy(10.dp),
                ) {
                    items(visibleFields, key = { it.code }) { field ->
                        val value = draft.attributes[field.code].orEmpty()
                        val showError = saveAttempted || touchedFields.contains(field.code)
                        val fieldError = if (showError) validation.attributeErrors[field.code] else null
                        val isAutofilled = draft.autoFilledAttributes.contains(field.code)

                        AttributeFieldEditor(
                            field = field,
                            value = value,
                            isAutofilled = isAutofilled,
                            errorMessage = fieldError,
                            onValueChange = { nextValue ->
                                val wasAutoFilled = draft.autoFilledAttributes.contains(field.code)
                                touchedFields = touchedFields + field.code
                                draft = draft.withAttribute(field.code, nextValue, userEdited = true)
                                TargetWizardAnalytics.attributeChanged(
                                    code = field.code,
                                    controlType = field.control.name.lowercase(),
                                    isAutoFilled = wasAutoFilled,
                                    toEmpty = nextValue.isBlank(),
                                )
                            },
                            onClear = {
                                val wasAutoFilled = draft.autoFilledAttributes.contains(field.code)
                                touchedFields = touchedFields + field.code
                                draft = draft.withAttribute(field.code, "", userEdited = true)
                                TargetWizardAnalytics.attributeChanged(
                                    code = field.code,
                                    controlType = field.control.name.lowercase(),
                                    isAutoFilled = wasAutoFilled,
                                    toEmpty = true,
                                )
                            },
                        )
                    }
                }

                if (optionalFields.size > visibleOptionalFields.size) {
                    TextButton(onClick = { showOptionalAttributes = !showOptionalAttributes }) {
                        Text(if (showOptionalAttributes) "Скрыть лишние" else "Показать ещё")
                    }
                }
            }

            if (schemaStatus == SchemaStatus.OK && unknownMapEntries.isNotEmpty()) {
                TextButton(onClick = { showUnknownMap = !showUnknownMap }) {
                    Text(if (showUnknownMap) "Скрыть map-поля" else "Advanced: Показать всё из map")
                }
                if (showUnknownMap) {
                    unknownMapEntries.forEach { (code, value) ->
                        OutlinedTextField(
                            value = value,
                            onValueChange = {
                                touchedFields = touchedFields + code
                                draft = draft.withAttribute(code, it, userEdited = true)
                                TargetWizardAnalytics.attributeChanged(
                                    code = code,
                                    controlType = "input",
                                    isAutoFilled = false,
                                    toEmpty = it.isBlank(),
                                )
                            },
                            label = { Text(code) },
                            singleLine = true,
                            modifier = Modifier.fillMaxWidth(),
                        )
                    }
                }
            }

            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically,
            ) {
                TextButton(
                    onClick = {
                        TargetWizardAnalytics.resetTapped()
                        draft = if (mode == TargetWizardMode.EDIT) initial.copy(autoFilledAttributes = emptySet()) else TrackTargetDraft()
                        touchedFields = emptySet()
                        saveAttempted = false
                    },
                ) {
                    Text("Сбросить")
                }
                Button(
                    enabled = validation.isValid,
                    onClick = {
                        TargetWizardAnalytics.saveTapped(validation.isValid)
                        saveAttempted = true
                        if (!validation.isValid) return@Button

                        val normalized = normalizeDraftForSave(draft)
                        closeWithReason("save") {
                            onSave(
                                TrackTargetWizardResult(
                                    draft = normalized,
                                    title = normalized.previewTitle(categories),
                                    dedupKey = normalized.dedupKeyOrNull(),
                                ),
                            )
                        }
                    },
                ) {
                    Text(if (mode == TargetWizardMode.CREATE) "Готово" else "Сохранить")
                }
            }
        }
    }

    if (showCategoryPicker) {
        CategoryLeafPickerSheet(
            categories = categories,
            selectedCode = draft.categoryCode,
            onDismiss = { showCategoryPicker = false },
            onSelect = { category ->
                showCategoryPicker = false
                draft = draft.copy(categoryCode = category.code, sourceLabel = null)
                touchedFields = touchedFields + "category"
                TargetWizardAnalytics.categoryPicked(category.code)
                TargetWizardAnalytics.targetChanged(draft)
            },
        )
    }
}

private fun normalizeDraftForSave(draft: TrackTargetDraft): TrackTargetDraft {
    val normalizedCategory = draft.categoryCode?.trim()?.uppercase()?.takeIf { it.isNotBlank() }
    val normalizedAttributes = sanitizeAttributes(draft.attributes)
    return when (draft.type) {
        TrackType.PRODUCT -> draft.copy(
            brand = draft.brand.trim(),
            model = draft.model.trim(),
            categoryCode = normalizedCategory,
            attributes = normalizedAttributes,
            autoFilledAttributes = draft.autoFilledAttributes.intersect(normalizedAttributes.keys),
        )
        TrackType.CATEGORY -> draft.copy(
            brand = "",
            model = "",
            categoryCode = normalizedCategory,
            attributes = normalizedAttributes,
            autoFilledAttributes = draft.autoFilledAttributes.intersect(normalizedAttributes.keys),
        )
        else -> draft.copy(
            categoryCode = normalizedCategory,
            attributes = normalizedAttributes,
            autoFilledAttributes = draft.autoFilledAttributes.intersect(normalizedAttributes.keys),
        )
    }
}

private fun sanitizeAttributes(values: Map<String, String>): Map<String, String> {
    if (values.isEmpty()) return emptyMap()
    return values.entries
        .mapNotNull { (key, value) ->
            val normalizedKey = key.trim()
            val normalizedValue = value.trim()
            if (normalizedKey.isBlank() || normalizedValue.isBlank()) null
            else normalizedKey to normalizedValue
        }
        .sortedBy { it.first.lowercase() }
        .toMap(LinkedHashMap())
}

private fun buildSchemaFields(profile: CategoryProfile): List<AttributeFieldSchema> {
    val attrsByCode = profile.attributes.associateBy { it.code }
    val categoryAttrByCode = profile.categoryAttributes.associateBy { it.attributeCode }
    val dictionaryByCode = profile.valueDictionaries.associateBy { it.attributeCode }

    val orderedCodes = buildList {
        profile.categoryAttributes
            .sortedBy { it.uiOrder }
            .forEach { add(it.attributeCode) }
        profile.attributes.forEach { def ->
            if (!contains(def.code)) add(def.code)
        }
    }

    return orderedCodes
        .mapNotNull { code ->
            val def = attrsByCode[code] ?: return@mapNotNull null
            val catAttr = categoryAttrByCode[code]
            val values = resolveValues(def, dictionaryByCode[code]?.entries?.map { it.canonicalCode }.orEmpty())
            val control = when {
                def.dataType == AttributeDataType.BOOL -> AttributeControl.CHIPS
                values.isNotEmpty() && values.size <= 6 -> AttributeControl.CHIPS
                values.isNotEmpty() -> AttributeControl.DROPDOWN
                else -> AttributeControl.INPUT
            }
            AttributeFieldSchema(
                code = code,
                title = def.title,
                required = def.requiredForSearch || (catAttr?.isRequiredForCategory == true),
                values = values,
                control = control,
                order = catAttr?.uiOrder ?: Int.MAX_VALUE,
            )
        }
        .sortedWith(
            compareByDescending<AttributeFieldSchema> { it.required }
                .thenBy { it.order }
                .thenBy { it.title.lowercase() },
        )
}

private fun resolveValues(def: AttributeDef, dictionaryValues: List<String>): List<String> {
    if (def.dataType == AttributeDataType.BOOL) return listOf("true", "false")
    if (dictionaryValues.isEmpty()) return emptyList()
    return dictionaryValues
        .map { it.trim() }
        .filter { it.isNotBlank() }
        .distinct()
}

private fun validateDraft(
    draft: TrackTargetDraft,
    fields: List<AttributeFieldSchema>,
    profile: CategoryProfile?,
): ValidationResult {
    val categoryError = if (
        (draft.type == TrackType.CATEGORY || draft.type == TrackType.PRODUCT) &&
        draft.categoryCode.isNullOrBlank()
    ) {
        "Выберите категорию"
    } else {
        null
    }

    val valuesByCode = draft.normalizedDraftAttributesForValidation()
    val fieldByCode = fields.associateBy { normalizeDraftAttributeCode(it.code) }
    val requiredErrors = draft.requiredAttributeCodes(profile)
        .mapNotNull { code ->
            val normalizedCode = normalizeDraftAttributeCode(code)
            val value = valuesByCode[normalizedCode]?.trim().orEmpty()
            if (value.isNotBlank()) {
                null
            } else {
                val field = fieldByCode[normalizedCode]
                val title = field?.title ?: code
                val errorKey = field?.code ?: code
                errorKey to "Заполните \"$title\""
            }
        }
        .toMap(LinkedHashMap())

    return ValidationResult(
        brandError = null,
        modelError = null,
        categoryError = categoryError,
        attributeErrors = requiredErrors,
    )
}

private fun TrackTargetDraft.withAttribute(
    code: String,
    value: String,
    userEdited: Boolean,
): TrackTargetDraft {
    val trimmedCode = code.trim()
    if (trimmedCode.isBlank()) return this
    val trimmedValue = value.trim()
    val nextAttributes = if (trimmedValue.isBlank()) {
        attributes - trimmedCode
    } else {
        attributes + (trimmedCode to trimmedValue)
    }
    val nextAutoFilled = if (userEdited) {
        autoFilledAttributes - trimmedCode
    } else {
        autoFilledAttributes
    }
    return copy(attributes = nextAttributes, autoFilledAttributes = nextAutoFilled)
}

@OptIn(ExperimentalLayoutApi::class)
@Composable
private fun AttributeFieldEditor(
    field: AttributeFieldSchema,
    value: String,
    isAutofilled: Boolean,
    errorMessage: String?,
    onValueChange: (String) -> Unit,
    onClear: () -> Unit,
) {
    Column(verticalArrangement = Arrangement.spacedBy(6.dp)) {
        val title = if (field.required) "${field.title} *" else field.title
        when (field.control) {
            AttributeControl.INPUT -> {
                OutlinedTextField(
                    value = value,
                    onValueChange = onValueChange,
                    label = { Text(title) },
                    supportingText = {
                        val hints = buildList {
                            add(field.code)
                            if (isAutofilled) add("автозаполнено")
                        }
                        Text(
                            text = hints.joinToString(" • "),
                            style = MaterialTheme.typography.bodySmall,
                            color = if (errorMessage != null) MaterialTheme.colorScheme.error else MaterialTheme.colorScheme.onSurfaceVariant,
                        )
                    },
                    singleLine = true,
                    isError = errorMessage != null,
                    modifier = Modifier.fillMaxWidth(),
                )
                if (value.isNotBlank()) {
                    TextButton(onClick = onClear) { Text("Очистить") }
                }
            }
            AttributeControl.DROPDOWN -> {
                DropdownAttributeField(
                    title = title,
                    code = field.code,
                    values = field.values,
                    value = value,
                    isAutofilled = isAutofilled,
                    isError = errorMessage != null,
                    onValueSelected = onValueChange,
                    onClear = onClear,
                )
            }
            AttributeControl.CHIPS -> {
                Text(
                    text = title,
                    style = MaterialTheme.typography.bodyMedium,
                )
                FlowRow(
                    horizontalArrangement = Arrangement.spacedBy(8.dp),
                    verticalArrangement = Arrangement.spacedBy(8.dp),
                ) {
                    field.values.forEach { option ->
                        TargetTypeChip(
                            label = option,
                            selected = value == option,
                            onClick = { onValueChange(option) },
                        )
                    }
                }
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically,
                ) {
                    val hints = buildList {
                        add(field.code)
                        if (isAutofilled) add("автозаполнено")
                    }
                    Text(
                        text = hints.joinToString(" • "),
                        style = MaterialTheme.typography.bodySmall,
                        color = if (errorMessage != null) MaterialTheme.colorScheme.error else MaterialTheme.colorScheme.onSurfaceVariant,
                    )
                    if (value.isNotBlank()) {
                        TextButton(onClick = onClear) { Text("Очистить") }
                    }
                }
            }
        }
        if (errorMessage != null) {
            Text(
                text = errorMessage,
                style = MaterialTheme.typography.bodySmall,
                color = MaterialTheme.colorScheme.error,
            )
        }
    }
}

@OptIn(ExperimentalMaterial3Api::class)
@Composable
private fun CategoryLeafPickerSheet(
    categories: List<Category>,
    selectedCode: String?,
    onDismiss: () -> Unit,
    onSelect: (Category) -> Unit,
) {
    val sheetState = rememberModalBottomSheetState(skipPartiallyExpanded = true)
    var query by remember { mutableStateOf("") }
    val filtered = remember(categories, query) {
        val q = query.trim().lowercase()
        if (q.isBlank()) categories
        else categories.filter {
            listOfNotNull(it.title, it.code).joinToString(" ").lowercase().contains(q)
        }
    }

    ModalBottomSheet(onDismissRequest = onDismiss, sheetState = sheetState) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(16.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp),
        ) {
            Text("Выбрать категорию", style = MaterialTheme.typography.titleMedium)
            OutlinedTextField(
                value = query,
                onValueChange = { query = it },
                label = { Text("Поиск") },
                singleLine = true,
                modifier = Modifier.fillMaxWidth(),
            )

            LazyColumn(
                modifier = Modifier
                    .fillMaxWidth()
                    .heightIn(max = 420.dp),
                verticalArrangement = Arrangement.spacedBy(6.dp),
            ) {
                items(filtered, key = { it.code }) { category ->
                    val selected = category.code == selectedCode
                    Surface(
                        shape = MaterialTheme.shapes.medium,
                        color = if (selected) {
                            MaterialTheme.colorScheme.primary.copy(alpha = 0.12f)
                        } else {
                            MaterialTheme.colorScheme.surfaceVariant.copy(alpha = 0.35f)
                        },
                        modifier = Modifier
                            .fillMaxWidth()
                            .clickable { onSelect(category) },
                    ) {
                        Column(
                            modifier = Modifier.padding(horizontal = 12.dp, vertical = 10.dp),
                            verticalArrangement = Arrangement.spacedBy(2.dp),
                        ) {
                            Text(
                                text = category.title ?: category.code,
                                style = MaterialTheme.typography.bodyMedium,
                                maxLines = 1,
                                overflow = TextOverflow.Ellipsis,
                            )
                            Text(
                                text = category.code,
                                style = MaterialTheme.typography.bodySmall,
                                color = MaterialTheme.colorScheme.onSurfaceVariant,
                            )
                        }
                    }
                }
            }
        }
    }
}
