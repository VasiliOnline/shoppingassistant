package com.example.shoppingassistant.feature.pages.trackeditems

import com.example.shoppingassistant.domain.tracks.Track
import com.example.shoppingassistant.domain.tracks.Freshness
import com.example.shoppingassistant.domain.tracks.RankedOffer
import com.example.shoppingassistant.domain.tracks.TrackOfferSort
import com.example.shoppingassistant.domain.tracks.TrackTop10

enum class TrackedItemsFilter { ALL, ACTIVE, PAUSED, ERROR }
enum class TrackedItemsSort { NEW_EVENTS, BEST_PRICE, AZ }

sealed interface TrackedItemsLoadState {
    data object Loading : TrackedItemsLoadState
    data class Content(val items: List<Track>) : TrackedItemsLoadState
    data object Empty : TrackedItemsLoadState
    data class Error(val message: String) : TrackedItemsLoadState
}

data class TrackedItemsState(
    val tracks: TrackedItemsLoadState = TrackedItemsLoadState.Loading,
    val filter: TrackedItemsFilter = TrackedItemsFilter.ALL,
    val sort: TrackedItemsSort = TrackedItemsSort.NEW_EVENTS,
    val searchQuery: String = "",
    val pausedExpanded: Boolean = false,
    val actionMessage: String? = null,
)

sealed interface Top10LoadState {
    data object Loading : Top10LoadState
    data class Data(
        val top10: TrackTop10,
        val isStale: Boolean,
        val isRefreshing: Boolean,
        val lastError: String? = null,
    ) : Top10LoadState
    data class Empty(
        val top10: TrackTop10,
        val isStale: Boolean,
        val isRefreshing: Boolean,
        val lastError: String? = null,
    ) : Top10LoadState
    data class FatalError(val message: String) : Top10LoadState
}

data class TrackTop10State(
    val track: Track? = null,
    val top10: Top10LoadState = Top10LoadState.Loading,
    val offers: List<RankedOffer> = emptyList(),
    val offersSort: TrackOfferSort = TrackOfferSort.PRICE_ASC,
    val offersLoading: Boolean = false,
    val offersCanLoadMore: Boolean = false,
    val offersError: String? = null,
    val freshness: Freshness? = null,
    val isOffline: Boolean = false,
    val actionMessage: String? = null,
)
