/**
 * Реализация UI-задачи "Забыл пароль": форма ввода email -> запрос на /api/auth/forgot-password.
 *
 * После успешной отправки:
 * - прокидываем onResetTokenReceived(resetToken) — в dev это вернёт токен, в прод может быть null;
 * - закрываем форму через onClose().
 */
@Composable
fun ProfileForgotPasswordTask(
    props: ProfileForgotPasswordProps,
) {
    var email by remember { mutableStateOf("") }
    var emailError by remember { mutableStateOf<String?>(null) }
    var errorText by remember { mutableStateOf<String?>(null) }
    var isLoading by remember { mutableStateOf(false) }
    var successMessage by remember { mutableStateOf<String?>(null) }

    val emailFocusRequester = remember { FocusRequester() }
    val scope = rememberCoroutineScope()

    val backgroundColor = AuthPalette.ScreenBackground

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(backgroundColor),
        contentAlignment = Alignment.Center,
    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 24.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp),
            horizontalAlignment = Alignment.CenterHorizontally,
        ) {
            Text(
                text = "Восстановление пароля",
                style = MaterialTheme.typography.titleLarge,
                color = Color.White,
            )

            AuthTextField(
                value = email,
                onValueChange = { newValue ->
                    email = newValue
                    emailError = null
                    errorText = null
                    successMessage = null
                },
                label = "Email",
                placeholder = "user@example.com",
                leadingIcon = null,
                isError = emailError != null,
                errorText = emailError,
                keyboardOptions = androidx.compose.foundation.text.KeyboardOptions(
                    keyboardType = KeyboardType.Email,
                    imeAction = ImeAction.Done,
                ),
                focusRequester = emailFocusRequester,
                validationState = AuthFieldValidationState.None,
            )

            if (errorText != null) {
                Text(
                    text = errorText!!,
                    color = MaterialTheme.colorScheme.error.copy(alpha = 0.9f),
                    style = MaterialTheme.typography.bodySmall,
                )
            }

            if (successMessage != null) {
                Text(
                    text = successMessage!!,
                    color = Color.White,
                    style = MaterialTheme.typography.bodySmall,
                )
            }

            Spacer(modifier = Modifier.height(4.dp))

            AuthPrimaryButton(
                text = "Отправить ссылку",
                onClick = {
                    val validationError = validateAuthEmail(email)
                    emailError = validationError
                    if (validationError != null) {
                        email = ""
                        emailFocusRequester.requestFocus()
                        return@AuthPrimaryButton
                    }

                    isLoading = true
                    errorText = null
                    successMessage = null

                    scope.launch {
                        try {
                            val token = props.onSubmit(email.trim())
                            props.onResetTokenReceived(token)
                            successMessage = "Если email существует, мы отправили ссылку для восстановления."
                            props.onClose()
                        } catch (t: Throwable) {
                            errorText = t.message
                                ?: "Не удалось отправить ссылку. Попробуйте ещё раз."
                        } finally {
                            isLoading = false
                        }
                    }
                },
                modifier = Modifier.fillMaxWidth(),
                enabled = !isLoading,
                isLoading = isLoading,
            )

            TextButton(
                onClick = { if (!isLoading) props.onClose() },
                enabled = !isLoading,
            ) {
                Text(
                    text = "Вернуться",
                    style = MaterialTheme.typography.bodySmall,
                    color = Color.White,
                )
            }
        }
    }
}
