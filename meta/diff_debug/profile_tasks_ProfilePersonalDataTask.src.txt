@Composable
fun ProfilePersonalDataTask(
    emails: SnapshotStateList<ProfileEmail>,
    phone: String,
    onPhoneChanged: (String) -> Unit,
    onBackClick: () -> Unit,
    onPasswordChanged: (String) -> Unit = {},
    onRequestDelete: () -> Unit = {},
) {
    var password by remember { mutableStateOf("") }
    var passwordError by remember { mutableStateOf<String?>(null) }
    var showAddEmailDialog by remember { mutableStateOf(false) }
    var showPhoneDialog by remember { mutableStateOf(false) }
    var showDeleteDialog by remember { mutableStateOf(false) }
    val passwordShortErrorText = stringResource(R.string.profile_auth_password_short)
    val passwordMinLengthErrorText = stringResource(R.string.profile_password_new_min)
    val passwordLettersDigitsErrorText = stringResource(R.string.profile_password_requirements)

    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(MaterialTheme.colorScheme.surface)
            .padding(horizontal = 16.dp, vertical = 12.dp),
        verticalArrangement = Arrangement.spacedBy(14.dp),
    ) {
        Row(
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.spacedBy(8.dp),
        ) {
            IconButton(onClick = onBackClick) {
                Icon(
                    imageVector = Icons.AutoMirrored.Outlined.ArrowBack,
                    contentDescription = stringResource(R.string.profile_back),
                    tint = MaterialTheme.colorScheme.onSurfaceVariant,
                )
            }
            Text(
                text = stringResource(R.string.profile_personal_title),
                style = MaterialTheme.typography.titleLarge,
                fontWeight = FontWeight.SemiBold,
            )
        }

        EmailSection(
            emails = emails,
            onAddClick = { showAddEmailDialog = true },
            onRemove = { email -> emails.remove(email) },
        )

        HorizontalDivider(Modifier, DividerDefaults.Thickness, DividerDefaults.color)

        PasswordSection(
            password = password,
            error = passwordError,
            onPasswordChange = {
                password = it
                passwordError = null
            },
            onSave = {
                passwordError = validatePasswordForProfile(
                    password = password,
                    shortError = passwordShortErrorText,
                    minLengthError = passwordMinLengthErrorText,
                    lettersDigitsError = passwordLettersDigitsErrorText,
                )
                if (passwordError == null) {
                    onPasswordChanged(password)
                }
            },

            )

        HorizontalDivider(Modifier, DividerDefaults.Thickness, DividerDefaults.color)

        PhoneSection(
            phone = phone,
            onChangeClick = { showPhoneDialog = true },
        )

        HorizontalDivider(Modifier, DividerDefaults.Thickness, DividerDefaults.color)

        Button(
            onClick = { showDeleteDialog = true },
            modifier = Modifier.fillMaxWidth(),
            colors = ButtonDefaults.buttonColors(
                containerColor = MaterialTheme.colorScheme.errorContainer,
                contentColor = MaterialTheme.colorScheme.onErrorContainer,
            ),
            shape = RoundedCornerShape(14.dp),
        ) {
            Icon(imageVector = Icons.Outlined.Delete, contentDescription = null)
            Spacer(modifier = Modifier.width(6.dp))
            Text(stringResource(R.string.profile_personal_delete_account))
        }
    }

    if (showAddEmailDialog) {
        AddEmailDialog(
            onDismiss = {
                showAddEmailDialog = false
            },
            onSubmit = { email ->
                val normalized = email.address.trim()
                val alreadyExists = emails.any { it.address.equals(normalized, ignoreCase = true) }
                if (!alreadyExists) {
                    val makePrimary = emails.none { it.isPrimary }
                    emails.add(email.copy(isPrimary = makePrimary))
                }
                showAddEmailDialog = false
            },
        )
    }

    if (showPhoneDialog) {
        PhoneChangeDialog(
            currentPhone = phone,
            onDismiss = { showPhoneDialog = false },
            onConfirm = { newPhone ->
                onPhoneChanged(newPhone)
                showPhoneDialog = false
            },
        )
    }

    if (showDeleteDialog) {
        DeleteAccountDialog(
            phone = phone,
            onDismiss = { showDeleteDialog = false },
            onConfirm = {
                onRequestDelete()
                showDeleteDialog = false
            },
        )
    }
}

@Composable
private fun EmailSection(
    emails: SnapshotStateList<ProfileEmail>,
    onAddClick: () -> Unit,
    onRemove: (ProfileEmail) -> Unit,
) {
    Column(
        modifier = Modifier.fillMaxWidth(),
        verticalArrangement = Arrangement.spacedBy(10.dp),
    ) {
        Text(
            text = stringResource(R.string.profile_personal_email_label),
            style = MaterialTheme.typography.titleMedium,
        )
        if (emails.isEmpty()) {
            Text(
                text = stringResource(R.string.profile_personal_emails_empty),
                style = MaterialTheme.typography.bodyMedium,
                color = MaterialTheme.colorScheme.onSurfaceVariant,
            )
        } else {
            emails.forEach { email ->
                EmailRow(email = email, onRemove = { onRemove(email) })
            }
        }
        OutlinedButton(
            onClick = onAddClick,
            modifier = Modifier.fillMaxWidth(),
            shape = RoundedCornerShape(12.dp),
        ) {
            Icon(imageVector = Icons.Outlined.Add, contentDescription = null)
            Spacer(modifier = Modifier.width(6.dp))
            Text(stringResource(R.string.profile_personal_email_add))
        }
    }
}

@Composable
private fun EmailRow(
    email: ProfileEmail,
    onRemove: () -> Unit,
) {
    Surface(
        modifier = Modifier.fillMaxWidth(),
        shape = RoundedCornerShape(12.dp),
        color = MaterialTheme.colorScheme.surfaceVariant.copy(alpha = 0.35f),
    ) {
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 12.dp, vertical = 10.dp),
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.spacedBy(12.dp),
        ) {
            Column(
                modifier = Modifier.weight(1f),
                verticalArrangement = Arrangement.spacedBy(4.dp),
            ) {
                Text(
                    text = email.address,
                    style = MaterialTheme.typography.bodyLarge,
                    fontWeight = FontWeight.SemiBold,
                )
                Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                    if (email.isPrimary) {
                        StatusChip(label = stringResource(R.string.profile_personal_email_primary))
                    }
                    if (email.isVerified) {
                        StatusChip(label = stringResource(R.string.profile_personal_email_verified))
                    }
                }
            }
            Button(
                onClick = onRemove,
                colors = ButtonDefaults.buttonColors(
                    containerColor = MaterialTheme.colorScheme.errorContainer,
                    contentColor = MaterialTheme.colorScheme.onErrorContainer,
                ),
                shape = RoundedCornerShape(10.dp),
            ) {
                Text(stringResource(R.string.profile_links_delete_action))
            }
        }
    }
}

@Composable
private fun AddEmailDialog(
    onDismiss: () -> Unit,
    onSubmit: (ProfileEmail) -> Unit,
) {
    var email by remember { mutableStateOf("") }
    var code by remember { mutableStateOf("") }
    var emailError by remember { mutableStateOf<String?>(null) }
    var codeRequested by remember { mutableStateOf(false) }
    val emailRequiredError = stringResource(R.string.profile_auth_email_required)
    val emailInvalidError = stringResource(R.string.profile_auth_email_invalid)
    val emailCodeInvalidErrorText = stringResource(R.string.profile_email_change_code_invalid)

    AlertDialog(
        onDismissRequest = onDismiss,
        confirmButton = {
            TextButton(
                onClick = {
                    if (!codeRequested) {
                        emailError = validateAuthEmail(email, emailRequiredError, emailInvalidError)
                        if (emailError == null) {
                            codeRequested = true
                        }
                    } else {
                        if (code.isBlank()) {
                            emailError = emailCodeInvalidErrorText
                        } else {
                            onSubmit(
                                ProfileEmail(
                                    address = email.trim(),
                                    isVerified = true,
                                ),
                            )
                        }
                    }
                },
            ) {
                Text(
                    if (codeRequested) {
                        stringResource(R.string.profile_email_add_confirm)
                    } else {
                        stringResource(R.string.profile_email_change_send)
                    },
                )
            }
        },
        dismissButton = {
            TextButton(onClick = onDismiss) {
                Text(stringResource(R.string.profile_links_cancel_action))
            }
        },
        title = { Text(stringResource(R.string.profile_email_add_title)) },
        text = {
            Column(verticalArrangement = Arrangement.spacedBy(10.dp)) {
                Text(
                    text = stringResource(R.string.profile_email_add_description),
                    style = MaterialTheme.typography.bodyMedium,
                )
                TextField(
                    value = email,
                    onValueChange = {
                        email = it
                        emailError = null
                    },
                    singleLine = true,
                    placeholder = { Text(stringResource(R.string.profile_email_placeholder)) },
                    enabled = !codeRequested,
                    isError = emailError != null && !codeRequested,
                    colors = lightFieldColors(),
                    shape = RoundedCornerShape(12.dp),
                )
                if (codeRequested) {
                    TextField(
                        value = code,
                        onValueChange = { code = it },
                        singleLine = true,
                        placeholder = { Text(stringResource(R.string.profile_email_change_code_placeholder)) },
                        colors = lightFieldColors(),
                        shape = RoundedCornerShape(12.dp),
                    )
                }
                if (emailError != null) {
                    Text(
                        text = emailError!!,
                        style = MaterialTheme.typography.bodySmall,
                        color = MaterialTheme.colorScheme.error,
                    )
                }
            }
        },
    )
}

@Composable

    phone: String,
    onDismiss: () -> Unit,

) {

    var code by remember { mutableStateOf("") }
    var error by remember { mutableStateOf<String?>(null) }
    var codeRequested by remember { mutableStateOf(false) }
    val phoneRequiredError = stringResource(R.string.profile_personal_phone_required)
    val phoneShortError = stringResource(R.string.profile_personal_phone_short)
    val deleteCodeRequiredErrorText = stringResource(R.string.profile_personal_delete_code_required)

    AlertDialog(
        onDismissRequest = onDismiss,
        confirmButton = {
            Button(
                onClick = {
                    if (!codeRequested) {
                        error = validatePhone(confirmationPhone, phoneRequiredError, phoneShortError)
                        if (error == null) {
                            codeRequested = true
                        }
                    } else {
                        if (code.isBlank()) {
                            error = deleteCodeRequiredErrorText

                            onConfirm()

                    }
                },
                colors = ButtonDefaults.buttonColors(
                    containerColor = MaterialTheme.colorScheme.error,
                    contentColor = MaterialTheme.colorScheme.onError,
                ),
            ) {
                Text(
                    if (codeRequested) {
                        stringResource(R.string.profile_personal_delete_confirm)
                    } else {
                        stringResource(R.string.profile_phone_send_code)

                )
            }
        },
        dismissButton = {
            TextButton(onClick = onDismiss) {
                Text(stringResource(R.string.profile_links_cancel_action))
            }
        },
        title = { Text(stringResource(R.string.profile_personal_delete_account)) },

                    text = stringResource(R.string.profile_personal_delete_description),

                    value = confirmationPhone,
                    onValueChange = {
                        confirmationPhone = it

                    singleLine = true,

                    colors = lightFieldColors(),
                    shape = RoundedCornerShape(12.dp),

            }
        },
    )
}

private fun validatePasswordForProfile(
    password: String,
    shortError: String,
    minLengthError: String,
    lettersDigitsError: String,
): String? {
    val base = validateAuthPassword(password, shortError)
    if (base != null) return base
    if (password.length < 8) return minLengthError
    val hasLetters = password.any { it.isLetter() }
    val hasDigits = password.any { it.isDigit() }
    if (!hasLetters || !hasDigits) return lettersDigitsError
    return null
}

private fun validatePhone(
    raw: String,
    emptyError: String,
    shortError: String,
): String? {
    val digits = raw.filter { it.isDigit() }
    if (digits.isEmpty()) return emptyError
    if (digits.length < 10) return shortError
    return null
}
