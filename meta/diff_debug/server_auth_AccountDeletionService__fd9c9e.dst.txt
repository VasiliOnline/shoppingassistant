package com.example.shoppingassistant.server.auth

import com.example.shoppingassistant.server.config.DeletionConfig
import com.example.shoppingassistant.server.db.AuthUsersTable
import com.example.shoppingassistant.server.db.DatabaseFactory
import com.example.shoppingassistant.server.offers.AlertsTable
import com.example.shoppingassistant.server.offers.OffersTable
import com.example.shoppingassistant.server.offers.SellerStatsTable
import com.example.shoppingassistant.server.offers.UserPreferencesTable
import com.example.shoppingassistant.server.offers.UserProfilesTable
import com.example.shoppingassistant.server.offers.UserReviewsTable
import java.security.SecureRandom
import java.util.Base64
import java.util.concurrent.TimeUnit
import org.jetbrains.exposed.sql.SqlExpressionBuilder.eq
import org.jetbrains.exposed.sql.and
import org.jetbrains.exposed.sql.deleteWhere
import org.jetbrains.exposed.sql.greaterEq
import org.jetbrains.exposed.sql.lessEq
import org.jetbrains.exposed.sql.or
import org.jetbrains.exposed.sql.selectAll
import org.jetbrains.exposed.sql.update

data class DeletionSchedule(
    val deleteAfter: Long,
    val restoreToken: String,
)

/**
 * Сервис отложенного удаления аккаунта с возможностью восстановления до TTL.
 */
class AccountDeletionService(
    private val config: DeletionConfig,
    private val sessionManager: SessionManager,
    private val random: SecureRandom = SecureRandom(),
) {

    suspend fun schedule(userId: Long): DeletionSchedule? {
        val now = System.currentTimeMillis()
        val deleteAfter = now + TimeUnit.SECONDS.toMillis(config.deleteTtlSeconds)
        val restoreToken = generateRestoreToken()

        val updated = DatabaseFactory.dbQuery {
            AuthUsersTable.update(
                where = { AuthUsersTable.id eq userId },
            ) {
                it[isDeleted] = true
                it[deletionRequestedAt] = now
                it[deletionAfter] = deleteAfter
                it[deletionRestoreToken] = restoreToken
            }
        }

        return if (updated > 0) {
            sessionManager.invalidateAllForUser(userId)
            DeletionSchedule(
                deleteAfter = deleteAfter,
                restoreToken = restoreToken,
            )
        } else {
            null
        }
    }

    suspend fun restore(token: String): Long? {
        val now = System.currentTimeMillis()
        val row = DatabaseFactory.dbQuery {
            AuthUsersTable
                .selectAll()
                .where {
                    (AuthUsersTable.deletionRestoreToken eq token) and
                        (AuthUsersTable.isDeleted eq true) and
                        (AuthUsersTable.deletionAfter greaterEq now)
                }
                .singleOrNull()
        } ?: return null

        val userId = row[AuthUsersTable.id]
        val restored = DatabaseFactory.dbQuery {
            AuthUsersTable.update({ AuthUsersTable.id eq userId }) {
                it[isDeleted] = false
                it[deletionRequestedAt] = null
                it[deletionAfter] = null
                it[deletionRestoreToken] = null
            }
        }
        return if (restored > 0) {
            userId
        } else {
            null
        }
    }

    /**
     * Безопасно удаляет аккаунты, у которых истёк TTL на восстановление.
     * Возвращает список очищенных userId.
     */
    suspend fun purgeExpired(now: Long = System.currentTimeMillis()): List<Long> {
        val expiredIds = DatabaseFactory.dbQuery {
            AuthUsersTable
                .selectAll()
                .where {
                    (AuthUsersTable.isDeleted eq true) and
                        (AuthUsersTable.deletionAfter lessEq now)
                }
                .map { it[AuthUsersTable.id] }
        }
        if (expiredIds.isEmpty()) return emptyList()

        DatabaseFactory.dbQuery {
            expiredIds.forEach { userId ->
                sessionManager.invalidateAllForUser(userId)
                OffersTable.deleteWhere { OffersTable.userId eq userId }
                UserPreferencesTable.deleteWhere { UserPreferencesTable.userId eq userId }
                SellerStatsTable.deleteWhere { SellerStatsTable.userId eq userId }
                UserProfilesTable.deleteWhere { UserProfilesTable.userId eq userId }
                UserReviewsTable.deleteWhere {
                    (UserReviewsTable.fromUserId eq userId) or (UserReviewsTable.toUserId eq userId)
                }
                AlertsTable.deleteWhere { AlertsTable.userId eq userId }
                AuthUsersTable.deleteWhere { AuthUsersTable.id eq userId }
            }
        }
        return expiredIds
    }

    private fun generateRestoreToken(): String {
        val bytes = ByteArray(24)
        random.nextBytes(bytes)
        return Base64.getUrlEncoder().withoutPadding().encodeToString(bytes)
    }
}
