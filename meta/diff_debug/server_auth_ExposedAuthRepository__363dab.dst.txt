package com.example.shoppingassistant.server.auth

import com.example.shoppingassistant.domain.auth.AuthRepository
import com.example.shoppingassistant.domain.model.AuthError
import com.example.shoppingassistant.domain.model.AuthResult
import com.example.shoppingassistant.domain.model.AuthUser
import com.example.shoppingassistant.server.db.AuthUsersTable
import com.example.shoppingassistant.server.db.DatabaseFactory
import org.jetbrains.exposed.sql.Op
import org.jetbrains.exposed.sql.ResultRow
import org.jetbrains.exposed.sql.SqlExpressionBuilder
import org.jetbrains.exposed.sql.SqlExpressionBuilder.eq
import org.jetbrains.exposed.sql.insert
import org.jetbrains.exposed.sql.select
import org.jetbrains.exposed.sql.selectAll
import java.util.Locale

/**
 * Реализация AuthRepository поверх Postgres (auth_users) через Exposed.
 *
 * Пароль храним как salted SHA-256:
 * столбец [AuthUsersTable.password] содержит строку "<saltBase64>:<hashBase64>".
 *
 * Глобального currentUserId здесь больше нет — сессии реализованы отдельно
 * через SessionManager и bearer-токен.
 */
class ExposedAuthRepository(
    private val passwordHasher: PasswordHasher,
) : AuthRepository {

    override suspend fun register(
        email: String,
        password: String,
        displayName: String?,
    ): AuthResult = DatabaseFactory.dbQuery {
        val normalizedEmail = normalizeEmail(email)

        val existing = AuthUsersTable
            .selectAll().where { AuthUsersTable.email eq normalizedEmail }
            .singleOrNull()

        if (existing != null) {
            return@dbQuery AuthResult.Error(AuthError.EMAIL_ALREADY_EXISTS)
        }

        val safeDisplayName = displayName?.takeIf { it.isNotBlank() }
        val passwordHash = passwordHasher.hash(password)

        val insertStatement = AuthUsersTable.insert { row ->
            row[AuthUsersTable.email] = normalizedEmail
            row[AuthUsersTable.password] = passwordHash
            row[AuthUsersTable.displayName] = safeDisplayName
            row[AuthUsersTable.phone] = null
            row[AuthUsersTable.avatarUrl] = null
            row[AuthUsersTable.city] = null
            row[AuthUsersTable.photoUrls] = emptyList()
        }

        val id = insertStatement[AuthUsersTable.id]
        val user = AuthUser(
            id = id,
            email = normalizedEmail,
            displayName = safeDisplayName ?: normalizedEmail,
            phone = null,
            avatarUrl = null,
            city = null,
            emailVerified = false,
            emailVerifiedAt = null,
            phoneVerifiedAt = null,
            createdAt = insertStatement[AuthUsersTable.createdAt],
            photos = emptyList(),
        )

        AuthResult.Success(user)
    }

    override suspend fun login(
        email: String,
        password: String,
    ): AuthResult = DatabaseFactory.dbQuery {
        val normalizedEmail = normalizeEmail(email)

        val row = AuthUsersTable
            .selectAll().where {
                (AuthUsersTable.email eq normalizedEmail) and (AuthUsersTable.isDeleted eq false)
            }
            .singleOrNull()
            ?: return@dbQuery AuthResult.Error(AuthError.INVALID_CREDENTIALS)

        val storedPassword = row[AuthUsersTable.password]
        if (!passwordHasher.verify(password, storedPassword)) {
            return@dbQuery AuthResult.Error(AuthError.INVALID_CREDENTIALS)
        }

        val user = row.toDomain()
        AuthResult.Success(user)
    }

    /**
     * В backend-реализации больше нет глобального "текущего пользователя".
     * Этот метод оставлен только ради совместимости с доменным контрактом
     * и всегда возвращает null. На HTTP-уровне используется getUserById(id).
     */
    override suspend fun getCurrentUser(): AuthUser? = null

    override suspend fun getUserById(id: Long): AuthUser? = DatabaseFactory.dbQuery {
        AuthUsersTable
            .selectAll().where { (AuthUsersTable.id eq id) and (AuthUsersTable.isDeleted eq false) }
            .singleOrNull()
            ?.toDomain()
    }

    override suspend fun logout() {
        // Ничего не делаем: завершение сессии обрабатывается на уровне токена
        // (SessionManager) в HTTP-слое.
    }

    // На backend-е bearer не хранится — возвращаем null для совместимости.
    override suspend fun currentToken(): String? = null

    override suspend fun changePassword(oldPassword: String, newPassword: String): AuthResult {
        // В backend-реализации смена пароля требует userId извне (сессия),
        // поэтому этот метод не используется. HTTP-слой реализует смену пароля
        // через PasswordResetService/SessionManager. Для совместимости вернём UNKNOWN.
        return AuthResult.Error(AuthError.UNKNOWN)
    }

    private fun normalizeEmail(email: String): String =
        email.trim().lowercase(Locale.ROOT)

    private fun ResultRow.toDomain(): AuthUser {
        return AuthUser(
            id = this[AuthUsersTable.id],
            email = this[AuthUsersTable.email],
            displayName = this[AuthUsersTable.displayName] ?: this[AuthUsersTable.email],
            phone = this[AuthUsersTable.phone],
            avatarUrl = this[AuthUsersTable.avatarUrl],
            city = this[AuthUsersTable.city],
            emailVerified = this[AuthUsersTable.emailVerified] || this[AuthUsersTable.emailVerifiedAt] != null,
            emailVerifiedAt = this[AuthUsersTable.emailVerifiedAt],
            phoneVerifiedAt = this[AuthUsersTable.phoneVerifiedAt],
            createdAt = this[AuthUsersTable.createdAt],
            photos = this[AuthUsersTable.photoUrls] ?: emptyList(),
        )
    }
}
