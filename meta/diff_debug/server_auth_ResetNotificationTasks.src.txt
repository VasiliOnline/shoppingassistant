package com.example.shoppingassistant.server.auth

/**
 * Контракт отправки письма со ссылкой на сброс пароля.
 */
interface ResetNotificationSender {
    /**
     * Отправляет пользователю reset-токен (обычно в виде ссылки).
     */
    fun sendResetLink(email: String, resetToken: String)
}

/**
 * Контракт отправки кода восстановления по телефону (SMS/Push).
 */
interface SmsResetNotificationSender {
    /**
        * Отправляет короткий код или токен по телефону.
        */
    fun sendResetCode(phone: String, resetToken: String)
}
