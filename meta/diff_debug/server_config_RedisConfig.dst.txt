package com.example.shoppingassistant.server.config

/**
 * Конфигурация подключения к Redis для хранения сессий.
 *
 * Читает значения из переменных окружения:
 * - REDIS_URL — URL подключения к Redis (например, "redis://localhost:6379/0");
 * - REDIS_SESSION_TTL_SECONDS — TTL сессии в секундах (опционально, по умолчанию 7 суток);
 * - REDIS_SESSION_KEY_PREFIX — префикс для ключей сессий (по умолчанию "session:").
 *
 * enabled == true только если явно указан REDIS_URL.
 */
data class RedisConfig(
    val enabled: Boolean,
    val url: String?,
    val sessionTtlSeconds: Long,
    val sessionKeyPrefix: String,
) {

    companion object {

        fun fromEnv(): RedisConfig {
            val rawUrl = System.getenv("REDIS_URL")?.trim()
            val url = rawUrl?.takeIf { it.isNotEmpty() }

            val ttl = System.getenv("REDIS_SESSION_TTL_SECONDS")
                ?.toLongOrNull()
                ?.takeIf { it > 0 }
                ?: (7L * 24L * 60L * 60L) // 7 суток по умолчанию

            val keyPrefix = System.getenv("REDIS_SESSION_KEY_PREFIX")
                ?.takeIf { it.isNotBlank() }
                ?: "session:"

            return RedisConfig(
                enabled = url != null,
                url = url,
                sessionTtlSeconds = ttl,
                sessionKeyPrefix = keyPrefix,
            )
        }
    }
}
