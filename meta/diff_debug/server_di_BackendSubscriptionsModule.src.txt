package com.example.shoppingassistant.server.di

import com.example.shoppingassistant.server.config.FcmConfig
import com.example.shoppingassistant.server.config.SubscriptionsDeliveryConfig
import com.example.shoppingassistant.server.config.SubscriptionsEngineConfig
import com.example.shoppingassistant.server.push.FcmLegacyPushSenderImpl
import com.example.shoppingassistant.server.push.FcmPushSender
import com.example.shoppingassistant.server.push.PushTokensRepository
import com.example.shoppingassistant.server.push.PushTokensRepositoryImpl
import com.example.shoppingassistant.server.subscriptions.CompositeSubscriptionNotificationsDeliveryRunnerImpl
import com.example.shoppingassistant.server.subscriptions.SubscriptionNotificationsDeliveryRunnerImpl
import com.example.shoppingassistant.server.subscriptions.SubscriptionNotificationsRepository
import com.example.shoppingassistant.server.subscriptions.SubscriptionNotificationsRepositoryImpl
import com.example.shoppingassistant.server.subscriptions.SubscriptionNotificationsDeliveryRunner
import com.example.shoppingassistant.server.subscriptions.SubscriptionNotificationsPushDeliveryRunnerImpl
import com.example.shoppingassistant.server.subscriptions.SubscriptionsEngineRunner
import com.example.shoppingassistant.server.subscriptions.SubscriptionsEngineRunnerImpl
import org.koin.dsl.module

val backendSubscriptionsModule = module {
    single { SubscriptionsEngineConfig.fromEnv() }
    single { SubscriptionsDeliveryConfig.fromEnv() }
    single { FcmConfig.fromEnv() }
    single<PushTokensRepository> { PushTokensRepositoryImpl() }
    single<FcmPushSender> { FcmLegacyPushSenderImpl(config = get()) }
    single<SubscriptionNotificationsRepository> { SubscriptionNotificationsRepositoryImpl() }
    single<SubscriptionsEngineRunner> { SubscriptionsEngineRunnerImpl(config = get()) }
    single<SubscriptionNotificationsDeliveryRunner> {
        CompositeSubscriptionNotificationsDeliveryRunnerImpl(
            runners = listOf(
                SubscriptionNotificationsDeliveryRunnerImpl(
                    config = get(),
                    emailConfig = get(),
                ),
                SubscriptionNotificationsPushDeliveryRunnerImpl(
                    config = get(),
                    fcmConfig = get(),
                    tokensRepo = get(),
                    sender = get(),
                ),
            ),
        )
    }
}
