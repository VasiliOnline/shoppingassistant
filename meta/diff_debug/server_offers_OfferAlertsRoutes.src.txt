package com.example.shoppingassistant.server.offers

import io.ktor.server.application.ApplicationCall
import io.ktor.server.application.call
import io.ktor.server.request.receive
import io.ktor.server.response.respond
import io.ktor.server.response.respondText
import io.ktor.server.routing.Route
import io.ktor.server.routing.delete
import io.ktor.server.routing.get
import io.ktor.server.routing.post
import io.ktor.server.routing.route
import kotlinx.serialization.Serializable
import com.example.shoppingassistant.server.auth.SessionManager
import io.ktor.http.HttpStatusCode
import org.koin.java.KoinJavaComponent

@Serializable
data class PriceAlertRequest(
    val offerId: String,
    val percentDrop: Int,
    val deliveryChannel: String? = null,
    val minIntervalMinutes: Int? = null,
)

@Serializable
data class PriceAlertResponse(
    val accepted: Boolean,
    val message: String,
)

fun Route.offerAlertsRoutes() {
    val alertsRepo = OfferAlertsRepository
    val sessionManager: SessionManager = KoinJavaComponent.get(SessionManager::class.java)

    val minDropPercent = 1
    val maxDropPercent = 50
    val defaultMinIntervalMinutes = 120
    val minIntervalMinutes = 10
    val maxIntervalMinutes = 60 * 24 * 7
    val defaultDeliveryChannel = "EMAIL"
    val allowedDeliveryChannels = setOf("EMAIL", "PUSH")

    route("/api/offers/alerts") {
        get {
            val userId = requireUserId(call, sessionManager) ?: return@get
            val items = alertsRepo.listAlerts(userId)
            call.respond(items)
        }
        delete("/{id}") {
            val userId = requireUserId(call, sessionManager) ?: return@delete
            val id = call.parameters["id"]?.toLongOrNull()
            if (id == null) {
                call.respondText("Bad request", status = io.ktor.http.HttpStatusCode.BadRequest)
                return@delete
            }
            alertsRepo.deactivateAlert(id, userId)
            call.respondText("OK")
        }
        post("/price-drop") {
            val req = call.receive<PriceAlertRequest>()
            val userId = requireUserId(call, sessionManager) ?: return@post
            val offerId = req.offerId.toLongOrNull()
            if (offerId == null) {
                call.respond(HttpStatusCode.BadRequest, PriceAlertResponse(accepted = false, message = "Bad request"))
                return@post
            }
            if (req.percentDrop !in minDropPercent..maxDropPercent) {
                call.respond(
                    HttpStatusCode.BadRequest,
                    PriceAlertResponse(
                        accepted = false,
                        message = "percentDrop must be in $minDropPercent..$maxDropPercent",
                    ),
                )
                return@post
            }

            val deliveryChannel = req.deliveryChannel
                ?.trim()
                ?.uppercase()
                ?.takeIf { it.isNotEmpty() }
                ?: defaultDeliveryChannel
            if (deliveryChannel !in allowedDeliveryChannels) {
                call.respond(
                    HttpStatusCode.BadRequest,
                    PriceAlertResponse(
                        accepted = false,
                        message = "deliveryChannel must be one of: ${allowedDeliveryChannels.joinToString()}",
                    ),
                )
                return@post
            }

            val effectiveMinIntervalMinutes = (req.minIntervalMinutes ?: defaultMinIntervalMinutes)
                .coerceIn(minIntervalMinutes, maxIntervalMinutes)

            val result = alertsRepo.upsertOfferPercentDropAlert(
                userId = userId,
                offerId = offerId,
                percentDrop = req.percentDrop,
                deliveryChannel = deliveryChannel,
                minIntervalMinutes = effectiveMinIntervalMinutes,
            )

            call.respond(
                status = if (result.created) HttpStatusCode.Created else HttpStatusCode.OK,
                message = PriceAlertResponse(
                    accepted = true,
                    message = if (result.created) {
                        "Подписка зарегистрирована: снижение не меньше ${req.percentDrop}% (оффер ${req.offerId})"
                    } else {
                        "Подписка обновлена: снижение не меньше ${req.percentDrop}% (оффер ${req.offerId})"
                    },
                ),
            )
        }
    }
}

private suspend fun requireUserId(call: ApplicationCall, sessionManager: SessionManager): Long? {
    val header = call.request.headers["Authorization"]
    val token = extractBearerToken(header)
    if (token == null) {
        call.respondText("Unauthorized", status = io.ktor.http.HttpStatusCode.Unauthorized)
        return null
    }
    val userId = sessionManager.getUserId(token)
    if (userId == null) {
        call.respondText("Unauthorized", status = io.ktor.http.HttpStatusCode.Unauthorized)
        return null
    }
    return userId
}

private fun extractBearerToken(header: String?): String? {
    if (header == null || !header.startsWith("Bearer ")) return null
    return header.removePrefix("Bearer ").trim().takeIf { it.isNotEmpty() }
}
