package com.example.shoppingassistant.server.offers

/**
 * Манифест задач серверного слоя для офферов (Postgres + Ktor).
 * Помогает держать скоупы работ явными и не смешивать с UI.
 */
enum class OfferBackendTaskId {
    SCHEMA,
    REPOSITORY,
    ROUTE,
    RANKING,
    ALERTS_API,
    ANALYTICS,
}

data class OfferBackendTask(
    val id: OfferBackendTaskId,
    val path: String,
    val description: String,
)

/**
 * План для backend-части офферов:
 * - SCHEMA: схемы/миграции Postgres (products/products_i18n/offers/user_prefs/reviews).
 * - REPOSITORY: реализация OfferRepository (Postgres → OfferFull).
 * - ROUTE: Ktor endpoint /api/offers/search с OfferSearchCriteria.
 * - RANKING: интеграция RankService при выдаче.
 * - ALERTS_API: подписки/alerts эндпоинты.
 * - ANALYTICS: отдача свежести/трендов.
 */
val offerBackendTasks = listOf(
    OfferBackendTask(
        id = OfferBackendTaskId.SCHEMA,
        path = "server/src/main/resources/db/migration",
        description = "Миграции Postgres: products/products_i18n/offers/user_prefs/user_reviews + индексы.",
    ),
    OfferBackendTask(
        id = OfferBackendTaskId.REPOSITORY,
        path = "server/src/main/kotlin/com/example/shoppingassistant/server/offers/OfferRepositoryImpl.kt",
        description = "Реализация OfferRepository: SQL к Postgres, мэппинг в OfferFull.",
    ),
    OfferBackendTask(
        id = OfferBackendTaskId.ROUTE,
        path = "server/src/main/kotlin/com/example/shoppingassistant/server/offers/OfferRoutes.kt",
        description = "Ktor /api/offers/search: принимает OfferSearchCriteria, вызывает OfferRepository + RankService.",
    ),
    OfferBackendTask(
        id = OfferBackendTaskId.RANKING,
        path = "server/src/main/kotlin/com/example/shoppingassistant/server/offers/OfferRanking.kt",
        description = "Слой ранжирования/сортировки (RankService) для офферов.",
    ),
    OfferBackendTask(
        id = OfferBackendTaskId.ALERTS_API,
        path = "server/src/main/kotlin/com/example/shoppingassistant/server/offers/OfferAlertsRoutes.kt",
        description = "Эндпоинты подписок/alerts на снижение цены.",
    ),
    OfferBackendTask(
        id = OfferBackendTaskId.ANALYTICS,
        path = "server/src/main/kotlin/com/example/shoppingassistant/server/offers/OfferAnalytics.kt",
        description = "Отдача свежести/трендов (price history, updatedAt) для вкладки Аналитика.",
    ),
)
