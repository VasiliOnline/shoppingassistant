package com.example.shoppingassistant.server.plugins

import com.example.shoppingassistant.server.config.ServerConfig
import io.ktor.http.HttpHeaders
import io.ktor.http.HttpMethod
import io.ktor.server.application.Application
import io.ktor.server.application.install
import io.ktor.server.plugins.cors.routing.CORS

/**
 * CORS-конфигурация: разрешённые origin берутся из ServerConfig.corsOrigins.
 * Если список пуст — CORS не настраиваем (по умолчанию закрыто).
 */
fun Application.configureCors(config: ServerConfig) {
    if (config.corsOrigins.isEmpty()) return

    install(CORS) {
        allowCredentials = true
        allowNonSimpleContentTypes = true
        methods.addAll(
            listOf(
                HttpMethod.Get,
                HttpMethod.Post,
                HttpMethod.Put,
                HttpMethod.Patch,
                HttpMethod.Delete,
                HttpMethod.Options,
            ),
        )
        headers.add(HttpHeaders.Authorization)
        headers.add(HttpHeaders.ContentType)

        config.corsOrigins.forEach { origin ->
            allowHost(
                host = origin,
                schemes = listOf("http", "https"),
            )
        }
    }
}
