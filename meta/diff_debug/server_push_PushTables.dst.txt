package com.example.shoppingassistant.server.push

import com.example.shoppingassistant.server.db.AuthUsersTable
import org.jetbrains.exposed.sql.ReferenceOption
import org.jetbrains.exposed.sql.Table

object PushTokensTable : Table("push_tokens") {
    val id = long("id").autoIncrement()
    val userId = long("user_id").references(AuthUsersTable.id, onDelete = ReferenceOption.CASCADE)
    val platform = varchar("platform", 16)
    val token = varchar("token", 512)
    val deviceId = varchar("device_id", 128).nullable()
    val isActive = bool("is_active").default(true)
    val createdAt = long("created_at").clientDefault { System.currentTimeMillis() }
    val lastSeenAt = long("last_seen_at").clientDefault { System.currentTimeMillis() }

    override val primaryKey = PrimaryKey(id)

    init {
        index(true, userId, platform, token)
        index(false, userId, isActive)
        index(false, platform, isActive)
    }
}
