package com.example.shoppingassistant.server.subscriptions

import com.example.shoppingassistant.server.db.AuthUsersTable
import com.example.shoppingassistant.server.offers.AlertsTable
import com.example.shoppingassistant.server.offers.OfferPriceHistoryTable
import com.example.shoppingassistant.server.offers.OffersTable
import org.jetbrains.exposed.sql.ReferenceOption
import org.jetbrains.exposed.sql.Table

object SubscriptionNotificationsTable : Table("subscription_notifications") {
    val id = long("id").autoIncrement()
    val userId = long("user_id").references(AuthUsersTable.id, onDelete = ReferenceOption.CASCADE)
    val alertId = long("alert_id").references(AlertsTable.id, onDelete = ReferenceOption.CASCADE)
    val offerId = long("offer_id").references(OffersTable.id, onDelete = ReferenceOption.CASCADE)
    val priceHistoryId =
        long("price_history_id").references(OfferPriceHistoryTable.id, onDelete = ReferenceOption.CASCADE)

    val createdAt = long("created_at").clientDefault { System.currentTimeMillis() }
    val priceCollectedAt = long("price_collected_at")
    val isRead = bool("is_read").default(false)

    val message = text("message")
    val dropPercent = double("drop_percent")
    val baselinePriceMinor = long("baseline_price_minor")
    val currentPriceMinor = long("current_price_minor")
    val currency = varchar("currency", 8)

    override val primaryKey = PrimaryKey(id)

    init {
        index(true, alertId, priceHistoryId)
        index(false, userId, createdAt)
        index(false, userId, isRead)
    }
}

object SubscriptionsEngineStateTable : Table("subscriptions_engine_state") {
    val key = varchar("key", 64)
    val lastHistoryId = long("last_history_id").default(0)
    val updatedAt = long("updated_at").clientDefault { System.currentTimeMillis() }

    override val primaryKey = PrimaryKey(key)
}
