package com.example.shoppingassistant.server.tracks

import com.example.shoppingassistant.server.auth.SessionManager
import com.example.shoppingassistant.server.tracks.top10.TrackTop10SnapshotRepository
import com.example.shoppingassistant.server.tracks.top10.refresh.TrackTop10RefreshService
import io.ktor.http.HttpStatusCode
import io.ktor.server.application.ApplicationCall
import io.ktor.server.application.call
import io.ktor.server.request.receive
import io.ktor.server.response.respond
import io.ktor.server.response.respondText
import io.ktor.server.routing.Route
import io.ktor.server.routing.delete
import io.ktor.server.routing.get
import io.ktor.server.routing.post
import io.ktor.server.routing.put
import io.ktor.server.routing.route
import kotlinx.coroutines.withTimeoutOrNull
import org.koin.java.KoinJavaComponent
import kotlin.math.max

fun Route.tracksRoutes() {
    val sessionManager: SessionManager = KoinJavaComponent.get(SessionManager::class.java)
    val tracksRepo: TracksRepository = KoinJavaComponent.get(TracksRepository::class.java)
    val eventsRepo: TrackEventsRepository = KoinJavaComponent.get(TrackEventsRepository::class.java)
    val snapshotRepo: TrackTop10SnapshotRepository = KoinJavaComponent.get(TrackTop10SnapshotRepository::class.java)
    val refreshService: TrackTop10RefreshService = KoinJavaComponent.get(TrackTop10RefreshService::class.java)

    route("/api/tracks") {
        get {
            val userId = requireUserId(call, sessionManager) ?: return@get
            call.respond(tracksRepo.list(userId))
        }

        post {
            val userId = requireUserId(call, sessionManager) ?: return@post
            val payload = call.receive<TrackCreateRequest>()
            when (val result = tracksRepo.create(userId, payload)) {
                is TrackCreateResult.Created -> call.respond(result.track)
                is TrackCreateResult.AlreadyExists -> call.respond(result.track)
                is TrackCreateResult.InvalidInput -> call.respond(
                    status = HttpStatusCode.BadRequest,
                    message = result,
                )
                is TrackCreateResult.LimitReached -> call.respond(
                    status = HttpStatusCode.TooManyRequests,
                    message = result,
                )
            }
        }

        get("/{id}") {
            val userId = requireUserId(call, sessionManager) ?: return@get
            val trackId = call.parameters["id"]?.toLongOrNull()
            if (trackId == null) {
                call.respondText("Bad request", status = HttpStatusCode.BadRequest)
                return@get
            }
            val track = tracksRepo.get(userId, trackId)
            if (track == null) {
                call.respondText("Not found", status = HttpStatusCode.NotFound)
                return@get
            }
            call.respond(track)
        }

        put("/{id}") {
            val userId = requireUserId(call, sessionManager) ?: return@put
            val trackId = call.parameters["id"]?.toLongOrNull()
            if (trackId == null) {
                call.respondText("Bad request", status = HttpStatusCode.BadRequest)
                return@put
            }
            val payload = call.receive<TrackUpdateRequest>()
            val updated = tracksRepo.update(userId, trackId, payload)
            if (updated == null) {
                call.respondText("Not found", status = HttpStatusCode.NotFound)
                return@put
            }
            call.respond(updated)
        }

        post("/{id}/pause") {
            val userId = requireUserId(call, sessionManager) ?: return@post
            val trackId = call.parameters["id"]?.toLongOrNull()
            if (trackId == null) {
                call.respondText("Bad request", status = HttpStatusCode.BadRequest)
                return@post
            }
            val updated = tracksRepo.setActive(userId, trackId, isActive = false)
            if (updated == null) {
                call.respondText("Not found", status = HttpStatusCode.NotFound)
                return@post
            }
            call.respond(updated)
        }

        post("/{id}/resume") {
            val userId = requireUserId(call, sessionManager) ?: return@post
            val trackId = call.parameters["id"]?.toLongOrNull()
            if (trackId == null) {
                call.respondText("Bad request", status = HttpStatusCode.BadRequest)
                return@post
            }
            val updated = tracksRepo.setActive(userId, trackId, isActive = true)
            if (updated == null) {
                call.respondText("Not found", status = HttpStatusCode.NotFound)
                return@post
            }
            call.respond(updated)
        }

        delete("/{id}") {
            val userId = requireUserId(call, sessionManager) ?: return@delete
            val trackId = call.parameters["id"]?.toLongOrNull()
            if (trackId == null) {
                call.respondText("Bad request", status = HttpStatusCode.BadRequest)
                return@delete
            }
            val deleted = tracksRepo.delete(userId, trackId)
            if (!deleted) {
                call.respondText("Not found", status = HttpStatusCode.NotFound)
                return@delete
            }
            call.respondText("OK")
        }

        get("/{id}/events") {
            val userId = requireUserId(call, sessionManager) ?: return@get
            val trackId = call.parameters["id"]?.toLongOrNull()
            if (trackId == null) {
                call.respondText("Bad request", status = HttpStatusCode.BadRequest)
                return@get
            }
            val limit = call.request.queryParameters["limit"]?.toIntOrNull() ?: 30
            val offset = call.request.queryParameters["offset"]?.toIntOrNull() ?: 0
            val page = eventsRepo.listEventsPage(userId, trackId, limit, offset)
            call.respond(page)
        }

        post("/events/{eventId}/read") {
            val userId = requireUserId(call, sessionManager) ?: return@post
            val eventId = call.parameters["eventId"]?.toLongOrNull()
            if (eventId == null) {
                call.respondText("Bad request", status = HttpStatusCode.BadRequest)
                return@post
            }
            val ok = eventsRepo.markRead(userId, eventId)
            if (!ok) {
                call.respondText("Not found", status = HttpStatusCode.NotFound)
                return@post
            }
            call.respondText("OK")
        }

        post("/{id}/events/read-all") {
            val userId = requireUserId(call, sessionManager) ?: return@post
            val trackId = call.parameters["id"]?.toLongOrNull()
            if (trackId == null) {
                call.respondText("Bad request", status = HttpStatusCode.BadRequest)
                return@post
            }
            val updated = eventsRepo.markAllRead(userId, trackId)
            call.respond(MarkAllReadResponse(updated = updated))
        }

        get("/{id}/top10") {
            val userId = requireUserId(call, sessionManager) ?: return@get
            val trackId = call.parameters["id"]?.toLongOrNull()
            if (trackId == null) {
                call.respondText("Bad request", status = HttpStatusCode.BadRequest)
                return@get
            }
            val forceRefresh = call.request.queryParameters["forceRefresh"]?.toBoolean() ?: false
            val preferCache = call.request.queryParameters["preferCache"]?.toBoolean() ?: true
            val now = System.currentTimeMillis()
            val cached = snapshotRepo.getSnapshot(userId, trackId)?.snapshot

            if (forceRefresh || !preferCache) {
                val refreshed = runCatching {
                    withTimeoutOrNull(2_500L) { refreshService.refreshNow(userId, trackId) }
                }.getOrNull()

                val snapshot = refreshed ?: cached
                if (snapshot != null) {
                    call.respond(withFreshness(snapshot, now))
                } else {
                    refreshService.enqueue(trackId)
                    call.respond(HttpStatusCode.Accepted, emptySnapshot(trackId, now))
                }
                return@get
            }

            if (cached != null) {
                call.respond(withFreshness(cached, now))
            } else {
                refreshService.enqueue(trackId)
                call.respond(emptySnapshot(trackId, now))
            }
        }

        post("/{id}/top10/refresh") {
            val userId = requireUserId(call, sessionManager) ?: return@post
            val trackId = call.parameters["id"]?.toLongOrNull()
            if (trackId == null) {
                call.respondText("Bad request", status = HttpStatusCode.BadRequest)
                return@post
            }
            refreshService.enqueue(trackId)
            val now = System.currentTimeMillis()
            val snapshot = snapshotRepo.getSnapshot(userId, trackId)?.snapshot
            val payload = snapshot?.let { withFreshness(it, now) } ?: emptySnapshot(trackId, now)
            call.respond(HttpStatusCode.Accepted, payload)
        }
    }
}

@kotlinx.serialization.Serializable
private data class MarkAllReadResponse(val updated: Int)

private suspend fun requireUserId(call: ApplicationCall, sessionManager: SessionManager): Long? {
    val header = call.request.headers["Authorization"]
    val token = extractBearerToken(header)
    if (token == null) {
        call.respondText("Unauthorized", status = HttpStatusCode.Unauthorized)
        return null
    }
    val userId = sessionManager.getUserId(token)
    if (userId == null) {
        call.respondText("Unauthorized", status = HttpStatusCode.Unauthorized)
        return null
    }
    return userId
}

private fun extractBearerToken(header: String?): String? {
    if (header == null || !header.startsWith("Bearer ")) return null
    return header.removePrefix("Bearer ").trim().takeIf { it.isNotEmpty() }
}

private fun withFreshness(snapshot: com.example.shoppingassistant.domain.tracks.TrackTop10, now: Long): com.example.shoppingassistant.domain.tracks.TrackTop10 {
    if (snapshot.computedAt <= 0L) return snapshot
    val freshness = max(0, ((now - snapshot.computedAt) / 1000L).toInt())
    return snapshot.copy(freshnessSec = freshness)
}

private fun emptySnapshot(trackId: Long, now: Long): com.example.shoppingassistant.domain.tracks.TrackTop10 {
    return com.example.shoppingassistant.domain.tracks.TrackTop10(
        trackId = trackId.toString(),
        computedAt = 0L,
        freshnessSec = Int.MAX_VALUE,
        items = emptyList(),
        explanation = com.example.shoppingassistant.domain.tracks.RankExplanation(
            summary = "Нет данных",
            details = listOf("Снимок ещё не построен"),
        ),
        sourceStamps = listOf(
            com.example.shoppingassistant.domain.tracks.SourceStamp(
                sourceType = "empty",
                sourceName = "empty",
                fetchedAt = now,
                ttlSec = 0,
            ),
        ),
    )
}
