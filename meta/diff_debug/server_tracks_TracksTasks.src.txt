package com.example.shoppingassistant.server.tracks

import com.example.shoppingassistant.domain.tracks.Track
import com.example.shoppingassistant.domain.tracks.TrackEvent
import com.example.shoppingassistant.domain.tracks.TrackFilters
import com.example.shoppingassistant.domain.tracks.TrackOfferSort
import com.example.shoppingassistant.domain.tracks.TrackOffersPage
import com.example.shoppingassistant.domain.tracks.TrackTarget
import com.example.shoppingassistant.domain.tracks.TrackType
import kotlinx.serialization.Serializable

@Serializable
data class TrackCreateRequest(
    val type: TrackType,
    val target: TrackTarget,
    val filters: TrackFilters = TrackFilters(),
    val title: String? = null,
)

@Serializable
data class TrackUpdateRequest(
    val title: String? = null,
    val filters: TrackFilters? = null,
    val isActive: Boolean? = null,
)

@Serializable
data class TrackTargetUpdateRequest(
    val type: TrackType,
    val target: TrackTarget,
    val title: String? = null,
)

@Serializable
sealed class TrackTargetUpdateResult {
    @Serializable data class Updated(val track: Track) : TrackTargetUpdateResult()
    @Serializable data class AlreadyExists(val track: Track) : TrackTargetUpdateResult()
    @Serializable data class InvalidInput(val reason: String) : TrackTargetUpdateResult()
    @Serializable data object NotFound : TrackTargetUpdateResult()
}

@Serializable
sealed class TrackCreateResult {
    @Serializable data class Created(val track: Track) : TrackCreateResult()
    @Serializable data class AlreadyExists(val track: Track) : TrackCreateResult()
    @Serializable data class LimitReached(val max: Int) : TrackCreateResult()
    @Serializable data class InvalidInput(val reason: String) : TrackCreateResult()
}

@Serializable
data class TrackEventsPage(
    val items: List<TrackEvent>,
    val limit: Int,
    val offset: Int,
    val canLoadMore: Boolean,
)

interface TracksRepository {
    suspend fun list(userId: Long): List<Track>
    suspend fun get(userId: Long, trackId: Long): Track?
    suspend fun create(userId: Long, request: TrackCreateRequest): TrackCreateResult
    suspend fun update(userId: Long, trackId: Long, request: TrackUpdateRequest): Track?
    suspend fun updateTarget(userId: Long, trackId: Long, request: TrackTargetUpdateRequest): TrackTargetUpdateResult
    suspend fun listOffers(
        userId: Long,
        trackId: Long,
        limit: Int,
        offset: Int,
        sort: TrackOfferSort,
    ): TrackOffersPage
    suspend fun setActive(userId: Long, trackId: Long, isActive: Boolean): Track?
    suspend fun delete(userId: Long, trackId: Long): Boolean
}

interface TrackEventsRepository {
    suspend fun listEventsPage(userId: Long, trackId: Long, limit: Int, offset: Int): TrackEventsPage
    suspend fun markRead(userId: Long, eventId: Long): Boolean
    suspend fun markAllRead(userId: Long, trackId: Long): Int
    suspend fun addEvent(
        userId: Long,
        trackId: Long,
        type: String,
        title: String,
        subtitle: String? = null,
        dedupKey: String,
    )
}
