package com.example.shoppingassistant.server.ugc

import com.example.shoppingassistant.domain.ugc.UgcMirrorResult
import io.ktor.http.HttpStatusCode
import io.ktor.server.application.call
import io.ktor.server.request.receive
import io.ktor.server.response.respond
import io.ktor.server.routing.Route
import io.ktor.server.routing.post
import kotlinx.serialization.Serializable
import org.koin.java.KoinJavaComponent

@Serializable
private data class UgcMirrorRequest(
    val url: String,
)

@Serializable
private data class UgcMirrorResponse(
    val sourceUrl: String? = null,
    val title: String? = null,
    val description: String? = null,
    val priceRaw: String? = null,
    val currency: String? = null,
    val imageUrls: List<String>? = null,
    val attributes: Map<String, String>? = null,
)

/**
 * HTTP-эндпойнты для UGC-зеркала.
 *
 * Маршрут: POST /api/ugc/mirror
 */
fun Route.ugcRoutes() {
    post("/api/ugc/mirror") {
        val request = runCatching { call.receive<UgcMirrorRequest>() }
            .getOrElse {
                call.respond(
                    status = HttpStatusCode.BadRequest,
                    message = mapOf(
                        "error" to "BAD_REQUEST",
                        "message" to "Некорректное тело запроса",
                    ),
                )
                return@post
            }

        val service: UgcMirrorService =
            KoinJavaComponent.get(UgcMirrorService::class.java)

        val result: UgcMirrorResult = runCatching { service.mirrorByUrl(request.url) }
            .getOrElse {
                call.respond(
                    status = HttpStatusCode.InternalServerError,
                    message = mapOf(
                        "error" to "INTERNAL_ERROR",
                        "message" to "Не удалось построить UGC-зеркало",
                    ),
                )
                return@post
            }

        // На уровне HTTP-ответа приводим доменную модель к DTO,
        // совместимому с MirrorResponse на клиенте.
        val attributes = (result.attributes as? Map<String, String>)

        val response = UgcMirrorResponse(
            sourceUrl = result.sourceUrl,
            title = result.title,
            description = result.description,
            priceRaw = result.priceRaw,
            currency = result.currency,
            imageUrls = result.imageUrls,
            attributes = attributes,
        )

        call.respond(
            status = HttpStatusCode.OK,
            message = response,
        )
    }
}
