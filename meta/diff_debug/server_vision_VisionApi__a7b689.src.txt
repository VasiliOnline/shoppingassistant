package com.example.shoppingassistant.server.vision

import com.example.shoppingassistant.domain.model.NormalizedQuery
import com.example.shoppingassistant.domain.vision.VisionNormalizeRequest as VisionNormalizePhotosRequest
import com.example.shoppingassistant.domain.vision.VisionNormalizeResult
import com.example.shoppingassistant.domain.vision.VisionConsumeRequest
import com.example.shoppingassistant.domain.vision.VisionUsageResult
import io.ktor.http.HttpStatusCode
import io.ktor.server.application.call
import io.ktor.server.request.receive
import io.ktor.server.response.respond
import io.ktor.server.routing.Route
import io.ktor.server.routing.post
import kotlinx.serialization.Serializable
import org.koin.java.KoinJavaComponent

@Serializable
private data class VisionNormalizeImageRequest(
    val imageBase64: String,
)

@Serializable
private data class VisionNormalizeResponse(
    val status: String,
    val normalized: NormalizedQuery? = null,
)

@Serializable
private data class VisionNormalizePhotosResponse(
    val status: String,
    val result: VisionNormalizeResult? = null,
)

@Serializable
private data class VisionUsageRequest(
    val userKey: String? = null,
)

fun Route.visionRoutes() {
    post("/api/vision/normalize") {
        val request = runCatching { call.receive<VisionNormalizeImageRequest>() }.getOrElse {
            call.respond(
                status = HttpStatusCode.BadRequest,
                message = mapOf("error" to "BAD_REQUEST", "message" to "Некорректное тело запроса"),
            )
            return@post
        }

        val service: VisionService = KoinJavaComponent.get(VisionService::class.java)
        val normalized = runCatching { service.normalizeImage(request.imageBase64) }.getOrNull()

        if (normalized == null) {
            call.respond(
                status = HttpStatusCode.BadRequest,
                message = VisionNormalizeResponse(status = "FAILED", normalized = null),
            )
            return@post
        }

        call.respond(
            status = HttpStatusCode.OK,
            message = VisionNormalizeResponse(status = "OK", normalized = normalized),
        )
    }

    post("/api/vision/normalizePhotos") {
        val request = runCatching { call.receive<VisionNormalizePhotosRequest>() }.getOrElse {
            call.respond(
                status = HttpStatusCode.BadRequest,
                message = mapOf("error" to "BAD_REQUEST", "message" to "Некорректное тело запроса"),
            )
            return@post
        }

        val service: VisionService = KoinJavaComponent.get(VisionService::class.java)
        val result = runCatching { service.normalizePhotos(request) }.getOrNull()

        call.respond(
            status = HttpStatusCode.OK,
            message = VisionNormalizePhotosResponse(
                status = if (result == null) "FAILED" else "OK",
                result = result,
            ),
        )
    }

    post("/api/vision/usage") {
        val request = runCatching { call.receive<VisionUsageRequest>() }.getOrElse {
            call.respond(
                status = HttpStatusCode.BadRequest,
                message = mapOf("error" to "BAD_REQUEST", "message" to "Некорректное тело запроса"),
            )
            return@post
        }

        val repo: VisionUsageRepositoryImpl = KoinJavaComponent.get(VisionUsageRepositoryImpl::class.java)
        val result = runCatching { repo.getUsage(request.userKey) }.getOrNull()
            ?: VisionUsageResult(status = com.example.shoppingassistant.domain.vision.VisionUsageStatus.ERROR)

        call.respond(
            status = HttpStatusCode.OK,
            message = result,
        )
    }

    post("/api/vision/consume") {
        val request = runCatching { call.receive<VisionConsumeRequest>() }.getOrElse {
            call.respond(
                status = HttpStatusCode.BadRequest,
                message = mapOf("error" to "BAD_REQUEST", "message" to "Некорректное тело запроса"),
            )
            return@post
        }

        val repo: VisionUsageRepositoryImpl = KoinJavaComponent.get(VisionUsageRepositoryImpl::class.java)
        val result = runCatching { repo.consume(request) }.getOrNull()
            ?: VisionUsageResult(status = com.example.shoppingassistant.domain.vision.VisionUsageStatus.ERROR)

        call.respond(
            status = HttpStatusCode.OK,
            message = result,
        )
    }
}
