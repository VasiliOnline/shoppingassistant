package com.example.shoppingassistant.push

import android.util.Log
import com.example.shoppingassistant.core.push.SubscriptionsNotificationsChannel
import com.example.shoppingassistant.core.push.SubscriptionsNotificationsSettings
import com.example.shoppingassistant.core.push.SubscriptionsNotificationsSettingsStorage
import com.example.shoppingassistant.core.push.SubscriptionsPushNotifier
import com.example.shoppingassistant.core.push.SubscriptionsPushStateStorage
import com.example.shoppingassistant.core.push.SubscriptionsPushTokensRemoteDataSource
import com.google.firebase.messaging.FirebaseMessagingService
import com.google.firebase.messaging.RemoteMessage
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.launch
import org.koin.core.context.GlobalContext
import java.util.Calendar
import kotlin.math.abs

class SubscriptionsFirebaseMessagingService : FirebaseMessagingService() {

    override fun onNewToken(token: String) {
        val koin = runCatching { GlobalContext.get() }.getOrNull()
        if (koin == null) {
            Log.w("SubscriptionsFCM", "Koin is not started yet; skip token registration")
            return
        }

        val registrar = koin.get<SubscriptionsPushTokensRemoteDataSource>()
        CoroutineScope(Dispatchers.IO).launch {
            runCatching { registrar.registerToken(platform = "ANDROID", token = token) }
                .onFailure { t ->
                    Log.w("SubscriptionsFCM", "Failed to register FCM token: ${t.message}", t)
                }
        }
    }

    override fun onMessageReceived(message: RemoteMessage) {
        val body = message.notification?.body
            ?: message.data["message"]
            ?: return

        val notificationId = message.data["notificationId"]?.toLongOrNull()
        val koin = runCatching { GlobalContext.get() }.getOrNull()

        if (notificationId != null && koin != null) {
            runCatching {
                val state = koin.get<SubscriptionsPushStateStorage>()
                val prev = state.getLastNotifiedId()
                if (notificationId > prev) state.setLastNotifiedId(notificationId)
            }
        }

        SubscriptionsNotificationsChannel.ensureCreated(this)

        val notifier = SubscriptionsPushNotifier(this)
        if (!notifier.canPostNotifications()) return

        val settings = runCatching {
            koin?.get<SubscriptionsNotificationsSettingsStorage>()?.get()
        }.getOrNull() ?: SubscriptionsNotificationsSettings()

        if (!settings.pushEnabled) return
        if (settings.quietHoursEnabled && isQuietNow(settings)) return

        val id = abs(notificationId?.hashCode() ?: (message.messageId?.hashCode() ?: body.hashCode()))
        notifier.notifyText(
            id = id,
            title = "Сработало отслеживание",
            message = body,
            settings = settings,
        )
    }

    private fun isQuietNow(settings: SubscriptionsNotificationsSettings): Boolean {
        val hour = Calendar.getInstance().get(Calendar.HOUR_OF_DAY)
        val start = settings.quietHoursStart.coerceIn(0, 23)
        val end = settings.quietHoursEnd.coerceIn(0, 23)

        return when {
            start == end -> true
            start < end -> hour in start until end
            else -> hour >= start || hour < end
        }
    }
}
