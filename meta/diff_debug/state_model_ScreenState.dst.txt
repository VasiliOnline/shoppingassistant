package com.example.shoppingassistant.feature.ui.state.model

import androidx.compose.runtime.Immutable
import androidx.compose.ui.graphics.vector.ImageVector

@Immutable
sealed interface ScreenState {
    @Immutable
    data class Loading(val phase: LoadingPhase) : ScreenState

    @Immutable
    data class Content(
        val isStale: Boolean = false,
        val lastUpdatedMillis: Long? = null,
    ) : ScreenState

    @Immutable
    data class Empty(
        val reason: EmptyReason,
        val title: String? = null,
        val message: String? = null,
        val primaryAction: StateAction? = null,
        val secondaryAction: StateAction? = null,
        val icon: ImageVector? = null,
    ) : ScreenState

    @Immutable
    data class Error(
        val message: String,
        val title: String? = null,
        val primaryAction: StateAction? = null,
        val secondaryAction: StateAction? = null,
        val icon: ImageVector? = null,
    ) : ScreenState

    @Immutable
    data class Offline(
        val mode: OfflineMode,
        val message: String? = null,
        val lastUpdatedMillis: Long? = null,
        val primaryAction: StateAction? = null,
    ) : ScreenState
}

enum class LoadingPhase {
    INITIAL,
    REFRESHING,
    PAGING,
}

enum class EmptyReason {
    NO_RESULTS,
    NO_HISTORY,
    NO_SUBSCRIPTIONS,
    NO_MESSAGES,
    NO_OFFERS,
    OTHER,
}

enum class OfflineMode {
    WITH_CACHE,
    NO_CACHE,
}

@Immutable
data class StateAction(
    val label: String,
    val onAction: () -> Unit,
)
