@Composable
fun SubscriptionsCreateMode(
    state: SubscriptionsCreateState,
    contentPadding: PaddingValues,
    onOpenCategory: () -> Unit,
    onSearchQueryChange: (String) -> Unit,
    onSuggestionPick: (MainSuggestItem) -> Unit,
    onOpenBrand: () -> Unit,
    onOpenModel: () -> Unit,
    onBrandSelected: (String) -> Unit,
    onModelSelected: (String) -> Unit,
    onOpenMoreAttributes: () -> Unit,
    onOpenGeneralParams: () -> Unit,
    onOpenConditions: () -> Unit,
    onDismissSheet: () -> Unit,
    onCategoryQueryChange: (String) -> Unit,
    onCategoryPathChange: (List<String>) -> Unit,
    onSelectCategoryLeaf: (String) -> Unit,
    onPickFilterAttr: (CreateFilterTarget, AttributeDef) -> Unit,
    onPickFilterValue: (CreateFilterTarget, AttributeDef, String) -> Unit,
    onClearFilterValue: (CreateFilterTarget, String) -> Unit,
    onUpdateConditions: (com.example.shoppingassistant.feature.pages.subscriptions.CreateConditions) -> Unit,
    onOpenExisting: () -> Unit,
    modifier: Modifier = Modifier,
) {
    val draft = state.draft
    val showKeyAttributes = state.attributeDefs.any { it.key.equals("brand", ignoreCase = true) || it.key.equals("model", ignoreCase = true) }
    val generalAvailable = state.generalAttributeDefs.isNotEmpty()
    val moreAvailable = state.moreAttributeDefs.isNotEmpty()

    LazyColumn(
        modifier = modifier
            .fillMaxSize()
            .padding(contentPadding)
            .padding(horizontal = LayoutDefaults.HorizontalPadding),
        verticalArrangement = Arrangement.spacedBy(12.dp),
        contentPadding = PaddingValues(vertical = LayoutDefaults.SectionSpacing),
    ) {
        item {
            TargetSummaryCard(draft)
        }
        item { SectionHeader("Категория") }
        item {
            CreateRow(
                title = "Категория",
                value = draft.categoryBreadcrumb ?: "Выберите конечную категорию",
                onClick = onOpenCategory,
                isError = state.validationErrorSection == CreateSection.CATEGORY,
            )
        }
        item { SectionHeader("Быстрый поиск") }
        item {
            InputRow(
                input = draft.searchQuery,
                onInputChange = onSearchQueryChange,
                onSubmit = { },
                suggestions = state.suggestions,
                onSuggestionClick = onSuggestionPick,
                hideSuggestions = false,
                hasSelection = draft.product != null,
                onPhotoClick = {},
                onLinkClick = {},
                onVoiceClick = {},
                onClear = { onSearchQueryChange("") },
                showMediaActions = false,
                placeholder = "Категория или товар",
                modifier = Modifier.fillMaxWidth(),
            )
        }
        if (showKeyAttributes) {
            item { SectionHeader("Ключевые характеристики") }
            val brandDef = state.attributeDefs.firstOrNull { it.key.equals("brand", ignoreCase = true) }
            val modelDef = state.attributeDefs.firstOrNull { it.key.equals("model", ignoreCase = true) }
            if (brandDef != null) {
                item {
                    CreateRow(
                        title = brandDef.title,
                        value = draft.brand ?: "Не выбрано",
                        onClick = onOpenBrand,
                        enabled = brandDef.options.isNotEmpty() || brandDef.allowedValues.isNotEmpty(),
                    )
                }
            }
            if (modelDef != null) {
                val modelEnabled = !draft.brand.isNullOrBlank()
                item {
                    CreateRow(
                        title = modelDef.title,
                        value = draft.model ?: if (modelEnabled) "Не выбрано" else "Сначала выберите бренд",
                        onClick = onOpenModel,
                        enabled = modelEnabled && (modelDef.options.isNotEmpty() || modelDef.allowedValues.isNotEmpty()),
                    )
                }
            }
        }
        item { SectionHeader("Ещё характеристики") }
        item {
            CreateRow(
                title = "Характеристики",
                value = if (moreAvailable) "Выбрать" else "Для категории нет характеристик",
                onClick = onOpenMoreAttributes,
                enabled = moreAvailable,
            )
        }
        if (draft.moreAttributes.isNotEmpty()) {
            item {
                SelectedAttributesSummary(
                    selected = draft.moreAttributes,
                    defs = state.moreAttributeDefs,
                    maxLines = 5,
                )
            }
        }
        if (generalAvailable) {
            item { SectionHeader("Общие параметры") }
            item {
                CreateRow(
                    title = "Общие параметры",
                    value = if (draft.generalAttributes.isEmpty()) "Выбрать" else "Выбрано: ${draft.generalAttributes.size}",
                    onClick = onOpenGeneralParams,
                    enabled = generalAvailable,
                )
            }
            if (draft.generalAttributes.isNotEmpty()) {
                item {
                    SelectedAttributesSummary(
                        selected = draft.generalAttributes,
                        defs = state.generalAttributeDefs,
                        maxLines = 5,
                    )
                }
            }
        }
        item { SectionHeader("Условия") }
        item {
            CreateRow(
                title = "Условия срабатывания",
                value = conditionsSummary(draft),
                onClick = onOpenConditions,
                isError = state.validationErrorSection == CreateSection.CONDITIONS,
            )
        }
        if (state.existingSubscription != null) {
            item {
                ExistingSubscriptionCard(onOpenExisting = onOpenExisting)
            }
        }
        item { Spacer(modifier = Modifier.size(4.dp)) }
    }

    when (val sheet = state.sheet) {
        CreateSheet.Category -> {
            CategoryLeafPickerSheet(
                categories = state.categories,
                query = state.categoryQuery,
                pathCodes = state.categoryPathCodes,
                selectedCategoryCode = draft.categoryCode,
                onQueryChange = onCategoryQueryChange,
                onPathChange = onCategoryPathChange,
                onSelectCategory = { category -> onSelectCategoryLeaf(category.code) },
                onDismiss = onDismissSheet,
            )
        }
        is CreateSheet.ValuePicker -> {
            val def = state.attributeDefs.firstOrNull { it.key == sheet.key }
            val selected = when (sheet.key.lowercase()) {
                "brand" -> draft.brand
                "model" -> draft.model
                else -> null
            }
            val options = remember(def) {
                val allowed = def?.allowedValues?.map { it.canonical }.orEmpty()
                if (allowed.isNotEmpty()) allowed else def?.options.orEmpty()
            }
            ValuePickerSheet(
                title = sheet.title,
                options = options,
                selected = selected,
                onSelect = { value ->
                    when (sheet.key.lowercase()) {
                        "brand" -> onBrandSelected(value)
                        "model" -> onModelSelected(value)
                    }
                },
                onDismiss = onDismissSheet,
            )
        }
        is CreateSheet.Filter -> {
            val target = sheet.target
            val defs = if (target == CreateFilterTarget.MORE) state.moreAttributeDefs else state.generalAttributeDefs
            val selected = if (target == CreateFilterTarget.MORE) draft.moreAttributes else draft.generalAttributes
            val filterState = if (target == CreateFilterTarget.MORE) state.moreFilterSheet else state.generalFilterSheet
            val current = filterState.currentAttrKey?.let { key -> defs.firstOrNull { it.key == key } }
            FilterSheet(
                visible = true,
                stage = filterState.stage,
                defs = defs,
                current = current,
                selected = selected,
                attrFacetCounts = emptyMap(),
                attrFacetCountsLoading = emptySet(),
                facetCounts = null,
                facetCountsLoading = false,
                onDismiss = onDismissSheet,
                onPickAttr = { def -> onPickFilterAttr(target, def) },
                onPickValue = { def, value -> onPickFilterValue(target, def, value) },
                onClearAttr = { def -> onClearFilterValue(target, def.key) },
            )
        }
        CreateSheet.Conditions -> {
            ConditionsSheet(
                conditions = draft.conditions,
                onChange = { updated -> onUpdateConditions(updated) },
                onDismiss = onDismissSheet,
            )
        }
        null -> Unit
    }
}

@Composable
private fun TargetSummaryCard(draft: SubscriptionsCreateDraft) {
    val summary = buildTargetSummary(draft)
    Surface(
        shape = MaterialTheme.shapes.large,
        color = MaterialTheme.colorScheme.surfaceVariant,
        modifier = Modifier.fillMaxWidth(),
    ) {
        Column(
            modifier = Modifier.padding(14.dp),
            verticalArrangement = Arrangement.spacedBy(6.dp),
        ) {
            Text(
                text = "Цель отслеживания",
                style = MaterialTheme.typography.labelMedium,
                color = MaterialTheme.colorScheme.onSurfaceVariant,
            )
            Text(
                text = summary,
                style = MaterialTheme.typography.titleMedium,
                maxLines = 1,
                overflow = TextOverflow.Ellipsis,
            )
        }
    }
}

private fun buildTargetSummary(draft: SubscriptionsCreateDraft): String {
    val productTitle = draft.product?.title
        ?: listOfNotNull(draft.brand?.takeIf { it.isNotBlank() }, draft.model?.takeIf { it.isNotBlank() })
            .joinToString(" ")
            .trim()
    val category = draft.categoryBreadcrumb
    return listOfNotNull(
        productTitle.takeIf { !it.isNullOrBlank() },
        category.takeIf { !it.isNullOrBlank() },
    ).joinToString(" · ").ifBlank { "Выберите категорию" }
}

@Composable
private fun SectionHeader(title: String) {
    Text(
        text = title,
        style = MaterialTheme.typography.titleSmall,
        color = MaterialTheme.colorScheme.primary,
    )
}

@Composable
private fun CreateRow(
    title: String,
    value: String,
    onClick: () -> Unit,
    enabled: Boolean = true,
    isError: Boolean = false,
) {
    Surface(
        shape = MaterialTheme.shapes.medium,
        color = if (isError) MaterialTheme.colorScheme.error.copy(alpha = 0.08f)
        else MaterialTheme.colorScheme.surfaceVariant.copy(alpha = 0.4f),
        modifier = Modifier
            .fillMaxWidth()
            .heightIn(min = 52.dp)
            .clickable(enabled = enabled, onClick = onClick),
    ) {
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 12.dp, vertical = 10.dp),
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.spacedBy(10.dp),
        ) {
            Column(modifier = Modifier.weight(1f)) {
                Text(text = title, style = MaterialTheme.typography.bodyMedium)
                Text(
                    text = value,
                    style = MaterialTheme.typography.bodySmall,
                    color = if (isError) MaterialTheme.colorScheme.error else MaterialTheme.colorScheme.onSurfaceVariant,
                    maxLines = 1,
                    overflow = TextOverflow.Ellipsis,
                )
            }
            Icon(
                imageVector = Icons.AutoMirrored.Outlined.KeyboardArrowRight,
                contentDescription = null,
                tint = if (enabled) MaterialTheme.colorScheme.onSurfaceVariant else MaterialTheme.colorScheme.onSurfaceVariant.copy(alpha = 0.4f),
            )
        }
    }
}

@Composable
private fun SelectedAttributesSummary(
    selected: Map<String, String>,
    defs: List<AttributeDef>,
    maxLines: Int,
) {
    val titleByKey = remember(defs) { defs.associate { it.key to it.title } }
    val entries = selected.entries
        .sortedBy { it.key.lowercase() }
        .map { (key, value) -> "${titleByKey[key] ?: key}: $value" }
    Column(verticalArrangement = Arrangement.spacedBy(4.dp)) {
        entries.take(maxLines).forEach { line ->
            Text(
                text = line,
                style = MaterialTheme.typography.bodySmall,
                color = MaterialTheme.colorScheme.onSurfaceVariant,
                maxLines = 1,
                overflow = TextOverflow.Ellipsis,
            )
        }
        if (entries.size > maxLines) {
            Text(
                text = "ещё ${entries.size - maxLines}",
                style = MaterialTheme.typography.bodySmall,
                color = MaterialTheme.colorScheme.onSurfaceVariant,
            )
        }
    }
}

private fun conditionsSummary(draft: SubscriptionsCreateDraft): String {
    val parts = mutableListOf<String>()
    val conditions = draft.conditions
    if (conditions.newOfferAppeared) parts += "Новый оффер"
    if (conditions.maxPriceEnabled && conditions.maxPriceMajor.isNotBlank()) {
        parts += "Цена ≤ ${conditions.maxPriceMajor.trim()} ${conditions.currency}"
    }
    if (conditions.dropPercentEnabled && conditions.dropPercent.isNotBlank()) {
        parts += "Падение ≥ ${conditions.dropPercent.trim()}%"
    }
    return if (parts.isEmpty()) "Не выбрано" else parts.joinToString(" · ")
}

@Composable
private fun ExistingSubscriptionCard(
    onOpenExisting: () -> Unit,
) {
    ElevatedCard(
        modifier = Modifier.fillMaxWidth(),
        shape = MaterialTheme.shapes.large,
    ) {
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(14.dp),
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.spacedBy(12.dp),
        ) {
            Icon(Icons.Outlined.Info, contentDescription = null)
            Column(modifier = Modifier.weight(1f)) {
                Text(
                    text = "Уже отслеживается",
                    style = MaterialTheme.typography.titleSmall,
                    fontWeight = FontWeight.SemiBold,
                )
                Text(
                    text = "Откройте подписку, чтобы изменить условия.",
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                )
            }
            TextButton(onClick = onOpenExisting) { Text("Открыть") }
        }
    }
}
