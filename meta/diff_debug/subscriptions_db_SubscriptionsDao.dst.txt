package com.example.shoppingassistant.core.data.subscriptions.db

import androidx.room.Dao
import androidx.room.Insert
import androidx.room.OnConflictStrategy
import androidx.room.Query
import androidx.room.Transaction

@Dao
interface SubscriptionsDao {

    @Transaction
    @Query("SELECT * FROM subscriptions ORDER BY updatedAtMillis DESC")
    suspend fun listSubscriptions(): List<SubscriptionWithConditions>

    @Query("SELECT COUNT(*) FROM subscriptions")
    suspend fun countSubscriptions(): Int

    @Query("SELECT id FROM subscriptions WHERE scope = :scope AND input = :input LIMIT 1")
    suspend fun findSubscriptionId(scope: String, input: String): Long?

    @Query("SELECT title FROM subscriptions WHERE id = :id LIMIT 1")
    suspend fun getSubscriptionTitle(id: Long): String?

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertSubscription(entity: SubscriptionEntity): Long

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertConditions(items: List<SubscriptionConditionEntity>)

    @Query("DELETE FROM subscription_conditions WHERE subscriptionId = :subscriptionId")
    suspend fun deleteConditions(subscriptionId: Long)

    @Query(
        """
            UPDATE subscriptions
            SET title = :title,
                minAlertIntervalMinutes = :minAlertIntervalMinutes,
                isActive = :isActive,
                updatedAtMillis = :updatedAtMillis
            WHERE id = :id
            """
    )
    suspend fun updateSubscriptionBase(
        id: Long,
        title: String,
        minAlertIntervalMinutes: Int,
        isActive: Boolean,
        updatedAtMillis: Long
    ): Int

    @Query("DELETE FROM subscriptions WHERE id = :id")
    suspend fun deleteSubscription(id: Long)

    @Query("SELECT * FROM subscription_notifications ORDER BY createdAtMillis DESC")
    suspend fun listNotifications(): List<SubscriptionNotificationEntity>

    @Query("SELECT * FROM subscription_notifications ORDER BY createdAtMillis DESC LIMIT :limit")
    suspend fun listNotificationsLimited(limit: Int): List<SubscriptionNotificationEntity>

    @Query(
        """
        DELETE FROM subscription_notifications
        WHERE id NOT IN (
            SELECT id FROM subscription_notifications
            ORDER BY createdAtMillis DESC
            LIMIT :keepLast
        )
        """
    )

    suspend fun trimNotifications(keepLast: Int)

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertNotification(entity: SubscriptionNotificationEntity): Long

    @Query("UPDATE subscription_notifications SET isRead = 1 WHERE id = :id")
    suspend fun markRead(id: Long)

    // ==========================
    // Production-grade атомарность
    // ==========================

    /**
     * Атомарное создание подписки: subscription -> conditions -> notification
     */
    @Transaction
    suspend fun createSubscription(
        subscription: SubscriptionEntity,
        conditions: List<SubscriptionConditionEntity>,
        notification: SubscriptionNotificationEntity
    ): Long {
        val id = insertSubscription(subscription)
        if (conditions.isNotEmpty()) {
            insertConditions(conditions.map { it.copy(subscriptionId = id) })
        }
        insertNotification(notification.copy(subscriptionId = id))
        return id
    }

    /**
     * Атомарное обновление: base -> replace conditions -> notification
     */
    @Transaction
    suspend fun updateSubscriptionWithConditions(
        id: Long,
        title: String,
        minAlertIntervalMinutes: Int,
        isActive: Boolean,
        updatedAtMillis: Long,
        conditions: List<SubscriptionConditionEntity>,
        notification: SubscriptionNotificationEntity
    ): Boolean {
        val updated = updateSubscriptionBase(
            id = id,
            title = title,
            minAlertIntervalMinutes = minAlertIntervalMinutes,
            isActive = isActive,
            updatedAtMillis = updatedAtMillis
        ) > 0
        if (!updated) return false

        deleteConditions(id)
        if (conditions.isNotEmpty()) {
            insertConditions(conditions.map { it.copy(subscriptionId = id) })
        }
        insertNotification(notification)
        return true
    }

    /**
     * Атомарное удаление: delete -> notification
     */
    @Transaction
    suspend fun deleteSubscriptionWithNotification(
        id: Long,
        notification: SubscriptionNotificationEntity
    ) {
        deleteSubscription(id)
        insertNotification(notification)
    }
}
