package com.example.shoppingassistant.core.data.subscriptions.db

import androidx.room.Dao
import androidx.room.Insert
import androidx.room.OnConflictStrategy
import androidx.room.Query
import androidx.room.Transaction

@Dao
interface SubscriptionsDao {

    @Transaction
    @Query("SELECT * FROM subscriptions WHERE ownerKey = :ownerKey ORDER BY updatedAtMillis DESC")
    suspend fun listSubscriptions(ownerKey: String): List<SubscriptionWithConditions>

    @Query("SELECT COUNT(*) FROM subscriptions WHERE ownerKey = :ownerKey")
    suspend fun countSubscriptions(ownerKey: String): Int

    @Query("SELECT id FROM subscriptions WHERE ownerKey = :ownerKey AND scope = :scope AND input = :input LIMIT 1")
    suspend fun findSubscriptionId(ownerKey: String, scope: String, input: String): Long?

    @Query("SELECT title FROM subscriptions WHERE ownerKey = :ownerKey AND id = :id LIMIT 1")
    suspend fun getSubscriptionTitle(ownerKey: String, id: Long): String?

    @Insert(onConflict = OnConflictStrategy.IGNORE)
    suspend fun insertSubscription(entity: SubscriptionEntity): Long

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertConditions(items: List<SubscriptionConditionEntity>)

    @Query("DELETE FROM subscription_conditions WHERE subscriptionId = :subscriptionId")
    suspend fun deleteConditions(subscriptionId: Long)

    @Query(
        """
            UPDATE subscriptions
            SET title = :title,
                minAlertIntervalMinutes = :minAlertIntervalMinutes,
                isActive = :isActive,
                updatedAtMillis = :updatedAtMillis
            WHERE ownerKey = :ownerKey AND id = :id
            """
    )
    suspend fun updateSubscriptionBase(
        ownerKey: String,
        id: Long,
        title: String,
        minAlertIntervalMinutes: Int,
        isActive: Boolean,
        updatedAtMillis: Long
    ): Int

    @Query("DELETE FROM subscriptions WHERE ownerKey = :ownerKey AND id = :id")
    suspend fun deleteSubscription(ownerKey: String, id: Long): Int

    @Query("SELECT * FROM subscription_notifications WHERE ownerKey = :ownerKey ORDER BY createdAtMillis DESC")
    suspend fun listNotifications(ownerKey: String): List<SubscriptionNotificationEntity>

    @Query(
        "SELECT * FROM subscription_notifications WHERE ownerKey = :ownerKey ORDER BY createdAtMillis DESC LIMIT :limit"
    )
    suspend fun listNotificationsLimited(ownerKey: String, limit: Int): List<SubscriptionNotificationEntity>

    @Query(
        "SELECT * FROM subscription_notifications WHERE ownerKey = :ownerKey ORDER BY createdAtMillis DESC LIMIT :limit OFFSET :offset"
    )
    suspend fun listNotificationsPage(
        ownerKey: String,
        limit: Int,
        offset: Int,
    ): List<SubscriptionNotificationEntity>

    @Query("SELECT COUNT(*) FROM subscription_notifications WHERE ownerKey = :ownerKey")
    suspend fun countNotifications(ownerKey: String): Int

    @Query(
        """
        DELETE FROM subscription_notifications
        WHERE ownerKey = :ownerKey AND id NOT IN (
            SELECT id FROM subscription_notifications
            WHERE ownerKey = :ownerKey
            ORDER BY createdAtMillis DESC
            LIMIT :keepLast
        )
        """
    )
    suspend fun trimNotifications(ownerKey: String, keepLast: Int)

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertNotification(entity: SubscriptionNotificationEntity): Long

    @Query("UPDATE subscription_notifications SET isRead = 1 WHERE ownerKey = :ownerKey AND id = :id")
    suspend fun markRead(ownerKey: String, id: Long)

    @Query("UPDATE subscription_notifications SET isRead = 1 WHERE ownerKey = :ownerKey AND isRead = 0")
    suspend fun markAllRead(ownerKey: String): Int

    // ==========================
    // Production-grade атомарность
    // ==========================

    /**
     * Атомарное создание подписки: subscription -> conditions -> notification
     */
    @Transaction
    suspend fun createSubscription(
        subscription: SubscriptionEntity,
        conditions: List<SubscriptionConditionEntity>,
        notification: SubscriptionNotificationEntity
    ): Long {
        val id = insertSubscription(subscription)
        if (id == -1L) return -1L
        if (conditions.isNotEmpty()) {
            insertConditions(conditions.map { it.copy(subscriptionId = id) })
        }
        insertNotification(notification.copy(subscriptionId = id))
        return id
    }

    /**
     * Атомарное обновление: base -> replace conditions -> notification
     */
    @Transaction
    suspend fun updateSubscriptionWithConditions(
        ownerKey: String,
        id: Long,
        title: String,
        minAlertIntervalMinutes: Int,
        isActive: Boolean,
        updatedAtMillis: Long,
        conditions: List<SubscriptionConditionEntity>,
        notification: SubscriptionNotificationEntity
    ): Boolean {
        val updated = updateSubscriptionBase(
            ownerKey = ownerKey,
            id = id,
            title = title,
            minAlertIntervalMinutes = minAlertIntervalMinutes,
            isActive = isActive,
            updatedAtMillis = updatedAtMillis
        ) > 0
        if (!updated) return false

        deleteConditions(id)
        if (conditions.isNotEmpty()) {
            insertConditions(conditions.map { it.copy(subscriptionId = id) })
        }
        insertNotification(notification)
        return true
    }

    /**
     * Атомарное удаление: delete -> notification
     */
    @Transaction
    suspend fun deleteSubscriptionWithNotification(
        ownerKey: String,
        id: Long,
        notification: SubscriptionNotificationEntity
    ) {
        val deleted = deleteSubscription(ownerKey = ownerKey, id = id)
        if (deleted > 0) {
            insertNotification(notification)
        }
    }
}
