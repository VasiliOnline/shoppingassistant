object TemplatePresetsSeed {
    val popular: List<TemplatePresetSpec> = buildPopularPresets()
}

private fun buildPopularPresets(): List<TemplatePresetSpec> {
    val presets = mutableListOf<TemplatePresetSpec>()
    presets += phonePresets()
    presets += laptopPresets()
    presets += strollerPresets()
    presets += coffeeMachinePresets()
    presets += sneakersPresets()
    return presets
}

private fun presetSpec(
    source: TemplatePresetSource,
    categoryCode: String,
    anchorType: TemplateAnchorType,
    anchorId: String,
    title: String,
    attrs: Map<String, String>,
    rank: Int,
): TemplatePresetSpec {
    val attrList = attrs.map { (key, value) ->
        TemplateSnapshotAttr(key = key, value = value)
    }
    return TemplatePresetSpec(
        source = source,
        title = title,
        rank = rank,
        snapshot = TemplateSnapshotData(
            anchorType = anchorType,
            anchorId = anchorId,
            categoryCode = categoryCode,
            attrs = attrList,
            mode = TemplateSnapshotMode.SEARCH,
        ),
    )
}
