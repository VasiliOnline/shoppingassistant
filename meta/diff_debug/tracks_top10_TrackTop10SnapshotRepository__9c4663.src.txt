package com.example.shoppingassistant.server.tracks.top10

import com.example.shoppingassistant.domain.tracks.RankExplanation
import com.example.shoppingassistant.domain.tracks.SourceStamp
import com.example.shoppingassistant.domain.tracks.TrackTop10
import com.example.shoppingassistant.server.db.DatabaseFactory
import com.example.shoppingassistant.server.tracks.TracksTable
import org.jetbrains.exposed.sql.ResultRow
import org.jetbrains.exposed.sql.SqlExpressionBuilder.eq
import org.jetbrains.exposed.sql.SqlExpressionBuilder.isNull
import org.jetbrains.exposed.sql.SqlExpressionBuilder.lessEq
import org.jetbrains.exposed.sql.SqlExpressionBuilder.plus
import org.jetbrains.exposed.sql.and
import org.jetbrains.exposed.sql.innerJoin
import org.jetbrains.exposed.sql.insert
import org.jetbrains.exposed.sql.or
import org.jetbrains.exposed.sql.selectAll
import org.jetbrains.exposed.sql.update

data class TrackTop10SnapshotRow(
    val trackId: Long,
    val snapshot: TrackTop10,
    val lockedBy: String?,
    val lockUntil: Long?,
    val lastAttemptAt: Long?,
    val lastSuccessAt: Long?,
    val failCount: Int,
    val nextRetryAt: Long?,
    val lastError: String?,
)

interface TrackTop10SnapshotRepository {
    suspend fun getSnapshot(userId: Long, trackId: Long): TrackTop10SnapshotRow?
    suspend fun upsertSnapshot(trackId: Long, snapshot: TrackTop10, now: Long)
    suspend fun tryLock(trackId: Long, workerId: String, lockTtlMs: Long): Boolean
    suspend fun unlock(trackId: Long, workerId: String)
    suspend fun markAttempt(trackId: Long, now: Long)
    suspend fun markSuccess(trackId: Long, now: Long)
    suspend fun markFailure(trackId: Long, now: Long, error: String?, nextRetryAt: Long)
    suspend fun scheduleRefresh(trackId: Long, at: Long)
}

class TrackTop10SnapshotRepositoryImpl(
    private val clock: () -> Long = { System.currentTimeMillis() },
) : TrackTop10SnapshotRepository {

    override suspend fun getSnapshot(userId: Long, trackId: Long): TrackTop10SnapshotRow? =
        DatabaseFactory.dbQuery {
            val row = TrackTop10SnapshotsTable
                .innerJoin(TracksTable, { TrackTop10SnapshotsTable.trackId }, { TracksTable.id })
                .selectAll()
                .where { (TracksTable.userId eq userId) and (TracksTable.id eq trackId) }
                .limit(1)
                .singleOrNull()
                ?: return@dbQuery null

            mapRow(row)
        }

    override suspend fun upsertSnapshot(trackId: Long, snapshot: TrackTop10, now: Long) {
        DatabaseFactory.dbQuery {
            val updated = TrackTop10SnapshotsTable.update(
                where = { TrackTop10SnapshotsTable.trackId eq trackId },
            ) { stmt ->
                stmt[computedAt] = snapshot.computedAt
                stmt[items] = snapshot.items
                stmt[explanation] = snapshot.explanation
                stmt[sourceStamps] = snapshot.sourceStamps
                stmt[freshnessSec] = snapshot.freshnessSec
            }
            if (updated == 0) {
                TrackTop10SnapshotsTable.insert { stmt ->
                    stmt[TrackTop10SnapshotsTable.trackId] = trackId
                    stmt[computedAt] = snapshot.computedAt
                    stmt[items] = snapshot.items
                    stmt[explanation] = snapshot.explanation
                    stmt[sourceStamps] = snapshot.sourceStamps
                    stmt[freshnessSec] = snapshot.freshnessSec
                    stmt[lockedBy] = null
                    stmt[lockUntil] = null
                    stmt[lastAttemptAt] = now
                    stmt[lastSuccessAt] = now
                    stmt[failCount] = 0
                    stmt[nextRetryAt] = null
                    stmt[lastError] = null
                }
            }
        }
    }

    override suspend fun tryLock(trackId: Long, workerId: String, lockTtlMs: Long): Boolean =
        DatabaseFactory.dbQuery {
            val now = clock()
            val lockUntilAt = now + lockTtlMs
            val updated = TrackTop10SnapshotsTable.update(
                where = {
                    (TrackTop10SnapshotsTable.trackId eq trackId) and
                        (TrackTop10SnapshotsTable.lockUntil.isNull() or (TrackTop10SnapshotsTable.lockUntil lessEq now))
                },
            ) { stmt ->
                stmt[lockedBy] = workerId
                stmt[lockUntil] = lockUntilAt
            }
            if (updated > 0) return@dbQuery true

            val emptySnapshot = emptySnapshot(trackId)
            val inserted = runCatching {
                TrackTop10SnapshotsTable.insert { stmt ->
                    stmt[TrackTop10SnapshotsTable.trackId] = trackId
                    stmt[computedAt] = emptySnapshot.computedAt
                    stmt[items] = emptySnapshot.items
                    stmt[explanation] = emptySnapshot.explanation
                    stmt[sourceStamps] = emptySnapshot.sourceStamps
                    stmt[freshnessSec] = emptySnapshot.freshnessSec
                    stmt[lockedBy] = workerId
                    stmt[lockUntil] = lockUntilAt
                    stmt[lastAttemptAt] = null
                    stmt[lastSuccessAt] = null
                    stmt[failCount] = 0
                    stmt[nextRetryAt] = null
                    stmt[lastError] = null
                }
                true
            }.getOrDefault(false)

            inserted
        }

    override suspend fun unlock(trackId: Long, workerId: String) {
        DatabaseFactory.dbQuery {
            TrackTop10SnapshotsTable.update(
                where = { (TrackTop10SnapshotsTable.trackId eq trackId) and (TrackTop10SnapshotsTable.lockedBy eq workerId) },
            ) { stmt ->
                stmt[lockedBy] = null
                stmt[lockUntil] = null
            }
        }
    }

    override suspend fun markAttempt(trackId: Long, now: Long) {
        DatabaseFactory.dbQuery {
            TrackTop10SnapshotsTable.update(
                where = { TrackTop10SnapshotsTable.trackId eq trackId },
            ) { stmt ->
                stmt[lastAttemptAt] = now
            }
        }
    }

    override suspend fun markSuccess(trackId: Long, now: Long) {
        DatabaseFactory.dbQuery {
            TrackTop10SnapshotsTable.update(
                where = { TrackTop10SnapshotsTable.trackId eq trackId },
            ) { stmt ->
                stmt[lastSuccessAt] = now
                stmt[failCount] = 0
                stmt[nextRetryAt] = null
                stmt[lastError] = null
            }
        }
    }

    override suspend fun markFailure(trackId: Long, now: Long, error: String?, nextRetryAt: Long) {
        val trimmed = error?.trim()?.take(512)
        DatabaseFactory.dbQuery {
            TrackTop10SnapshotsTable.update(
                where = { TrackTop10SnapshotsTable.trackId eq trackId },
            ) { stmt ->
                stmt[lastAttemptAt] = now
                stmt[failCount] = TrackTop10SnapshotsTable.failCount + 1
                stmt[TrackTop10SnapshotsTable.nextRetryAt] = nextRetryAt
                stmt[lastError] = trimmed
            }
        }
    }

    override suspend fun scheduleRefresh(trackId: Long, at: Long) {
        DatabaseFactory.dbQuery {
            val updated = TrackTop10SnapshotsTable.update(
                where = { TrackTop10SnapshotsTable.trackId eq trackId },
            ) { stmt ->
                stmt[nextRetryAt] = at
            }
            if (updated == 0) {
                val emptySnapshot = emptySnapshot(trackId)
                TrackTop10SnapshotsTable.insert { stmt ->
                    stmt[TrackTop10SnapshotsTable.trackId] = trackId
                    stmt[computedAt] = emptySnapshot.computedAt
                    stmt[items] = emptySnapshot.items
                    stmt[explanation] = emptySnapshot.explanation
                    stmt[sourceStamps] = emptySnapshot.sourceStamps
                    stmt[freshnessSec] = emptySnapshot.freshnessSec
                    stmt[lockedBy] = null
                    stmt[lockUntil] = null
                    stmt[lastAttemptAt] = null
                    stmt[lastSuccessAt] = null
                    stmt[failCount] = 0
                    stmt[TrackTop10SnapshotsTable.nextRetryAt] = at
                    stmt[lastError] = null
                }
            }
        }
    }

    private fun mapRow(row: ResultRow): TrackTop10SnapshotRow {
        val snapshot = TrackTop10(
            trackId = row[TrackTop10SnapshotsTable.trackId].toString(),
            computedAt = row[TrackTop10SnapshotsTable.computedAt],
            freshnessSec = row[TrackTop10SnapshotsTable.freshnessSec],
            items = row[TrackTop10SnapshotsTable.items],
            explanation = row[TrackTop10SnapshotsTable.explanation],
            sourceStamps = row[TrackTop10SnapshotsTable.sourceStamps],
        )
        return TrackTop10SnapshotRow(
            trackId = row[TrackTop10SnapshotsTable.trackId],
            snapshot = snapshot,
            lockedBy = row[TrackTop10SnapshotsTable.lockedBy],
            lockUntil = row[TrackTop10SnapshotsTable.lockUntil],
            lastAttemptAt = row[TrackTop10SnapshotsTable.lastAttemptAt],
            lastSuccessAt = row[TrackTop10SnapshotsTable.lastSuccessAt],
            failCount = row[TrackTop10SnapshotsTable.failCount],
            nextRetryAt = row[TrackTop10SnapshotsTable.nextRetryAt],
            lastError = row[TrackTop10SnapshotsTable.lastError],
        )
    }

    private fun emptySnapshot(trackId: Long): TrackTop10 {
        return TrackTop10(
            trackId = trackId.toString(),
            computedAt = 0L,
            freshnessSec = Int.MAX_VALUE,
            items = emptyList(),
            explanation = RankExplanation(
                summary = "Нет данных",
                details = listOf("Снимок ещё не построен"),
            ),
            sourceStamps = listOf(
                SourceStamp(
                    sourceType = "empty",
                    sourceName = "empty",
                    fetchedAt = 0L,
                    ttlSec = 0,
                ),
            ),
        )
    }
}
