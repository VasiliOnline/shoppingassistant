package com.example.shoppingassistant.core.data.ugc.draft

import com.example.shoppingassistant.domain.ugc.draft.DraftOffer
import kotlinx.serialization.json.Json
import kotlinx.serialization.encodeToString
import kotlinx.serialization.decodeFromString

internal object DraftOfferJsonCodec {
    private val json = Json {
        ignoreUnknownKeys = true
        explicitNulls = false
        encodeDefaults = true
    }

    fun encode(draft: DraftOffer): String =
        json.encodeToString(DraftOffer.serializer(), draft)

    fun decode(raw: String): DraftOffer? =
        runCatching { json.decodeFromString(DraftOffer.serializer(), raw) }.getOrNull()
}
