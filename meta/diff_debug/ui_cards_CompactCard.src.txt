package com.example.shoppingassistant.feature.ui.cards

import androidx.compose.foundation.ExperimentalFoundationApi
import androidx.compose.foundation.combinedClickable
import androidx.compose.foundation.interaction.MutableInteractionSource
import androidx.compose.foundation.layout.Arrangement
import androidx.compose.foundation.layout.Box
import androidx.compose.foundation.layout.Column
import androidx.compose.foundation.layout.Row
import androidx.compose.foundation.layout.Spacer
import androidx.compose.foundation.layout.height
import androidx.compose.foundation.layout.fillMaxWidth
import androidx.compose.foundation.layout.padding
import androidx.compose.foundation.layout.size
import androidx.compose.foundation.shape.RoundedCornerShape
import androidx.compose.material.icons.Icons
import androidx.compose.material.icons.outlined.MoreVert
import androidx.compose.material3.DropdownMenu
import androidx.compose.material3.DropdownMenuItem

import androidx.compose.ui.text.style.TextOverflow

@OptIn(ExperimentalFoundationApi::class)
@Composable
fun CompactCard(
    ui: CompactCardUi,
    modifier: Modifier = Modifier,
    density: CardDensity = CardDensity.Dense,
    photoPeekEnabled: Boolean = false,
    overflowActions: List<CompactOverflowAction> = CardActionRules.compactOverflowActions,
) {
    val metrics = CardTokens.metrics(density)
    val shape = RoundedCornerShape(metrics.cornerRadius)
    val pressFeedback = rememberPressFeedbackTask()
    val interactionSource = remember { MutableInteractionSource() }
    val hasOverflow = ui.onOverflowAction != null && overflowActions.isNotEmpty()
    var overflowExpanded by remember { mutableStateOf(false) }

    Surface(
        color = MaterialTheme.colorScheme.surface,
        shape = shape,
        modifier = modifier
            .then(with(pressFeedback) { Modifier.pressEffect(interactionSource) })
            .combinedClickable(
                interactionSource = interactionSource,
                indication = null,
                onClick = ui.onOpenDetails,
                onLongClick = {
                    if (hasOverflow) {
                        overflowExpanded = true
                        ui.onOverflowOpen?.invoke()
                    }
                },
            ),
    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(metrics.contentPadding),
            verticalArrangement = Arrangement.spacedBy(metrics.rowGap),
        ) {
            Box {
                CardMediaPreview(
                    media = ui.media,
                    photoCount = ui.photoCount,
                    photoPeekEnabled = photoPeekEnabled,
                    onOpenDetails = ui.onOpenDetails,
                    onPhotoPeekOpen = null,
                    onPhotoPeekSwipe = null,
                    onPhotoPeekOpenDetails = null,
                    density = density,
                )
                if (ui.showSave && ui.onToggleSave != null) {
                    Box(
                        modifier = Modifier.align(Alignment.TopEnd),
                    ) {
                        CardSaveToggle(
                            isSaved = ui.isSaved,
                            onToggle = ui.onToggleSave,
                            iconSize = metrics.iconSize,
                            hitTarget = metrics.iconHitTarget,
                        )
                    }
                }
            }

            Row(verticalAlignment = Alignment.CenterVertically) {
                Text(
                    text = ui.priceText,
                    style = MaterialTheme.typography.titleMedium,
                    color = MaterialTheme.colorScheme.onSurface,
                    maxLines = 1,
                    overflow = TextOverflow.Ellipsis,
                    modifier = Modifier.weight(1f),
                )
                if (hasOverflow) {
                    IconButton(
                        onClick = {
                            overflowExpanded = true
                            ui.onOverflowOpen?.invoke()
                        },
                        modifier = Modifier.size(metrics.iconHitTarget),
                    ) {
                        Icon(
                            imageVector = Icons.Outlined.MoreVert,
                            contentDescription = stringResource(R.string.card_overflow_label),
                            tint = MaterialTheme.colorScheme.onSurfaceVariant,
                        )
                    }
                }
            }

            Text(
                text = ui.title,
                style = MaterialTheme.typography.bodySmall,
                color = MaterialTheme.colorScheme.onSurface,
                maxLines = 1,
                overflow = TextOverflow.Ellipsis,
            )

            ui.badge?.let { badge ->
                CardBadgesRow(
                    badges = listOf(badge),
                    maxCount = 1,
                    density = density,
                )
            } ?: Spacer(modifier = Modifier.height(metrics.badgeHeight))
        }
    }

    if (hasOverflow) {
        CompactOverflowMenu(
            expanded = overflowExpanded,
            onDismiss = { overflowExpanded = false },
            actions = overflowActions,
            onAction = {
                overflowExpanded = false
                ui.onOverflowAction?.invoke(it)
            },
        )
    }
}

@Composable
private fun CompactOverflowMenu(
    expanded: Boolean,
    onDismiss: () -> Unit,
    actions: List<CompactOverflowAction>,
    onAction: (CompactOverflowAction) -> Unit,
) {
    DropdownMenu(
        expanded = expanded,
        onDismissRequest = onDismiss,
    ) {
        actions.forEach { action ->
            DropdownMenuItem(
                text = { Text(compactActionLabel(action)) },
                onClick = { onAction(action) },
            )
        }
    }
}

@Composable
private fun compactActionLabel(action: CompactOverflowAction): String = when (action) {
    CompactOverflowAction.Share -> stringResource(R.string.card_action_share)
    CompactOverflowAction.Hide -> stringResource(R.string.card_action_hide)
}
