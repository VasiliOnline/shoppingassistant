package com.example.shoppingassistant.feature.ui.cards

import androidx.compose.foundation.combinedClickable

@OptIn(ExperimentalFoundationApi::class)
@Composable
fun OfferCard(
    ui: OfferCardUi,
    modifier: Modifier = Modifier,
    density: CardDensity = CardDensity.Regular,
    photoPeekEnabled: Boolean = false,
    overflowActions: List<OfferOverflowAction> = CardActionRules.offerOverflowActions(includeCopyLink = true),
) {
    val metrics = CardTokens.metrics(density)
    val shape = RoundedCornerShape(metrics.cornerRadius)
    val pressFeedback = rememberPressFeedbackTask()
    val interactionSource = remember { MutableInteractionSource() }
    var overflowExpanded by remember { mutableStateOf(false) }

    Surface(
        color = MaterialTheme.colorScheme.surface,
        shape = shape,
        modifier = modifier
            .then(with(pressFeedback) { Modifier.pressEffect(interactionSource) })
            .combinedClickable(
                interactionSource = interactionSource,
                indication = null,
                onClick = ui.onOpenDetails,
                onLongClick = {
                    if (overflowActions.isNotEmpty()) {
                        overflowExpanded = true
                        ui.onOverflowOpen?.invoke()
                    }
                },
            ),
    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(metrics.contentPadding),
            verticalArrangement = Arrangement.spacedBy(metrics.rowGap),
        ) {
            CardMediaPreview(
                media = ui.media,
                photoCount = ui.photoCount,
                photoPeekEnabled = photoPeekEnabled,
                onOpenDetails = ui.onOpenDetails,
                onPhotoPeekOpen = ui.onPhotoPeekOpen,
                onPhotoPeekSwipe = ui.onPhotoPeekSwipe,
                onPhotoPeekOpenDetails = ui.onPhotoPeekOpenDetails,
                density = density,
            )

            Row(
                horizontalArrangement = Arrangement.spacedBy(8.dp),
            ) {
                Text(
                    text = ui.title,
                    style = MaterialTheme.typography.titleSmall,
                    color = MaterialTheme.colorScheme.onSurface,
                    maxLines = 2,
                    overflow = TextOverflow.Ellipsis,
                    modifier = Modifier.weight(1f),
                )
                SaveToggle(
                    isSaved = ui.isSaved,
                    onToggle = ui.onToggleSave,
                    iconSize = metrics.iconSize,
                    hitTarget = metrics.iconHitTarget,
                )
                if (overflowActions.isNotEmpty()) {
                    IconButton(
                        onClick = {
                            overflowExpanded = true
                            ui.onOverflowOpen?.invoke()
                        },
                        modifier = Modifier.size(metrics.iconHitTarget),
                    ) {
                        Icon(
                            imageVector = Icons.Outlined.MoreVert,
                            contentDescription = stringResource(R.string.card_overflow_label),
                            tint = MaterialTheme.colorScheme.onSurfaceVariant,
                        )
                    }
                }
            }

            Text(
                text = ui.priceText,
                style = MaterialTheme.typography.titleMedium,
                color = MaterialTheme.colorScheme.onSurface,
                maxLines = 1,
                overflow = TextOverflow.Ellipsis,
            )

            MetaLine(
                location = ui.locationText,
                trust = selectTrustLabel(ui.trust),
            )

            CardBadgesRow(
                badges = ui.badges,
                maxCount = 2,
                density = density,
            )
        }
    }

    OfferOverflowMenu(
        expanded = overflowExpanded,
        onDismiss = { overflowExpanded = false },
        actions = overflowActions,
        onAction = {
            overflowExpanded = false
            ui.onOverflowAction(it)
        },
    )
}

@Composable
private fun SaveToggle(
    isSaved: Boolean,
    onToggle: () -> Unit,
    iconSize: androidx.compose.ui.unit.Dp,
    hitTarget: androidx.compose.ui.unit.Dp,
) {
    CardSaveToggle(
        isSaved = isSaved,
        onToggle = onToggle,
        iconSize = iconSize,
        hitTarget = hitTarget,
        modifier = Modifier,
    )
}

@Composable
private fun MetaLine(
    location: String?,
    trust: String?,
    modifier: Modifier = Modifier,
) {
    val parts = listOfNotNull(
        location?.takeIf { it.isNotBlank() },
        trust?.takeIf { it.isNotBlank() },
    )
    val text = parts.joinToString(" • ")
    Text(
        text = if (text.isBlank()) " " else text,
        style = MaterialTheme.typography.bodySmall,
        color = if (text.isBlank()) Color.Transparent else MaterialTheme.colorScheme.onSurfaceVariant,
        maxLines = 1,
        overflow = TextOverflow.Ellipsis,
        modifier = modifier,
    )
}

@Composable
private fun OfferOverflowMenu(
    expanded: Boolean,
    onDismiss: () -> Unit,
    actions: List<OfferOverflowAction>,
    onAction: (OfferOverflowAction) -> Unit,
) {
    DropdownMenu(
        expanded = expanded,
        onDismissRequest = onDismiss,
    ) {
        actions.forEach { action ->
            DropdownMenuItem(
                text = { Text(offerActionLabel(action)) },
                onClick = { onAction(action) },
            )
        }
    }
}

@Composable
private fun offerActionLabel(action: OfferOverflowAction): String = when (action) {
    OfferOverflowAction.Share -> stringResource(R.string.card_action_share)
    OfferOverflowAction.Hide -> stringResource(R.string.card_action_hide)
    OfferOverflowAction.Report -> stringResource(R.string.card_action_report)
    OfferOverflowAction.CopyLink -> stringResource(R.string.card_action_copy_link)
}
