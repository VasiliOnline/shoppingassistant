@Composable
fun EdgeMenuEditor(
    layout: MenuLayout,
    config: MenuLayoutConfig,
    catalog: List<MenuActionDescriptor>,
    onLayoutChange: (MenuLayout) -> Unit,
    onDone: () -> Unit,
    modifier: Modifier = Modifier,
) {
    val haptics = LocalHapticFeedback.current
    val density = LocalDensity.current
    val context = LocalContext.current
    val scope = rememberCoroutineScope()
    val slotBounds = remember { mutableStateMapOf<MenuSlotRef, Rect>() }
    val snackbarHostState = remember { SnackbarHostState() }
    var deleteBounds by remember { mutableStateOf<Rect?>(null) }
    var dragInfo by remember { mutableStateOf<DragInfo?>(null) }
    var pickerTarget by remember { mutableStateOf<MenuSlotRef?>(null) }
    var pickerVisible by rememberSaveable { mutableStateOf(true) }
    var pendingUndo by remember { mutableStateOf<MenuLayout?>(null) }
    var lastHoverSlot by remember { mutableStateOf<MenuSlotRef?>(null) }

    LaunchedEffect(Unit) {
        haptics.performHapticFeedback(androidx.compose.ui.hapticfeedback.HapticFeedbackType.LongPress)
    }

    fun updateDrag(info: DragInfo) {
        dragInfo = info
        val target = slotBounds.entries.firstOrNull { it.value.contains(info.position) }?.key
        if (target != null && target != lastHoverSlot) {
            haptics.performHapticFeedback(HapticFeedbackType.TextHandleMove)
        }
        lastHoverSlot = target
    }

    Box(modifier = modifier.fillMaxSize()) {
        Box(
            modifier = Modifier
                .fillMaxSize()
                .background(MaterialTheme.colorScheme.onSurface.copy(alpha = 0.12f)),
        )

        Surface(
            shape = MaterialTheme.shapes.large,
            color = MaterialTheme.colorScheme.scrim.copy(alpha = 0.75f),
            shadowElevation = 10.dp,
            modifier = Modifier
                .align(Alignment.TopCenter)
                .padding(horizontal = 16.dp, vertical = 10.dp)
                .fillMaxWidth(),
        ) {
            Column(
                modifier = Modifier.padding(horizontal = 14.dp, vertical = 10.dp),
                verticalArrangement = Arrangement.spacedBy(6.dp),
            ) {
                Row(verticalAlignment = Alignment.CenterVertically) {
                    Text(
                        text = "Редактирование меню",
                        style = MaterialTheme.typography.titleMedium,
                        color = MaterialTheme.colorScheme.inverseOnSurface,
                    )
                    Spacer(modifier = Modifier.weight(1f))
                    TextButton(onClick = onDone) {
                        Text(
                            text = "Готово",
                            color = MaterialTheme.colorScheme.inverseOnSurface,
                        )
                    }
                }
                Text(
                    text = "Нажмите пустой слот, чтобы добавить. Удерживайте и перетащите для перемещения.",
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.inverseOnSurface,
                    maxLines = 2,
                    overflow = TextOverflow.Ellipsis,
                )
            }
        }

        fun removeFromSlot(ref: MenuSlotRef, action: MenuActionId) {
            if (layout.pinned.contains(action)) {
                Toast.makeText(context, "Нельзя убрать закрепленную иконку", Toast.LENGTH_SHORT).show()
                return
            }
            val before = layout
            val outcome = MenuLayoutOps.applyDrop(
                layout = layout,
                config = config,
                actionId = action,
                from = ref,
                to = null,
            )
            onLayoutChange(outcome.layout)
            if (outcome.removed != null) {
                pendingUndo = before
                scope.launch {
                    val result = snackbarHostState.showSnackbar("Иконка удалена", "Отменить")
                    if (result == androidx.compose.material3.SnackbarResult.ActionPerformed) {
                        pendingUndo?.let { onLayoutChange(it) }
                    }
                    pendingUndo = null
                }
            }
        }

        SlotPanel(
            edge = MenuEdge.RIGHT,
            layout = layout,
            slots = config.rightSlots,
            catalog = catalog,
            slotBackgroundAlpha = layout.slotBackgroundAlpha,
            slotBackgroundBlur = layout.slotBackgroundBlur,
            dragInfo = dragInfo,
            selectedSlot = pickerTarget,
            slotBounds = slotBounds,
            onEmptyClick = {
                pickerTarget = it
                pickerVisible = true
            },
            onOccupiedClick = {
                pickerTarget = it
                pickerVisible = true
            },
            onDragStart = {
                dragInfo = it
                lastHoverSlot = null
                haptics.performHapticFeedback(HapticFeedbackType.LongPress)
            },
            onDragUpdate = { updateDrag(it) },
            onDragEnd = {
                dragInfo?.let { info ->
                    applyDragDrop(
                        layout = layout,
                        config = config,
                        dragInfo = info,
                        slotBounds = slotBounds,
                        deleteBounds = deleteBounds,
                        onApply = { outcome, deleted ->
                            val isPinnedBlocked = outcome.layout == layout && layout.pinned.contains(info.actionId)
                            if (isPinnedBlocked) {
                                Toast.makeText(context, "Нельзя убрать закрепленную иконку", Toast.LENGTH_SHORT).show()
                            } else {
                                val before = layout
                                if (outcome.layout != layout) {
                                    haptics.performHapticFeedback(
                                        if (deleted) HapticFeedbackType.LongPress else HapticFeedbackType.TextHandleMove,
                                    )
                                }
                                onLayoutChange(outcome.layout)
                                if (deleted && outcome.removed != null) {
                                    pendingUndo = before
                                    scope.launch {
                                        val result = snackbarHostState.showSnackbar("Иконка удалена", "Отменить")
                                        if (result == androidx.compose.material3.SnackbarResult.ActionPerformed) {
                                            pendingUndo?.let { onLayoutChange(it) }
                                        }
                                        pendingUndo = null
                                    }
                                }
                            }
                        },
                    )
                }
                dragInfo = null
                lastHoverSlot = null
            },
            onRemove = ::removeFromSlot,
            modifier = Modifier
                .align(Alignment.CenterEnd)
                .padding(end = 12.dp)
                .zIndex(1f),
        )

        SlotPanel(
            edge = MenuEdge.BOTTOM,
            layout = layout,
            slots = config.bottomSlots,
            catalog = catalog,
            slotBackgroundAlpha = layout.slotBackgroundAlpha,
            slotBackgroundBlur = layout.slotBackgroundBlur,
            dragInfo = dragInfo,
            selectedSlot = pickerTarget,
            slotBounds = slotBounds,
            onEmptyClick = {
                pickerTarget = it
                pickerVisible = true
            },
            onOccupiedClick = {
                pickerTarget = it
                pickerVisible = true
            },
            onDragStart = {
                dragInfo = it
                lastHoverSlot = null
                haptics.performHapticFeedback(HapticFeedbackType.LongPress)
            },
            onDragUpdate = { updateDrag(it) },
            onDragEnd = {
                dragInfo?.let { info ->
                    applyDragDrop(
                        layout = layout,
                        config = config,
                        dragInfo = info,
                        slotBounds = slotBounds,
                        deleteBounds = deleteBounds,
                        onApply = { outcome, deleted ->
                            val isPinnedBlocked = outcome.layout == layout && layout.pinned.contains(info.actionId)
                            if (isPinnedBlocked) {
                                Toast.makeText(context, "Нельзя убрать закрепленную иконку", Toast.LENGTH_SHORT).show()
                            } else {
                                val before = layout
                                if (outcome.layout != layout) {
                                    haptics.performHapticFeedback(
                                        if (deleted) HapticFeedbackType.LongPress else HapticFeedbackType.TextHandleMove,
                                    )
                                }
                                onLayoutChange(outcome.layout)
                                if (deleted && outcome.removed != null) {
                                    pendingUndo = before
                                    scope.launch {
                                        val result = snackbarHostState.showSnackbar("Иконка удалена", "Отменить")
                                        if (result == androidx.compose.material3.SnackbarResult.ActionPerformed) {
                                            pendingUndo?.let { onLayoutChange(it) }
                                        }
                                        pendingUndo = null
                                    }
                                }
                            }
                        },
                    )
                }
                dragInfo = null
                lastHoverSlot = null
            },
            onRemove = ::removeFromSlot,
            modifier = Modifier
                .align(Alignment.BottomCenter)
                .padding(bottom = 12.dp)
                .zIndex(1f),
        )

        AnimatedVisibility(visible = dragInfo != null) {
            DeleteZone(
                onBounds = { deleteBounds = it },
                modifier = Modifier
                    .align(Alignment.TopCenter)
                    .zIndex(1f),
            )
        }

        dragInfo?.let { info ->
            DragGhost(info = info, catalog = catalog, density = density)
        }

        SnackbarHost(
            hostState = snackbarHostState,
            modifier = Modifier.align(Alignment.BottomCenter),
        )

        if (pickerVisible) {
            MenuActionPalette(
                catalog = catalog,
                layout = layout,
                selectedSlot = pickerTarget,
                onSelect = { action ->
                    val target = pickerTarget ?: findFirstEmptySlot(layout, config)
                    if (target == null) {
                        Toast.makeText(context, "Выберите слот для замены", Toast.LENGTH_SHORT).show()
                        return@MenuActionPalette
                    }
                    val targetAction = actionAt(layout, target)
                    if (targetAction != null && layout.pinned.contains(targetAction) && targetAction != action) {
                        Toast.makeText(context, "Нельзя заменить закрепленную иконку", Toast.LENGTH_SHORT).show()
                        return@MenuActionPalette
                    }
                    val before = layout
                    val from = findSlot(layout, action)
                    val outcome = MenuLayoutOps.applyDrop(
                        layout = layout,
                        config = config,
                        actionId = action,
                        from = from,
                        to = target,
                    )
                    onLayoutChange(outcome.layout)
                    pickerTarget = null
                    pickerVisible = false
                    if (outcome.removed != null && outcome.removed != action) {
                        pendingUndo = before
                        scope.launch {
                            val result = snackbarHostState.showSnackbar("Иконка удалена", "Отменить")
                            if (result == androidx.compose.material3.SnackbarResult.ActionPerformed) {
                                pendingUndo?.let { onLayoutChange(it) }
                            }
                            pendingUndo = null
                        }
                    }
                },
                onStyleChange = { alpha, blur ->
                    onLayoutChange(layout.copy(slotBackgroundAlpha = alpha, slotBackgroundBlur = blur))
                },
                onDismiss = {
                    pickerVisible = false
                    pickerTarget = null
                },
                modifier = Modifier
                    .align(Alignment.Center)
                    .padding(horizontal = 20.dp)
                    .zIndex(2f),
            )
        }

    }
}

@Composable
private fun SlotPanel(
    edge: MenuEdge,
    layout: MenuLayout,
    slots: Int,
    catalog: List<MenuActionDescriptor>,
    slotBackgroundAlpha: Float,
    slotBackgroundBlur: Float,
    dragInfo: DragInfo?,
    selectedSlot: MenuSlotRef?,
    slotBounds: MutableMap<MenuSlotRef, Rect>,
    onEmptyClick: (MenuSlotRef) -> Unit,
    onOccupiedClick: (MenuSlotRef) -> Unit,
    onDragStart: (DragInfo) -> Unit,
    onDragUpdate: (DragInfo) -> Unit,
    onDragEnd: () -> Unit,
    onRemove: (MenuSlotRef, MenuActionId) -> Unit,
    modifier: Modifier = Modifier,
) {
    val arrangement = if (edge == MenuEdge.RIGHT) Arrangement.spacedBy(10.dp) else Arrangement.spacedBy(12.dp)
    val panelAlpha = slotBackgroundAlpha.coerceIn(0f, 1f)
    val panelColor = MaterialTheme.colorScheme.surfaceVariant.copy(alpha = panelAlpha)
    val panelBorder = MaterialTheme.colorScheme.outline.copy(alpha = 0.4f)
    val panelShape = MaterialTheme.shapes.large
    val panelModifier = modifier.wrapContentSize()

    if (edge == MenuEdge.RIGHT) {
        Box(
            modifier = panelModifier
                .border(1.dp, panelBorder, panelShape),
        ) {
            if (panelAlpha > 0f || slotBackgroundBlur > 0f) {
                Box(
                    modifier = Modifier
                        .matchParentSize()
                        .blur(slotBackgroundBlur.dp)
                        .background(panelColor, panelShape),
                )
            }
            Column(
                modifier = Modifier
                    .padding(8.dp)
                    .width(IntrinsicSize.Min),
                verticalArrangement = arrangement,
            ) {
                for (index in 0 until slots) {
                    val action = layout.right.getOrNull(index)
                    val descriptor = catalog.firstOrNull { it.id == action }
                    SlotItem(
                        ref = MenuSlotRef(edge, index),
                        action = action,
                        descriptor = descriptor,
                        slotBackgroundAlpha = slotBackgroundAlpha,
                        slotBackgroundBlur = slotBackgroundBlur,
                        dragInfo = dragInfo,
                        isTarget = selectedSlot == MenuSlotRef(edge, index),
                        slotBounds = slotBounds,
                        canRemove = action != null && !layout.pinned.contains(action),
                        onRemove = onRemove,
                        onEmptyClick = onEmptyClick,
                        onOccupiedClick = onOccupiedClick,
                        onDragStart = onDragStart,
                        onDragUpdate = onDragUpdate,
                        onDragEnd = onDragEnd,
                    )
                }
            }
        }
    } else {
        Box(
            modifier = panelModifier
                .border(1.dp, panelBorder, panelShape),
        ) {
            if (panelAlpha > 0f || slotBackgroundBlur > 0f) {
                Box(
                    modifier = Modifier
                        .matchParentSize()
                        .blur(slotBackgroundBlur.dp)
                        .background(panelColor, panelShape),
                )
            }
            Row(
                modifier = Modifier
                    .padding(8.dp)
                    .width(IntrinsicSize.Min),
                horizontalArrangement = arrangement,
            ) {
                for (index in 0 until slots) {
                    val action = layout.bottom.getOrNull(index)
                    val descriptor = catalog.firstOrNull { it.id == action }
                    SlotItem(
                        ref = MenuSlotRef(edge, index),
                        action = action,
                        descriptor = descriptor,
                        slotBackgroundAlpha = slotBackgroundAlpha,
                        slotBackgroundBlur = slotBackgroundBlur,
                        dragInfo = dragInfo,
                        isTarget = selectedSlot == MenuSlotRef(edge, index),
                        slotBounds = slotBounds,
                        canRemove = action != null && !layout.pinned.contains(action),
                        onRemove = onRemove,
                        onEmptyClick = onEmptyClick,
                        onOccupiedClick = onOccupiedClick,
                        onDragStart = onDragStart,
                        onDragUpdate = onDragUpdate,
                        onDragEnd = onDragEnd,
                    )
                }
            }
        }
    }
}

private data class DragInfo(
    val actionId: MenuActionId,
    val from: MenuSlotRef,
    val position: Offset = Offset.Zero,
)

@Composable
private fun SlotItem(
    ref: MenuSlotRef,
    action: MenuActionId?,
    descriptor: MenuActionDescriptor?,
    slotBackgroundAlpha: Float,
    slotBackgroundBlur: Float,
    dragInfo: DragInfo?,
    isTarget: Boolean,
    slotBounds: MutableMap<MenuSlotRef, Rect>,
    canRemove: Boolean,
    onRemove: (MenuSlotRef, MenuActionId) -> Unit,
    onEmptyClick: (MenuSlotRef) -> Unit,
    onOccupiedClick: (MenuSlotRef) -> Unit,
    onDragStart: (DragInfo) -> Unit,
    onDragUpdate: (DragInfo) -> Unit,
    onDragEnd: () -> Unit,
) {
    val interaction = remember { MutableInteractionSource() }
    val shape = MaterialTheme.shapes.large
    val bounds = slotBounds[ref]
    val isHover = dragInfo?.let { info -> bounds?.contains(info.position) == true } == true
    val borderColor = when {
        isHover -> MaterialTheme.colorScheme.primary
        isTarget -> MaterialTheme.colorScheme.primary.copy(alpha = 0.7f)
        action == null -> MaterialTheme.colorScheme.outlineVariant
        else -> MaterialTheme.colorScheme.outline
    }
    val baseAlpha = slotBackgroundAlpha.coerceIn(0f, 1f)
    val hoverBoost = if (isHover || isTarget) 0.18f else 0f
    val slotSurface = if (action == null) {
        MaterialTheme.colorScheme.surfaceVariant
    } else {
        MaterialTheme.colorScheme.surface
    }
    val baseSurface = slotSurface.copy(alpha = (baseAlpha + hoverBoost).coerceIn(0f, 1f))
    val backgroundColor = slotSurface.copy(alpha = baseAlpha)

    Box(
        modifier = Modifier
            .size(56.dp)
            .background(baseSurface, shape)
            .border(width = if (isHover || isTarget) 2.dp else 1.dp, color = borderColor, shape = shape)
            .onGloballyPositioned { coordinates -> slotBounds[ref] = coordinates.boundsInWindow() }
            .pointerInput(action, descriptor) {
                if (action == null || descriptor == null) return@pointerInput
                detectDragGesturesAfterLongPress(
                    onDragStart = { offset ->
                        val itemBounds = slotBounds[ref]
                        if (itemBounds == null) {
                            onDragStart(DragInfo(action, ref, offset))
                        } else {
                            onDragStart(DragInfo(action, ref, itemBounds.topLeft + offset))
                        }
                    },
                    onDrag = { change, _ ->
                        val itemBounds = slotBounds[ref] ?: return@detectDragGesturesAfterLongPress
                        onDragUpdate(DragInfo(action, ref, itemBounds.topLeft + change.position))
                    },
                    onDragEnd = { onDragEnd() },
                    onDragCancel = { onDragEnd() },
                )
            }
            .clickable(
                interactionSource = interaction,
                indication = null,
            ) {
                if (action == null) {
                    onEmptyClick(ref)
                } else {
                    onOccupiedClick(ref)
                }
            }
            .padding(10.dp),
        contentAlignment = Alignment.Center,
    ) {
        if (slotBackgroundBlur > 0f && baseAlpha > 0f) {
            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .blur(slotBackgroundBlur.dp)
                    .background(backgroundColor, shape),
            )
        }
        if (action == null || descriptor == null) {
            Icon(
                imageVector = Icons.Outlined.Add,
                contentDescription = "Пустой слот",
                tint = MaterialTheme.colorScheme.onSurface,
                modifier = Modifier.size(22.dp),
            )
        } else {
            Box(modifier = Modifier.fillMaxSize()) {
                Icon(
                    imageVector = descriptor.icon,
                    contentDescription = descriptor.title,
                    tint = MaterialTheme.colorScheme.onSurface,
                    modifier = Modifier
                        .align(Alignment.Center)
                        .size(22.dp),
                )
                if (canRemove) {
                    IconButton(
                        onClick = { onRemove(ref, action) },
                        modifier = Modifier
                            .align(Alignment.TopEnd)
                            .size(22.dp),
                    ) {
                        Icon(
                            imageVector = Icons.Outlined.Close,
                            contentDescription = "Удалить",
                            tint = MaterialTheme.colorScheme.onSurfaceVariant,
                            modifier = Modifier.size(14.dp),
                        )
                    }
                }
            }
        }
    }
}

@Composable
private fun DragGhost(
    info: DragInfo,
    catalog: List<MenuActionDescriptor>,
    density: Density,
) {
    val descriptor = catalog.firstOrNull { it.id == info.actionId } ?: return
    val x = with(density) { info.position.x.toDp() }
    val y = with(density) { info.position.y.toDp() }
    Box(
        modifier = Modifier
            .fillMaxSize()
            .padding(0.dp)
            .zIndex(3f),
    ) {
        Surface(
            shape = MaterialTheme.shapes.medium,
            color = MaterialTheme.colorScheme.surfaceVariant,
            shadowElevation = 6.dp,
            modifier = Modifier
                .align(Alignment.TopStart)
                .padding(start = x, top = y),
        ) {
            Icon(
                imageVector = descriptor.icon,
                contentDescription = descriptor.title,
                tint = MaterialTheme.colorScheme.onSurface,
                modifier = Modifier
                    .padding(8.dp)
                    .size(20.dp),
            )
        }
    }
}

@Composable
private fun DeleteZone(
    onBounds: (Rect) -> Unit,
    modifier: Modifier = Modifier,
) {
    Surface(
        shape = MaterialTheme.shapes.medium,
        color = MaterialTheme.colorScheme.errorContainer,
        modifier = modifier
            .padding(top = 10.dp)
            .onGloballyPositioned { coordinates -> onBounds(coordinates.boundsInWindow()) },
    ) {
        Row(
            modifier = Modifier
                .padding(horizontal = 16.dp, vertical = 8.dp),
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.spacedBy(8.dp),
        ) {
            Icon(
                imageVector = Icons.Outlined.Delete,
                contentDescription = null,
                tint = MaterialTheme.colorScheme.onErrorContainer,
            )
            Text(
                text = "Удалить",
                style = MaterialTheme.typography.bodyMedium,
                color = MaterialTheme.colorScheme.onErrorContainer,
            )
        }
    }
}

@Composable
private fun MenuActionPalette(
    catalog: List<MenuActionDescriptor>,
    layout: MenuLayout,
    selectedSlot: MenuSlotRef?,
    onSelect: (MenuActionId) -> Unit,
    onStyleChange: (Float, Float) -> Unit,
    onDismiss: () -> Unit,
    modifier: Modifier = Modifier,
) {
    var settingsExpanded by remember { mutableStateOf(false) }
    var alpha by remember(layout) { mutableStateOf(layout.slotBackgroundAlpha) }
    var blur by remember(layout) { mutableStateOf(layout.slotBackgroundBlur) }

    fun updateStyle(nextAlpha: Float = alpha, nextBlur: Float = blur) {
        alpha = nextAlpha
        blur = nextBlur
        onStyleChange(nextAlpha, nextBlur)
    }

    val availableItems = remember(catalog, layout) {
        catalog.filterNot { isUsed(layout, it.id) }
    }

    Surface(
        shape = MaterialTheme.shapes.extraLarge,
        color = MaterialTheme.colorScheme.surface,
        tonalElevation = 4.dp,
        shadowElevation = 6.dp,
        modifier = modifier.sizeIn(maxWidth = 360.dp, maxHeight = 520.dp),
    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 16.dp, vertical = 12.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp),
        ) {
            Row(verticalAlignment = Alignment.CenterVertically) {
                Column(modifier = Modifier.weight(1f)) {
                    Text(
                        text = "Доступные действия",
                        style = MaterialTheme.typography.titleMedium,
                    )
                    selectedSlot?.let { slot ->
                        Text(
                            text = if (slot.edge == MenuEdge.RIGHT) {
                                "Слот справа - ${slot.index + 1}"
                            } else {
                                "Слот снизу - ${slot.index + 1}"
                            },
                            style = MaterialTheme.typography.bodySmall,
                            color = MaterialTheme.colorScheme.onSurfaceVariant,
                            maxLines = 1,
                            overflow = TextOverflow.Ellipsis,
                        )
                    }
                }
                IconButton(onClick = { settingsExpanded = !settingsExpanded }) {
                    Icon(
                        imageVector = Icons.Outlined.MoreVert,
                        contentDescription = "Настройки отображения",
                    )
                }
                IconButton(onClick = onDismiss) {
                    Icon(
                        imageVector = Icons.Outlined.Close,
                        contentDescription = "Закрыть",
                    )
                }
            }

            AnimatedVisibility(visible = settingsExpanded) {
                Column(verticalArrangement = Arrangement.spacedBy(8.dp)) {
                    Text(
                        text = "Фон линий",
                        style = MaterialTheme.typography.bodyMedium,
                    )
                    Slider(
                        value = alpha,
                        onValueChange = { updateStyle(nextAlpha = it) },
                        valueRange = 0f..0.4f,
                    )
                    Row(verticalAlignment = Alignment.CenterVertically) {
                        Checkbox(
                            checked = blur > 0f,
                            onCheckedChange = { checked ->
                                val next = if (checked) {
                                    if (blur <= 0f) 12f else blur
                                } else {
                                    0f
                                }
                                updateStyle(nextBlur = next)
                            },
                            colors = CheckboxDefaults.colors(
                                checkedColor = MaterialTheme.colorScheme.primary,
                                uncheckedColor = MaterialTheme.colorScheme.outline,
                            ),
                        )
                        Text(
                            text = "Блюр фона",
                            style = MaterialTheme.typography.bodyMedium,
                        )
                    }
                    if (blur > 0f) {
                        Slider(
                            value = blur,
                            onValueChange = { updateStyle(nextBlur = it) },
                            valueRange = 0f..24f,
                        )
                    }
                }
            }

            LazyColumn(verticalArrangement = Arrangement.spacedBy(10.dp)) {
                items(availableItems) { item ->
                    val used = isUsed(layout, item.id)
                    val enabled = item.isAvailable
                    Surface(
                        shape = MaterialTheme.shapes.medium,
                        color = MaterialTheme.colorScheme.surfaceVariant,
                        modifier = Modifier
                            .fillMaxWidth()
                            .clickable(enabled = enabled) { onSelect(item.id) },
                    ) {
                        Row(
                            modifier = Modifier
                                .fillMaxWidth()
                                .padding(horizontal = 14.dp, vertical = 12.dp),
                            verticalAlignment = Alignment.CenterVertically,
                            horizontalArrangement = Arrangement.spacedBy(12.dp),
                        ) {
                            Icon(
                                imageVector = item.icon,
                                contentDescription = item.title,
                                tint = if (enabled) MaterialTheme.colorScheme.onSurface else MaterialTheme.colorScheme.onSurfaceVariant,
                            )
                            Column(modifier = Modifier.weight(1f)) {
                                Text(
                                    text = item.title,
                                    style = MaterialTheme.typography.bodyLarge,
                                    color = if (enabled) MaterialTheme.colorScheme.onSurface else MaterialTheme.colorScheme.onSurfaceVariant,
                                    maxLines = 1,
                                    overflow = TextOverflow.Ellipsis,
                                )
                                when {
                                    !enabled && item.unavailableReason != null -> Text(
                                        text = item.unavailableReason,
                                        style = MaterialTheme.typography.bodySmall,
                                        color = MaterialTheme.colorScheme.onSurfaceVariant,
                                    )
                                    used -> Text(
                                        text = "Уже на панели",
                                        style = MaterialTheme.typography.bodySmall,
                                        color = MaterialTheme.colorScheme.onSurfaceVariant,
                                    )
                                }
                            }
                        }
                    }
                }
            }
            if (availableItems.isEmpty()) {
                Text(
                    text = "Все действия уже размещены",
                    style = MaterialTheme.typography.bodyMedium,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                )
            }
        }
    }
}

private fun applyDragDrop(
    layout: MenuLayout,
    config: MenuLayoutConfig,
    dragInfo: DragInfo,
    slotBounds: Map<MenuSlotRef, Rect>,
    deleteBounds: Rect?,
    onApply: (MenuDropOutcome, Boolean) -> Unit,
) {
    val target = slotBounds.entries.firstOrNull { it.value.contains(dragInfo.position) }?.key
    val shouldDelete = deleteBounds?.contains(dragInfo.position) == true || target == null
    val outcome = MenuLayoutOps.applyDrop(
        layout = layout,
        config = config,
        actionId = dragInfo.actionId,
        from = dragInfo.from,
        to = if (shouldDelete) null else target,
    )
    onApply(outcome, shouldDelete)
}


private fun actionAt(layout: MenuLayout, ref: MenuSlotRef): MenuActionId? {
    return when (ref.edge) {
        MenuEdge.RIGHT -> layout.right.getOrNull(ref.index)
        MenuEdge.BOTTOM -> layout.bottom.getOrNull(ref.index)
    }
}

private fun findFirstEmptySlot(layout: MenuLayout, config: MenuLayoutConfig): MenuSlotRef? {
    val rightIndex = layout.right.indexOfFirst { it == null }
    if (rightIndex >= 0 && rightIndex < config.rightSlots) {
        return MenuSlotRef(MenuEdge.RIGHT, rightIndex)
    }
    val bottomIndex = layout.bottom.indexOfFirst { it == null }
    if (bottomIndex >= 0 && bottomIndex < config.bottomSlots) {
        return MenuSlotRef(MenuEdge.BOTTOM, bottomIndex)
    }
    return null
}

private fun findSlot(layout: MenuLayout, actionId: MenuActionId): MenuSlotRef? {
    val rightIndex = layout.right.indexOf(actionId)
    if (rightIndex >= 0) return MenuSlotRef(MenuEdge.RIGHT, rightIndex)
    val bottomIndex = layout.bottom.indexOf(actionId)
    if (bottomIndex >= 0) return MenuSlotRef(MenuEdge.BOTTOM, bottomIndex)
    return null
}

private fun isUsed(layout: MenuLayout, actionId: MenuActionId): Boolean {
    return layout.right.contains(actionId) || layout.bottom.contains(actionId)
}
