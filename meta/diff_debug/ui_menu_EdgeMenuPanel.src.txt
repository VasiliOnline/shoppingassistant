@Composable
fun EdgeActivationStrips(
    onEnterEdit: () -> Unit,
    modifier: Modifier = Modifier,
) {
    Box(modifier = modifier.fillMaxSize()) {
        Box(
            modifier = Modifier
                .align(Alignment.CenterEnd)
                .width(10.dp)
                .fillMaxHeight()
                .alpha(0.06f)
                .background(MaterialTheme.colorScheme.onSurfaceVariant)
                .pointerInput(Unit) {
                    detectTapGestures(onLongPress = { onEnterEdit() })
                }
                .semantics { contentDescription = "Правый край: редактировать меню" },
        )
        Box(
            modifier = Modifier
                .align(Alignment.BottomCenter)
                .height(10.dp)
                .fillMaxWidth()
                .alpha(0.06f)
                .background(MaterialTheme.colorScheme.onSurfaceVariant)
                .pointerInput(Unit) {
                    detectTapGestures(onLongPress = { onEnterEdit() })
                }
                .semantics { contentDescription = "Нижний край: редактировать меню" },
        )
    }
}

@Composable
fun EdgeMenuPanel(
    layout: MenuLayout,
    config: MenuLayoutConfig,
    catalog: List<MenuActionDescriptor>,
    badges: Map<MenuActionId, Int> = emptyMap(),
    onActionClick: (MenuActionId) -> Unit,
    onEnterEdit: (() -> Unit)? = null,
    modifier: Modifier = Modifier,
    alpha: Float = 1f,
) {
    Box(modifier = modifier.fillMaxSize()) {
        Column(
            modifier = Modifier
                .align(Alignment.CenterEnd)
                .padding(end = 10.dp)
                .alpha(alpha),
            verticalArrangement = Arrangement.spacedBy(8.dp),
        ) {
            (0 until config.rightSlots).forEach { index ->
                val action = layout.right.getOrNull(index) ?: return@forEach
                val descriptor = catalog.firstOrNull { it.id == action } ?: return@forEach
                EdgeMenuItem(
                    action = descriptor,
                    enabled = descriptor.isAvailable,
                    slotBackgroundAlpha = layout.slotBackgroundAlpha,
                    slotBackgroundBlur = layout.slotBackgroundBlur,
                    badgeCount = badges[action] ?: 0,
                    onClick = { onActionClick(action) },
                    onLongClick = onEnterEdit,
                )
            }
        }

        Row(
            modifier = Modifier
                .align(Alignment.BottomCenter)
                .padding(bottom = 10.dp)
                .alpha(alpha),
            horizontalArrangement = Arrangement.spacedBy(8.dp),
        ) {
            (0 until config.bottomSlots).forEach { index ->
                val action = layout.bottom.getOrNull(index) ?: return@forEach
                val descriptor = catalog.firstOrNull { it.id == action } ?: return@forEach
                EdgeMenuItem(
                    action = descriptor,
                    enabled = descriptor.isAvailable,
                    slotBackgroundAlpha = layout.slotBackgroundAlpha,
                    slotBackgroundBlur = layout.slotBackgroundBlur,
                    badgeCount = badges[action] ?: 0,
                    onClick = { onActionClick(action) },
                    onLongClick = onEnterEdit,
                )
            }
        }
    }
}

@Composable
private fun EdgeMenuItem(
    action: MenuActionDescriptor,
    enabled: Boolean,
    slotBackgroundAlpha: Float,
    slotBackgroundBlur: Float,
    badgeCount: Int,
    onClick: () -> Unit,
    onLongClick: (() -> Unit)?,
) {
    val shape = RoundedCornerShape(18.dp)
    val backgroundAlpha = slotBackgroundAlpha.coerceIn(0f, 1f)
    val backgroundColor = MaterialTheme.colorScheme.surface.copy(alpha = backgroundAlpha)
    Surface(
        shape = shape,
        color = MaterialTheme.colorScheme.surface.copy(alpha = 0f),
        modifier = Modifier
            .size(52.dp)
            .alpha(if (enabled) 1f else 0.5f)
            .combinedClickable(
                enabled = enabled,
                onClick = onClick,
                onLongClick = { onLongClick?.invoke() },
            ),
    ) {
        Box(contentAlignment = Alignment.Center) {
            if (backgroundAlpha > 0f || slotBackgroundBlur > 0f) {
                Box(
                    modifier = Modifier
                        .fillMaxSize()
                        .blur(slotBackgroundBlur.dp)
                        .background(backgroundColor, shape),
                )
            }
            Icon(
                imageVector = action.icon,
                contentDescription = action.title,
                tint = MaterialTheme.colorScheme.onSurface,
                modifier = Modifier.size(22.dp),
            )
            if (badgeCount > 0) {
                val label = if (badgeCount > 99) "99+" else badgeCount.toString()
                Surface(
                    color = MaterialTheme.colorScheme.primary,
                    shape = RoundedCornerShape(10.dp),
                    modifier = Modifier
                        .align(Alignment.TopEnd)
                        .padding(top = 4.dp, end = 4.dp),
                ) {
                    Text(
                        text = label,
                        style = MaterialTheme.typography.labelSmall,
                        color = MaterialTheme.colorScheme.onPrimary,
                        modifier = Modifier.padding(horizontal = 4.dp, vertical = 1.dp),
                    )
                }
            }
        }
    }
}
