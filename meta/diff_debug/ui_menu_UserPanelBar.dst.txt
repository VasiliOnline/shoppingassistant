private val RailExpandedWidth = 56.dp

@Composable
fun UserPanelBar(
    panel: UserPanel,
    modeCatalog: List<PanelItemDescriptor<ModeKey>>,
    actionCatalog: List<PanelItemDescriptor<ActionKey>>,
    currentMode: ModeKey,
    badges: Map<ActionKey, Int> = emptyMap(),
    onModeClick: (ModeKey) -> Unit,
    onActionClick: (ActionKey) -> Unit,
    onEnterEdit: () -> Unit,
    modifier: Modifier = Modifier,
    alpha: Float = 1f,
) {
    var railExpanded by rememberSaveable { mutableStateOf(false) }
    val railAlignment = if (panel.handedness == Handedness.LEFT) {
        Alignment.CenterStart
    } else {
        Alignment.CenterEnd
    }
    val railPadding = if (panel.handedness == Handedness.LEFT) PaddingValues(start = 8.dp) else PaddingValues(end = 8.dp)
    val railButtonAlignment = if (panel.handedness == Handedness.LEFT) Alignment.BottomStart else Alignment.BottomEnd
    val railButtonPadding = if (panel.handedness == Handedness.LEFT) PaddingValues(start = 12.dp) else PaddingValues(end = 12.dp)
    val panelAlpha = alpha.coerceIn(0f, 1f)

    val primarySlot = panel.sideActions.firstOrNull { it.enabled && it.key != null }
    val primaryKey = primarySlot?.key
    val primaryDescriptor = actionCatalog.firstOrNull { it.key == primaryKey }

    Box(modifier = modifier.fillMaxSize()) {
        if (railExpanded) {
            // Scrim area to collapse the rail when tapping outside.
            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .padding(bottom = LayoutDefaults.BottomNavHeight)
                    .combinedClickable(
                        onClick = { railExpanded = false },
                        onLongClick = {},
                    ),
            )

            val items = panel.sideActions.mapNotNull { slot ->
                val key = slot.key ?: return@mapNotNull null
                val descriptor = actionCatalog.firstOrNull { it.key == key } ?: return@mapNotNull null
                Triple(slot, key, descriptor)
            }

            Surface(
                modifier = Modifier
                    .align(railAlignment)
                    .padding(railPadding)
                    .fillMaxHeight()
                    .width(RailExpandedWidth)
                    .alpha(panelAlpha),
                tonalElevation = 4.dp,
                color = MaterialTheme.colorScheme.surfaceVariant,
                shape = MaterialTheme.shapes.large,
            ) {
                Column(
                    modifier = Modifier
                        .fillMaxHeight()
                        .padding(vertical = 12.dp),
                    verticalArrangement = Arrangement.spacedBy(8.dp),
                    horizontalAlignment = Alignment.CenterHorizontally,
                ) {
                    // Keep actions away from the status bar and close to the bottom corner.
                    Spacer(modifier = Modifier.weight(1f))
                    items.forEach { (slot, key, descriptor) ->
                        val isEnabled = slot.enabled && descriptor.isAvailable
                        RailActionItem(
                            descriptor = descriptor,
                            enabled = isEnabled,
                            badgeCount = badges[key] ?: 0,
                            onClick = {
                                if (isEnabled) {
                                    onActionClick(key)
                                    railExpanded = false
                                }
                            },
                            onLongPress = {},
                        )
                    }
                }
            }
        } else if (primaryDescriptor != null) {
            CollapsedRailButton(
                descriptor = primaryDescriptor,
                enabled = primarySlot?.enabled == true && primaryDescriptor.isAvailable,
                onClick = { railExpanded = true },
                modifier = Modifier
                    .align(railButtonAlignment)
                    .padding(railButtonPadding)
                    .padding(bottom = LayoutDefaults.BottomNavHeight + 12.dp)
                    .alpha(panelAlpha),
            )
        }

        NavigationBar(
            modifier = Modifier
                .align(Alignment.BottomCenter)
                .height(LayoutDefaults.BottomNavHeight)
                .alpha(panelAlpha),
            tonalElevation = 3.dp,
            containerColor = MaterialTheme.colorScheme.surface,
            windowInsets = WindowInsets(0, 0, 0, 0),
        ) {
            panel.bottomModes.forEach { slot ->
                val key = slot.key ?: return@forEach
                val descriptor = modeCatalog.firstOrNull { it.key == key } ?: return@forEach
                val enabled = slot.enabled && descriptor.isAvailable
                NavigationBarItem(
                    selected = key == currentMode,
                    onClick = {
                        if (enabled) {
                            railExpanded = false
                            onModeClick(key)
                        }
                    },
                    icon = {
                        Icon(
                            imageVector = descriptor.icon,
                            contentDescription = descriptor.title,
                            modifier = Modifier.size(24.dp),
                        )
                    },
                    label = {
                        Text(
                            text = descriptor.title,
                            fontSize = 11.sp,
                            maxLines = 1,
                            overflow = TextOverflow.Ellipsis,
                        )
                    },
                    alwaysShowLabel = true,
                    enabled = enabled,
                )
            }
        }
    }
}

@Composable
private fun CollapsedRailButton(
    descriptor: PanelItemDescriptor<ActionKey>,
    enabled: Boolean,
    onClick: () -> Unit,
    modifier: Modifier = Modifier,
) {
    Surface(
        modifier = modifier
            .size(48.dp)
            .combinedClickable(
                onClick = onClick,
                onLongClick = {},
                enabled = enabled,
            ),
        tonalElevation = 4.dp,
        color = MaterialTheme.colorScheme.surfaceVariant,
        shape = MaterialTheme.shapes.large,
    ) {
        Box(modifier = Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
            Icon(
                imageVector = descriptor.icon,
                contentDescription = descriptor.title,
                tint = if (enabled) MaterialTheme.colorScheme.onSurface else MaterialTheme.colorScheme.onSurfaceVariant,
                modifier = Modifier.size(24.dp),
            )
        }
    }
}

@Composable
private fun RailActionItem(
    descriptor: PanelItemDescriptor<ActionKey>,
    enabled: Boolean,
    badgeCount: Int,
    onClick: () -> Unit,
    onLongPress: () -> Unit,
) {
    val badgeLabel = when {
        badgeCount <= 0 -> null
        badgeCount > 99 -> "99+"
        else -> badgeCount.toString()
    }
    Box(
        modifier = Modifier
            .size(48.dp)
            .combinedClickable(
                onClick = onClick,
                onLongClick = onLongPress,
                enabled = enabled,
            ),
        contentAlignment = Alignment.Center,
    ) {
        if (badgeLabel != null) {
            BadgedBox(
                badge = {
                    Badge {
                        Text(text = badgeLabel, fontSize = 10.sp, maxLines = 1)
                    }
                },
            ) {
                Icon(
                    imageVector = descriptor.icon,
                    contentDescription = descriptor.title,
                    tint = if (enabled) MaterialTheme.colorScheme.onSurface else MaterialTheme.colorScheme.onSurfaceVariant,
                    modifier = Modifier.size(24.dp),
                )
            }
        } else {
            Icon(
                imageVector = descriptor.icon,
                contentDescription = descriptor.title,
                tint = if (enabled) MaterialTheme.colorScheme.onSurface else MaterialTheme.colorScheme.onSurfaceVariant,
                modifier = Modifier.size(24.dp),
            )
        }
    }
}
