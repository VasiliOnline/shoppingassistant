@Composable
fun StateHost(
    state: ScreenState,
    contentPadding: PaddingValues,
    modifier: Modifier = Modifier,
    screenName: String? = null,
    showAppendOverlay: Boolean = true,
    loadingContent: (@Composable (PaddingValues) -> Unit)? = null,
    content: @Composable (PaddingValues) -> Unit,
) {
    val reduceMotion = rememberReduceMotionEnabled()
    val transitionDuration = if (reduceMotion) 80 else 180
    val bannerDuration = if (reduceMotion) 80 else 200
    val initialDelayMs = 120L
    val minInitialMs = 400L
    val targetState by rememberUpdatedState(state)
    var renderState by remember { mutableStateOf(state) }
    var initialShownAt by remember { mutableStateOf<Long?>(null) }
    val metricsSampled = remember(screenName) {
        screenName != null && FlowMetrics.shouldSampleState()
    }
    var lastStateKey by remember { mutableStateOf<String?>(null) }
    var lastStateStartMs by remember { mutableStateOf<Long?>(null) }

    LaunchedEffect(state) {
        if (state is ScreenState.Loading && state.phase == LoadingPhase.INITIAL) {
            if (renderState is ScreenState.Loading && (renderState as ScreenState.Loading).phase == LoadingPhase.INITIAL) {
                return@LaunchedEffect
            }
            initialShownAt = null
            val startedAt = SystemClock.elapsedRealtime()
            if (initialDelayMs > 0) {
                kotlinx.coroutines.delay(initialDelayMs)
            }
            if (targetState is ScreenState.Loading &&
                (targetState as ScreenState.Loading).phase == LoadingPhase.INITIAL
            ) {
                renderState = state
                initialShownAt = SystemClock.elapsedRealtime().coerceAtLeast(startedAt)
            }
        } else {
            val shownAt = initialShownAt
            if (renderState is ScreenState.Loading &&
                (renderState as ScreenState.Loading).phase == LoadingPhase.INITIAL &&
                shownAt != null
            ) {
                val elapsed = SystemClock.elapsedRealtime() - shownAt
                val remaining = minInitialMs - elapsed
                if (remaining > 0) {
                    kotlinx.coroutines.delay(remaining)
                }
            }
            renderState = state
            initialShownAt = null
        }
    }

    LaunchedEffect(renderState) {
        if (screenName.isNullOrBlank()) return@LaunchedEffect
        val key = buildStateKey(renderState)
        val now = SystemClock.elapsedRealtime()
        if (lastStateKey != null && lastStateStartMs != null && lastStateKey != key) {
            FlowMetrics.markStateDuration(
                sampled = metricsSampled,
                screen = screenName,
                type = lastStateKey ?: "",
                durationMs = now - (lastStateStartMs ?: now),
            )
        }
        if (lastStateKey != key) {
            val cacheAge = resolveCacheAge(renderState)
            FlowMetrics.markStateShown(
                sampled = metricsSampled,
                screen = screenName,
                type = stateType(renderState),
                subType = stateSubType(renderState),
                reason = stateReason(renderState),
                wasCacheShown = cacheAge != null,
                cacheAgeMs = cacheAge,
            )
            lastStateKey = key
            lastStateStartMs = now
        }
    }

    Box(modifier = modifier.fillMaxSize()) {
        AnimatedContent(
            targetState = renderState,
            transitionSpec = {
                val skip = isOverlayState(initialState) && isOverlayState(targetState)
                val spec = tween<Float>(durationMillis = if (skip) 0 else transitionDuration)
                fadeIn(spec) togetherWith fadeOut(spec)
            },
            label = "state-host",
        ) { target ->
            Box(modifier = Modifier.fillMaxSize()) {
                when (target) {
                    is ScreenState.Loading -> when (target.phase) {
                        LoadingPhase.INITIAL -> {
                            if (loadingContent != null) {
                                loadingContent(contentPadding)
                            } else {
                                LoadingSkeleton(contentPadding = contentPadding)
                            }
                        }
                        LoadingPhase.REFRESHING -> {
                            content(contentPadding)
                            RefreshingIndicator(
                                modifier = Modifier
                                    .align(Alignment.TopCenter)
                                    .padding(top = 8.dp),
                            )
                        }
                        LoadingPhase.PAGING,
                        LoadingPhase.APPEND,
                        -> {
                            content(contentPadding)
                            if (showAppendOverlay) {
                                PagingLoadingItem(
                                    modifier = Modifier
                                        .align(Alignment.BottomCenter)
                                        .padding(bottom = contentPadding.calculateBottomPadding() + 8.dp),
                                )
                            }
                        }
                    }
                    is ScreenState.Content -> {
                        content(contentPadding)
                        renderStaleIndicator(
                            target = target,
                            modifier = Modifier
                                .align(Alignment.TopCenter)
                                .padding(top = 8.dp),
                            reduceMotion = reduceMotion,
                            bannerDuration = bannerDuration,
                            screenName = screenName,
                        )
                    }
                    is ScreenState.Stale -> {
                        content(contentPadding)
                        renderStaleIndicator(
                            target = target,
                            modifier = Modifier
                                .align(Alignment.TopCenter)
                                .padding(top = 8.dp),
                            reduceMotion = reduceMotion,
                            bannerDuration = bannerDuration,
                            screenName = screenName,
                        )
                    }
                    is ScreenState.Empty -> {
                        val (title, message) = resolveEmptyStrings(target)
                        EmptyStateBlock(
                            title = title,
                            message = message,
                            primaryAction = wrapAction(target.primaryAction, screenName),
                            secondaryAction = wrapAction(target.secondaryAction, screenName),
                            icon = target.icon,
                            contentPadding = contentPadding,
                        )
                    }
                    is ScreenState.Error -> {
                        val title = target.title ?: stringResource(R.string.state_error_title)
                        ErrorStateBlock(
                            title = title,
                            message = target.message,
                            primaryAction = wrapAction(target.primaryAction, screenName),
                            secondaryAction = wrapAction(target.secondaryAction, screenName),
                            icon = target.icon,
                            contentPadding = contentPadding,
                        )
                    }
                    is ScreenState.Offline -> {
                        when (target.mode) {
                            OfflineMode.WITH_CACHE -> {
                                content(contentPadding)
                                AnimatedVisibility(
                                    visible = true,
                                    enter = if (reduceMotion) fadeIn(tween(bannerDuration)) else slideInVertically(
                                        animationSpec = tween(bannerDuration),
                                    ) { -it / 2 },
                                    exit = if (reduceMotion) fadeOut(tween(bannerDuration)) else slideOutVertically(
                                        animationSpec = tween(bannerDuration),
                                    ) { -it / 2 },
                                ) {
                                    OfflineBanner(
                                        message = target.message ?: stringResource(R.string.state_offline_banner_title),
                                        lastUpdatedMillis = target.lastUpdatedMillis,
                                        action = wrapAction(target.primaryAction, screenName),
                                        modifier = Modifier
                                            .padding(
                                                horizontal = LayoutDefaults.HorizontalPadding,
                                                vertical = 8.dp,
                                            ),
                                    )
                                }
                            }
                            OfflineMode.NO_CACHE -> {
                                OfflineStateBlock(
                                    title = stringResource(R.string.state_offline_no_cache_title),
                                    message = target.message ?: stringResource(R.string.state_offline_no_cache_body),
                                    primaryAction = wrapAction(target.primaryAction, screenName),
                                    secondaryAction = wrapAction(target.secondaryAction, screenName),
                                    icon = null,
                                    contentPadding = contentPadding,
                                )
                            }
                        }
                    }
                }
            }
        }
    }
}

@Composable
private fun resolveEmptyStrings(state: ScreenState.Empty): Pair<String, String> {
    val title = state.title
    val message = state.message
    if (title != null && message != null) return title to message

    val defaultTitle = when (state.reason) {
        EmptyReason.FIRST_USE -> stringResource(R.string.state_empty_first_use_title)
        EmptyReason.NO_RESULTS -> stringResource(R.string.state_empty_no_results_title)
        EmptyReason.FILTERS_TOO_STRICT -> stringResource(R.string.state_empty_filters_too_strict_title)
        EmptyReason.NO_HISTORY -> stringResource(R.string.state_empty_no_history_title)
        EmptyReason.NO_SUBSCRIPTIONS -> stringResource(R.string.state_empty_no_subscriptions_title)
        EmptyReason.NO_MESSAGES -> stringResource(R.string.state_empty_no_messages_title)
        EmptyReason.NO_OFFERS -> stringResource(R.string.state_empty_no_offers_title)
        EmptyReason.OTHER -> stringResource(R.string.state_empty_other_title)
    }
    val defaultMessage = when (state.reason) {
        EmptyReason.FIRST_USE -> stringResource(R.string.state_empty_first_use_body)
        EmptyReason.NO_RESULTS -> stringResource(R.string.state_empty_no_results_body)
        EmptyReason.FILTERS_TOO_STRICT -> stringResource(R.string.state_empty_filters_too_strict_body)
        EmptyReason.NO_HISTORY -> stringResource(R.string.state_empty_no_history_body)
        EmptyReason.NO_SUBSCRIPTIONS -> stringResource(R.string.state_empty_no_subscriptions_body)
        EmptyReason.NO_MESSAGES -> stringResource(R.string.state_empty_no_messages_body)
        EmptyReason.NO_OFFERS -> stringResource(R.string.state_empty_no_offers_body)
        EmptyReason.OTHER -> stringResource(R.string.state_empty_other_body)
    }
    return (title ?: defaultTitle) to (message ?: defaultMessage)
}

@Composable
private fun renderStaleIndicator(
    target: ScreenState,
    modifier: Modifier,
    reduceMotion: Boolean,
    bannerDuration: Int,
    screenName: String?,
) {
    val staleInfo = when (target) {
        is ScreenState.Content -> if (target.isStale || target.lastUpdatedMillis != null) {
            StaleInfo(
                lastUpdatedMillis = target.lastUpdatedMillis,
                message = target.staleMessage,
                action = target.staleAction,
            )
        } else null
        is ScreenState.Stale -> StaleInfo(
            lastUpdatedMillis = target.lastUpdatedMillis,
            message = target.message,
            action = target.action,
        )
        else -> null
    } ?: return

    val label = staleInfo.message ?: run {
        val updatedAt = staleInfo.lastUpdatedMillis?.let { millis ->
            val formatted = java.text.DateFormat.getDateTimeInstance(
                java.text.DateFormat.SHORT,
                java.text.DateFormat.SHORT,
            ).format(java.util.Date(millis))
            stringResource(R.string.state_last_updated, formatted)
        }
        updatedAt ?: stringResource(R.string.state_stale_generic)
    }

    AnimatedVisibility(
        visible = true,
        enter = if (reduceMotion) fadeIn(tween(bannerDuration)) else slideInVertically(
            animationSpec = tween(bannerDuration),
        ) { -it / 2 },
        exit = if (reduceMotion) fadeOut(tween(bannerDuration)) else slideOutVertically(
            animationSpec = tween(bannerDuration),
        ) { -it / 2 },
    ) {
        StaleIndicator(
            message = label,
            action = wrapAction(staleInfo.action, screenName),
            modifier = modifier,
        )
    }
}

private data class StaleInfo(
    val lastUpdatedMillis: Long?,
    val message: String?,
    val action: StateAction?,
)

private fun resolveCacheAge(state: ScreenState): Long? = when (state) {
    is ScreenState.Offline -> state.lastUpdatedMillis?.let { System.currentTimeMillis() - it }
    is ScreenState.Content -> state.lastUpdatedMillis?.let { System.currentTimeMillis() - it }
    is ScreenState.Stale -> state.lastUpdatedMillis?.let { System.currentTimeMillis() - it }
    else -> null
}

private fun buildStateKey(state: ScreenState): String =
    listOfNotNull(stateType(state), stateSubType(state), stateReason(state)).joinToString("|")

private fun stateType(state: ScreenState): String = when (state) {
    is ScreenState.Loading -> "loading"
    is ScreenState.Empty -> "empty"
    is ScreenState.Error -> "error"
    is ScreenState.Offline -> "offline"
    is ScreenState.Stale -> "stale"
    is ScreenState.Content -> if (state.isStale || state.lastUpdatedMillis != null) "stale" else "content"
}

private fun stateSubType(state: ScreenState): String? = when (state) {
    is ScreenState.Loading -> when (state.phase) {
        LoadingPhase.INITIAL -> "initial"
        LoadingPhase.REFRESHING -> "refresh"
        LoadingPhase.APPEND, LoadingPhase.PAGING -> "append"
    }
    is ScreenState.Empty -> state.reason.name.lowercase()
    is ScreenState.Error -> when (state.type) {
        ErrorType.RETRYABLE -> "retryable"
        ErrorType.NON_RETRYABLE -> "non_retryable"
        ErrorType.AUTH_REQUIRED -> "auth_required"
    }
    is ScreenState.Offline -> when (state.mode) {
        OfflineMode.WITH_CACHE -> "with_cache"
        OfflineMode.NO_CACHE -> "no_cache"
    }
    is ScreenState.Stale -> "content"
    is ScreenState.Content -> if (state.isStale || state.lastUpdatedMillis != null) "content" else null
}

private fun stateReason(state: ScreenState): String? = null

private fun isOverlayState(state: ScreenState): Boolean = when (state) {
    is ScreenState.Content -> true
    is ScreenState.Stale -> true
    is ScreenState.Loading -> state.phase != LoadingPhase.INITIAL
    is ScreenState.Offline -> state.mode == OfflineMode.WITH_CACHE
    else -> false
}

private fun wrapAction(
    action: StateAction?,
    screenName: String?,
): StateAction? {
    if (action == null) return null
    if (screenName.isNullOrBlank()) return action
    return action.copy(onAction = {
        val event = when (action.type) {
            StateActionType.RETRY -> "retry_tap"
            StateActionType.RESET_FILTERS -> "reset_filters_tap"
            StateActionType.CHANGE_QUERY -> "change_query_tap"
            StateActionType.CREATE -> "create_tap"
            StateActionType.LOGIN -> "login_tap"
            StateActionType.OPEN_SETTINGS -> "open_settings_tap"
            StateActionType.GENERIC -> null
        }
        if (event != null) {
            FlowMetrics.markStateAction(
                screen = screenName,
                type = event,
            )
        }
        action.onAction()
    })
}
