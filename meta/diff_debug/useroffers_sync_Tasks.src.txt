package com.example.shoppingassistant.feature.pages.useroffers.sync

import com.example.shoppingassistant.domain.offers.TrackedOfferInput
import kotlinx.coroutines.flow.StateFlow

enum class UserOfferSyncStatus {
    CREATED,
    SYNCING,
    SYNCED,
    FAILED,
}

data class UserOfferSyncEntry(
    val localId: String,
    val payload: TrackedOfferInput,
    val status: UserOfferSyncStatus,
    val serverId: String? = null,
    val errorMessage: String? = null,
    val attempts: Int = 0,
    val updatedAtMillis: Long = System.currentTimeMillis(),
)

interface UserOffersSyncQueueStore {
    val entries: StateFlow<List<UserOfferSyncEntry>>
    fun enqueue(entry: UserOfferSyncEntry)
    fun update(entry: UserOfferSyncEntry)
    fun remove(localId: String)
    fun replaceAll(entries: List<UserOfferSyncEntry>)
}

interface UserOffersSyncTask {
    suspend fun enqueueAndSync(localId: String, payload: TrackedOfferInput): com.example.shoppingassistant.domain.offers.CreateTrackedOfferResult
    suspend fun syncPending()
}
