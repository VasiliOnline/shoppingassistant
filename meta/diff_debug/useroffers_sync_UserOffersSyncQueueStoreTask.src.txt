package com.example.shoppingassistant.feature.pages.useroffers.sync

import android.content.Context
import com.example.shoppingassistant.domain.ingest.SourceType
import com.example.shoppingassistant.domain.offers.OfferCategory
import com.example.shoppingassistant.domain.offers.TrackedOfferInput
import com.example.shoppingassistant.domain.offers.TrackedOfferSource
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import org.json.JSONArray
import org.json.JSONObject

class UserOffersSyncQueueStoreTask(
    context: Context,
) : UserOffersSyncQueueStore {
    private val prefs = context.getSharedPreferences("user_offers_sync_queue", Context.MODE_PRIVATE)
    private val _entries = MutableStateFlow<List<UserOfferSyncEntry>>(emptyList())

    override val entries: StateFlow<List<UserOfferSyncEntry>> = _entries

    init {
        loadFromStorage()
    }

    override fun enqueue(entry: UserOfferSyncEntry) {
        val updated = upsert(entry, _entries.value)
        updateItems(updated)
    }

    override fun update(entry: UserOfferSyncEntry) {
        val updated = upsert(entry, _entries.value)
        updateItems(updated)
    }

    override fun remove(localId: String) {
        updateItems(_entries.value.filterNot { it.localId == localId })
    }

    override fun replaceAll(entries: List<UserOfferSyncEntry>) {
        updateItems(entries)
    }

    private fun updateItems(entries: List<UserOfferSyncEntry>) {
        val deduped = entries.distinctBy { it.localId }
        _entries.value = deduped
        persist(deduped)
    }

    private fun upsert(entry: UserOfferSyncEntry, entries: List<UserOfferSyncEntry>): List<UserOfferSyncEntry> {
        val idx = entries.indexOfFirst { it.localId == entry.localId }
        return if (idx >= 0) {
            entries.toMutableList().apply { set(idx, entry) }
        } else {
            listOf(entry) + entries
        }
    }

    private fun loadFromStorage() {
        val raw = prefs.getString(KEY_ENTRIES, null) ?: return
        runCatching {
            val array = JSONArray(raw)
            val parsed = buildList {
                for (i in 0 until array.length()) {
                    val obj = array.optJSONObject(i) ?: continue
                    parseEntry(obj)?.let { add(it) }
                }
            }
            _entries.value = parsed
        }
    }

    private fun persist(entries: List<UserOfferSyncEntry>) {
        runCatching {
            val array = JSONArray()
            entries.forEach { array.put(encodeEntry(it)) }
            prefs.edit().putString(KEY_ENTRIES, array.toString()).apply()
        }
    }

    private fun encodeEntry(entry: UserOfferSyncEntry): JSONObject = JSONObject().apply {
        put("localId", entry.localId)
        put("status", entry.status.name)
        put("serverId", entry.serverId)
        put("errorMessage", entry.errorMessage)
        put("attempts", entry.attempts)
        put("updatedAtMillis", entry.updatedAtMillis)
        put("payload", encodePayload(entry.payload))
    }

    private fun encodePayload(payload: TrackedOfferInput): JSONObject = JSONObject().apply {
        put("userId", payload.userId)
        put("title", payload.title)
        put("category", payload.category.name)
        put("brand", payload.brand)
        put("model", payload.model)
        put("primaryAttribute", payload.primaryAttribute)
        put("priceValue", payload.priceValue)
        put("currency", payload.currency)
        put("description", payload.description)
        put("imageUrls", JSONArray(payload.imageUrls))
        put("attributes", JSONObject(payload.attributes))
        put("source", encodeSource(payload.source))
    }

    private fun encodeSource(source: TrackedOfferSource): JSONObject = JSONObject().apply {
        put("sourceType", source.sourceType.name)
        put("sourceId", source.sourceId)
        put("url", source.url)
        put("canonicalUrl", source.canonicalUrl)
        put("listingId", source.listingId)
        put("domainName", source.domainName)
        put("sourceIconUrl", source.sourceIconUrl)
    }

    private fun parseEntry(obj: JSONObject): UserOfferSyncEntry? {
        val localId = obj.optString("localId").takeIf { it.isNotBlank() } ?: return null
        val status = obj.optString("status").toSyncStatus()
        val payloadObj = obj.optJSONObject("payload") ?: return null
        val payload = parsePayload(payloadObj) ?: return null
        return UserOfferSyncEntry(
            localId = localId,
            payload = payload,
            status = status,
            serverId = obj.optString("serverId").takeIf { it.isNotBlank() },
            errorMessage = obj.optString("errorMessage").takeIf { it.isNotBlank() },
            attempts = obj.optInt("attempts", 0).coerceAtLeast(0),
            updatedAtMillis = obj.optLong("updatedAtMillis").takeIf { it > 0 } ?: System.currentTimeMillis(),
        )
    }

    private fun parsePayload(obj: JSONObject): TrackedOfferInput? {
        val userId = obj.optString("userId").takeIf { it.isNotBlank() } ?: return null
        val title = obj.optString("title").takeIf { it.isNotBlank() } ?: return null
        val category = obj.optString("category").toOfferCategory()
        val priceValue = obj.optDouble("priceValue").takeIf { !it.isNaN() && it > 0 } ?: return null
        val currency = obj.optString("currency").ifBlank { "USD" }
        val sourceObj = obj.optJSONObject("source") ?: return null
        val source = parseSource(sourceObj) ?: return null
        val imageUrls = obj.optJSONArray("imageUrls").toStringList()
        val attributes = obj.optJSONObject("attributes").toStringMap()

        return TrackedOfferInput(
            userId = userId,
            title = title,
            category = category,
            brand = obj.optString("brand").takeIf { it.isNotBlank() },
            model = obj.optString("model").takeIf { it.isNotBlank() },
            primaryAttribute = obj.optString("primaryAttribute").takeIf { it.isNotBlank() },
            priceValue = priceValue,
            currency = currency,
            imageUrls = imageUrls,
            description = obj.optString("description").takeIf { it.isNotBlank() },
            attributes = attributes,
            source = source,
        )
    }

    private fun parseSource(obj: JSONObject): TrackedOfferSource? {
        val type = obj.optString("sourceType").toSourceType()
        val url = obj.optString("url").takeIf { it.isNotBlank() } ?: return null
        return TrackedOfferSource(
            sourceType = type,
            sourceId = obj.optString("sourceId").takeIf { it.isNotBlank() },
            url = url,
            canonicalUrl = obj.optString("canonicalUrl").takeIf { it.isNotBlank() },
            listingId = obj.optString("listingId").takeIf { it.isNotBlank() },
            domainName = obj.optString("domainName").takeIf { it.isNotBlank() },
            sourceIconUrl = obj.optString("sourceIconUrl").takeIf { it.isNotBlank() },
        )
    }

    private fun String.toSyncStatus(): UserOfferSyncStatus = runCatching {
        UserOfferSyncStatus.valueOf(this)
    }.getOrDefault(UserOfferSyncStatus.CREATED)

    private fun String.toOfferCategory(): OfferCategory = runCatching {
        OfferCategory.valueOf(this)
    }.getOrDefault(OfferCategory.OTHER)

    private fun String.toSourceType(): SourceType = runCatching {
        SourceType.valueOf(this)
    }.getOrDefault(SourceType.UNKNOWN)

    private fun JSONArray?.toStringList(): List<String> {
        if (this == null) return emptyList()
        return buildList {
            for (i in 0 until length()) {
                val value = optString(i).takeIf { it.isNotBlank() }
                if (value != null) add(value)
            }
        }
    }

    private fun JSONObject?.toStringMap(): Map<String, String> {
        if (this == null) return emptyMap()
        val map = linkedMapOf<String, String>()
        keys().forEach { key ->
            val value = optString(key).takeIf { it.isNotBlank() } ?: return@forEach
            map[key] = value
        }
        return map
    }

    private companion object {
        const val KEY_ENTRIES = "entries"
    }
}
