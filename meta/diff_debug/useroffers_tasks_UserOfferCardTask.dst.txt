@Composable
fun UserOfferCard(
    offer: UserOfferCardUi,
    layout: UserOffersLayout,
    actions: List<UserOfferAction>,
    onAction: (UserOfferAction) -> Unit,
    onOpen: () -> Unit,
    selectionMode: Boolean = false,
    selected: Boolean = false,
    onToggleSelect: () -> Unit = {},
    onLongPress: () -> Unit = {},
    modifier: Modifier = Modifier,
) {
    when (layout) {
        UserOffersLayout.HORIZONTAL -> UserOfferCardHorizontal(
            offer = offer,
            actions = actions,
            onAction = onAction,
            onOpen = onOpen,
            selectionMode = selectionMode,
            selected = selected,
            onToggleSelect = onToggleSelect,
            onLongPress = onLongPress,
            modifier = modifier,
        )
        UserOffersLayout.VERTICAL -> UserOfferCardVertical(
            offer = offer,
            actions = actions,
            onAction = onAction,
            onOpen = onOpen,
            selectionMode = selectionMode,
            selected = selected,
            onToggleSelect = onToggleSelect,
            onLongPress = onLongPress,
            modifier = modifier,
        )
    }
}

@Composable
private fun UserOfferCardHorizontal(
    offer: UserOfferCardUi,
    actions: List<UserOfferAction>,
    onAction: (UserOfferAction) -> Unit,
    onOpen: () -> Unit,
    selectionMode: Boolean,
    selected: Boolean,
    onToggleSelect: () -> Unit,
    onLongPress: () -> Unit,
    modifier: Modifier = Modifier,
) {
    val clickModifier = Modifier.combinedClickable(
        onClick = { if (selectionMode) onToggleSelect() else onOpen() },
        onLongClick = onLongPress,
    )
    Card(
        modifier = modifier
            .fillMaxWidth()
            .heightIn(min = 280.dp)
            .then(clickModifier),
        shape = RoundedCornerShape(18.dp),
        colors = CardDefaults.cardColors(containerColor = MaterialTheme.colorScheme.surface),
        elevation = CardDefaults.cardElevation(defaultElevation = 2.dp),
        border = BorderStroke(1.dp, MaterialTheme.colorScheme.outlineVariant.copy(alpha = 0.45f)),
    ) {
        Box {
            Column(verticalArrangement = Arrangement.spacedBy(0.dp)) {
                OfferCover(
                    offer = offer,
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(180.dp),
                    corner = RoundedCornerShape(18.dp),
                )
                Column(
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(horizontal = 14.dp, vertical = 12.dp),
                    verticalArrangement = Arrangement.spacedBy(8.dp),
                ) {
                    HeaderRow(
                        offer = offer,
                        actions = actions,
                        onAction = onAction,
                    )
                    OfferMetaRow(offer)
                    MetricsRow(offer)
                }
            }
            StatusStripe(
                status = offer.status,
                modifier = Modifier.align(Alignment.CenterStart),
            )
            SelectionBadge(
                selected = selected,
                visible = selectionMode,
                modifier = Modifier.align(Alignment.TopStart),
            )
        }
    }
}

@Composable
private fun OfferMetaRow(offer: UserOfferCardUi) {
    val createdAt = offer.publishedAtMillis ?: offer.updatedAtMillis
    val createdLabel = createdAt?.let { "Создано ${formatDateTime(it)}" }
    val completedLabel = offer.completedAtMillis?.let { "Завершено ${formatDateTime(it)}" }
    val updatedLabel = offer.updatedAtMillis
        ?.takeIf { createdAt == null || it != createdAt }
        ?.let { "Обновлено ${formatDateTime(it)}" }
    val showRemoval = offer.status == UserOfferStatus.ACTIVE || offer.status == UserOfferStatus.PAUSED
    val removalLabel = if (showRemoval) rememberRemovalLabel(offer.expiresAtMillis) else null
    val removalUrgent = showRemoval && isRemovalUrgent(offer.expiresAtMillis)
    val sourceUpdatedLabel = rememberSourceUpdatedLabel(offer.sourceUpdatedAtMillis)
    val sourceLabel = buildSourceLabel(offer.sourceName, sourceUpdatedLabel)
    Column(
        verticalArrangement = Arrangement.spacedBy(6.dp),
    ) {
        Row(
            horizontalArrangement = Arrangement.spacedBy(6.dp),
            verticalAlignment = Alignment.CenterVertically,
        ) {
            StatusBadge(
                label = statusShortLabel(offer.status),
                color = statusColor(offer.status),
            )
            when (offer.publicationStatus) {
                UserOfferPublicationStatus.ON_MODERATION -> StatusBadge(
                    label = "На модерации",
                    color = MaterialTheme.colorScheme.tertiary,
                )
                UserOfferPublicationStatus.ERROR -> StatusBadge(
                    label = "Ошибка публикации",
                    color = MaterialTheme.colorScheme.error,
                )
                UserOfferPublicationStatus.PUBLISHED -> Unit
            }
            if (offer.isPromoted) {
                StatusBadge(
                    label = "Продвигается",
                    color = MaterialTheme.colorScheme.secondary,
                )
            }
        }
        createdLabel?.let {
            MetaLine(icon = Icons.Outlined.Event, text = it)
        }
        updatedLabel?.let {
            MetaLine(icon = Icons.Outlined.Update, text = it)
        }
        sourceLabel?.let {
            SourceMetaLine(
                iconUrl = offer.sourceIconUrl,
                text = it,
            )
        }
        completedLabel?.let {
            MetaLine(icon = Icons.Outlined.CheckCircle, text = it)
        }
        removalLabel?.let {
            MetaLine(
                icon = Icons.Outlined.Timer,
                text = it,
                highlight = removalUrgent,
            )
        }
    }
}

@Composable
private fun MetaLine(
    icon: ImageVector,
    text: String,
    highlight: Boolean = false,
    modifier: Modifier = Modifier,
) {
    val color = if (highlight) {

    Row(

        verticalAlignment = Alignment.CenterVertically,
    ) {

            imageVector = icon,
            contentDescription = null,
            tint = color,
            modifier = Modifier.size(14.dp),
        )

            text = text,
            style = MaterialTheme.typography.bodySmall,
            color = color,
        )
    }
}

) {
    val color = if (highlight) {
        MaterialTheme.colorScheme.error
    } else {
        MaterialTheme.colorScheme.onSurfaceVariant
    }
    Row(
        modifier = modifier,
        horizontalArrangement = Arrangement.spacedBy(6.dp),
        verticalAlignment = Alignment.CenterVertically,
    ) {
        if (iconUrl.isNullOrBlank()) {
            Icon(
                imageVector = Icons.Outlined.Language,
                contentDescription = null,
                tint = color,
                modifier = Modifier.size(14.dp),

            AsyncImage(
                model = iconUrl,
                contentDescription = null,
                modifier = Modifier
                    .size(14.dp)

                contentScale = ContentScale.Crop,
            )
        }
        Text(
            text = text,
            style = MaterialTheme.typography.bodySmall,
            color = color,
        )
    }
}

@Composable
private fun MetricsRow(offer: UserOfferCardUi) {
    val hasMetrics = offer.viewsCount != null || offer.favoritesCount != null || offer.messagesCount != null
    val healthLabel = when (offer.health) {
        UserOfferHealth.BELOW_AVERAGE -> "ниже среднего"
        UserOfferHealth.OK -> "норм"
        null -> null
    }
    if (!hasMetrics && healthLabel == null) return

    Row(
        modifier = Modifier.fillMaxWidth(),
        horizontalArrangement = Arrangement.spacedBy(10.dp),
        verticalAlignment = Alignment.CenterVertically,
    ) {
        offer.viewsCount?.let {
            MetricItem(icon = Icons.Outlined.Visibility, value = it.toString())
        }
        offer.favoritesCount?.let {
            MetricItem(icon = Icons.Outlined.FavoriteBorder, value = it.toString())
        }
        offer.messagesCount?.let {
            MetricItem(icon = Icons.Outlined.ChatBubbleOutline, value = it.toString())
        }
        healthLabel?.let {
            StatusBadge(
                label = it,
                color = if (offer.health == UserOfferHealth.BELOW_AVERAGE) {
                    MaterialTheme.colorScheme.error
                } else {
                    MaterialTheme.colorScheme.secondary
                },
            )
        }
    }
}

@Composable
private fun ActionsMenu(
    offer: UserOfferCardUi,
    actions: List<UserOfferAction>,
    onAction: (UserOfferAction) -> Unit,
) {
    var expanded by remember { mutableStateOf(false) }
    Box {
        IconButton(
            onClick = { expanded = true },
            modifier = Modifier.size(42.dp),
        ) {
            Icon(
                imageVector = Icons.Outlined.MoreVert,
                contentDescription = "Действия",
                tint = MaterialTheme.colorScheme.onSurface,
                modifier = Modifier.size(26.dp),
            )
        }

            actions.forEach { action ->

                    onClick = {
                        expanded = false
                        onAction(action)
                    },
                )
            }
        }
    }
}

@Composable
private fun OfferCover(
    offer: UserOfferCardUi,
    modifier: Modifier = Modifier,
    corner: RoundedCornerShape = RoundedCornerShape(14.dp),
) {
    val painter = if (offer.coverUrl.isNullOrBlank()) {
        rememberAsyncImagePainter(model = Icons.Outlined.Timer)
    } else {
        rememberAsyncImagePainter(model = offer.coverUrl)
    }
    Box(
        modifier = modifier
            .clip(corner)
            .background(MaterialTheme.colorScheme.surfaceVariant.copy(alpha = 0.4f)),
        contentAlignment = Alignment.BottomStart,
    ) {
        if (offer.coverUrl.isNullOrBlank()) {
            Image(
                painter = painter,
                contentDescription = null,
                modifier = Modifier
                    .fillMaxSize()
                    .padding(18.dp),
                alpha = 0.4f,
            )
        } else {
            AsyncImage(
                model = offer.coverUrl,
                contentDescription = "Фото товара",
                modifier = Modifier.fillMaxSize(),
                contentScale = ContentScale.Crop,
            )
            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .background(
                        Brush.verticalGradient(
                            listOf(
                                Color.Transparent,
                                Color.Black.copy(alpha = 0.35f),
                            )
                        )
                    ),
            )
        }

        val badge = rememberRemovalBadge(
            expiresAtMillis = offer.expiresAtMillis,
            status = offer.status,
        )
        if (badge != null) {
            Row(
                modifier = Modifier
                    .padding(12.dp)
                    .background(
                        color = MaterialTheme.colorScheme.surface.copy(alpha = 0.85f),
                        shape = RoundedCornerShape(12.dp),
                    )
                    .padding(horizontal = 10.dp, vertical = 6.dp),
                horizontalArrangement = Arrangement.spacedBy(6.dp),
                verticalAlignment = Alignment.CenterVertically,
            ) {
                Icon(
                    imageVector = Icons.Outlined.Timer,
                    contentDescription = null,
                    tint = MaterialTheme.colorScheme.onSurface,
                    modifier = Modifier.size(16.dp),
                )
                Text(
                    text = badge,
                    style = MaterialTheme.typography.labelMedium,
                    color = MaterialTheme.colorScheme.onSurface,
                )
            }
        }
    }
}

@Composable
private fun rememberRemovalLabel(expiresAtMillis: Long?): String? {
    if (expiresAtMillis == null) return null
    var label by remember(expiresAtMillis) {
        mutableStateOf(removalLabel(expiresAtMillis))
    }
    LaunchedEffect(expiresAtMillis) {
        while (true) {
            label = removalLabel(expiresAtMillis)
            delay(removalUpdateDelay(expiresAtMillis))
        }
    }
    return label
}

@Composable
private fun rememberRemovalBadge(
    expiresAtMillis: Long?,
    status: UserOfferStatus,
): String? {
    if (expiresAtMillis == null) return null
    if (status == UserOfferStatus.FINISHED || status == UserOfferStatus.ARCHIVED) return null
    var label by remember(expiresAtMillis, status) {
        mutableStateOf(removalBadgeLabel(expiresAtMillis))
    }
    LaunchedEffect(expiresAtMillis, status) {
        while (true) {
            label = removalBadgeLabel(expiresAtMillis)
            delay(removalUpdateDelay(expiresAtMillis))
        }
    }
    return label
}

private fun removalLabel(expiresAtMillis: Long): String {
    val minutesLeft = remainingMinutes(expiresAtMillis)
    if (minutesLeft <= 0) return "Снято"
    if (minutesLeft < 60) return "Снимется через $minutesLeft мин"
    val hoursLeft = (minutesLeft + 59) / 60
    if (hoursLeft < 24) return "Снимется через ${pluralize(hoursLeft, "час", "часа", "часов")}"
    val daysLeft = (hoursLeft + 23) / 24
    return "Снимется через ${pluralize(daysLeft, "день", "дня", "дней")}"
}

private fun removalBadgeLabel(expiresAtMillis: Long): String {
    val minutesLeft = remainingMinutes(expiresAtMillis)
    if (minutesLeft <= 0) return "Снято"
    if (minutesLeft < 60) return "Снятие через $minutesLeft мин"
    val hoursLeft = (minutesLeft + 59) / 60
    return "Снятие через ${pluralize(hoursLeft, "час", "часа", "часов")}"
}

private fun removalUpdateDelay(expiresAtMillis: Long): Long {
    val minutesLeft = remainingMinutes(expiresAtMillis)
    return if (minutesLeft < 60) 60_000L else 30 * 60_000L
}

private fun remainingMinutes(expiresAtMillis: Long): Long {
    val diff = expiresAtMillis - System.currentTimeMillis()
    if (diff <= 0) return 0L
    return max(1L, diff / 60_000L)
}

private fun isRemovalUrgent(expiresAtMillis: Long?): Boolean {
    if (expiresAtMillis == null) return false
    return expiresAtMillis - System.currentTimeMillis() <= 60 * 60_000L
}

@Composable
private fun rememberSourceUpdatedLabel(updatedAtMillis: Long?): String? {
    if (updatedAtMillis == null) return null
    var label by remember(updatedAtMillis) {
        mutableStateOf(relativeUpdatedLabel(updatedAtMillis))
    }
    LaunchedEffect(updatedAtMillis) {
        while (true) {
            label = relativeUpdatedLabel(updatedAtMillis)
            delay(relativeUpdateDelay(updatedAtMillis))
        }
    }
    return label
}

private fun relativeUpdatedLabel(updatedAtMillis: Long): String {
    val minutes = elapsedMinutes(updatedAtMillis)
    if (minutes <= 0L) return "обновлено только что"
    if (minutes < 60) return "обновлено $minutes мин назад"
    val hours = (minutes + 59) / 60
    if (hours < 24) return "обновлено ${pluralize(hours, "час", "часа", "часов")} назад"
    val days = (hours + 23) / 24
    return "обновлено ${pluralize(days, "день", "дня", "дней")} назад"
}

private fun relativeUpdateDelay(updatedAtMillis: Long): Long {
    val minutes = elapsedMinutes(updatedAtMillis)
    return when {
        minutes < 60 -> 60_000L
        minutes < 24 * 60 -> 60 * 60_000L
        else -> 24 * 60 * 60_000L
    }
}

private fun elapsedMinutes(updatedAtMillis: Long): Long {
    val diff = System.currentTimeMillis() - updatedAtMillis
    if (diff <= 0) return 0L
    return diff / 60_000L
}

private fun statusShortLabel(status: UserOfferStatus): String = when (status) {
    UserOfferStatus.ACTIVE -> "Активно"
    UserOfferStatus.PAUSED -> "Приостановлено"
    UserOfferStatus.DRAFT -> "Черновик"
    UserOfferStatus.FINISHED -> "Завершено"
    UserOfferStatus.ARCHIVED -> "В архиве"
}

private fun buildSourceLabel(sourceName: String?, sourceUpdatedLabel: String?): String? {
    val name = sourceName?.trim()?.ifBlank { null }
    val updated = sourceUpdatedLabel?.trim()?.ifBlank { null }
    if (name == null && updated == null) return null
    return when {
        name != null && updated != null -> "Источник: $name · $updated"
        name != null -> "Источник: $name"
        else -> "Источник: $updated"
    }
}

private fun formatDateTime(epochMillis: Long): String {
    val sdf = SimpleDateFormat("dd MMM, HH:mm", Locale.getDefault())
    return sdf.format(Date(epochMillis))
}

private fun pluralize(value: Long, one: String, few: String, many: String): String {
    val mod10 = (value % 10).toInt()
    val mod100 = (value % 100).toInt()
    val word = when {
        mod10 == 1 && mod100 != 11 -> one
        mod10 in 2..4 && mod100 !in 12..14 -> few
        else -> many
    }
    return "$value $word"
}
