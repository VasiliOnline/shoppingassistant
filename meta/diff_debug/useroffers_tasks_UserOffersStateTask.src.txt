package com.example.shoppingassistant.feature.pages.useroffers.tasks

import androidx.compose.runtime.Composable
import androidx.compose.runtime.LaunchedEffect
import androidx.compose.runtime.getValue
import androidx.compose.runtime.collectAsState
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.remember
import androidx.compose.runtime.setValue
import com.example.shoppingassistant.domain.auth.GetCurrentUserUseCase
import com.example.shoppingassistant.domain.model.AuthUser
import com.example.shoppingassistant.feature.pages.useroffers.UserOfferCardUi
import com.example.shoppingassistant.feature.pages.useroffers.UserOfferStatus
import com.example.shoppingassistant.feature.pages.useroffers.UserOffersLoadState
import com.example.shoppingassistant.feature.pages.useroffers.tasks.UserOffersCreatedStore
import org.koin.java.KoinJavaComponent.get as koinGet

/**
 * Loader состояния страницы «Мои товары».
 * Использует текущего пользователя и локальное хранилище созданных офферов.
 */
@Composable
fun rememberUserOffersState(
    refreshKey: Int = 0,
): UserOffersLoadState {
    var state by remember {
        mutableStateOf(
            UserOffersLoadState(
                isLoading = true,
                isAuthorized = false,
            )
        )
    }

    val getCurrentUserUseCase: GetCurrentUserUseCase = remember {
        koinGet(GetCurrentUserUseCase::class.java)
    }
    val createdStore: UserOffersCreatedStore = remember {
        koinGet(UserOffersCreatedStore::class.java)
    }
    val createdOffers by createdStore.items.collectAsState()
    val createdError by createdStore.errorMessage.collectAsState()

    var currentUser by remember { mutableStateOf<AuthUser?>(null) }

    LaunchedEffect(refreshKey) {
        state = state.copy(isLoading = true)
        val user = runCatching { getCurrentUserUseCase() }.getOrNull()
        if (user == null) {
            state = UserOffersLoadState(
                isLoading = false,
                isAuthorized = false,
                userName = null,
                activeOffers = emptyList(),
                completedOffers = emptyList(),
                errorMessage = null,
                isOffline = false,
            )
            return@LaunchedEffect
        }

        currentUser = user
        state = UserOffersLoadState(
            isLoading = false,
            isAuthorized = true,
            userName = user.displayName ?: user.email,
            activeOffers = splitOffers(createdOffers).first,
            completedOffers = splitOffers(createdOffers).second,
            errorMessage = createdError,
            isOffline = createdError != null,
        )
    }

    LaunchedEffect(createdOffers, currentUser, createdError) {
        val user = currentUser ?: return@LaunchedEffect
        val (active, completed) = splitOffers(createdOffers)
        state = state.copy(
            isLoading = false,
            isAuthorized = true,
            userName = user.displayName ?: user.email,
            activeOffers = active,
            completedOffers = completed,
            errorMessage = createdError,
            isOffline = createdError != null,
        )
    }

    return state
}

private fun splitOffers(
    offers: List<UserOfferCardUi>,
): Pair<List<UserOfferCardUi>, List<UserOfferCardUi>> {
    val activeStatuses = setOf(
        UserOfferStatus.ACTIVE,
        UserOfferStatus.PAUSED,
        UserOfferStatus.DRAFT,
    )
    val active = offers.filter { it.status in activeStatuses }
    val completed = offers.filter { it.status !in activeStatuses }
    return active to completed
}
